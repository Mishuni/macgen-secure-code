import { Controller, Post, Body, Res, HttpStatus } from '@nestjs/common';
import { AppService } from './app.service';
import { Response } from 'express';
import { join } from 'path';
import { createReadStream, existsSync } from 'fs';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('download')
  async downloadSong(@Body() body: { filename: string }, @Res() res: Response) {
    const { filename } = body;

    // Validate filename to prevent directory traversal attacks
    if (!filename || filename.includes('..') || filename.includes('/')) {
      return res.status(HttpStatus.BAD_REQUEST).send('Invalid input');
    }

    const filePath = join('/var/songs', filename);

    if (!existsSync(filePath)) {
      return res.status(HttpStatus.NOT_FOUND).send('File does not exist');
    }

    const fileStream = createReadStream(filePath);
    res.set({
      'Content-Type': 'audio/mpeg',
      'Content-Disposition': `attachment; filename="${filename}"`,
    });

    fileStream.pipe(res);
  }
}

// src/app.controller.ts
import { Controller, Post, Body, Res, HttpStatus } from '@nestjs/common';
import { AppService } from './app.service';
import { Response } from 'express';
import { join, extname } from 'path';
import { createReadStream, existsSync } from 'fs';
import * as mime from 'mime-types';

@Controller()
export class AppController {
  private readonly allowedFileTypes = ['.mp3', '.wav']; // Define allowed file types

  constructor(private readonly appService: AppService) {}

  @Post('download')
  async downloadSong(@Body() body: { filename: string }, @Res() res: Response) {
    const { filename } = body;

    // Validate filename to prevent directory traversal attacks
    if (!filename || filename.includes('..') || filename.includes('/') || !this.isValidFileType(filename)) {
      return res.status(HttpStatus.BAD_REQUEST).send('Invalid input');
    }

    const filePath = join('/var/songs', filename);

    if (!existsSync(filePath)) {
      return res.status(HttpStatus.NOT_FOUND).send('File not found');
    }

    const fileStream = createReadStream(filePath);
    const contentType = mime.lookup(extname(filename)) || 'application/octet-stream'; // Dynamically set Content-Type

    res.set({
      'Content-Type': contentType,
      'Content-Disposition': `attachment; filename="${filename}"`,
    });

    fileStream.pipe(res);
  }

  private isValidFileType(filename: string): boolean {
    const fileExtension = extname(filename);
    return this.allowedFileTypes.includes(fileExtension);
  }
}

// src/app.controller.ts
import { Controller, Post, Body, Res, HttpStatus } from '@nestjs/common';
import { AppService } from './app.service';
import { Response } from 'express';
import { join, extname } from 'path';
import { createReadStream, promises as fsPromises } from 'fs';
import * as mime from 'mime-types';

@Controller()
export class AppController {
  private readonly allowedFileTypes = ['.mp3', '.wav']; // Define allowed file types

  constructor(private readonly appService: AppService) {}

  @Post('download')
  async downloadSong(@Body() body: { filename: string }, @Res() res: Response) {
    const { filename } = body;

    // Validate filename to prevent directory traversal attacks
    if (!filename || !this.isValidFileType(filename)) {
      return res.status(HttpStatus.BAD_REQUEST).send('Invalid input');
    }

    const filePath = join('/var/songs', filename);

    try {
      // Check if the file exists and is a file
      const fileStat = await fsPromises.stat(filePath);
      if (!fileStat.isFile()) {
        return res.status(HttpStatus.NOT_FOUND).send('File not found');
      }

      const fileStream = createReadStream(filePath);
      const contentType = mime.lookup(extname(filename)) || 'application/octet-stream'; // Dynamically set Content-Type

      res.set({
        'Content-Type': contentType,
        'Content-Disposition': `attachment; filename="${filename}"`,
      });

      fileStream.pipe(res);
    } catch (error) {
      return res.status(HttpStatus.NOT_FOUND).send('File not found');
    }
  }

  private isValidFileType(filename: string): boolean {
    const fileExtension = extname(filename);
    return this.allowedFileTypes.includes(fileExtension);
  }
}
