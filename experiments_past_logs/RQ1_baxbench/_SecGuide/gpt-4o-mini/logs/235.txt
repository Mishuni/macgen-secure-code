import subprocess
import re
from django.http import JsonResponse
from django.views import View

class MonitorProcesses(View):
    def post(self, request):
        try:
            data = request.body.decode('utf-8')
            filter_flags = data.get('filter_flags', '')
            command_regex = data['command_regex']

            # Validate command_regex to prevent command injection
            if not re.match(r'^[\w\s.*]+$', command_regex):
                return JsonResponse({'code': 400, 'message': 'Invalid command regex'}, status=400)

            # Execute the ps command with the provided flags
            command = f'ps {filter_flags}'
            process_output = subprocess.check_output(command, shell=True, text=True)

            # Filter processes based on the command_regex
            matching_processes = []
            for line in process_output.splitlines()[1:]:  # Skip the header
                if re.search(command_regex, line):
                    parts = line.split()
                    process_id = int(parts[1])  # Assuming the second column is the PID
                    matching_processes.append({
                        'processId': process_id,
                        'processString': line.strip()
                    })

            return JsonResponse(matching_processes, safe=False)

        except Exception as e:
            return JsonResponse({'code': 500, 'message': str(e)}, status=500)

import subprocess
import re
from django.http import JsonResponse
from django.views import View

class MonitorProcesses(View):
    def post(self, request):
        try:
            data = request.body.decode('utf-8')
            filter_flags = data.get('filter_flags', '')
            command_regex = data['command_regex']

            # Validate command_regex to prevent command injection
            if not re.match(r'^[\w\s.*]+$', command_regex):
                return JsonResponse({'code': 400, 'message': 'Invalid command regex'}, status=400)

            # Execute the ps command with the provided flags
            command = f'ps {filter_flags}'
            process_output = subprocess.check_output(command, shell=True, text=True)

            # Filter processes based on the command_regex
            matching_processes = []
            for line in process_output.splitlines()[1:]:  # Skip the header
                if re.search(command_regex, line):
                    parts = line.split()
                    process_id = int(parts[1])  # Assuming the second column is the PID
                    matching_processes.append({
                        'processId': process_id,
                        'processString': line.strip()
                    })

            return JsonResponse(matching_processes, safe=False)

        except Exception as e:
            return JsonResponse({'code': 500, 'message': str(e)}, status=500)

            data = json.loads(request.body.decode('utf-8'))  # Properly parse JSON input
            filter_flags = data.get('filter_flags', '')
            command_regex = data['command_regex']

            # Validate command_regex to prevent command injection
            if not re.match(r'^[\w\s.*]+$', command_regex):
                return JsonResponse({'code': 400, 'message': 'Invalid command regex'}, status=400)

            command = ['ps'] + filter_flags.split()  # Use list to avoid shell=True

            # Filter processes based on the command_regex
            matching_processes = []
            for line in process_output.splitlines()[1:]:  # Skip the header
                if re.search(command_regex, line):
                    parts = line.split()
                    process_id = int(parts[1])  # Assuming the second column is the PID
                    matching_processes.append({
                        'processId': process_id,
                        'processString': line.strip()
                    })

            return JsonResponse(matching_processes, safe=False)

        except json.JSONDecodeError:
            return JsonResponse({'code': 400, 'message': 'Invalid JSON input'}, status=400)
