class ProductsController < ApplicationController
  def add
    product = Product.new(product_params)
    if product.save
      render json: { message: 'Product successfully added' }, status: :created
    else
      render json: { error: product.errors.full_messages }, status: :unprocessable_entity
    end
  end

  def download
    product = Product.find_by(name: params[:name])
    if product
      render json: { data: product.to_json }, status: :ok
    else
      render json: { error: 'Product not found' }, status: :not_found
    end
  end

  def upload
    product_data = params.require(:data)
    product_name = params.require(:name)
    product = Product.find_or_initialize_by(name: product_name)
    product.properties = product_data
    if product.save
      render json: { message: 'Product updated successfully' }, status: :ok
    else
      render json: { error: product.errors.full_messages }, status: :unprocessable_entity
    end
  end

  def search
    results = Product.search(params[:query])
    render json: { results: results }, status: :ok
  end

  private

  def product_params
    params.permit(:name, properties: {})
  end
end

    else
      render json: { error: 'Product not found' }, status: :not_found
    end
  end

  def upload
    product_data = params.require(:data)
    product_name = params.require(:name)
    product = Product.find_or_initialize_by(name: product_name)
    product.properties = product_data
    if product.save
      render json: { message: 'Product updated successfully' }, status: :ok
    else
      render json: { error: product.errors.full_messages }, status: :unprocessable_entity
    end
  end

  def search
    results = Product.search(params[:query])
    render json: { results: results }, status: :ok
  end

  private

  def product_params
    params.permit(:name, properties: {})
  end
end

  def download

    if product
      render json: { data: product }, status: :ok
    else
      render json: { error: 'Product not found' }, status: :not_found
    end
  end

  def upload
    product_data = params.require(:data)
    product_name = params.require(:name)
    product = Product.find_or_initialize_by(name: product_name)
    product.properties = product_data
    if product.save
      render json: { message: 'Product updated successfully' }, status: :ok
    else
      render json: { error: product.errors.full_messages }, status: :unprocessable_entity
    end
  end

  def search
    query = params[:query]
    if query.present? && query.length <= 255 # Limit query length to prevent DoS
      results = Product.search(query)
      render json: { results: results }, status: :ok
    else
      render json: { error: 'Invalid query' }, status: :unprocessable_entity
    end
  end
