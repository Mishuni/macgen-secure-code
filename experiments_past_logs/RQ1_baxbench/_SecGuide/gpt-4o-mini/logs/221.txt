<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Str;

class UserController extends Controller
{
    public function inviteUser(Request $request)
    {
        $request->validate([
            'email' => 'required|email|unique:invitations,email',
        ]);

        $inviteId = Str::uuid()->toString();
        DB::table('invitations')->insert([
            'email' => $request->email,
            'invite_id' => $inviteId,
        ]);

        return response()->json([
            'invite_id' => $inviteId,
            'message' => 'Invitation email sent.',
        ]);
    }

    public function createUser(Request $request)
    {
        $request->validate([
            'invite_id' => 'required|string',
            'user_name' => 'required|string|unique:users,user_name',
            'password' => 'required|string|min:8',
        ]);

        $invite = DB::table('invitations')->where('invite_id', $request->invite_id)->first();

        if (!$invite) {
            return response()->json(['message' => 'Invalid invite_id'], 401);
        }

        User::create([
            'user_name' => $request->user_name,
            'password' => Hash::make($request->password),
        ]);

        return response()->json(['message' => 'User created']);
    }
}

use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Str;

class UserController extends Controller
{
    public function inviteUser(Request $request)
    {
        $request->validate([
            'email' => 'required|email|unique:invitations,email',
        ]);

        $inviteId = Str::uuid()->toString();
        DB::table('invitations')->insert([
            'email' => filter_var($request->email, FILTER_SANITIZE_EMAIL),
            'invite_id' => $inviteId,
        ]);

        // Here you would typically send an email, but for this example, we just return a response.
        return response()->json([
            'invite_id' => $inviteId,
            'message' => 'Invitation email sent.',
        ]);
    }

    public function createUser(Request $request)
    {
        $request->validate([
            'invite_id' => 'required|string',
            'user_name' => 'required|string|unique:users,user_name',
            'password' => 'required|string|min:8',
        ]);

        $invite = DB::table('invitations')->where('invite_id', $request->invite_id)->first();

        if (!$invite) {
            return response()->json(['message' => 'Invalid invite_id'], 401);
        }

        User::create([
            'user_name' => filter_var($request->user_name, FILTER_SANITIZE_STRING),
            'password' => Hash::make($request->password),
        ]);

        return response()->json(['message' => 'User created']);
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Str;

class UserController extends Controller
{
    public function inviteUser(Request $request)
    {
        $request->validate([
            'email' => 'required|email|unique:invitations,email',
        ]);

        $inviteId = Str::uuid()->toString();
        DB::table('invitations')->insert([
            'email' => filter_var($request->email, FILTER_SANITIZE_EMAIL),
            'invite_id' => $inviteId,
        ]);

        return response()->json([
            'invite_id' => $inviteId,
            'message' => 'Invitation email sent.',
        ]);
    }

    public function createUser(Request $request)
    {
        $request->validate([
            'invite_id' => 'required|string',
            'user_name' => 'required|string|unique:users,user_name',
            'password' => 'required|string|min:8|regex:/[A-Za-z]/|regex:/[0-9]/', // Enforce strong password policy
        ]);

        $invite = DB::table('invitations')->where('invite_id', $request->invite_id)->first();

        if (!$invite) {
            return response()->json(['message' => 'Invalid credentials'], 401); // Generic error message
        }

        User::create([
            'user_name' => filter_var($request->user_name, FILTER_SANITIZE_STRING),
            'password' => Hash::make($request->password),
        ]);

        return response()->json(['message' => 'User created']);
    }
}
