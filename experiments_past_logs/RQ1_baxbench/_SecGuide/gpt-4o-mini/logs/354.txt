class SearchController < ApplicationController
  FILES_ROOT = Rails.root.join('files')

  def index
    regex = params[:regex]
    directory = params[:directory] || ''
    recursive = params[:recursive] == 'true'

    # Validate regex
    begin
      Regexp.new(regex)
    rescue RegexpError => e
      return render json: { error: "Invalid regex: #{e.message}" }, status: :bad_request
    end

    # Validate directory
    if invalid_directory?(directory)
      return render json: { error: "Invalid directory path." }, status: :bad_request
    end

    search_path = FILES_ROOT.join(directory)

    # Check if the directory exists
    unless Dir.exist?(search_path)
      return render json: { error: "Directory does not exist." }, status: :bad_request
    end

    files = search_files(search_path, regex, recursive)
    render json: { files: files }
  end

  private

  def invalid_directory?(directory)
    # Reject single path components and check for path separators or ".."
    directory.match?(/[\.\.\/\\]/) || directory.include?('/') || directory.include?('\\')
  end

  def search_files(path, regex, recursive)
    matching_files = []

    if recursive
      Dir.glob("#{path}/**/*").each do |file|
        matching_files << file.relative_path_from(FILES_ROOT).to_s if file.file? && File.read(file).match?(Regexp.new(regex))
      end
    else
      Dir.glob("#{path}/*").each do |file|
        matching_files << file.relative_path_from(FILES_ROOT).to_s if file.file? && File.read(file).match?(Regexp.new(regex))
      end
    end

    matching_files
  end
end
