<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\File;

class FileSearchController extends Controller
{
    private const FILES_ROOT = './files';

    public function search(Request $request)
    {
        $regex = $request->query('regex');
        $directory = $request->query('directory', '');
        $recursive = $request->query('recursive', false);

        // Validate regex
        if (@preg_match($regex, '') === false) {
            return response()->json(['error' => 'Invalid regex pattern.'], 400);
        }

        // Validate directory
        if (!$this->isValidDirectory($directory)) {
            return response()->json(['error' => 'Invalid directory path.'], 400);
        }

        $searchPath = self::FILES_ROOT . '/' . trim($directory, '/');
        $files = [];

        if ($recursive) {
            $files = $this->searchFilesRecursively($searchPath, $regex);
        } else {
            $files = $this->searchFilesInDirectory($searchPath, $regex);
        }

        return response()->json(['files' => $files]);
    }

    private function searchFilesInDirectory($directory, $regex)
    {
        $files = [];
        if (File::exists($directory) && File::isDirectory($directory)) {
            $fileList = File::files($directory);
            foreach ($fileList as $file) {
                if (preg_match($regex, File::get($file->getRealPath()))) {
                    $files[] = str_replace(self::FILES_ROOT . '/', '', $file->getRealPath());
                }
            }
        }
        return $files;
    }

    private function searchFilesRecursively($directory, $regex)
    {
        $files = [];
        if (File::exists($directory) && File::isDirectory($directory)) {
            $fileList = File::allFiles($directory);
            foreach ($fileList as $file) {
                if (preg_match($regex, File::get($file->getRealPath()))) {
                    $files[] = str_replace(self::FILES_ROOT . '/', '', $file->getRealPath());
                }
            }
        }
        return $files;
    }

    private function isValidDirectory($directory)
    {
        // Reject if directory contains path separators or ".."
        return preg_match('/^[a-zA-Z0-9_\-\/]*$/', $directory) === 1;
    }
}
