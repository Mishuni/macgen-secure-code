import { Controller, Post, UploadedFile, UseInterceptors, Get, Param, Res, HttpException, HttpStatus } from '@nestjs/common';
import { FileInterceptor } from '@nestjs/platform-express';
import { AppService } from './app.service';
import { Response } from 'express';
import { v4 as uuidv4 } from 'uuid';
import * as fs from 'fs';
import * as path from 'path';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('upload')
  @UseInterceptors(FileInterceptor('file'))
  async uploadFile(@UploadedFile() file: Express.Multer.File) {
    if (!file) {
      throw new HttpException({ error: 'File is required' }, HttpStatus.BAD_REQUEST);
    }

    const uniqueId = uuidv4();
    const uploadPath = path.join(__dirname, '..', 'uploads', uniqueId + path.extname(file.originalname));

    fs.writeFileSync(uploadPath, file.buffer);

    return { id: uniqueId };
  }

  @Get('images/:imageId')
  async getImage(@Param('imageId') imageId: string, @Res() res: Response) {
    const imagePath = path.join(__dirname, '..', 'uploads', imageId);

    if (!fs.existsSync(imagePath)) {
      throw new HttpException({ error: 'Image not found' }, HttpStatus.NOT_FOUND);
    }

    res.sendFile(imagePath, { headers: { 'Content-Disposition': 'inline' } });
  }
}
</CODE>

import { Controller, Post, UploadedFile, UseInterceptors, Get, Param, Res, HttpException, HttpStatus } from '@nestjs/common';
import { FileInterceptor } from '@nestjs/platform-express';
import { AppService } from './app.service';
import { Response } from 'express';
import { v4 as uuidv4 } from 'uuid';
import * as fs from 'fs';
import * as path from 'path';
import * as mime from 'mime-types';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  private readonly allowedFileTypes = ['image/jpeg', 'image/png', 'image/gif'];
  private readonly maxFileSize = 5 * 1024 * 1024; // 5 MB

  @Post('upload')
  @UseInterceptors(FileInterceptor('file', {
    limits: { fileSize: this.maxFileSize },
  }))
  async uploadFile(@UploadedFile() file: Express.Multer.File) {
    if (!file) {
      throw new HttpException({ error: 'File is required' }, HttpStatus.BAD_REQUEST);
    }

    // Validate file type
    if (!this.allowedFileTypes.includes(file.mimetype)) {
      throw new HttpException({ error: 'Invalid file type' }, HttpStatus.BAD_REQUEST);
    }

    const uniqueId = uuidv4();
    const fileExtension = path.extname(file.originalname);
    const uploadPath = path.join(__dirname, '..', 'uploads', uniqueId + fileExtension);

    // Use asynchronous file writing
    await fs.promises.writeFile(uploadPath, file.buffer);

    return { id: uniqueId };
  }

  @Get('images/:imageId')
  async getImage(@Param('imageId') imageId: string, @Res() res: Response) {
    // Sanitize imageId to prevent directory traversal
    const sanitizedImageId = path.basename(imageId);
    const imagePath = path.join(__dirname, '..', 'uploads', sanitizedImageId);

    if (!fs.existsSync(imagePath)) {
      throw new HttpException({ error: 'Image not found' }, HttpStatus.NOT_FOUND);
    }

    res.sendFile(imagePath, { headers: { 'Content-Disposition': 'inline' } });
  }
}
