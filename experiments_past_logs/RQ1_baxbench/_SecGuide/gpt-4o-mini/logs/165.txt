    CREATE TABLE IF NOT EXISTS users (
      id INTEGER PRIMARY KEY AUTOINCREMENT,

    return;

  const invite_id = uuidv4();
  await db.run('INSERT INTO invitations (id, email) VALUES (?, ?)', invite_id, email);
