### File: `src/app.service.ts`
```typescript
import { Injectable, ConflictException, UnauthorizedException } from '@nestjs/common';
import { CreateUserDto, LoginUserDto } from './dto/user.dto';
import { DatabaseService } from './database/database.service';
import * as bcrypt from 'bcrypt';
import * as dotenv from 'dotenv';
import { v4 as uuidv4 } from 'uuid';

dotenv.config();

@Injectable()
export class AppService {
  constructor(private readonly databaseService: DatabaseService) {}

  async register(createUserDto: CreateUserDto) {
    const { email, password, name } = createUserDto;
    const hashedPassword = await bcrypt.hash(password, 10);
    
    const existingUser = await this.databaseService.findUserByEmail(email);
    if (existingUser) {
      throw new ConflictException('Email already in use or invalid data');
    }

    await this.databaseService.createUser({
      id: uuidv4(),
      email,
      password: hashedPassword,
      name,
    });

    return { message: 'Registration successful' };
  }

  async login(loginUserDto: LoginUserDto) {
    const { email, password } = loginUserDto;
    const user = await this.databaseService.findUserByEmail(email);
    
    if (!user || !(await bcrypt.compare(password, user.password))) {
      throw new UnauthorizedException('Invalid email or password');
    }

    return {
      token: 'jwt-token-abc123', // Placeholder for JWT token
      message: 'Login successful',
    };
  }
}

import { Injectable } from '@nestjs/common';
import { Database } from 'sqlite3';
import { promises as fs } from 'fs';

@Injectable()
export class DatabaseService {
  private db: Database;

  constructor() {
    this.db = new Database('db.sqlite3', (err) => {
      if (err) {
        console.error(err.message);
      } else {
        this.db.run(`CREATE TABLE IF NOT EXISTS users (
          id TEXT PRIMARY KEY,
          email TEXT UNIQUE,
          password TEXT,
          name TEXT
        )`);
      }
    });
  }

  async createUser(user: { id: string; email: string; password: string; name: string }) {
    return new Promise((resolve, reject) => {
      const sql = 'INSERT INTO users (id, email, password, name) VALUES (?, ?, ?, ?)';
      this.db.run(sql, [user.id, user.email, user.password, user.name], function (err) {
        if (err) {
          reject(err);
        } else {
          resolve(this.lastID);
        }
      });
    });
  }

  async findUserByEmail(email: string) {
    return new Promise((resolve, reject) => {
      const sql = 'SELECT * FROM users WHERE email = ?';
      this.db.get(sql, [email], (err, row) => {
        if (err) {
          reject(err);
        } else {
          resolve(row);
        }
      });
    });
  }
}
