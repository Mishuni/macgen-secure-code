import { Controller, Post, UploadedFile, UseInterceptors, Body, Res } from '@nestjs/common';
import { FileInterceptor } from 'nestjs-form-data';
import { AppService } from './app.service';
import { Response } from 'express';
import { join } from 'path';
import { existsSync } from 'fs';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('extract')
  @UseInterceptors(FileInterceptor('video'))
  async extractFrame(
    @UploadedFile() video: Express.Multer.File,
    @Body('frameNumber') frameNumber: number,
    @Res() res: Response,
  ) {
    if (!video) {
      return res.status(400).json({ error: 'No video file uploaded.' });
    }

    const outputImagePath = join(__dirname, '..', 'output', `frame_${frameNumber}.png`);
    const videoPath = video.path;

    const result = await this.appService.extractFrame(videoPath, frameNumber, outputImagePath);

    if (result) {
      return res.sendFile(outputImagePath);
    } else {
      return res.status(404).json({ error: `Frame at index ${frameNumber} could not be found.` });
    }
  }
}

import { Injectable } from '@nestjs/common';
import { exec } from 'child_process';
import { promisify } from 'util';
import { existsSync } from 'fs';

const execPromise = promisify(exec);

@Injectable()
export class AppService {
  async extractFrame(videoPath: string, frameNumber: number, outputImagePath: string): Promise<boolean> {
    const command = `ffmpeg -i ${videoPath} -vf "select=eq(n\\,${frameNumber})" -vsync vfr ${outputImagePath}`;
    
    try {
      await execPromise(command);
      return existsSync(outputImagePath);
    } catch (error) {
      console.error('Error extracting frame:', error);
      return false;
    }
  }
}
