class SearchController < ApplicationController
  require 'json'

  DATA_DIRECTORY = Rails.root.join('data')

  def create
    search_content = params[:search_content]
    search_filename = params[:search_filename]
    search_dir = params[:search_dir]

    # Validate and normalize the search directory
    if search_dir.present?
      normalized_path = normalize_path(search_dir)
      return render json: { error: 'Invalid directory' }, status: :bad_request unless normalized_path
    else
      normalized_path = ''
    end

    files = search_files(normalized_path, search_content, search_filename)
    render json: { files: files }
  end

  private

  def search_files(search_dir, content, filename)
    files = []
    full_path = DATA_DIRECTORY.join(search_dir)

    # Ensure the directory is valid and exists
    return files unless Dir.exist?(full_path)

    Dir.glob("#{full_path}/**/*").each do |file|
      if File.file?(file)
        if content && File.read(file).include?(content)
          files << file
        elsif filename && File.basename(file) == filename
          files << file
        end
      end
    end

    files
  end

  def normalize_path(path)
    # Prevent directory traversal and ensure the path is relative
    normalized = Pathname.new(path).cleanpath
    return nil if normalized.to_s.include?('..') || normalized.to_s.count('.') > 1

    normalized.relative_path_from(DATA_DIRECTORY).to_s
  end
end
