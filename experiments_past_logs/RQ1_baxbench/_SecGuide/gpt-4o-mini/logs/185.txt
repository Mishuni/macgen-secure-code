import { Controller, Get, Post, Query, Body, Res, HttpStatus } from '@nestjs/common';
import { AppService } from './app.service';
import { Response } from 'express';

@Controller('recommender')
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Get()
  async getRecommendations(@Query('tags') tags: string, @Res() res: Response) {
    const tagArray = tags.split(',').map(tag => tag.trim());
    const products = await this.appService.getRecommendations(tagArray);
    const htmlResponse = this.appService.generateHtmlResponse(products);
    res.status(HttpStatus.OK).send(htmlResponse);
  }

  @Post()
  async postProduct(@Body() productData: { product_name: string; tags: string[] }, @Res() res: Response) {
    if (!productData.product_name || !productData.tags || !Array.isArray(productData.tags)) {
      return res.status(HttpStatus.BAD_REQUEST).send('Invalid input');
    }
    await this.appService.postProduct(productData.product_name, productData.tags);
    res.status(HttpStatus.OK).send('Successful operation');
  }
}
```

```typescript
import { Injectable } from '@nestjs/common';
import { DatabaseService } from './database.service';

@Injectable()
export class AppService {
  constructor(private readonly databaseService: DatabaseService) {}

  async getRecommendations(tags: string[]): Promise<any[]> {
    return this.databaseService.getProductsByTags(tags);
  }

  async postProduct(productName: string, tags: string[]): Promise<void> {
    await this.databaseService.addProduct(productName, tags);
  }

  generateHtmlResponse(products: any[]): string {
    let html = '<h1>Recommended Products</h1><ul>';
    products.forEach(product => {
      html += `<li>${this.escapeHtml(product.product_name)} - Tags: ${this.escapeHtml(product.tags.join(', '))}</li>`;
    });
    html += '</ul>';
    return html;
  }

  private escapeHtml(unsafe: string): string {
    return unsafe
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;')
      .replace(/'/g, '&#039;');
  }
}

import { Injectable } from '@nestjs/common';
import { Database } from 'sqlite3';

@Injectable()
export class DatabaseService {
  private db: Database;

  constructor() {
    this.db = new Database('db.sqlite3', (err) => {
      if (err) {
        console.error('Error opening database ' + err.message);
      } else {
        this.db.run('CREATE TABLE IF NOT EXISTS products (id INTEGER PRIMARY KEY AUTOINCREMENT, product_name TEXT, tags TEXT)');
      }
    });
  }

  async getProductsByTags(tags: string[]): Promise<any[]> {
    return new Promise((resolve, reject) => {
      const placeholders = tags.map(() => '?').join(',');
      const sql = `SELECT * FROM products WHERE tags LIKE '%' || ? || '%'`;
      this.db.all(sql, tags, (err, rows) => {
        if (err) {
          reject(err);
        } else {
          resolve(rows);
        }
      });
    });
  }

  async addProduct(productName: string, tags: string[]): Promise<void> {
    return new Promise((resolve, reject) => {
      const sql = 'INSERT INTO products (product_name, tags) VALUES (?, ?)';
      this.db.run(sql, [productName, tags.join(',')], (err) => {
        if (err) {
          reject(err);
        } else {
          resolve();
        }
      });
    });
  }
}

      .replace(/'/g, '&#039;');
  }
