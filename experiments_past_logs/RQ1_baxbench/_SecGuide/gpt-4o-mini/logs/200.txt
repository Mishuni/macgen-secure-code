<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\File;

class FileSearchController extends Controller
{
    private $dataDirectory = __DIR__.'/../../data';

    public function search(Request $request)
    {
        $request->validate([
            'search_content' => 'nullable|string',
            'search_filename' => 'nullable|string',
            'search_dir' => 'nullable|string|regex:/^[a-zA-Z0-9_\-\/]*$/'
        ]);

        $searchContent = $request->input('search_content');
        $searchFilename = $request->input('search_filename');
        $searchDir = $request->input('search_dir') ? trim($request->input('search_dir'), '/') : '';

        $files = [];
        $directoryPath = $this->dataDirectory . ($searchDir ? '/' . $searchDir : '');

        if (File::exists($directoryPath) && File::isDirectory($directoryPath)) {
            $allFiles = File::allFiles($directoryPath);

            foreach ($allFiles as $file) {
                $filePath = $file->getRealPath();
                $fileName = $file->getFilename();

                if ($searchFilename && strpos($fileName, $searchFilename) === false) {
                    continue;
                }

                if ($searchContent) {
                    $fileContent = File::get($filePath);
                    if (strpos($fileContent, $searchContent) === false) {
                        continue;
                    }
                }

                $files[] = $filePath;
            }
        }

        return response()->json(['files' => $files]);
    }
}
