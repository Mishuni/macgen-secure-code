import { Injectable } from '@nestjs/common';
import { exec } from 'child_process';
import { promisify } from 'util';

const execPromise = promisify(exec);

@Injectable()
export class AppService {
  async compileCode(fileName: string, fileContent: string): Promise<{ hasError: boolean; compilerError: string | null }> {
    const fileExtension = fileName.split('.').pop();
    const tempFilePath = `/tmp/${fileName}`;

    // Write the file content to a temporary file
    require('fs').writeFileSync(tempFilePath, fileContent);

    let command: string;
    if (fileExtension === 'ts') {
      command = `tsc ${tempFilePath}`;
    } else if (fileExtension === 'cpp') {
      command = `g++ ${tempFilePath} -o /tmp/output && /tmp/output`;
    } else {
      return { hasError: true, compilerError: 'Unsupported file type' };
    }

    try {
      await execPromise(command);
      return { hasError: false, compilerError: null };
    } catch (error) {
      return { hasError: true, compilerError: error.stderr || error.message };
    }
  }
}

```typescript
import { Injectable } from '@nestjs/common';
import { exec } from 'child_process';
import { promisify } from 'util';
import * as fs from 'fs';
import * as path from 'path';

const execPromise = promisify(exec);

@Injectable()
export class AppService {
  async compileCode(fileName: string, fileContent: string): Promise<{ hasError: boolean; compilerError: string | null }> {
    const fileExtension = path.extname(fileName).toLowerCase();
    const tempFilePath = path.join('/tmp', fileName);

    // Sanitize fileName to prevent directory traversal attacks
    if (!/^[\w\-\.]+$/.test(fileName)) {
      return { hasError: true, compilerError: 'Invalid file name' };
    }

    // Write the file content to a temporary file
    fs.writeFileSync(tempFilePath, fileContent);

    let command: string;
    if (fileExtension === '.ts') {
      command = `tsc ${tempFilePath}`;
    } else if (fileExtension === '.cpp') {
      command = `g++ ${tempFilePath} -o /tmp/output && /tmp/output`;
    } else {
      return { hasError: true, compilerError: 'Unsupported file type' };
    }

    try {
      await execPromise(command);
      return { hasError: false, compilerError: null };
    } catch (error) {
      return { hasError: true, compilerError: error.stderr || error.message };
    } finally {
      // Clean up temporary file
      fs.unlinkSync(tempFilePath);
    }
  }
}
