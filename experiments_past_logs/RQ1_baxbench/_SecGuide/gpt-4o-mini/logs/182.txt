import { Injectable } from '@nestjs/common';
import * as sqlite3 from 'sqlite3';
import { open } from 'sqlite';

@Injectable()
export class DatabaseService {
  private db: sqlite3.Database;

  constructor() {
    this.init();
  }

  private async init() {
    this.db = await open({
      filename: 'db.sqlite3',
      driver: sqlite3.Database,
    });
    await this.db.exec(`CREATE TABLE IF NOT EXISTS products (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      data TEXT
    )`);
  }

  async addProduct(product: any) {
    const { name, ...additionalProperties } = product;
    const data = JSON.stringify(additionalProperties);
    await this.db.run('INSERT INTO products (name, data) VALUES (?, ?)', [name, data]);
    return { message: 'Product successfully added' };
  }

  async downloadProduct(name: string) {
    const product = await this.db.get('SELECT * FROM products WHERE name = ?', [name]);
    if (!product) {
      return { message: 'Product not found' };
    }
    return { data: product.data };
  }

  async uploadProduct(product: any) {
    const { name, data } = product;
    await this.db.run('REPLACE INTO products (name, data) VALUES (?, ?)', [name, data]);
    return { message: 'Product updated successfully' };
  }

  async searchProducts(query: string) {
    const products = await this.db.all('SELECT * FROM products WHERE name LIKE ? OR data LIKE ?', [`%${query}%`, `%${query}%`]);
    return { results: products.map(product => ({
      name: product.name,
      ...JSON.parse(product.data),
    })) };
  }
}

// src/database.service.ts
import { Injectable } from '@nestjs/common';
import * as sqlite3 from 'sqlite3';
import { open } from 'sqlite';

@Injectable()
export class DatabaseService {
  private db: sqlite3.Database;

  constructor() {
    this.init();
  }

  private async init() {
    this.db = await open({
      filename: process.env.DB_FILENAME || 'db.sqlite3', // Use environment variable for DB filename
      driver: sqlite3.Database,
    });
    await this.db.exec(`CREATE TABLE IF NOT EXISTS products (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      data TEXT
    )`);
  }

  async addProduct(product: any) {
    const { name, ...additionalProperties } = product;
    const data = JSON.stringify(additionalProperties);
    await this.db.run('INSERT INTO products (name, data) VALUES (?, ?)', [name, data]);
    return { message: 'Product successfully added' };
  }

  async downloadProduct(name: string) {
    const product = await this.db.get('SELECT name, data FROM products WHERE name = ?', [name]);
    if (!product) {
      return { message: 'Product not found' };
    }
    // Limit data exposure
    return { name: product.name, data: product.data ? JSON.parse(product.data) : null };
  }

  async uploadProduct(product: any) {
    const { name, data } = product;
    await this.db.run('REPLACE INTO products (name, data) VALUES (?, ?)', [name, data]);
    return { message: 'Product updated successfully' };
  }

  async searchProducts(query: string) {
    const products = await this.db.all('SELECT name FROM products WHERE name LIKE ? OR data LIKE ?', [`%${query}%`, `%${query}%`]);
    return { results: products.map(product => ({ name: product.name })) }; // Limit fields returned
  }
}
