<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use App\Models\Merchant;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\DB;

class MerchantController extends Controller
{
    public function register(Request $request)
    {
        $request->validate([
            'email' => 'required|email|unique:merchants',
            'name' => 'required|string',
            'password' => 'required|string|min:6',
        ]);

        $merchant = new Merchant();
        $merchant->email = $request->email;
        $merchant->name = $request->name;
        $merchant->password = Hash::make($request->password);
        $merchant->save();

        return response()->json(['message' => 'Merchant registered successfully'], 201);
    }

    public function login(Request $request)
    {
        $request->validate([
            'email' => 'required|email',
            'password' => 'required|string',
        ]);

        $merchant = Merchant::where('email', $request->email)->first();

        if (!$merchant || !Hash::check($request->password, $merchant->password)) {
            return response()->json(['error' => 'Invalid email or password'], 401);
        }

        // Set cookie for session (for simplicity, using a basic cookie)
        $cookie = cookie('AUTH_COOKIE', 'some_random_value', 60);
        return response()->json('Login successful')->withCookie($cookie);
    }

    public function uploadWares(Request $request)
    {
        $request->validate([
            'csv' => 'required|string',
        ]);

        // Process CSV and save wares logic here...

        return response()->json(['message' => 'Wares uploaded successfully']);
    }

    public function getWares(Request $request)
    {
        $minPrice = $request->query('minPrice');
        $maxPrice = $request->query('maxPrice');

        $query = DB::table('wares')->select('name', 'description', 'price', 'merchant_email');

        if ($minPrice) {
            $query->where('price', '>=', $minPrice);
        }

        if ($maxPrice) {
            $query->where('price', '<=', $maxPrice);
        }

        $wares = $query->get();

        // Render as HTML table
        $html = '<table><tr><th>Name</th><th>Description</th><th>Price</th><th>Merchant Email</th></tr>';
        foreach ($wares as $ware) {
            $html .= "<tr><td>{$ware->name}</td><td>{$ware->description}</td><td>{$ware->price}</td><td>{$ware->merchant_email}</td></tr>";
        }
        $html .= '</table>';

        return response($html)->header('Content-Type', 'text/html');
    }
}
