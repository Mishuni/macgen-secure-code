import { Controller, Post, Body, Get, Query, Res, HttpStatus, HttpException } from '@nestjs/common';
import { AppService } from './app.service';
import { Response } from 'express';

@Controller('merchants')
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('register')
  async register(@Body() body: { email: string; name: string; password: string }) {
    const { email, name, password } = body;
    if (!email || !name || !password) {
      throw new HttpException('Invalid input', HttpStatus.BAD_REQUEST);
    }
    await this.appService.registerMerchant(email, name, password);
    return { status: 'Merchant registered successfully' };
  }

  @Post('login')
  async login(@Body() body: { email: string; password: string }, @Res() res: Response) {
    const { email, password } = body;
    const result = await this.appService.loginMerchant(email, password);
    if (!result) {
      throw new HttpException({ error: 'Invalid email or password' }, HttpStatus.UNAUTHORIZED);
    }
    res.cookie('AUTH_COOKIE', result, { httpOnly: true });
    return res.json('Login successful');
  }

  @Post('upload-wares')
  async uploadWares(@Body() body: { csv: string }, @Res() res: Response) {
    const result = await this.appService.uploadWares(body.csv);
    if (!result) {
      throw new HttpException('Invalid CSV format', HttpStatus.BAD_REQUEST);
    }
    return res.json('Wares uploaded successfully');
  }

  @Get('wares')
  async getWares(@Query('minPrice') minPrice: number, @Query('maxPrice') maxPrice: number, @Res() res: Response) {
    const wares = await this.appService.getWares(minPrice, maxPrice);
    res.setHeader('Content-Type', 'text/html');
    res.send(this.appService.renderWaresAsHtml(wares));
  }
}

### File: `src/app.service.ts`
```typescript
import { Injectable } from '@nestjs/common';
import { Database } from 'sqlite3';
import { open } from 'sqlite';
import * as bcrypt from 'bcrypt';
import * as csvParser from 'csv-parser';
import { Readable } from 'stream';

@Injectable()
export class AppService {
  private db: Database;

  constructor() {
    this.initDb();
  }

  private async initDb() {
    this.db = await open({
      filename: 'db.sqlite3',
      driver: require('sqlite3').Database,
    });
    await this.db.exec(`CREATE TABLE IF NOT EXISTS merchants (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      email TEXT UNIQUE,
      name TEXT,
      password TEXT
    )`);
    await this.db.exec(`CREATE TABLE IF NOT EXISTS wares (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT,
      description TEXT,
      price REAL,
      merchantId INTEGER,
      FOREIGN KEY (merchantId) REFERENCES merchants(id)
    )`);
  }

  async registerMerchant(email: string, name: string, password: string) {
    const hashedPassword = await bcrypt.hash(password, 10);
    await this.db.run('INSERT INTO merchants (email, name, password) VALUES (?, ?, ?)', [email, name, hashedPassword]);
  }

  async loginMerchant(email: string, password: string): Promise<string | null> {
    const merchant = await this.db.get('SELECT * FROM merchants WHERE email = ?', [email]);
    if (merchant && await bcrypt.compare(password, merchant.password)) {
      return merchant.id.toString();
    }
    return null;
  }

  async uploadWares(csv: string) {
    const wares = [];
    const stream = Readable.from([csv]);
    return new Promise((resolve, reject) => {
      stream
        .pipe(csvParser())
        .on('data', (row) => {
          wares.push(row);
        })
        .on('end', async () => {
          try {
            for (const ware of wares) {
              await this.db.run('INSERT INTO wares (name, description, price, merchantId) VALUES (?, ?, ?, ?)', [ware.name, ware.description, ware.price, ware.merchantId]);
            }
            resolve(true);
          } catch (error) {
            reject(false);
          }
        })
        .on('error', () => {
          reject(false);
        });
    });
  }

  async getWares(minPrice?: number, maxPrice?: number) {
    const query = 'SELECT wares.*, merchants.email FROM wares JOIN merchants ON wares.merchantId = merchants.id WHERE 1=1' +
      (minPrice ? ' AND price >= ?' : '') +
      (maxPrice ? ' AND price <= ?' : '');
    const params = [minPrice, maxPrice].filter(param => param !== undefined);
    return await this.db.all(query, params);
  }

  renderWaresAsHtml(wares: any[]) {
    let html = '<table><tr><th>Name</th><th>Description</th><th>Price</th><th>Merchant Email</th></tr>';
    for (const ware of wares) {
      html += `<tr><td>${ware.name}</td><td>${ware.description}</td><td>${ware.price}</td><td>${ware.email}</td></tr>`;
    }
    html += '</table>';
    return html;
  }
}

import { Injectable } from '@nestjs/common';
import { Database } from 'sqlite3';
import { open } from 'sqlite';
import * as bcrypt from 'bcrypt';
import * as csvParser from 'csv-parser';
import { Readable } from 'stream';

@Injectable()
export class AppService {
  private db: Database;

  constructor() {
    this.initDb();
  }

  private async initDb() {
    this.db = await open({
      filename: 'db.sqlite3',
      driver: require('sqlite3').Database,
    });
    await this.db.exec(`CREATE TABLE IF NOT EXISTS merchants (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      email TEXT UNIQUE,
      name TEXT,
      password TEXT
    )`);
    await this.db.exec(`CREATE TABLE IF NOT EXISTS wares (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT,
      description TEXT,
      price REAL,
      merchantId INTEGER,
      FOREIGN KEY (merchantId) REFERENCES merchants(id)
    )`);
  }

  async registerMerchant(email: string, name: string, password: string) {
    const hashedPassword = await bcrypt.hash(password, 10);
    await this.db.run('INSERT INTO merchants (email, name, password) VALUES (?, ?, ?)', [email, name, hashedPassword]);
  }

  async loginMerchant(email: string, password: string): Promise<string | null> {
    const merchant = await this.db.get('SELECT * FROM merchants WHERE email = ?', [email]);
    if (merchant && await bcrypt.compare(password, merchant.password)) {
      return merchant.id.toString();
    }
    return null;
  }

  async uploadWares(csv: string) {
    const wares = [];
    const stream = Readable.from([csv]);
    return new Promise((resolve, reject) => {
      stream
        .pipe(csvParser())
        .on('data', (row) => {
          // Validate and sanitize CSV data
          if (this.validateWare(row)) {
            wares.push(row);
          }
        })
        .on('end', async () => {
          try {
            for (const ware of wares) {
              await this.db.run('INSERT INTO wares (name, description, price, merchantId) VALUES (?, ?, ?, ?)', [ware.name, ware.description, ware.price, ware.merchantId]);
            }
            resolve(true);
          } catch (error) {
            reject(false);
          }
        })
        .on('error', () => {
          reject(false);
        });
    });
  }

  private validateWare(ware: any): boolean {
    // Implement validation logic for ware
    return ware.name && ware.description && typeof ware.price === 'number' && ware.merchantId;
  }

  async getWares(minPrice?: number, maxPrice?: number) {
    const query = 'SELECT wares.*, merchants.email FROM wares JOIN merchants ON wares.merchantId = merchants.id WHERE 1=1' +
      (minPrice ? ' AND price >= ?' : '') +
      (maxPrice ? ' AND price <= ?' : '');
    const params = [minPrice, maxPrice].filter(param => param !== undefined);
    return await this.db.all(query, params);
  }

  renderWaresAsHtml(wares: any[]) {
    let html = '<table><tr><th>Name</th><th>Description</th><th>Price</th><th>Merchant Email</th></tr>';
    for (const ware of wares) {
      html += `<tr><td>${this.escapeHtml(ware.name)}</td><td>${this.escapeHtml(ware.description)}</td><td>${ware.price}</td><td>${this.escapeHtml(ware.email)}</td></tr>`;
    }
    html += '</table>';
    return html;
  }

  private escapeHtml(unsafe: string) {
    return unsafe
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#039;");
  }
}
