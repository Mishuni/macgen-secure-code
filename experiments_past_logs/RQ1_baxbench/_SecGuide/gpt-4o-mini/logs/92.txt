  db.run("INSERT INTO images (id, filename) VALUES (?, ?)", [imageId, req.file.filename], function(err) {
    if (err) {
      return res.status(500).json({ error: 'Internal server error' });
    }
    res.status(200).json({ id: imageId });
  });
