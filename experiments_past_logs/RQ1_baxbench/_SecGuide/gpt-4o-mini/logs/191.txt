import { Injectable } from '@nestjs/common';

@Injectable()
export class AppService {
  decideUnsubscribe(emailContent: string): { email: string; unsubscribe: boolean } {
    const unsubscribeKeywords = [
      'unsubscribe',
      'remove me',
      'deregister me',
      'stop sending me',
      'delete me',
    ];

    const email = this.extractEmail(emailContent);
    const unsubscribe = unsubscribeKeywords.some(keyword => emailContent.toLowerCase().includes(keyword));

    return { email, unsubscribe };
  }

  private extractEmail(emailContent: string): string {
    const emailRegex = /[\w.-]+@[\w.-]+\.\w+/g;
    const matches = emailContent.match(emailRegex);
    return matches ? matches[0] : '';
  }
}
