function isValidPath(inputPath) {
    // Reject paths with ".." or any path separators
    return !inputPath.includes('..') && !inputPath.includes('/') && !inputPath.includes('\\') && inputPath.split('.').length <= 2;
}

function searchFiles(searchContent, searchFilename, searchDir) {
    const results = [];
    const dirPath = path.join(DATA_DIR, searchDir || '');

    if (!fs.existsSync(dirPath) || !fs.lstatSync(dirPath).isDirectory()) {
        return results; // Return empty if directory does not exist
    }

    const files = fs.readdirSync(dirPath);
    for (const file of files) {
        const filePath = path.join(dirPath, file);
        if (fs.lstatSync(filePath).isDirectory()) {
            // Recursively search in subdirectories
            results.push(...searchFiles(searchContent, searchFilename, path.join(searchDir || '', file)));
        } else {
            let contentMatches = true;
            let filenameMatches = true;

            if (searchContent) {
                const fileContent = fs.readFileSync(filePath, 'utf-8');
                contentMatches = fileContent.includes(searchContent);
            }

            if (searchFilename) {
                filenameMatches = file.startsWith(searchFilename);
            }

            if (contentMatches && filenameMatches) {
                results.push(filePath);
            }
        }
    }
    return results;
}
