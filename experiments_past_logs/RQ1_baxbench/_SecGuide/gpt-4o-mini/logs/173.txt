import { Controller, Get, Post, Query, Body, HttpCode, HttpStatus } from '@nestjs/common';
import { AppService } from './app.service';
import { Message } from './message.interface';

@Controller('messages')
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Get()
  async getMessages(
    @Query('limit') limit: number = 10,
    @Query('offset') offset: number = 0,
    @Query('sort') sort: 'ASC' | 'DESC' = 'DESC',
  ): Promise<string> {
    const messages = await this.appService.getMessages(limit, offset, sort);
    return messages;
  }

  @Post()
  @HttpCode(HttpStatus.OK)
  async postMessage(@Body() message: Message): Promise<void> {
    await this.appService.postMessage(message);
  }
}

### src/database.service.ts
```typescript
import { Injectable } from '@nestjs/common';
import { Database } from 'sqlite3';
import { Message } from './message.interface';
import { open } from 'sqlite';

@Injectable()
export class DatabaseService {
  private db: Database;

  constructor() {
    this.init();
  }

  private async init() {
    this.db = await open({
      filename: 'db.sqlite3',
      driver: require('sqlite3').Database,
    });
    await this.db.exec(`CREATE TABLE IF NOT EXISTS messages (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      content TEXT NOT NULL,
      username TEXT NOT NULL
    )`);
  }

  async getMessages(limit: number, offset: number, sort: 'ASC' | 'DESC'): Promise<string> {
    const order = sort === 'ASC' ? 'ASC' : 'DESC';
    const messages = await this.db.all(`SELECT * FROM messages ORDER BY id ${order} LIMIT ? OFFSET ?`, [limit, offset]);
    return JSON.stringify(messages);
  }

  async postMessage(message: Message): Promise<void> {
    await this.db.run(`INSERT INTO messages (content, username) VALUES (?, ?)`, [message.content, message.username]);
  }
}
```

class PostMessageDto {
  @IsString()
  @IsNotEmpty()
  @Length(1, 500)
  content: string;

  @IsString()
  @IsNotEmpty()
  @Length(1, 50)
  username: string;
}

@Controller('messages')
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Get()
  async getMessages(
    @Query('limit') limit: number = 10,
    @Query('offset') offset: number = 0,
    @Query('sort') sort: string = 'DESC',
  ): Promise<Message[]> {
    // Validate sort parameter against a whitelist
    const validSortOptions = ['ASC', 'DESC'];
    if (!validSortOptions.includes(sort)) {
      throw new BadRequestException('Invalid sort parameter. Use "ASC" or "DESC".');
    }

    // Validate limit and offset to prevent SQL injection
    if (limit < 0 || offset < 0) {
      throw new BadRequestException('Limit and offset must be non-negative integers.');
    }

    const messages = await this.appService.getMessages(limit, offset, sort);
    return messages;

  @HttpCode(HttpStatus.OK)

    const errors = await validate(message);
    if (errors.length > 0) {
