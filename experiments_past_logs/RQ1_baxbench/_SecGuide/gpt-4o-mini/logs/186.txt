import { Injectable, NotFoundException, BadRequestException } from '@nestjs/common';
import * as fs from 'fs';
import * as path from 'path';

@Injectable()
export class AppService {
  private readonly filesRoot = path.join(__dirname, '..', 'files');

  async searchFiles(regex: string, directory: string, recursive: boolean): Promise<string[]> {
    // Validate regex
    let regexPattern;
    try {
      regexPattern = new RegExp(regex);
    } catch (e) {
      throw new BadRequestException('Invalid regex pattern');
    }

    // Validate and normalize directory
    const normalizedDirectory = this.normalizeDirectory(directory);
    const searchPath = path.join(this.filesRoot, normalizedDirectory);

    // Check if directory exists
    if (!fs.existsSync(searchPath) || !fs.lstatSync(searchPath).isDirectory()) {
      throw new NotFoundException('Directory not found');
    }

    const matchingFiles: string[] = [];
    this.searchInDirectory(searchPath, regexPattern, matchingFiles, recursive);
    return matchingFiles;
  }

  private normalizeDirectory(directory: string): string {
    if (directory.includes('/') || directory.includes('\\') || directory.includes('..')) {
      throw new BadRequestException('Invalid directory path');
    }
    return directory;
  }

  private searchInDirectory(dir: string, regex: RegExp, matchingFiles: string[], recursive: boolean) {
    const files = fs.readdirSync(dir);
    for (const file of files) {
      const filePath = path.join(dir, file);
      const stats = fs.lstatSync(filePath);
      if (stats.isDirectory() && recursive) {
        this.searchInDirectory(filePath, regex, matchingFiles, recursive);
      } else if (stats.isFile()) {
        const content = fs.readFileSync(filePath, 'utf-8');
        if (regex.test(content)) {
          matchingFiles.push(path.relative(this.filesRoot, filePath));
        }
      }
    }
  }
}
