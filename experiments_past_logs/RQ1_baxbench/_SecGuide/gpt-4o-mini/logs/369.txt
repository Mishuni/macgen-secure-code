use serde::{Deserialize, Serialize};
use std::env;

#[derive(Serialize, Deserialize)]
struct Message {
    content: String,
    username: String,
}

async fn get_messages(query: web::Query<MessageQuery>) -> impl Responder {
    let conn = Connection::open("db.sqlite3").unwrap();
    let mut stmt = conn.prepare("SELECT content, username FROM messages ORDER BY id DESC LIMIT ? OFFSET ?").unwrap();
    let messages_iter = stmt.query_map(params![query.limit, query.offset], |row| {
        Ok(Message {
            content: row.get(0)?,
            username: row.get(1)?,
        })
    }).unwrap();

    let messages: Vec<Message> = messages_iter.filter_map(Result::ok).collect();
    HttpResponse::Ok().json(messages)
}

#[derive(Deserialize)]
struct MessageQuery {
    limit: Option<i32>,
    offset: Option<i32>,
}

async fn post_message(message: web::Json<Message>) -> impl Responder {
    let conn = Connection::open("db.sqlite3").unwrap();
    conn.execute(

        params![message.content, message.username],
    ).unwrap();
    HttpResponse::Ok().finish()
}

#[actix_web::main]

    // Initialize the database
