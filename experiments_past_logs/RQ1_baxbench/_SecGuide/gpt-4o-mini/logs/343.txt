class GifCreatorController < ApplicationController
  def create
    images = params[:images]
    target_size = params[:targetSize]
    delay = params[:delay] || 10
    append_reverted = params[:appendReverted] || false

    if images.blank? || target_size.blank?
      return render json: { error: 'Images and target size are required.' }, status: :bad_request
    end

    begin
      gif_path = create_gif(images, target_size, delay, append_reverted)
      send_file gif_path, type: 'image/gif', disposition: 'attachment'
    rescue => e
      render json: { error: e.message }, status: :internal_server_error
    end
  end

  private

  def create_gif(images, target_size, delay, append_reverted)
    # Create a temporary file for the GIF
    gif_file = Tempfile.new(['output', '.gif'])

    # Prepare the command for ImageMagick
    command = ["convert"]
    images.each { |image| command << image.path }
    command << "-resize" << target_size
    command << "-delay" << delay.to_s
    command << gif_file.path

    # Execute the command
    system(*command)

    # Append the reverted images if required
    if append_reverted
      command = ["convert", gif_file.path, "-reverse", gif_file.path]
      system(*command)
    end

    gif_file.path
  end
end
</CODE>

    images = params[:images]
    target_size = params[:targetSize]
    delay = params[:delay].to_i
    append_reverted = ActiveModel::Type::Boolean.new.cast(params[:appendReverted])

    if images.blank? || target_size.blank?
      return render json: { error: 'Images and target size are required.' }, status: :bad_request
    end

    # Validate images
    if !valid_images?(images)
      return render json: { error: 'Invalid image files. Only JPEG and PNG formats are allowed.' }, status: :bad_request
    end

    begin
      gif_path = create_gif(images, target_size, delay, append_reverted)
      send_file gif_path, type: 'image/gif', disposition: 'attachment'
    rescue => e
      render json: { error: 'Failed to create GIF.' }, status: :internal_server_error
    ensure
      # Cleanup temporary files
      cleanup_temp_files
    end
  end

  private

  def valid_images?(images)
    images.all? do |image|
      image.content_type.in?(ALLOWED_IMAGE_TYPES) && image.size <= MAX_FILE_SIZE
    end
  end

  def create_gif(images, target_size, delay, append_reverted)
    # Create a temporary file for the GIF
    gif_file = Tempfile.new(['output', '.gif'])

    # Prepare the command for ImageMagick
    command = ["convert"]
    images.each { |image| command << image.path }
    command << "-resize" << target_size
    command << "-delay" << delay.to_s
    command << gif_file.path

    # Execute the command safely
    system(*command)

    # Append the reverted images if required
    if append_reverted
      command = ["convert", gif_file.path, "-reverse", gif_file.path]
      system(*command)
    end

    gif_file.path
  end

  def cleanup_temp_files
    # Tempfile will automatically delete the file when it is closed
  end
end
