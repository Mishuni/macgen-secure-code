    end
  end

class SongsController < ApplicationController
  # Use an environment variable for the songs directory
  SONGS_DIRECTORY = ENV['SONGS_DIRECTORY'] || '/var/songs'

  def download
    filename = params[:filename]

    # Validate the filename to prevent directory traversal attacks
    if filename.blank? || !valid_filename?(filename)
      render json: { error: 'Invalid input' }, status: :bad_request
      return
    end

    file_path = File.join(SONGS_DIRECTORY, filename)

    # Check if the file exists and validate the file type
    if File.exist?(file_path) && valid_file_type?(file_path)
      send_file file_path, type: 'audio/mpeg', disposition: 'inline'
    else
      render json: { error: 'Requested file is not available' }, status: :not_found
    end
  end

  private

  def valid_filename?(filename)
    # Allow only alphanumeric characters, underscores, hyphens, and dots
    filename.match?(/\A[\w\-.]+\z/)
  end

  def valid_file_type?(file_path)
    # Check the file extension to ensure it's a valid audio file
    valid_extensions = %w[.mp3 .wav .ogg]
    valid_extensions.include?(File.extname(file_path).downcase)
  end
end
