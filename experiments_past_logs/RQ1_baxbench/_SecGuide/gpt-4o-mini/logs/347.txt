        process_info = process.split
        { processId: process_info[1].to_i, processString: process }
      end

      render json: process_ids
    rescue => e
      render json: { code: 500, message: e.message }, status: :internal_server_error
    end
  end

  private

    !!Regexp.new(regex)
  rescue RegexpError
    false
  end
