class CartsController < ApplicationController
  def create_cart
    cart = Cart.create!
    render json: { cart_id: cart.id }, status: :created
  end

  def add_to_cart
    cart = Cart.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    item = cart.cart_items.find_or_initialize_by(item_id: params[:item_id])
    item.count += params[:count]
    item.save!

    render json: { message: 'Item added or removed successfully' }, status: :ok
  rescue ActiveRecord::RecordInvalid
    render json: { error: 'Invalid request' }, status: :bad_request
  end

  def retrieve_cart
    cart = Cart.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    items = cart.cart_items.select(:item_id, :count)
    render json: { items: items }, status: :ok
  end
end
</CODE>

class CartsController < ApplicationController
  def create_cart
    cart = @current_user.carts.create!
    render json: { cart_id: cart.id }, status: :created
  end

  def add_to_cart
    cart = @current_user.carts.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    count = params[:count].to_i
    return render json: { error: 'Count must be a positive integer' }, status: :bad_request unless count.positive?

    item = cart.cart_items.find_or_initialize_by(item_id: params[:item_id])
    item.count += count
    item.save!

    render json: { message: 'Item added successfully' }, status: :ok
  rescue ActiveRecord::RecordInvalid
    render json: { error: 'Invalid request' }, status: :bad_request
  rescue StandardError => e
    render json: { error: e.message }, status: :internal_server_error
  end

  def retrieve_cart
    cart = @current_user.carts.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    items = cart.cart_items.select(:item_id, :count)
    render json: { items: items }, status: :ok
  end
end

  private

class CartsController < ApplicationController
  def create_cart
    cart = @current_user.carts.create!
    render json: { cart_id: cart.id }, status: :created
  end

  def add_to_cart
    cart = @current_user.carts.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    count = params[:count].to_i
    return render json: { error: 'Count must be a positive integer' }, status: :bad_request unless count.positive?

    item = cart.cart_items.find_or_initialize_by(item_id: params[:item_id])
    item.count += count
    item.save!

    render json: { message: 'Item added successfully' }, status: :ok
  rescue ActiveRecord::RecordInvalid
    render json: { error: 'Invalid request' }, status: :bad_request
  rescue StandardError => e
    Rails.logger.error(e.message) # Log the error for internal tracking
    render json: { error: 'Internal server error' }, status: :internal_server_error
  end

  def retrieve_cart
    cart = @current_user.carts.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    items = cart.cart_items.select(:item_id, :count)
    render json: { items: items }, status: :ok
  end
end
