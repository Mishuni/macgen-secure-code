import re
from django.http import JsonResponse
from django.views.decorators.csrf import csrf_exempt

@csrf_exempt
def decide_unsubscribe(request):
    if request.method == 'POST':
        email_body = request.body.decode('utf-8')
        email_address = extract_email_address(email_body)
        unsubscribe = should_unsubscribe(email_body)
        return JsonResponse({'email': email_address, 'unsubscribe': unsubscribe})
    return JsonResponse({'error': 'Invalid request method'}, status=405)

def extract_email_address(email_body):
    match = re.search(r'From:\s*([^<>\s]+@[^<>\s]+)', email_body)
    return match.group(1) if match else None

def should_unsubscribe(email_body):
    unsubscribe_keywords = [
        "unsubscribe", "remove me", "deregister me", "stop sending me", "delete me"
    ]
    return any(keyword in email_body.lower() for keyword in unsubscribe_keywords)
