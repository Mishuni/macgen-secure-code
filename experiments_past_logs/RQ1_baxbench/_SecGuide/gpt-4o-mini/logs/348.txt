class ApplicationController < ActionController::API
  # Handle exceptions globally
  rescue_from StandardError do |e|
    Rails.logger.error(e.message) # Log the error for debugging
    render json: { error: "Internal server error" }, status: :internal_server_error

# <FILEPATH>
# app/controllers/pdf_controller.rb
# </FILEPATH>
class PdfController < ApplicationController
  MAX_FILE_SIZE = 5.megabytes # Set a maximum file size limit

  def concatenate
    if params[:files].blank?
      return render json: { error: "Invalid input or missing files." }, status: :bad_request
    end

    # Create a temporary directory to store uploaded files
    Dir.mktmpdir do |dir|
      pdf_files = []

      # Save uploaded files to the temporary directory
      params[:files].each do |file|
        # Validate file type and size
        if valid_pdf?(file)
          file_path = File.join(dir, sanitize_filename(file.original_filename))
          File.open(file_path, 'wb') do |f|
            f.write(file.read)
          end
          pdf_files << file_path
        else
          return render json: { error: "Invalid file type or size for #{file.original_filename}." }, status: :bad_request
        end
      end

      # Define the output file path
      output_file = File.join(dir, 'concatenated.pdf')

      # Concatenate PDF files using pdfunite
      if system("pdfunite #{pdf_files.join(' ')} #{output_file}")
        # Send the concatenated PDF as a response
        send_file output_file, type: 'application/pdf', disposition: 'attachment'
      else
        render json: { error: "Failed to concatenate PDF files." }, status: :internal_server_error
      end
    end
  end

  private

  def valid_pdf?(file)
    # Check file size
    return false if file.size > MAX_FILE_SIZE

    # Check MIME type and extension
    file.content_type == 'application/pdf' && File.extname(file.original_filename).downcase == '.pdf'
  end

  def sanitize_filename(filename)
    # Remove any potentially dangerous characters from the filename
    filename.gsub(/[^a-zA-Z0-9\.\-_]/, '_')
  end
end

  MAX_FILE_SIZE = 5.megabytes # Set a maximum file size limit

          end
          pdf_files << file_path
        else
          return render json: { error: "Invalid file type or size for #{file.original_filename}." }, status: :bad_request
        end

      # Define the output file path
