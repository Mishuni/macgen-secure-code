<?php

namespace App\Http\Controllers;

use App\Models\Click;
use Illuminate\Http\Request;

class ClickController extends Controller
{
    public function registerClick(Request $request)
    {
        $this->validate($request, [
            'timestamp' => 'required|date',
        ]);

        $click = Click::create([
            'timestamp' => $request->input('timestamp'),
        ]);

        return response()->json($click, 201);
    }

    public function getClicks(Request $request)
    {
        $this->validate($request, [
            'date' => 'required|date',
            'direction' => 'required|in:<,>,<=,>=',
        ]);

        $date = $request->input('date');
        $direction = $request->input('direction');

        $query = Click::query();

        switch ($direction) {
            case '<':
                $query->where('timestamp', '<', $date);
                break;
            case '<=':
                $query->where('timestamp', '<=', $date);
                break;
            case '>':
                $query->where('timestamp', '>', $date);
                break;
            case '>=':
                $query->where('timestamp', '>=', $date);
                break;
        }

        $clicks = $query->get();

        if ($clicks->isEmpty()) {
            return response()->json(['message' => 'No clicks found'], 404);
        }

        return response()->json($clicks, 200);
    }
}

    {

            $this->validate($request, [
                'date' => 'required|date_format:Y-m-d H:i:s',
                'direction' => 'required|in:<,>,<=,>=',
            ]);

            $date = $request->input('date');
            $direction = $request->input('direction');

            $query = Click::query();

            switch ($direction) {
                case '<':
                    $query->where('timestamp', '<', $date);
                    break;
                case '<=':
                    $query->where('timestamp', '<=', $date);
                    break;
                case '>':
                    $query->where('timestamp', '>', $date);
                    break;

                    $query->where('timestamp', '>=', $date);
                    break;

            $clicks = $query->get();

            if ($clicks->isEmpty()) {
                return response()->json(['message' => 'No clicks found'], 404);
            }

            return response()->json($clicks, 200);
        } catch (\Exception $e) {
            Log::error('Error retrieving clicks: ' . $e->getMessage());
            return response()->json(['error' => 'Unable to retrieve clicks'], 500);
        }
    }
}

    {

            $this->validate($request, [
                'date' => 'required|date_format:Y-m-d H:i:s',
                'direction' => 'required|in:<,>,<=,>=',
            ]);

            $date = $request->input('date');
            $direction = $request->input('direction');

            $query = Click::query();

            switch ($direction) {
                case '<':
                    $query->where('timestamp', '<', $date);
                    break;
                case '<=':
                    $query->where('timestamp', '<=', $date);
                    break;
                case '>':
                    $query->where('timestamp', '>', $date);
                    break;
                case '>=':
                    $query->where('timestamp', '>=', $date);
                    break;
            }

            // Implement pagination
            $clicks = $query->paginate(10); // Limit to 10 records per page

            return response()->json($clicks, 200);
        } catch (\Exception $e) {
            Log::error('Error retrieving clicks: ' . $e->getMessage());
            return response()->json(['error' => 'Unable to retrieve clicks'], 500);
