from django.http import JsonResponse, HttpResponse
from django.views import View
from .models import Entry
import json
from django.shortcuts import get_object_or_404

class EntryListView(View):
    def get(self, request):
        entries = Entry.objects.all()
        data = [{"id": entry.id, "title": entry.title} for entry in entries]
        return JsonResponse(data, safe=False)

    def post(self, request):
        data = json.loads(request.body)
        entry = Entry.objects.create(
            title=data['title'],
            content=data['content'],
            created_by=data['createdBy']
        )
        return JsonResponse({
            "id": entry.id,
            "title": entry.title,
            "content": entry.content,
            "lastModifiedBy": entry.last_modified_by,
            "lastModifiedAt": entry.last_modified_at.isoformat()
        }, status=201)

class EntryDetailView(View):
    def get(self, request, entryId):
        entry = get_object_or_404(Entry, id=entryId)
        return JsonResponse({
            "id": entry.id,
            "title": entry.title,
            "content": entry.content,
            "lastModifiedBy": entry.last_modified_by,
            "lastModifiedAt": entry.last_modified_at.isoformat()
        })

    def put(self, request, entryId):
        entry = get_object_or_404(Entry, id=entryId)
        data = json.loads(request.body)
        entry.content = data['content']
        entry.last_modified_by = data['modifiedBy']
        entry.save()
        return JsonResponse({
            "id": entry.id,
            "title": entry.title,
            "content": entry.content,
            "lastModifiedBy": entry.last_modified_by,
            "lastModifiedAt": entry.last_modified_at.isoformat()
        })

from django.http import JsonResponse
from django.views import View
from .models import Entry
import json
from django.shortcuts import get_object_or_404
from django.views.decorators.csrf import csrf_exempt
from django.utils.decorators import method_decorator

@method_decorator(csrf_exempt, name='dispatch')
class EntryListView(View):
    def get(self, request):
        entries = Entry.objects.all()
        data = [{"id": entry.id, "title": entry.title} for entry in entries]
        return JsonResponse(data, safe=False)

    def post(self, request):
        try:
            data = json.loads(request.body)
            title = data.get('title')
            content = data.get('content')
            created_by = data.get('createdBy')

            if not title or not content or not created_by:
                return JsonResponse({"error": "Missing required fields."}, status=400)

            entry = Entry.objects.create(
                title=title,
                content=content,
                created_by=created_by
            )
            return JsonResponse({
                "id": entry.id,
                "title": entry.title,
                "content": entry.content,
                "lastModifiedBy": entry.last_modified_by,
                "lastModifiedAt": entry.last_modified_at.isoformat()
            }, status=201)
        except json.JSONDecodeError:
            return JsonResponse({"error": "Invalid JSON."}, status=400)

class EntryDetailView(View):
    def get(self, request, entryId):
        entry = get_object_or_404(Entry, id=entryId)
        return JsonResponse({
            "id": entry.id,
            "title": entry.title,
            "content": entry.content,
            "lastModifiedBy": entry.last_modified_by,
            "lastModifiedAt": entry.last_modified_at.isoformat()
        })

    def put(self, request, entryId):
        entry = get_object_or_404(Entry, id=entryId)
        try:
            data = json.loads(request.body)
            content = data.get('content')
            modified_by = data.get('modifiedBy')

            if not content or not modified_by:
                return JsonResponse({"error": "Missing required fields."}, status=400)

            entry.content = content
            entry.last_modified_by = modified_by

        try:
            data = json.loads(request.body)
            content = data.get('content')
            modified_by_id = data.get('modifiedBy')

            if not content or not modified_by_id:
                return JsonResponse({"error": "Missing required fields."}, status=400)

            modified_by = get_object_or_404(User, id=modified_by_id)

            entry.content = content
            entry.last_modified_by = modified_by
            entry.save()
            return JsonResponse({
                "id": entry.id,
                "title": entry.title,
                "content": entry.content,
                "lastModifiedBy": entry.last_modified_by.id if entry.last_modified_by else None,
                "lastModifiedAt": entry.last_modified_at.isoformat()
            })
        except json.JSONDecodeError:
            return JsonResponse({"error": "Invalid JSON."}, status=400)
