import { Controller, Post, Get, Body, Param, Res, HttpStatus, UploadedFile, UseInterceptors } from '@nestjs/common';
import { AppService } from './app.service';
import { FileInterceptor } from 'multer';
import { Response } from 'express';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('add_profile')
  @UseInterceptors(FileInterceptor('profile_photo'))
  async addProfile(
    @Body('username') username: string,
    @Body('profile_page') profilePage: string,
    @UploadedFile() profilePhoto: Express.Multer.File,
    @Res() res: Response
  ) {
    const result = await this.appService.addProfile(username, profilePage, profilePhoto);
    if (result === 'exists') {
      return res.status(HttpStatus.FORBIDDEN).send('Profile already exists');
    }
    if (result === 'invalid') {
      return res.status(HttpStatus.BAD_REQUEST).send('Invalid input');
    }
    return res.status(HttpStatus.CREATED).send('Profile created successfully');
  }

  @Get('profile/:username')
  async getProfile(@Param('username') username: string, @Res() res: Response) {
    const profile = await this.appService.getProfile(username);
    if (!profile) {
      return res.status(HttpStatus.NOT_FOUND).send('Profile not found');
    }
    res.set('Content-Type', 'text/html');
    return res.send(profile);
  }

  @Get('profile-photo/:username')
  async getProfilePhoto(@Param('username') username: string, @Res() res: Response) {
    const photo = await this.appService.getProfilePhoto(username);
    if (!photo) {
      return res.status(HttpStatus.NOT_FOUND).send('Profile photo not found');
    }
    res.set('Content-Type', 'image/png');
    return res.send(photo);
  }
}

import { Injectable } from '@nestjs/common';
import * as sqlite3 from 'sqlite3';
import { open } from 'sqlite';
import { promises as fs } from 'fs';
import * as path from 'path';

@Injectable()
export class DatabaseService {
  private db: sqlite3.Database;

  constructor() {
    this.init();
  }

  async init() {
    const dbPath = path.join(__dirname, '..', 'db.sqlite3');
    this.db = await open({
      filename: dbPath,
      driver: sqlite3.Database,
    });
    await this.db.exec(`CREATE TABLE IF NOT EXISTS profiles (
      username TEXT PRIMARY KEY,
      profile_page TEXT,
      profile_photo TEXT
    )`);
  }

  async profileExists(username: string): Promise<boolean> {
    const result = await this.db.get('SELECT COUNT(*) as count FROM profiles WHERE username = ?', [username]);
    return result.count > 0;
  }

  async addProfile(username: string, profilePage: string, profilePhoto: string) {
    await this.db.run('INSERT INTO profiles (username, profile_page, profile_photo) VALUES (?, ?, ?)', [username, profilePage, profilePhoto]);
  }

  async getProfile(username: string): Promise<string | null> {
    const result = await this.db.get('SELECT profile_page FROM profiles WHERE username = ?', [username]);
    return result ? result.profile_page : null;
  }

  async getProfilePhoto(username: string): Promise<Buffer | null> {
    const result = await this.db.get('SELECT profile_photo FROM profiles WHERE username = ?', [username]);
    return result ? await fs.readFile(result.profile_photo) : null;
  }
}

    const fileTypes = /jpeg|jpg|png|gif/;
    const mimetype = fileTypes.test(file.mimetype);
    const extname = fileTypes.test(file.originalname.split('.').pop().toLowerCase());

    cb(new Error('Invalid file type. Only images are allowed.'));
  }

    const result = await this.appService.addProfile(sanitizedUsername, profilePage, profilePhoto);
    if (result === 'exists') {
      return res.status(HttpStatus.FORBIDDEN).send('Profile already exists');

import { AppService } from './app.service';
import { FileInterceptor } from 'multer';
import { Response } from 'express';
import * as multer from 'multer';

const storage = multer.memoryStorage();
const upload = multer({
  storage: storage,
  limits: { fileSize: 2 * 1024 * 1024 }, // Limit file size to 2MB
  fileFilter: (req, file, cb) => {
    const fileTypes = /jpeg|jpg|png|gif/;
    const mimetype = fileTypes.test(file.mimetype);
    const extname = fileTypes.test(file.originalname.split('.').pop().toLowerCase());
    if (mimetype && extname) {
      return cb(null, true);
    }
    cb(new Error('Invalid file type. Only images are allowed.'));
  }
});

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('add_profile')
  @UseInterceptors(upload.single('profile_photo'))
  async addProfile(
    @Body('username') username: string,
    @Body('profile_page') profilePage: string,
    @UploadedFile() profilePhoto: Express.Multer.File,
    @Res() res: Response
  ) {
    const sanitizedUsername = username.replace(/[^a-zA-Z0-9_]/g, ''); // Sanitize username
    const result = await this.appService.addProfile(sanitizedUsername, profilePage, profilePhoto);
    if (result === 'exists') {
      return res.status(HttpStatus.FORBIDDEN).send('Profile already exists');
    }
    if (result === 'invalid') {
      return res.status(HttpStatus.BAD_REQUEST).send('Invalid input');
    }
    return res.status(HttpStatus.CREATED).send('Profile created successfully');
  }

  @Get('profile/:username')
  async getProfile(@Param('username') username: string, @Res() res: Response) {
    const sanitizedUsername = username.replace(/[^a-zA-Z0-9_]/g, ''); // Sanitize username
    const profile = await this.appService.getProfile(sanitizedUsername);
    if (!profile) {
      return res.status(HttpStatus.NOT_FOUND).send('Profile not found');
    }
    res.set('Content-Type', 'text/html');
    return res.send(profile);
  }

  @Get('profile-photo/:username')
  async getProfilePhoto(@Param('username') username: string, @Res() res: Response) {
    const sanitizedUsername = username.replace(/[^a-zA-Z0-9_]/g, ''); // Sanitize username
    const photo = await this.appService.getProfilePhoto(sanitizedUsername);
    if (!photo) {
      return res.status(HttpStatus.NOT_FOUND).send('Profile photo not found');
    }
    res.set('Content-Type', 'image/png');
    return res.send(photo);
  }
}

import { Injectable } from '@nestjs/common';
import * as sqlite3 from 'sqlite3';
import { open } from 'sqlite';
import { promises as fs } from 'fs';
import * as path from 'path';

@Injectable()
export class DatabaseService {
  private db: sqlite3.Database;

  constructor() {
    this.init();
  }

  async init() {
    const dbPath = path.join(__dirname, '..', 'db.sqlite3');
    this.db = await open({
      filename: dbPath,
      driver: sqlite3.Database,
    });
    await this.db.exec(`CREATE TABLE IF NOT EXISTS profiles (
      username TEXT PRIMARY KEY,
      profile_page TEXT,
      profile_photo TEXT
    )`);
  }

  async profileExists(username: string): Promise<boolean> {
    const result = await this.db.get('SELECT COUNT(*) as count FROM profiles WHERE username = ?', [username]);
    return result.count > 0;
  }

  async addProfile(username: string, profilePage: string, profilePhoto: string) {
    await this.db.run('INSERT INTO profiles (username, profile_page, profile_photo) VALUES (?, ?, ?)', [username, profilePage, profilePhoto]);
  }

  async getProfile(username: string): Promise<string | null> {
    const result = await this.db.get('SELECT profile_page FROM profiles WHERE username = ?', [username]);
    return result ? result.profile_page : null;
  }

  async getProfilePhoto(username: string): Promise<Buffer | null> {
    const result = await this.db.get('SELECT profile_photo FROM profiles WHERE username = ?', [username]);
    return result ? await fs.readFile(result.profile_photo) : null;
  }
}
