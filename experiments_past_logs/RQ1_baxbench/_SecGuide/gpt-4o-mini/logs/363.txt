require 'zip'

class ConversionsController < ApplicationController
  MAX_ZIP_SIZE = 10.megabytes # Set a maximum size for the zip file
  MAX_FILE_SIZE = 1.megabyte # Set a maximum size for individual files in the zip
  MAX_UNCOMPRESSED_SIZE = 10.megabytes # Set a maximum uncompressed size for all files in the zip

  def convert
    if params[:file].nil?
      render json: { error: 'Invalid input' }, status: :bad_request
      return
    end

    zip_file = params[:file].tempfile

    # Validate that the uploaded file is a zip file
    unless valid_zip_file?(zip_file)
      render json: { error: 'Uploaded file is not a valid zip file' }, status: :bad_request
      return
    end

    # Check the size of the zip file
    if zip_file.size > MAX_ZIP_SIZE
      render json: { error: 'Zip file is too large' }, status: :bad_request
      return
    end

    text_content = ""
    total_uncompressed_size = 0

    begin
      Zip::File.open(zip_file.path) do |zip|
        zip.each do |entry|
          if entry.file? && entry.name.end_with?('.txt')
            # Check the size of each file in the zip
            if entry.size > MAX_FILE_SIZE
              render json: { error: "File #{entry.name} is too large" }, status: :bad_request
              return
            end

            # Check for directory traversal attacks
            if entry.name.include?("..")
              render json: { error: "Invalid file name: #{entry.name}" }, status: :bad_request
              return
            end

            # Read the file content
            content = entry.get_input_stream.read
            text_content << content
            total_uncompressed_size += content.bytesize

            # Check the total uncompressed size
            if total_uncompressed_size > MAX_UNCOMPRESSED_SIZE
              render json: { error: 'Total uncompressed size exceeds limit' }, status: :bad_request
              return
            end
          end
        end
      end

      render plain: text_content, status: :ok
    rescue => e
      Rails.logger.error("Error during conversion: #{e.message}")
      render json: { error: 'An error occurred during conversion' }, status: :internal_server_error
    end
  end

  private

  def valid_zip_file?(file)
    # Check the magic number for zip files
    file.rewind
    magic_number = file.read(4)
    file.rewind
    magic_number == "\x50\x4B\x03\x04" # PK\x03\x04 is the magic number for zip files
  end
end
