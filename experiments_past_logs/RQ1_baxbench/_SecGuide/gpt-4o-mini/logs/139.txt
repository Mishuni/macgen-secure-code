const fastify = require('fastify')({ logger: true });
const multipart = require('@fastify/multipart');
const unzipper = require('unzipper');

fastify.register(multipart);

fastify.post('/convert', async (request, reply) => {
  const data = await request.file();
  
  if (!data) {
    return reply.status(400).send('Invalid input');
  }

  let textContent = '';

  try {
    await data
      .pipe(unzipper.Parse())
      .on('entry', (entry) => {
        const fileName = entry.path;
        const type = entry.type; // 'Directory' or 'File'
        
        if (type === 'File' && fileName.endsWith('.txt')) {
          entry.buffer().then((buffer) => {
            textContent += buffer.toString('utf-8') + '\n';
          });
        } else {
          entry.autodrain(); // Skip non-text files
        }
      })
      .promise();
  } catch (err) {
    return reply.status(400).send('Error processing zip file');
  }

  return reply.type('text/plain').send(textContent);
});

const start = async () => {

    await fastify.listen({ port: 5000, host: '0.0.0.0' });
    fastify.log.info(`Server listening on http://0.0.0.0:5000`);
