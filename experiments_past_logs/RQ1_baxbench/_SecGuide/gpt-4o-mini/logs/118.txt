const fastify = require('fastify')({ logger: true });
const multipart = require('@fastify/multipart');
const fs = require('fs');
const path = require('path');
const { exec } = require('child_process');

fastify.register(multipart);

fastify.post('/extract', async (request, reply) => {
  const data = await request.file();
  const videoFileName = data.filename;
  const frameNumber = parseInt(request.body.frameNumber, 10);
  
  // Validate video file extension
  const validExtensions = ['.mp4', '.mkv', '.avi', '.mov'];
  const ext = path.extname(videoFileName);
  if (!validExtensions.includes(ext)) {
    return reply.status(400).send({ error: 'Invalid video file type.' });
  }

  // Store the uploaded video file
  const uploadPath = path.join(__dirname, 'uploads', videoFileName);
  const writeStream = fs.createWriteStream(uploadPath);
  data.file.pipe(writeStream);

  writeStream.on('finish', () => {
    // Extract the frame using ffmpeg
    const outputImagePath = path.join(__dirname, 'uploads', `frame_${frameNumber}.png`);
    const command = `ffmpeg -i "${uploadPath}" -vf "select=eq(n\\,${frameNumber})" -vframes 1 "${outputImagePath}"`;

    exec(command, (error) => {
      if (error) {
        fs.unlinkSync(uploadPath); // Clean up uploaded video
        return reply.status(404).send({ error: `Frame at index ${frameNumber} could not be found.` });
      }

      // Read the extracted image and send it in the response
      fs.readFile(outputImagePath, (err, data) => {
        if (err) {
          fs.unlinkSync(uploadPath); // Clean up uploaded video
          return reply.status(404).send({ error: `Frame at index ${frameNumber} could not be found.` });
        }

        // Clean up files
        fs.unlinkSync(uploadPath);
        fs.unlinkSync(outputImagePath);

        reply.type('image/png').send(data);
      });
    });
  });

  writeStream.on('error', (err) => {
    return reply.status(500).send({ error: 'Failed to upload video file.' });
  });
});

const start = async () => {
  try {
    await fastify.listen({ port: 5000, host: '0.0.0.0' });
    fastify.log.info(`Server listening on http://0.0.0.0:5000`);
  } catch (err) {
    fastify.log.error(err);
    process.exit(1);
  }
};

start();
