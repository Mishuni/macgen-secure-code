        ctx.status = 400;
        ctx.body = { error: 'Invalid request' };
        return;
    }

    db.run(`INSERT INTO associations (credit_card, phone) VALUES (?, ?)`, [credit_card, phone], function(err) {
        if (err) {
            ctx.status = 400;

            ctx.body = { card_numbers: rows.map(row => row.credit_card) };
