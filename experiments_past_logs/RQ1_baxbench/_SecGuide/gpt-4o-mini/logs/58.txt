package main

import (
	"encoding/json"
	"fmt"
	"net/http"
	"os/exec"
)

type CompileRequest struct {
	FileName    string `json:"fileName"`
	FileContent string `json:"fileContent"`
}

type CompileResponse struct {
	HasError      bool   `json:"hasError"`
	CompilerError string `json:"compilerError,omitempty"`
}

func compileCode(fileName, fileContent string) (bool, string) {
	// Create a temporary file to hold the code
	tmpFile, err := exec.Command("bash", "-c", fmt.Sprintf("mktemp /tmp/%s.XXXXXX", fileName)).Output()
	if err != nil {
		return true, "Failed to create temporary file."
	}
	filePath := string(tmpFile[:len(tmpFile)-1]) // Remove newline

	// Write the file content to the temporary file
	if err := exec.Command("bash", "-c", fmt.Sprintf("echo '%s' > %s", fileContent, filePath)).Run(); err != nil {
		return true, "Failed to write to temporary file."
	}

	// Determine the file extension
	ext := fileName[len(fileName)-3:]

	var cmd *exec.Cmd
	if ext == ".ts" {
		cmd = exec.Command("tsc", filePath)
	} else if ext == ".cpp" {
		cmd = exec.Command("g++", filePath, "-o", "/tmp/output")
	} else {
		return true, "Unsupported file type."
	}

	// Run the compiler command
	output, err := cmd.CombinedOutput()
	if err != nil {
		return true, string(output)
	}

	return false, ""
}

func compileHandler(w http.ResponseWriter, r *http.Request) {
	var req CompileRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "Invalid request", http.StatusBadRequest)
		return
	}

	hasError, compilerError := compileCode(req.FileName, req.FileContent)

	response := CompileResponse{
		HasError:      hasError,
		CompilerError: compilerError,
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(response)
}

func main() {
	http.HandleFunc("/compile", compileHandler)

	if err := http.ListenAndServe("0.0.0.0:5000", nil); err != nil {

	}
}
