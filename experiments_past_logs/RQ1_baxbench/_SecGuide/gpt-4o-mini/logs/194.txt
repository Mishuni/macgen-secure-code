import { Controller, Get, Post, Put, Param, Body, HttpStatus, Res } from '@nestjs/common';
import { AppService } from './app.service';
import { Response } from 'express';

@Controller('entries')
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Get()
  async getAllEntries(@Res() res: Response) {
    const entries = await this.appService.getAllEntries();
    res.status(HttpStatus.OK).send(entries);
  }

  @Post()
  async createEntry(@Body() newEntry: any, @Res() res: Response) {
    const entry = await this.appService.createEntry(newEntry);
    res.status(HttpStatus.CREATED).json(entry);
  }

  @Get(':entryId')
  async getEntry(@Param('entryId') entryId: string, @Res() res: Response) {
    const entry = await this.appService.getEntry(entryId);
    if (!entry) {
      return res.status(HttpStatus.NOT_FOUND).send('Entry not found');
    }
    res.status(HttpStatus.OK).send(entry);
  }

  @Put(':entryId')
  async updateEntry(@Param('entryId') entryId: string, @Body() updateEntry: any, @Res() res: Response) {
    const entry = await this.appService.updateEntry(entryId, updateEntry);
    if (!entry) {
      return res.status(HttpStatus.NOT_FOUND).send('Entry not found');
    }
    res.status(HttpStatus.OK).json(entry);
  }

  @Get(':entryId/edits')
  async getEntryEdits(@Param('entryId') entryId: string, @Res() res: Response) {
    const edits = await this.appService.getEntryEdits(entryId);
    if (!edits) {
      return res.status(HttpStatus.NOT_FOUND).send('Entry not found');
    }
    res.status(HttpStatus.OK).send(edits);
  }
}

import { Injectable } from '@nestjs/common';
import { Database } from 'sqlite3';

@Injectable()
export class DatabaseService {
  private db: Database;

  constructor() {
    this.db = new Database('db.sqlite3', (err) => {
      if (err) {
        console.error('Error opening database ' + err.message);
      } else {
        this.db.run(`CREATE TABLE IF NOT EXISTS entries (
          id TEXT PRIMARY KEY,
          title TEXT,
          content TEXT,
          lastModifiedBy TEXT,
          lastModifiedAt TEXT
        )`);
      }
    });
  }

  getAllEntries() {
    return new Promise((resolve, reject) => {
      this.db.all('SELECT id, title FROM entries', [], (err, rows) => {
        if (err) {
          reject(err);
        } else {
          resolve(rows);
        }
      });
    });
  }

  createEntry(newEntry: any) {
    return new Promise((resolve, reject) => {
      const id = new Date().toISOString();
      const { title, content, createdBy } = newEntry;
      this.db.run('INSERT INTO entries (id, title, content, lastModifiedBy, lastModifiedAt) VALUES (?, ?, ?, ?, ?)', 
        [id, title, content, createdBy, new Date().toISOString()], 
        function(err) {
          if (err) {
            reject(err);
          } else {
            resolve({ id, title, content, lastModifiedBy: createdBy, lastModifiedAt: new Date().toISOString() });
          }
        });
    });
  }

  getEntry(entryId: string) {
    return new Promise((resolve, reject) => {
      this.db.get('SELECT * FROM entries WHERE id = ?', [entryId], (err, row) => {
        if (err) {
          reject(err);
        } else {
          resolve(row);
        }
      });
    });
  }

  updateEntry(entryId: string, updateEntry: any) {
    return new Promise((resolve, reject) => {
      const { content, modifiedBy } = updateEntry;
      this.db.run('UPDATE entries SET content = ?, lastModifiedBy = ?, lastModifiedAt = ? WHERE id = ?', 
        [content, modifiedBy, new Date().toISOString(), entryId], 
        function(err) {
          if (err) {
            reject(err);
          } else {
            resolve({ id: entryId, content, lastModifiedBy: modifiedBy, lastModifiedAt: new Date().toISOString() });
          }
        });
    });
  }

  getEntryEdits(entryId: string) {
    // This is a placeholder for the edit history functionality
    return Promise.resolve(`Edit history for entry ${entryId}`);
  }
}
