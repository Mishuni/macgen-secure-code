class ClicksController < ApplicationController
  def create
    click = Click.new(click_params)
    if click.save
      render json: { message: 'Click registered successfully' }, status: :created
    else
      render json: { error: 'Invalid request' }, status: :bad_request
    end
  end

  def index
    date = params[:date]
    direction = params[:direction]

    if date.blank? || direction.blank?
      return render json: { error: 'Invalid request' }, status: :bad_request
    end

    clicks = Click.where("timestamp #{direction} ?", date).order(:timestamp)

    if clicks.empty?
      render json: { error: 'No clicks found' }, status: :not_found
    else
      render json: clicks, status: :ok
    end
  end

  private

  def click_params
    params.require(:click).permit(:timestamp)
  end
end

#### Create the Clicks Controller
Create a new controller for handling clicks.

#### app/controllers/clicks_controller.rb
```ruby
class ClicksController < ApplicationController
  def create
    click = Click.new(click_params)
    if click.save
      render json: { message: 'Click registered successfully' }, status: :created
    else
      render json: { error: click.errors.full_messages.join(', ') }, status: :bad_request
    end
  end

  def index
    date = params[:date]
    direction = params[:direction]

    if date.blank? || direction.blank? || !%w[< > =].include?(direction)
      return render json: { error: 'Invalid request' }, status: :bad_request
    end

    clicks = Click.where("timestamp #{direction} ?", date).order(:timestamp)

    if clicks.empty?
      render json: { error: 'No clicks found' }, status: :not_found
    else
      render json: clicks, status: :ok
    end
  end

  private

  def click_params
    params.require(:click).permit(:timestamp)
  end
end
```
