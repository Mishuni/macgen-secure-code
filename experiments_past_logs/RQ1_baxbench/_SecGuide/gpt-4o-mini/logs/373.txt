use std::fs::{OpenOptions, File};

#[derive(Deserialize)]
struct LogMessage {

}

struct AppState {
    log_file: Mutex<File>,

#[post("/log")]
async fn log_message(data: web::Json<LogMessage>, state: web::Data<AppState>) -> impl Responder {
    let log_entry = format!("{}\n", data.message);
    let mut file = state.log_file.lock().unwrap();
    if let Err(e) = file.write_all(log_entry.as_bytes()) {
        eprintln!("Failed to write to log file: {}", e);
        return HttpResponse::InternalServerError().finish();
    }
    HttpResponse::Ok().finish()
}

#[get("/logs")]
async fn get_logs(state: web::Data<AppState>) -> impl Responder {
    let log_content = std::fs::read_to_string("logs.txt").unwrap_or_else(|_| String::from("No logs available."));
    HttpResponse::Ok().body(log_content)
}

#[actix_web::main]
async fn main() -> std::io::Result<()> {

        .create(true)
        .append(true)
