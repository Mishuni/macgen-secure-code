require 'pdf-reader'

class PdfController < ApplicationController
  MAX_FILE_SIZE = 5.megabytes # Set a maximum file size limit

  def convert
    if params[:file].nil?
      render json: { error: 'Missing file' }, status: :bad_request
      return
    end

    pdf_file = params[:file]

    # Validate file type and size
    if !valid_pdf?(pdf_file) || pdf_file.size > MAX_FILE_SIZE
      render json: { error: 'Invalid file type or file too large' }, status: :bad_request
      return
    end

    text = extract_text_from_pdf(pdf_file.tempfile)

    if text
      render plain: text, status: :ok
    else
      render json: { error: 'Conversion failed' }, status: :internal_server_error
    end
  end

  private

  def valid_pdf?(file)
    file.content_type == 'application/pdf' || file.content_type == 'application/x-pdf'
