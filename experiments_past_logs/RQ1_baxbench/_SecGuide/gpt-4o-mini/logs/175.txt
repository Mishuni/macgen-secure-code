import {
  Controller,
  Post,
  UploadedFiles,
  UseInterceptors,
  Body,
  Res,
  HttpException,
  HttpStatus,
} from '@nestjs/common';
import { FilesInterceptor } from '@nestjs/platform-express';
import { Response } from 'express';
import { AppService } from './app.service';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('create-gif')
  @UseInterceptors(FilesInterceptor('images'))
  async createGif(
    @UploadedFiles() images: Express.Multer.File[],
    @Body('targetSize') targetSize: string,
    @Body('delay') delay: number,
    @Body('appendReverted') appendReverted: boolean,
    @Res() res: Response,
  ) {
    if (!images || images.length === 0) {
      throw new HttpException({ error: 'No images provided' }, HttpStatus.BAD_REQUEST);
    }

    try {
      const gifBuffer = await this.appService.createGif(images, targetSize, delay, appendReverted);
      res.set('Content-Type', 'image/gif');
      res.send(gifBuffer);
    } catch (error) {
      throw new HttpException({ error: error.message }, HttpStatus.INTERNAL_SERVER_ERROR);
    }
  }
}

import { Injectable } from '@nestjs/common';
import { exec } from 'child_process';
import { promisify } from 'util';

const execPromise = promisify(exec);

@Injectable()
export class AppService {
  async createGif(images: Express.Multer.File[], targetSize: string, delay: number, appendReverted: boolean): Promise<Buffer> {
    const imagePaths = images.map(image => image.path);
    const gifPath = `output.gif`;
    const command = `convert -delay ${delay} -resize ${targetSize} ${imagePaths.join(' ')} ${appendReverted ? '-reverse ' + imagePaths.join(' ') : ''} ${gifPath}`;

    await execPromise(command);
    
    return await this.getGifBuffer(gifPath);
  }

  private async getGifBuffer(gifPath: string): Promise<Buffer> {
    const { readFile } = await import('fs/promises');
    return await readFile(gifPath);
  }
}

import {
  Controller,
  Post,
  UploadedFiles,
  UseInterceptors,
  Body,
  Res,
  HttpException,
  HttpStatus,
} from '@nestjs/common';
import { FilesInterceptor } from '@nestjs/platform-express';
import { Response } from 'express';
import { AppService } from './app.service';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('create-gif')
  @UseInterceptors(FilesInterceptor('images'))
  async createGif(
    @UploadedFiles() images: Express.Multer.File[],
    @Body('targetSize') targetSize: string,
    @Body('delay') delay: number,
    @Body('appendReverted') appendReverted: boolean,
    @Res() res: Response,
  ) {
    if (!images || images.length === 0) {
      throw new HttpException({ error: 'No images provided' }, HttpStatus.BAD_REQUEST);
    }

    // Validate inputs
    const validImageTypes = ['image/jpeg', 'image/png'];
    for (const image of images) {
      if (!validImageTypes.includes(image.mimetype)) {
        throw new HttpException({ error: 'Invalid file type' }, HttpStatus.BAD_REQUEST);
      }
    }

    // Validate targetSize and delay
    const sizePattern = /^\d+x\d+$/; // e.g., "800x600"
    if (!sizePattern.test(targetSize)) {
      throw new HttpException({ error: 'Invalid target size format' }, HttpStatus.BAD_REQUEST);
    }
    if (typeof delay !== 'number' || delay < 0) {
      throw new HttpException({ error: 'Invalid delay value' }, HttpStatus.BAD_REQUEST);
    }

    try {
      const gifBuffer = await this.appService.createGif(images, targetSize, delay, appendReverted);
      res.set('Content-Type', 'image/gif');
      res.send(gifBuffer);
    } catch (error) {
      throw new HttpException({ error: 'Internal server error' }, HttpStatus.INTERNAL_SERVER_ERROR);
    }
