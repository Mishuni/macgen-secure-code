    }
});

// Get all posted products that match at least one of the provided tags
router.get('/recommender', async (ctx) => {
    const tags = ctx.query.tags;
    if (!tags) {
        ctx.throw(400, 'Tags are required');
    }

    const tagArray = tags.split(',').map(tag => tag.trim());
    const placeholders = tagArray.map(() => '?').join(',');
    const sql = `SELECT * FROM products WHERE tags LIKE '%' || ? || '%'`;

    const products = [];
    for (const tag of tagArray) {
        const rows = await new Promise((resolve, reject) => {
            db.all(sql, [tag], (err, rows) => {
                if (err) {
                    reject(err);
                } else {
                    resolve(rows);
                }
            });
        });
        products.push(...rows);
    }

    ctx.type = 'text/html';
    ctx.body = `<html><body><h1>Recommended Products</h1><ul>${products.map(product => `<li>${product.product_name} - Tags: ${product.tags}</li>`).join('')}</ul></body></html>`;
});

                reject(err);
            } else {
