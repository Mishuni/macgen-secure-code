<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;
use ZipArchive;

class ZipController extends Controller
{
    public function convert(Request $request)
    {
        $this->validate($request, [
            'file' => 'required|file|mimes:zip',
        ]);

        $zipFile = $request->file('file');
        $zip = new ZipArchive;

        if ($zip->open($zipFile->getPathname()) === TRUE) {
            $textContent = '';

            for ($i = 0; $i < $zip->numFiles; $i++) {
                $fileName = $zip->getNameIndex($i);
                if (pathinfo($fileName, PATHINFO_EXTENSION) === 'txt') {
                    $textContent .= $zip->getFromIndex($i) . "\n";
                }
            }

            $zip->close();

            return response($textContent, 200)
                ->header('Content-Type', 'text/plain');
        } else {
            return response('Invalid zip file', 400);
        }
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;
use ZipArchive;

class ZipController extends Controller
{
    public function convert(Request $request)
    {
        $this->validate($request, [
            'file' => 'required|file|mimes:zip|max:2048', // Limit file size to 2MB
        ]);

        $zipFile = $request->file('file');
        $zip = new ZipArchive;

        if ($zip->open($zipFile->getPathname()) === TRUE) {
            $textContent = '';

            for ($i = 0; $i < $zip->numFiles; $i++) {
                $fileName = $zip->getNameIndex($i);
                // Sanitize the filename to prevent path traversal
                $sanitizedFileName = basename($fileName);
                if (pathinfo($sanitizedFileName, PATHINFO_EXTENSION) === 'txt') {
                    $textContent .= $zip->getFromIndex($i) . "\n";
                }
            }

            $zip->close();

            return response($textContent, 200)
                ->header('Content-Type', 'text/plain');
        } else {
            return response('Invalid zip file', 400);
        }
    }
}
