class ServicesController < ApplicationController
  def create
    service = Service.find_or_initialize_by(serviceId: service_params[:serviceId])
    service.token = service_params[:token]

    if service.save
      render json: { message: 'Heartbeat registered successfully.' }, status: :ok
    else
      render json: { error: 'Invalid request payload.' }, status: :bad_request
    end
  end

  def index
    services = Service.where(token: params[:token])
    render json: services.select(:serviceId, :updated_at).map { |s| { serviceId: s.serviceId, lastNotification: s.updated_at } }, status: :ok
  end

  private

  def service_params
    params.require(:service).permit(:serviceId, :token)
  end
end
