const express = require('express');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = 5000;
const FILES_ROOT = path.join(__dirname, 'files');

app.get('/search', (req, res) => {
    const { regex, directory = '', recursive = false } = req.query;

    // Validate regex
    let regexPattern;
    try {
        regexPattern = new RegExp(regex);
    } catch (e) {
        return res.status(400).json({ error: 'Invalid regex pattern.' });
    }

    // Validate directory
    const normalizedDirectory = path.normalize(directory).replace(/^(\.\.(\/|\\|$)|\/|\\)/, '');
    const searchPath = path.join(FILES_ROOT, normalizedDirectory);

    // Check if the directory is accessible
    if (!fs.existsSync(searchPath) || !fs.lstatSync(searchPath).isDirectory()) {
        return res.status(400).json({ error: 'Invalid directory path.' });
    }

    const matchingFiles = [];

    const searchFiles = (dir) => {
        const files = fs.readdirSync(dir);
        for (const file of files) {
            const filePath = path.join(dir, file);
            if (fs.lstatSync(filePath).isDirectory() && recursive) {
                searchFiles(filePath);
            } else {
                const content = fs.readFileSync(filePath, 'utf-8');
                if (regexPattern.test(content)) {
                    matchingFiles.push(path.relative(searchPath, filePath));
                }
            }
        }
    };

    searchFiles(searchPath);

    return res.status(200).json({ files: matchingFiles });
});

app.listen(PORT, '0.0.0.0', () => {
    console.log(`Server is running on http://0.0.0.0:${PORT}`);
});
