import zipfile
import os
from django.http import HttpResponse, JsonResponse
from django.views.decorators.csrf import csrf_exempt

@csrf_exempt
def convert_zip_to_text(request):
    if request.method == 'POST' and request.FILES.get('file'):
        zip_file = request.FILES['file']
        if zip_file.name.endswith('.zip'):
            try:
                with zipfile.ZipFile(zip_file, 'r') as zip_ref:
                    text_content = []
                    for file_info in zip_ref.infolist():
                        if file_info.filename.endswith('.txt'):
                            with zip_ref.open(file_info) as file:
                                text_content.append(file.read().decode('utf-8'))
                    return HttpResponse('\n'.join(text_content), content_type='text/plain')
            except zipfile.BadZipFile:
                return JsonResponse({'error': 'Invalid zip file'}, status=400)
        else:
            return JsonResponse({'error': 'File is not a zip file'}, status=400)
    return JsonResponse({'error': 'Invalid input'}, status=400)

import zipfile
import os
from django.http import HttpResponse, JsonResponse
from django.views.decorators.csrf import csrf_protect

@csrf_protect
def convert_zip_to_text(request):
    if request.method == 'POST' and request.FILES.get('file'):
        zip_file = request.FILES['file']
        if zip_file.name.endswith('.zip'):
            try:
                with zipfile.ZipFile(zip_file, 'r') as zip_ref:
                    text_content = []
                    for file_info in zip_ref.infolist():
                        # Sanitize the filename to prevent directory traversal
                        safe_name = os.path.basename(file_info.filename)
                        if safe_name.endswith('.txt'):
                            with zip_ref.open(file_info) as file:
                                text_content.append(file.read().decode('utf-8'))
                    return HttpResponse('\n'.join(text_content), content_type='text/plain')
            except zipfile.BadZipFile:
                return JsonResponse({'error': 'Invalid zip file'}, status=400)
        else:
            return JsonResponse({'error': 'File is not a zip file'}, status=400)
    return JsonResponse({'error': 'Invalid input'}, status=400)
