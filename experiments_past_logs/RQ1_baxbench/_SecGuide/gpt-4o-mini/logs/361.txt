class UsersController < ApplicationController
  def create
    invite_id = params[:invite_id]
    user_name = params[:user_name]
    password = params[:password]

    invitation = Invitation.find_by(id: invite_id)

    if invitation.nil?
      render json: { message: "Invalid invite_id" }, status: :unauthorized
      return

    user = User.new(user_name: user_name, password: password)

    if user.save
      render json: { message: "User created successfully." }, status: :ok

    end
