<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;
use Symfony\Component\HttpFoundation\Response;

class PDFController extends Controller
{
    public function convert(Request $request)
    {
        // Validate the request
        $request->validate([
            'file' => 'required|file|mimes:pdf|max:2048',
        ]);

        // Store the uploaded file temporarily
        $file = $request->file('file');
        $filename = $this->sanitizeFileName($file->getClientOriginalName());
        $path = $file->storeAs('uploads', $filename);

        // Convert PDF to text using pdftotext
        $text = $this->pdfToText(storage_path('app/' . $path));

        // Delete the temporary file
        Storage::delete($path);

        return response($text, Response::HTTP_OK)
            ->header('Content-Type', 'text/plain');
    }

    private function pdfToText($filePath)
    {
        $output = [];
        $returnVar = 0;
        exec("pdftotext " . escapeshellarg($filePath) . " -", $output, $returnVar);

        if ($returnVar !== 0) {
            return response()->json(['error' => 'Conversion failed'], Response::HTTP_INTERNAL_SERVER_ERROR);
        }

        return implode("\n", $output);
    }

    private function sanitizeFileName($filename)
    {
        // Remove any unwanted characters and ensure a safe filename
        return preg_replace('/[^a-zA-Z0-9_\-\.]/', '_', $filename);
    }
}
