require 'find'

class SearchController < ApplicationController
  FILES_ROOT = './files'.freeze

  def search
    regex_pattern = params[:regex]
    directory = params[:directory] || ''
    recursive = ActiveModel::Type::Boolean.new.cast(params[:recursive])

    # Validate regex pattern
    begin
      regex = Regexp.new(regex_pattern)
    rescue RegexpError => e
      render json: { error: "Invalid regex pattern: #{e.message}" }, status: :bad_request and return
    end

    # Validate and sanitize directory path
    sanitized_directory = sanitize_directory_path(directory)
    unless sanitized_directory
      render json: { error: 'Invalid directory path' }, status: :bad_request and return
    end

    # Search files
    files = search_files(regex, sanitized_directory, recursive)
    render json: { files: files }, status: :ok
  rescue StandardError => e
    render json: { error: "Internal server error: #{e.message}" }, status: :internal_server_error
  end

  private

  def sanitize_directory_path(directory)
    # Ensure the directory path is safe and within the FILES_ROOT
    normalized_path = File.expand_path(File.join(FILES_ROOT, directory))
    return nil unless normalized_path.start_with?(File.expand_path(FILES_ROOT))
    return nil if directory.include?('..') || directory.include?('/') || directory.include?('\\')

    normalized_path
  end

  def search_files(regex, directory, recursive)
    files = []
    search_path = File.join(FILES_ROOT, directory)

    if recursive
      Find.find(search_path) do |path|
        next unless File.file?(path)
        files << path.sub("#{FILES_ROOT}/", '') if file_matches_regex?(path, regex)
      end
    else
      Dir.foreach(search_path) do |filename|
        next if filename == '.' || filename == '..'
        path = File.join(search_path, filename)
        files << path.sub("#{FILES_ROOT}/", '') if File.file?(path) && file_matches_regex?(path, regex)
      end
    end

    files
  end

  def file_matches_regex?(file_path, regex)
    File.open(file_path, 'r') do |file|
      file.each_line do |line|
        return true if line.match?(regex)
      end
    end
    false
  end
end

require 'find'
require 'timeout'

class SearchController < ApplicationController
  FILES_ROOT = './files'.freeze
  REGEX_TIMEOUT = 1 # seconds
  MAX_FILES = 1000 # Limit the number of files processed

  def search
    regex_pattern = params[:regex]
    directory = params[:directory] || ''
    recursive = ActiveModel::Type::Boolean.new.cast(params[:recursive])

    # Validate regex pattern with timeout
    begin
      regex = Timeout.timeout(REGEX_TIMEOUT) { Regexp.new(regex_pattern) }
    rescue Timeout::Error
      render json: { error: "Regex pattern is too complex" }, status: :bad_request and return
    rescue RegexpError
      render json: { error: "Invalid regex pattern" }, status: :bad_request and return
    end

    # Validate and sanitize directory path
    sanitized_directory = sanitize_directory_path(directory)
    unless sanitized_directory
      render json: { error: 'Invalid directory path' }, status: :bad_request and return
    end

    # Search files
    files = search_files(regex, sanitized_directory, recursive)
    render json: { files: files }, status: :ok
  rescue StandardError => e
    Rails.logger.error("Internal server error: #{e.message}")
    render json: { error: "Internal server error" }, status: :internal_server_error
  end

  private

  def sanitize_directory_path(directory)
    # Ensure the directory path is safe and within the FILES_ROOT
    normalized_path = File.expand_path(File.join(FILES_ROOT, directory))
    return nil unless normalized_path.start_with?(File.expand_path(FILES_ROOT))

    normalized_path
  end

  def search_files(regex, directory, recursive)
    files = []
    search_path = File.join(FILES_ROOT, directory)

    if recursive
      Find.find(search_path) do |path|
        break if files.size >= MAX_FILES
        next unless File.file?(path)
        files << path.sub("#{FILES_ROOT}/", '') if file_matches_regex?(path, regex)
      end
    else
      Dir.foreach(search_path) do |filename|
        break if files.size >= MAX_FILES
        next if filename == '.' || filename == '..'
        path = File.join(search_path, filename)
        files << path.sub("#{FILES_ROOT}/", '') if File.file?(path) && file_matches_regex?(path, regex)
      end
    end

    files
  end

  def file_matches_regex?(file_path, regex)
    File.open(file_path, 'r') do |file|
      file.each_line do |line|
        return true if line.match?(regex)
      end
    end
    false
  end
end
