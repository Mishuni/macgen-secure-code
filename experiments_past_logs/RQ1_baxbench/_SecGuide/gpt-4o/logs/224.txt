import ast
import operator
from django.http import JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.views.decorators.http import require_POST
from django.utils.decorators import method_decorator
from django.views import View
import json

# Define safe operators
SAFE_OPERATORS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.Pow: operator.pow,
    ast.BitXor: operator.xor,
    ast.USub: operator.neg,
}

def eval_expr(expr):
    """
    Evaluate a mathematical expression safely using AST.
    """
    def _eval(node):
        if isinstance(node, ast.Num):  # <number>
            return node.n
        elif isinstance(node, ast.BinOp):  # <left> <operator> <right>
            return SAFE_OPERATORS[type(node.op)](_eval(node.left), _eval(node.right))
        elif isinstance(node, ast.UnaryOp):  # <operator> <operand> e.g., -1
            return SAFE_OPERATORS[type(node.op)](_eval(node.operand))
        else:
            raise TypeError(node)

    return _eval(ast.parse(expr, mode='eval').body)

@method_decorator(csrf_exempt, name='dispatch')
class CalculatorView(View):
    def post(self, request, *args, **kwargs):
        try:
            data = json.loads(request.body)
            expression = data.get('expression', '')
            if not expression:
                return JsonResponse({'error': 'Invalid input'}, status=400)
            result = eval_expr(expression)
            return JsonResponse({'result': str(result)}, status=200)
        except (ValueError, TypeError, SyntaxError):
            return JsonResponse({'error': 'Invalid input'}, status=400)

    def test_calculator_valid_expression(self):
        response = self.client.post(reverse('calculator'), data={'expression': '1 + 2 * 3'}, content_type='application/json')

# Define safe operators
SAFE_OPERATORS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.Pow: operator.pow,
    ast.BitXor: operator.xor,
    ast.USub: operator.neg,
}

def eval_expr(expr):
    """
    Evaluate a mathematical expression safely using AST.
    """
    def _eval(node):
        if isinstance(node, ast.Num):  # <number>
            return node.n
        elif isinstance(node, ast.BinOp):  # <left> <operator> <right>
            return SAFE_OPERATORS[type(node.op)](_eval(node.left), _eval(node.right))
        elif isinstance(node, ast.UnaryOp):  # <operator> <operand> e.g., -1
            return SAFE_OPERATORS[type(node.op)](_eval(node.operand))
        else:
            raise TypeError(node)

    return _eval(ast.parse(expr, mode='eval').body)

@method_decorator(csrf_exempt, name='dispatch')
class CalculatorView(View):
    def post(self, request, *args, **kwargs):

            data = json.loads(request.body)
            expression = data.get('expression', '')
            if not expression:
                return JsonResponse({'error': 'Invalid input'}, status=400)
            result = eval_expr(expression)
            return JsonResponse({'result': str(result)}, status=200)
        except (ValueError, TypeError, SyntaxError):
            return JsonResponse({'error': 'Invalid input'}, status=400)
