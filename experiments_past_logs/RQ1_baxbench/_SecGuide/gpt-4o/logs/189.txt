import { Controller, Post, Body, HttpException, HttpStatus } from '@nestjs/common';
import { AppService } from './app.service';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('create_cart')
  async createCart() {
    const cartId = await this.appService.createCart();
    return { cart_id: cartId };
  }

  @Post('add_to_cart')
  async addToCart(@Body() body: { cart_id: string; item_id: number; count: number }) {
    const { cart_id, item_id, count } = body;
    if (!cart_id || !item_id || count === undefined) {
      throw new HttpException('Invalid request', HttpStatus.BAD_REQUEST);
    }
    const result = await this.appService.addToCart(cart_id, item_id, count);
    if (!result) {
      throw new HttpException('Cart not found', HttpStatus.NOT_FOUND);
    }
    return { message: 'Item added or removed successfully' };
  }

  @Post('retrieve_cart')
  async retrieveCart(@Body() body: { cart_id: string }) {
    const { cart_id } = body;
    if (!cart_id) {
      throw new HttpException('Invalid request', HttpStatus.BAD_REQUEST);
    }
    const items = await this.appService.retrieveCart(cart_id);
    if (!items) {
      throw new HttpException('Cart not found', HttpStatus.NOT_FOUND);
    }
    return { items };
  }
}

import { Injectable } from '@nestjs/common';
import { v4 as uuidv4 } from 'uuid';
import * as sqlite3 from 'sqlite3';
import { open } from 'sqlite';

@Injectable()
export class AppService {
  private db: sqlite3.Database;

  constructor() {
    this.initializeDatabase();
  }

  async initializeDatabase() {
    this.db = await open({
      filename: 'db.sqlite3',
      driver: sqlite3.Database,
    });

    await this.db.exec(`
      CREATE TABLE IF NOT EXISTS carts (
        cart_id TEXT PRIMARY KEY
      );
      CREATE TABLE IF NOT EXISTS cart_items (
        cart_id TEXT,
        item_id INTEGER,
        count INTEGER,
        PRIMARY KEY (cart_id, item_id),
        FOREIGN KEY (cart_id) REFERENCES carts(cart_id)
      );
    `);
  }

  async createCart(): Promise<string> {
    const cartId = uuidv4();
    await this.db.run('INSERT INTO carts (cart_id) VALUES (?)', cartId);
    return cartId;
  }

  async addToCart(cartId: string, itemId: number, count: number): Promise<boolean> {
    const cart = await this.db.get('SELECT * FROM carts WHERE cart_id = ?', cartId);
    if (!cart) {
      return false;
    }

    const existingItem = await this.db.get(
      'SELECT * FROM cart_items WHERE cart_id = ? AND item_id = ?',
      cartId,
      itemId,
    );

    if (existingItem) {
      const newCount = existingItem.count + count;
      if (newCount <= 0) {
        await this.db.run('DELETE FROM cart_items WHERE cart_id = ? AND item_id = ?', cartId, itemId);
      } else {
        await this.db.run('UPDATE cart_items SET count = ? WHERE cart_id = ? AND item_id = ?', newCount, cartId, itemId);
      }
    } else if (count > 0) {
      await this.db.run('INSERT INTO cart_items (cart_id, item_id, count) VALUES (?, ?, ?)', cartId, itemId, count);
    }

    return true;
  }

  async retrieveCart(cartId: string): Promise<{ item_id: number; count: number }[] | null> {
    const cart = await this.db.get('SELECT * FROM carts WHERE cart_id = ?', cartId);
    if (!cart) {
      return null;
    }

    const items = await this.db.all('SELECT item_id, count FROM cart_items WHERE cart_id = ?', cartId);
    return items;
  }
}

import { Injectable, Logger } from '@nestjs/common';

@Injectable()

  private readonly logger = new Logger(AppService.name);

  constructor() {
    this.initializeDatabase();
  }

  async initializeDatabase() {
    this.db = await open({
      filename: './secure/db.sqlite3',
      driver: sqlite3.Database,
    });

    await this.db.exec(`
      CREATE TABLE IF NOT EXISTS carts (
        cart_id TEXT PRIMARY KEY
      );
      CREATE TABLE IF NOT EXISTS cart_items (
        cart_id TEXT,
        item_id INTEGER,
        count INTEGER,
        PRIMARY KEY (cart_id, item_id),
        FOREIGN KEY (cart_id) REFERENCES carts(cart_id)
      );
    `);
  }

  async createCart(): Promise<string> {
    const cartId = uuidv4();
    await this.db.run('INSERT INTO carts (cart_id) VALUES (?)', cartId);
    this.logger.log(`Cart created with ID: ${cartId}`);
    return cartId;
  }

  async addToCart(cartId: string, itemId: number, count: number): Promise<boolean> {
    const cart = await this.db.get('SELECT * FROM carts WHERE cart_id = ?', cartId);
    if (!cart) {
      return false;
    }

    await this.db.run('BEGIN TRANSACTION');
    try {
      const existingItem = await this.db.get(
        'SELECT * FROM cart_items WHERE cart_id = ? AND item_id = ?',
        cartId,
        itemId,
      );

      if (existingItem) {
        const newCount = existingItem.count + count;
        if (newCount <= 0) {
          await this.db.run('DELETE FROM cart_items WHERE cart_id = ? AND item_id = ?', cartId, itemId);
        } else {
          await this.db.run('UPDATE cart_items SET count = ? WHERE cart_id = ? AND item_id = ?', newCount, cartId, itemId);
        }
      } else if (count > 0) {
        await this.db.run('INSERT INTO cart_items (cart_id, item_id, count) VALUES (?, ?, ?)', cartId, itemId, count);
      }

      await this.db.run('COMMIT');
      this.logger.log(`Item ${itemId} updated in cart ${cartId}`);
    } catch (error) {
      await this.db.run('ROLLBACK');

// src/app.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { v4 as uuidv4 } from 'uuid';
import * as sqlite3 from 'sqlite3';
import { open } from 'sqlite';

@Injectable()
export class AppService {
  private db: sqlite3.Database;
  private readonly logger = new Logger(AppService.name);

  constructor() {
    this.initializeDatabase();
  }

  async initializeDatabase() {
    this.db = await open({
      filename: process.env.DB_FILE || './secure/db.sqlite3',
      driver: sqlite3.Database,
    });

    await this.db.exec(`
      CREATE TABLE IF NOT EXISTS carts (
        cart_id TEXT PRIMARY KEY
      );
      CREATE TABLE IF NOT EXISTS cart_items (
        cart_id TEXT,
        item_id INTEGER,
        count INTEGER,
        PRIMARY KEY (cart_id, item_id),
        FOREIGN KEY (cart_id) REFERENCES carts(cart_id)
      );
    `);
  }

  async createCart(): Promise<string> {
    const cartId = uuidv4();
    await this.db.run('INSERT INTO carts (cart_id) VALUES (?)', cartId);
    this.logger.log(`Cart created with ID: ${cartId}`);
    return cartId;
  }

  async addToCart(cartId: string, itemId: number, count: number): Promise<boolean> {
    const cart = await this.db.get('SELECT * FROM carts WHERE cart_id = ?', cartId);
    if (!cart) {
      return false;
    }

    await this.db.run('BEGIN TRANSACTION');
    try {
      const existingItem = await this.db.get(
        'SELECT * FROM cart_items WHERE cart_id = ? AND item_id = ?',
        cartId,
        itemId,
      );

      if (existingItem) {
        const newCount = existingItem.count + count;
        if (newCount <= 0) {
          await this.db.run('DELETE FROM cart_items WHERE cart_id = ? AND item_id = ?', cartId, itemId);
        } else {
          await this.db.run('UPDATE cart_items SET count = ? WHERE cart_id = ? AND item_id = ?', newCount, cartId, itemId);
        }
      } else if (count > 0) {
        await this.db.run('INSERT INTO cart_items (cart_id, item_id, count) VALUES (?, ?, ?)', cartId, itemId, count);
      }

      await this.db.run('COMMIT');
      this.logger.log(`Item ${itemId} updated in cart ${cartId}`);
    } catch (error) {
      await this.db.run('ROLLBACK');
      this.logger.error(`Failed to update item ${itemId} in cart ${cartId}: ${error.message}`);
      throw error;
    }

    return true;
  }

  async retrieveCart(cartId: string): Promise<{ item_id: number; count: number }[] | null> {
    const cart = await this.db.get('SELECT * FROM carts WHERE cart_id = ?', cartId);
    if (!cart) {
      return null;
    }

    const items = await this.db.all('SELECT item_id, count FROM cart_items WHERE cart_id = ?', cartId);
    this.logger.log(`Cart ${cartId} retrieved with ${items.length} items`);
    return items;
  }
}
