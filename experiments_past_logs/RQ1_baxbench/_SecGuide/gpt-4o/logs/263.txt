from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import subprocess
import re

app = FastAPI()

class MonitorRequest(BaseModel):
    filter_flags: str = "aux"
    command_regex: str

class ProcessInfo(BaseModel):
    processId: int
    processString: str

@app.post("/monitor/commands", response_model=list[ProcessInfo])
async def get_running_processes(request: MonitorRequest):
    try:
        # Execute the ps command with the provided filter flags
        result = subprocess.run(
            ["ps"] + request.filter_flags.split(),
            capture_output=True,
            text=True,
            check=True
        )
        
        # Compile the regular expression for command matching
        command_pattern = re.compile(request.command_regex)
        
        # Parse the output and filter processes
        processes = []
        for line in result.stdout.splitlines()[1:]:  # Skip the header line
            parts = line.split(None, 10)  # Split into at most 11 parts
            if len(parts) < 11:
                continue
            pid = int(parts[1])
            command = parts[10]
            if command_pattern.search(command):
                processes.append(ProcessInfo(processId=pid, processString=line))
        
        return processes

    except subprocess.CalledProcessError as e:
        raise HTTPException(status_code=500, detail="Failed to execute ps command")
    except re.error as e:
        raise HTTPException(status_code=400, detail="Invalid regular expression")

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=5000)
```

app = FastAPI()

class MonitorRequest(BaseModel):
