<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;
use Symfony\Component\Process\Exception\ProcessFailedException;
use Symfony\Component\Process\Process;

class GifController extends Controller
{
    public function createGif(Request $request)
    {
        $this->validate($request, [
            'images' => 'required|array',
            'images.*' => 'required|file|mimes:jpeg,png',
            'targetSize' => 'required|string|regex:/^\d+x\d+$/',
            'delay' => 'integer|min:10',
            'appendReverted' => 'boolean',
        ]);

        $images = $request->file('images');
        $targetSize = $request->input('targetSize');
        $delay = $request->input('delay', 10);
        $appendReverted = $request->input('appendReverted', false);

        $imagePaths = [];
        foreach ($images as $image) {
            $path = $image->store('temp');
            $imagePaths[] = storage_path('app/' . $path);
        }

        if ($appendReverted) {
            $imagePaths = array_merge($imagePaths, array_reverse($imagePaths));
        }

        $outputPath = storage_path('app/temp/output.gif');
        $command = array_merge(['convert', '-delay', $delay, '-resize', $targetSize], $imagePaths, [$outputPath]);

        $process = new Process($command);
        $process->run();

        foreach ($imagePaths as $path) {
            unlink($path);
        }

        if (!$process->isSuccessful()) {
            return response()->json(['error' => 'Failed to create GIF'], 500);
        }

        return response()->file($outputPath)->deleteFileAfterSend(true);
    }
}

    public function createGif(Request $request)
    {
        $this->validate($request, [
            'images' => 'required|array|max:10', // Limit to 10 images
            'images.*' => 'required|file|mimes:jpeg,png|max:2048', // Limit size to 2MB per image
            'targetSize' => 'required|string|regex:/^\d+x\d+$/',
            'delay' => 'integer|min:10',
            'appendReverted' => 'boolean',
        ]);

        $images = $request->file('images');
        $targetSize = $request->input('targetSize');
        $delay = $request->input('delay', 10);
        $appendReverted = $request->input('appendReverted', false);

        $imagePaths = [];
        foreach ($images as $image) {
            $path = $image->store('temp');
            $imagePaths[] = storage_path('app/' . $path);
        }

        if ($appendReverted) {
            $imagePaths = array_merge($imagePaths, array_reverse($imagePaths));
        }

        $outputPath = storage_path('app/temp/' . uniqid('output_', true) . '.gif');
        $command = array_merge(['convert', '-delay', $delay, '-resize', $targetSize], $imagePaths, [$outputPath]);

            if (file_exists($path)) {
                unlink($path);
            }

        if (!$process->isSuccessful()) {
            Log::error('Failed to create GIF', ['error' => $process->getErrorOutput()]);
