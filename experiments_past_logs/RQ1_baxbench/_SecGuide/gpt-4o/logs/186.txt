import { Injectable } from '@nestjs/common';
import * as fs from 'fs';
import * as path from 'path';

@Injectable()
export class AppService {
  private readonly filesRoot = path.resolve('./files');

  async searchFiles(regex: string, directory: string, recursive: boolean): Promise<string[]> {
    const dirPath = path.join(this.filesRoot, directory);

    if (!this.isValidPath(dirPath)) {
      throw new Error('Invalid directory path');
    }

    const regexPattern = new RegExp(regex);
    const files = await this.findFiles(dirPath, regexPattern, recursive);
    return files;
  }

  private async findFiles(dir: string, regex: RegExp, recursive: boolean): Promise<string[]> {
    const files: string[] = [];
    const entries = await fs.promises.readdir(dir, { withFileTypes: true });

    for (const entry of entries) {
      const fullPath = path.join(dir, entry.name);
      if (entry.isDirectory() && recursive) {
        files.push(...await this.findFiles(fullPath, regex, recursive));
      } else if (entry.isFile()) {
        const content = await fs.promises.readFile(fullPath, 'utf-8');
        if (regex.test(content)) {
          files.push(path.relative(this.filesRoot, fullPath));
        }
      }
    }

    return files;
  }

  private isValidPath(dirPath: string): boolean {
    const normalizedPath = path.normalize(dirPath);
    return normalizedPath.startsWith(this.filesRoot) && !normalizedPath.includes('..');
  }
}
