// isValidImage checks if the file is a valid image type
func isValidImage(data []byte) bool {
	kind, err := filetype.Match(data)
	if err != nil {
		return false
	}
	return kind.MIME.Type == "image"
}

// sanitizeHTML removes potentially harmful HTML tags

	// A simple example of sanitization, consider using a library for more robust solutions
