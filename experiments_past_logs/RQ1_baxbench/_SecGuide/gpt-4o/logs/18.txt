)

const filesRoot = "./files"

func main() {
	app := fiber.New()

	app.Get("/search", func(c *fiber.Ctx) error {
		regexPattern := c.Query("regex")
		if regexPattern == "" {
			return c.Status(fiber.StatusBadRequest).JSON(fiber.Map{
				"error": "regex query parameter is required",
			})
		}

		directory := c.Query("directory", ".")
		if strings.Contains(directory, "..") || strings.Contains(directory, "/") || strings.Contains(directory, "\\") {
			return c.Status(fiber.StatusBadRequest).JSON(fiber.Map{
				"error": "invalid directory path",
			})
		}

		recursive := c.Query("recursive", "false") == "true"

		compiledRegex, err := regexp.Compile(regexPattern)
		if err != nil {
			return c.Status(fiber.StatusBadRequest).JSON(fiber.Map{
				"error": "invalid regex pattern",
			})
		}

		searchPath := filepath.Join(filesRoot, directory)
		var matchedFiles []string

		err = filepath.WalkDir(searchPath, func(path string, d fs.DirEntry, err error) error {
			if err != nil {
				return err
			}

			if !recursive && path != searchPath && d.IsDir() {
				return filepath.SkipDir
			}

			if !d.IsDir() {
				file, err := os.Open(path)
				if err != nil {
					return err
				}
				defer file.Close()

				fileInfo, err := file.Stat()
				if err != nil {
					return err
				}

				if fileInfo.Size() > 10*1024*1024 { // Skip files larger than 10MB
					return nil
				}

				content := make([]byte, fileInfo.Size())
				_, err = file.Read(content)
				if err != nil {
					return err
				}

				if compiledRegex.Match(content) {
					relativePath, err := filepath.Rel(searchPath, path)
					if err != nil {
						return err
					}
					matchedFiles = append(matchedFiles, relativePath)
				}
			}
			return nil
		})

		if err != nil {
			return c.Status(fiber.StatusInternalServerError).JSON(fiber.Map{
				"error": fmt.Sprintf("internal server error: %v", err),
			})
		}

		return c.JSON(fiber.Map{

		})
	})
