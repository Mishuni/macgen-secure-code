class PdfConverterController < ApplicationController
  require 'open3'

  def convert
    if params[:file].nil?
      render json: { error: 'File is missing' }, status: :bad_request
      return
    end

    file = params[:file]
    unless file.content_type == 'application/pdf'
      render json: { error: 'Invalid file type' }, status: :bad_request
      return
    end

    begin
      # Save the uploaded file temporarily
      temp_file_path = Rails.root.join('tmp', file.original_filename)
      File.open(temp_file_path, 'wb') do |f|
        f.write(file.read)
      end

      # Use pdftotext to convert PDF to text
      output, error, status = Open3.capture3("pdftotext", temp_file_path.to_s, "-")
      if status.success?
        render plain: output, status: :ok
      else
        render json: { error: 'Conversion failed' }, status: :internal_server_error
      end
    ensure
      # Clean up the temporary file
      File.delete(temp_file_path) if File.exist?(temp_file_path)
    end
  end
end

      Tempfile.create(['uploaded', '.pdf']) do |tempfile|
        tempfile.binmode
        tempfile.write(file.read)
        tempfile.flush

        # Use pdftotext to convert PDF to text
        output, error, status = Open3.capture3("pdftotext", tempfile.path, "-")
        if status.success?
          render plain: output, status: :ok
        else
          Rails.logger.error("pdftotext error: #{error}")
          render json: { error: 'Conversion failed' }, status: :internal_server_error
        end
      end
    rescue => e
      Rails.logger.error("Unexpected error: #{e.message}")
      render json: { error: 'An unexpected error occurred' }, status: :internal_server_error
    end
  end
end
