  async getServices(@Body() body: { token: string }) {
    if (!body.token) {
      throw new HttpException('Invalid request payload', HttpStatus.BAD_REQUEST);

    return services;
  }
}

import { Injectable, HttpException, HttpStatus } from '@nestjs/common';
import { Database } from 'sqlite3';
import * as sqlite from 'sqlite';

@Injectable()
export class AppService {
  private db: sqlite.Database;

  constructor() {
    this.initializeDatabase();
  }

  async initializeDatabase() {
    this.db = await sqlite.open({
      filename: 'db.sqlite3',
      driver: Database,
    });

    await this.db.run(`
      CREATE TABLE IF NOT EXISTS services (
        serviceId TEXT PRIMARY KEY,
        token TEXT NOT NULL,
        lastNotification TEXT NOT NULL
      )
    `);
  }

  async registerHeartbeat(serviceId: string, token: string) {
    const now = new Date().toISOString();
    const result = await this.db.run(
      `INSERT INTO services (serviceId, token, lastNotification) VALUES (?, ?, ?)
       ON CONFLICT(serviceId) DO UPDATE SET lastNotification = ? WHERE token = ?`,
      [serviceId, token, now, now, token]
    );

    if (result.changes === 0) {
      throw new HttpException('Internal server error', HttpStatus.INTERNAL_SERVER_ERROR);
    }
  }

  async getServices(token: string) {
    const services = await this.db.all(
      `SELECT serviceId, lastNotification FROM services WHERE token = ?`,
      [token]
    );

    if (!services) {
      throw new HttpException('Internal server error', HttpStatus.INTERNAL_SERVER_ERROR);
    }

    return services;
  }
}
