import { Controller, Post, Body, Get, Query, Res, Req, HttpStatus } from '@nestjs/common';
import { AppService } from './app.service';
import { Response, Request } from 'express';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('/merchants/register')
  async registerMerchant(@Body() body: any, @Res() res: Response) {
    const { email, name, password } = body;
    if (!email || !name || !password) {
      return res.status(HttpStatus.BAD_REQUEST).json({ error: 'Invalid input' });
    }
    const result = await this.appService.registerMerchant(email, name, password);
    if (result) {
      return res.status(HttpStatus.CREATED).json({ message: 'Merchant registered successfully' });
    }
    return res.status(HttpStatus.BAD_REQUEST).json({ error: 'Registration failed' });
  }

  @Post('/merchants/login')
  async loginMerchant(@Body() body: any, @Res() res: Response) {
    const { email, password } = body;
    if (!email || !password) {
      return res.status(HttpStatus.UNAUTHORIZED).json({ error: 'Invalid email or password' });
    }
    const result = await this.appService.loginMerchant(email, password);
    if (result) {
      res.cookie('AUTH_COOKIE', result, { httpOnly: true });
      return res.status(HttpStatus.OK).json({ message: 'Login successful' });
    }
    return res.status(HttpStatus.UNAUTHORIZED).json({ error: 'Invalid email or password' });
  }

  @Post('/merchants/upload-wares')
  async uploadWares(@Body() body: any, @Req() req: Request, @Res() res: Response) {
    const { csv } = body;
    const authCookie = req.cookies['AUTH_COOKIE'];
    if (!authCookie) {
      return res.status(HttpStatus.UNAUTHORIZED).json({ error: 'Unauthorized' });
    }
    const result = await this.appService.uploadWares(authCookie, csv);
    if (result) {
      return res.status(HttpStatus.OK).json({ message: 'Wares uploaded successfully' });
    }
    return res.status(HttpStatus.BAD_REQUEST).json({ error: 'Invalid CSV format' });
  }

  @Get('/wares')
  async getWares(@Query('minPrice') minPrice: number, @Query('maxPrice') maxPrice: number, @Req() req: Request, @Res() res: Response) {
    const authCookie = req.cookies['AUTH_COOKIE'];
    if (!authCookie) {
      return res.status(HttpStatus.UNAUTHORIZED).json({ error: 'Unauthorized' });
    }
    const wares = await this.appService.getWares(minPrice, maxPrice);
    res.setHeader('Content-Type', 'text/html');
    return res.status(HttpStatus.OK).send(wares);
  }
}

import { Injectable } from '@nestjs/common';
import * as bcrypt from 'bcrypt';
import * as jwt from 'jsonwebtoken';
import { Database } from 'sqlite3';
import { v4 as uuidv4 } from 'uuid';
import * as dotenv from 'dotenv';

dotenv.config();

@Injectable()
export class AppService {
  private db: Database;

  constructor() {
    this.db = new Database('db.sqlite3', (err) => {
      if (err) {
        console.error('Error opening database', err.message);
      } else {
        this.db.run(
          `CREATE TABLE IF NOT EXISTS merchants (
            id TEXT PRIMARY KEY,
            email TEXT UNIQUE,
            name TEXT,
            password TEXT
          )`
        );
        this.db.run(
          `CREATE TABLE IF NOT EXISTS wares (
            id TEXT PRIMARY KEY,
            merchantId TEXT,
            name TEXT,
            description TEXT,
            price REAL,
            FOREIGN KEY (merchantId) REFERENCES merchants(id)
          )`
        );
      }
    });
  }

  async registerMerchant(email: string, name: string, password: string): Promise<boolean> {
    const hashedPassword = await bcrypt.hash(password, 10);
    const id = uuidv4();
    return new Promise((resolve, reject) => {
      this.db.run(
        `INSERT INTO merchants (id, email, name, password) VALUES (?, ?, ?, ?)`,
        [id, email, name, hashedPassword],
        function (err) {
          if (err) {
            console.error('Error registering merchant', err.message);
            resolve(false);
          } else {
            resolve(true);
          }
        }
      );
    });
  }

  async loginMerchant(email: string, password: string): Promise<string | null> {
    return new Promise((resolve, reject) => {
      this.db.get(
        `SELECT * FROM merchants WHERE email = ?`,
        [email],
        async (err, row) => {
          if (err || !row) {
            console.error('Error logging in merchant', err ? err.message : 'No merchant found');
            resolve(null);
          } else {
            const match = await bcrypt.compare(password, row.password);
            if (match) {
              const token = jwt.sign({ id: row.id }, process.env.APP_SECRET, { expiresIn: '1h' });
              resolve(token);
            } else {
              resolve(null);
            }
          }
        }
      );
    });
  }

  async uploadWares(authToken: string, csv: string): Promise<boolean> {
    try {
      const decoded: any = jwt.verify(authToken, process.env.APP_SECRET);
      const merchantId = decoded.id;
      const wares = csv.split('\n').map(line => line.split(','));
      for (const [name, description, price] of wares) {
        const id = uuidv4();
        this.db.run(
          `INSERT INTO wares (id, merchantId, name, description, price) VALUES (?, ?, ?, ?, ?)`,
          [id, merchantId, name, description, parseFloat(price)],
          function (err) {
            if (err) {
              console.error('Error uploading wares', err.message);
              throw new Error('Invalid CSV format');
            }
          }
        );
      }
      return true;
    } catch (error) {
      console.error('Error verifying token or uploading wares', error.message);
      return false;
    }
  }

  async getWares(minPrice: number, maxPrice: number): Promise<string> {
    return new Promise((resolve, reject) => {
      let query = `SELECT wares.name, wares.price, wares.description, merchants.email FROM wares JOIN merchants ON wares.merchantId = merchants.id WHERE 1=1`;
      const params: any[] = [];
      if (minPrice !== undefined) {
        query += ` AND wares.price >= ?`;
        params.push(minPrice);
      }
      if (maxPrice !== undefined) {
        query += ` AND wares.price <= ?`;
        params.push(maxPrice);
      }
      this.db.all(query, params, (err, rows) => {
        if (err) {
          console.error('Error retrieving wares', err.message);
          resolve('<p>Error retrieving wares</p>');
        } else {
          let html = '<table><tr><th>Name</th><th>Price</th><th>Description</th><th>Merchant Email</th></tr>';
          rows.forEach(row => {
            html += `<tr><td>${this.escapeHtml(row.name)}</td><td>${row.price}</td><td>${this.escapeHtml(row.description)}</td><td>${this.escapeHtml(row.email)}</td></tr>`;
          });
          html += '</table>';
          resolve(html);
        }
      });
    });
  }

  private escapeHtml(unsafe: string): string {
    return unsafe.replace(/[&<>"']/g, function (m) {
      switch (m) {
        case '&':
          return '&amp;';
        case '<':
          return '&lt;';
        case '>':
          return '&gt;';
        case '"':
          return '&quot;';
        case "'":
          return '&#039;';
        default:
          return m;
      }
    });
  }
}

import * as bcrypt from 'bcrypt';
import * as jwt from 'jsonwebtoken';
import { Database } from 'sqlite3';
import { v4 as uuidv4 } from 'uuid';
import * as dotenv from 'dotenv';

dotenv.config();

@Injectable()
export class AppService {
  private db: Database;

  constructor() {
    this.db = new Database('db.sqlite3', (err) => {
      if (err) {
        console.error('Error opening database', err.message);
      } else {
        this.db.run(
          `CREATE TABLE IF NOT EXISTS merchants (
            id TEXT PRIMARY KEY,
            email TEXT UNIQUE,
            name TEXT,
            password TEXT
          )`
        );
        this.db.run(
          `CREATE TABLE IF NOT EXISTS wares (
            id TEXT PRIMARY KEY,
            merchantId TEXT,
            name TEXT,
            description TEXT,
            price REAL,
            FOREIGN KEY (merchantId) REFERENCES merchants(id)
          )`
        );
      }
    });
  }

  async registerMerchant(email: string, name: string, password: string): Promise<boolean> {
    const hashedPassword = await bcrypt.hash(password, 10);
    const id = uuidv4();
    return new Promise((resolve, reject) => {
      this.db.run(
        `INSERT INTO merchants (id, email, name, password) VALUES (?, ?, ?, ?)`,
        [id, email, name, hashedPassword],
        function (err) {
          if (err) {
            console.error('Error registering merchant', err.message);
            resolve(false);
          } else {
            resolve(true);
          }
        }
      );
    });
  }

  async loginMerchant(email: string, password: string): Promise<string | null> {
    return new Promise((resolve, reject) => {
      this.db.get(
        `SELECT * FROM merchants WHERE email = ?`,
        [email],
        async (err, row) => {
          if (err || !row) {
            console.error('Error logging in merchant', err ? err.message : 'No merchant found');
            resolve(null);
          } else {
            const match = await bcrypt.compare(password, row.password);
            if (match) {
              const token = jwt.sign({ id: row.id }, process.env.APP_SECRET, { expiresIn: '1h' });
              resolve(token);
            } else {
              resolve(null);
            }
          }
        }
      );
    });
  }

  async uploadWares(authToken: string, csv: string): Promise<boolean> {
    try {
      const decoded: any = jwt.verify(authToken, process.env.APP_SECRET);
      const merchantId = decoded.id;
      const wares = csv.split('\n').map(line => line.split(','));
      for (const [name, description, price] of wares) {
        const id = uuidv4();
        this.db.run(
          `INSERT INTO wares (id, merchantId, name, description, price) VALUES (?, ?, ?, ?, ?)`,
          [id, merchantId, name, description, parseFloat(price)],
          function (err) {
            if (err) {
              console.error('Error uploading wares', err.message);
              throw new Error('Invalid CSV format');
            }
          }
        );
      }
      return true;
    } catch (error) {
      console.error('Error verifying token or uploading wares', error.message);
      return false;
    }
  }

  async getWares(minPrice: number, maxPrice: number): Promise<string> {
    return new Promise((resolve, reject) => {
      let query = `SELECT wares.name, wares.price, wares.description, merchants.email FROM wares JOIN merchants ON wares.merchantId = merchants.id WHERE 1=1`;
      const params: any[] = [];
      if (minPrice !== undefined) {
        query += ` AND wares.price >= ?`;
        params.push(minPrice);
      }
      if (maxPrice !== undefined) {
        query += ` AND wares.price <= ?`;
        params.push(maxPrice);
      }
      this.db.all(query, params, (err, rows) => {
        if (err) {
          console.error('Error retrieving wares', err.message);
          resolve('<p>Error retrieving wares</p>');
        } else {
          let html = '<table><tr><th>Name</th><th>Price</th><th>Description</th><th>Merchant Email</th></tr>';
          rows.forEach(row => {
            html += `<tr><td>${this.escapeHtml(row.name)}</td><td>${row.price}</td><td>${this.escapeHtml(row.description)}</td><td>${this.escapeHtml(row.email)}</td></tr>`;
          });
          html += '</table>';
          resolve(html);
        }
      });
    });
  }

  private escapeHtml(unsafe: string): string {
    return unsafe.replace(/[&<>"']/g, function (m) {
      switch (m) {
        case '&':
          return '&amp;';
        case '<':
          return '&lt;';
        case '>':
          return '&gt;';
        case '"':
          return '&quot;';
        case "'":
          return '&#039;';
        default:
          return m;
      }
    });
  }
}

          } else {
            resolve(true);
          }
        }
      );
    });
  }

  async loginMerchant(email: string, password: string): Promise<string | null> {
    return new Promise((resolve, reject) => {
      this.db.get(
        `SELECT * FROM merchants WHERE email = ?`,
        [email],
        async (err, row) => {
          if (err || !row) {
            console.error('Error logging in merchant', err ? err.message : 'No merchant found');
            resolve(null);
          } else {
            const match = await bcrypt.compare(password, row.password);
            if (match) {
              const token = jwt.sign({ id: row.id }, process.env.APP_SECRET, { expiresIn: '1h' });
              resolve(token);
            } else {
              resolve(null);
            }
          }
        }
      );
    });
  }

  async uploadWares(authToken: string, csv: string): Promise<boolean> {
    try {
      const decoded: any = jwt.verify(authToken, process.env.APP_SECRET);
      const merchantId = decoded.id;
      const wares = csv.split('\n').map(line => line.split(','));
      for (const [name, description, price] of wares) {
        const id = uuidv4();
        this.db.run(
          `INSERT INTO wares (id, merchantId, name, description, price) VALUES (?, ?, ?, ?, ?)`,
          [id, merchantId, name, description, parseFloat(price)],
          function (err) {
            if (err) {
              console.error('Error uploading wares', err.message);
              throw new Error('Invalid CSV format');
            }
          }
        );
      }
      return true;
    } catch (error) {
      console.error('Error verifying token or uploading wares', error.message);
      return false;
    }
  }

  async getWares(minPrice: number, maxPrice: number): Promise<any[]> {
    return new Promise((resolve, reject) => {
      let query = `SELECT wares.name, wares.price, wares.description, merchants.email FROM wares JOIN merchants ON wares.merchantId = merchants.id WHERE 1=1`;
      const params: any[] = [];
      if (minPrice !== undefined) {
        query += ` AND wares.price >= ?`;
        params.push(minPrice);
      }
      if (maxPrice !== undefined) {
        query += ` AND wares.price <= ?`;
        params.push(maxPrice);
      }
      this.db.all(query, params, (err, rows) => {
        if (err) {
          console.error('Error retrieving wares', err.message);
          resolve([]);
        } else {
          resolve(rows);
        }
      });
    });
  }
