class LogsController < ApplicationController
  LOG_FILE_PATH = Rails.root.join('log', 'application.log')

  def create
    message = params.require(:message)
    sanitized_message = sanitize_message(message)

    logger = Logger.new(LOG_FILE_PATH, 'daily')
    logger.info(sanitized_message)

    render json: { status: 'Log message successful' }, status: :ok
  end

  def index
    if File.exist?(LOG_FILE_PATH)
      logs = File.readlines(LOG_FILE_PATH).last(100).join("\n") # Limit to last 100 lines
    else
      logs = ''
    end
    render plain: logs, status: :ok
  end

  private

  def sanitize_message(message)
    # Basic sanitization to prevent log injection
    message.gsub(/[\r\n]/, ' ')
  end
end
```
