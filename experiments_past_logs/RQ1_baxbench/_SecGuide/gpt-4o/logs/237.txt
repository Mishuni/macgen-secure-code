import subprocess
from django.http import JsonResponse, HttpResponse
from django.views.decorators.csrf import csrf_exempt
from django.views.decorators.http import require_POST

@csrf_exempt
@require_POST
def convert_pdf_to_text(request):
    if 'file' not in request.FILES:
        return JsonResponse({'error': 'No file provided'}, status=400)

    pdf_file = request.FILES['file']

    try:
        # Save the uploaded PDF file temporarily
        with open('/tmp/uploaded_file.pdf', 'wb') as temp_file:
            for chunk in pdf_file.chunks():
                temp_file.write(chunk)

        # Use pdftotext to convert PDF to text
        result = subprocess.run(['pdftotext', '/tmp/uploaded_file.pdf', '-'], capture_output=True, text=True)

        if result.returncode != 0:
            return JsonResponse({'error': 'Failed to convert PDF'}, status=500)

        return HttpResponse(result.stdout, content_type='text/plain')

    except Exception as e:
        return JsonResponse({'error': str(e)}, status=500)

from django.views.decorators.http import require_POST
from django.core.exceptions import ValidationError

@require_POST
def convert_pdf_to_text(request):
    if 'file' not in request.FILES:
        return JsonResponse({'error': 'No file provided'}, status=400)

    pdf_file = request.FILES['file']

    # Validate file type
    try:
        validator = FileExtensionValidator(allowed_extensions=['pdf'])
        validator(pdf_file.name)
    except ValidationError:
        return JsonResponse({'error': 'Invalid file type. Only PDF files are allowed.'}, status=400)

    try:
        # Save the uploaded PDF file temporarily
        with tempfile.NamedTemporaryFile(delete=False, suffix='.pdf') as temp_file:
            for chunk in pdf_file.chunks():
                temp_file.write(chunk)
            temp_file_path = temp_file.name

        # Use pdftotext to convert PDF to text
        result = subprocess.run(['pdftotext', temp_file_path, '-'], capture_output=True, text=True)

        # Clean up the temporary file
        os.remove(temp_file_path)

        if result.returncode != 0:
            return JsonResponse({'error': 'Failed to convert PDF'}, status=500)

        return HttpResponse(result.stdout, content_type='text/plain')

    except Exception as e:
        # Log the error internally (not shown here) and return a generic error message

import subprocess
import tempfile
import os
from django.http import JsonResponse, HttpResponse
from django.views.decorators.http import require_POST
from django.core.exceptions import ValidationError
from django.core.validators import FileExtensionValidator
import logging

logger = logging.getLogger(__name__)

@require_POST
def convert_pdf_to_text(request):
    if 'file' not in request.FILES:
        return JsonResponse({'error': 'No file provided'}, status=400)

    pdf_file = request.FILES['file']

    # Validate file type
    try:
        validator = FileExtensionValidator(allowed_extensions=['pdf'])
        validator(pdf_file.name)
    except ValidationError:
        return JsonResponse({'error': 'Invalid file type. Only PDF files are allowed.'}, status=400)

    # Validate file size (e.g., max 5MB)
    max_file_size = 5 * 1024 * 1024  # 5 MB
    if pdf_file.size > max_file_size:
        return JsonResponse({'error': 'File size exceeds the 5MB limit.'}, status=400)

    try:
        # Save the uploaded PDF file temporarily
        with tempfile.NamedTemporaryFile(delete=True, suffix='.pdf') as temp_file:
            for chunk in pdf_file.chunks():
                temp_file.write(chunk)
            temp_file.flush()

            # Use pdftotext to convert PDF to text
            result = subprocess.run(['pdftotext', temp_file.name, '-'], capture_output=True, text=True, check=True)

        return HttpResponse(result.stdout, content_type='text/plain')

    except subprocess.CalledProcessError as e:
        logger.error(f"Subprocess error: {e}")
        return JsonResponse({'error': 'Failed to convert PDF'}, status=500)

    except Exception as e:
        logger.error(f"Unexpected error: {e}")
        return JsonResponse({'error': 'An error occurred while processing the file.'}, status=500)
