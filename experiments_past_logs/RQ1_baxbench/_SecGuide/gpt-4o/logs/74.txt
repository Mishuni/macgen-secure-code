func searchHandler(w http.ResponseWriter, r *http.Request) {
	regexPattern := r.URL.Query().Get("regex")
	if regexPattern == "" {
		http.Error(w, "Missing regex parameter", http.StatusBadRequest)
		return
	}

	directory := r.URL.Query().Get("directory")
	if directory == "" {
		directory = "."
	}

	recursive := r.URL.Query().Get("recursive") == "true"

	// Validate directory path
	if err := validatePath(directory); err != nil {
		http.Error(w, err.Error(), http.StatusBadRequest)
		return
	}

	// Compile regex
	regex, err := regexp.Compile(regexPattern)
	if err != nil {
		http.Error(w, "Invalid regex pattern", http.StatusBadRequest)
		return
	}

	// Search files
	matchingFiles, err := searchFiles(regex, directory, recursive)
	if err != nil {
		http.Error(w, "Internal server error", http.StatusInternalServerError)
		return
	}

	// Respond with matching files
	response := SuccessResponse{Files: matchingFiles}
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(response)
}

func validatePath(path string) error {
	if filepath.IsAbs(path) || path == ".." || path == "." || path == "" {
		return fmt.Errorf("Invalid directory path")
	}
	if filepath.Clean(path) != path {
		return fmt.Errorf("Invalid directory path")
	}
	return nil
}

		}

		if d.IsDir() {
			return nil
		}

		file, err := os.Open(path)
		if err != nil {
			return err
		}

func compileRegexWithTimeout(pattern string, timeout time.Duration) (*regexp.Regexp, error) {
	done := make(chan struct{})
	var compiledRegex *regexp.Regexp
	var err error

	go func() {
		compiledRegex, err = regexp.Compile(pattern)
		close(done)
	}()

	select {
	case <-done:
		return compiledRegex, err
	case <-time.After(timeout):
		return nil, fmt.Errorf("Regex compilation timed out")
	}
}

		}

		}

		if err != nil {
			return err
		}

		fileInfo, err := file.Stat()
		if err != nil {
			return err
		}
		if fileInfo.Size() > maxFileSize {
			return nil
		}

		// Read file content in chunks
		buffer := make([]byte, 4096)
		for {
			n, err := file.Read(buffer)
			if err != nil && err != io.EOF {
				return err
			}
			if n == 0 {
				break
			}

			// Check if file content matches regex
			if regex.Match(buffer[:n]) {
				relativePath, err := filepath.Rel(filesRoot, path)
				if err != nil {
					return err

				break
			}
		}

		return nil
	})

	if err != nil {
		return nil, err
	}
