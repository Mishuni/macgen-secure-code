import { Controller, Get, Post, Body, Query, Res, HttpStatus } from '@nestjs/common';
import { AppService } from './app.service';
import { Response } from 'express';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Get('messages')
  async getMessages(
    @Query('limit') limit: number = 10,
    @Query('offset') offset: number = 0,
    @Query('sort') sort: string = 'DESC',
    @Res() res: Response
  ) {
    const messages = await this.appService.getMessages(limit, offset, sort);
    res.status(HttpStatus.OK).send(messages);
  }

  @Post('messages')
  async postMessage(@Body() message: { content: string; username: string }, @Res() res: Response) {
    if (!message.content || !message.username) {
      return res.status(HttpStatus.BAD_REQUEST).send('Invalid input');
    }
    await this.appService.postMessage(message);
    res.status(HttpStatus.OK).send('Message posted successfully');
  }
}

import { Injectable } from '@nestjs/common';
import { Database } from 'sqlite3';
import * as sqlite from 'sqlite';

@Injectable()
export class AppService {
  private db: sqlite.Database;

  constructor() {
    this.initializeDatabase();
  }

  async initializeDatabase() {
    this.db = await sqlite.open({
      filename: 'db.sqlite3',
      driver: Database,
    });

    await this.db.run(`
      CREATE TABLE IF NOT EXISTS messages (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        content TEXT NOT NULL,
        username TEXT NOT NULL,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP
      )
    `);
  }

  async getMessages(limit: number, offset: number, sort: string): Promise<any> {
    const order = sort.toUpperCase() === 'ASC' ? 'ASC' : 'DESC';
    const messages = await this.db.all(
      `SELECT * FROM messages ORDER BY created_at ${order} LIMIT ? OFFSET ?`,
      [limit, offset]
    );
    return messages;
  }

  async postMessage(message: { content: string; username: string }) {
    await this.db.run(
      `INSERT INTO messages (content, username) VALUES (?, ?)`,
      [message.content, message.username]
    );
  }
}

import { Controller, Get, Post, Body, Query, Res, HttpStatus } from '@nestjs/common';
import { AppService } from './app.service';
import { Response } from 'express';
import { IsString, Length } from 'class-validator';
import { validateOrReject } from 'class-validator';

class PostMessageDto {
  @IsString()
  @Length(1, 500)
  content: string;

  @IsString()
  @Length(1, 100)
  username: string;
}

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Get('messages')
  async getMessages(
    @Query('limit') limit: number = 10,
    @Query('offset') offset: number = 0,
    @Query('sort') sort: string = 'DESC',
    @Res() res: Response
  ) {
    try {
      const messages = await this.appService.getMessages(limit, offset, sort);
      res.status(HttpStatus.OK).send(messages);
    } catch (error) {
      res.status(HttpStatus.INTERNAL_SERVER_ERROR).send('Error retrieving messages');
    }
  }

  @Post('messages')
  async postMessage(@Body() message: PostMessageDto, @Res() res: Response) {
    try {
      await validateOrReject(message);
      await this.appService.postMessage(message);
      res.status(HttpStatus.CREATED).send('Message posted successfully');
    } catch (error) {
      res.status(HttpStatus.BAD_REQUEST).send('Invalid input');
    }
  }
}

import * as sqlite from 'sqlite';
import * as path from 'path';

@Injectable()
export class AppService {
  private db: sqlite.Database;

  constructor() {
    this.initializeDatabase();
  }

  async initializeDatabase() {
    const dbPath = process.env.DB_PATH || path.resolve(__dirname, '../db.sqlite3');
    this.db = await sqlite.open({
      filename: dbPath,
      driver: Database,
    });

    await this.db.run(`
      CREATE TABLE IF NOT EXISTS messages (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        content TEXT NOT NULL,
        username TEXT NOT NULL,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP
      )
    `);
  }

  async getMessages(limit: number, offset: number, sort: string): Promise<any> {
    const order = sort.toUpperCase() === 'ASC' ? 'ASC' : 'DESC';
    const messages = await this.db.all(
      `SELECT * FROM messages ORDER BY created_at ${order} LIMIT ? OFFSET ?`,
      [limit, offset]
    );
    return messages;
  }

  async postMessage(message: { content: string; username: string }) {
    await this.db.run(
      `INSERT INTO messages (content, username) VALUES (?, ?)`,
      [message.content, message.username]
    );
  }
}
