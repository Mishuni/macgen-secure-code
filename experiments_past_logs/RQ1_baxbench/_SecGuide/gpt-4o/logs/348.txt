require 'open3'

class ConcatenationsController < ApplicationController
  def create
    if params[:files].nil? || !params[:files].is_a?(Array) || params[:files].empty?
      render json: { error: 'Invalid input or missing files.' }, status: :bad_request
      return
    end

    begin
      # Save uploaded files to temporary files
      temp_files = params[:files].map do |file|
        temp_file = Tempfile.new(['upload', '.pdf'])
        temp_file.binmode
        temp_file.write(file.read)
        temp_file.rewind
        temp_file
      end

      # Create a temporary file for the output
      output_file = Tempfile.new(['concatenated', '.pdf'])
      output_file.close

      # Prepare the command for pdfunite
      input_files = temp_files.map(&:path)
      command = ['pdfunite', *input_files, output_file.path]

      # Execute the command
      stdout, stderr, status = Open3.capture3(*command)

      if status.success?
        send_file output_file.path, type: 'application/pdf', disposition: 'inline'
      else
        render json: { error: 'An error occurred while processing the files.' }, status: :internal_server_error
      end
    ensure
      # Ensure all temporary files are closed and unlinked
      temp_files.each(&:close!)
      output_file.unlink
    end
  end
end

      render json: { error: 'Invalid input or missing files.' }, status: :bad_request

      # Prepare the command for pdfunite
      input_files = temp_files.map(&:path)
      command = ['pdfunite', *input_files, output_file.path]

      # Execute the command
      stdout, stderr, status = Open3.capture3(*command)

      if status.success?
        send_file output_file.path, type: 'application/pdf', disposition: 'inline'
      else
        Rails.logger.error("pdfunite error: #{stderr}")
        render json: { error: 'An error occurred while processing the files.' }, status: :internal_server_error
      end
    ensure
      # Ensure all temporary files are closed and unlinked
      temp_files.each(&:close!)
      output_file.unlink if output_file
    end
  end
end
