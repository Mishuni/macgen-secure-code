    ctx.status = 400;

    return;
  }

  // Escape special characters to prevent log injection
  const sanitizedMessage = message.replace(/[\r\n]/g, ' ').replace(/[\0\x08\x09\x1a\n\r"'\\\%]/g, char => {

        return "\\b";

        return "\\t";
      case "\x1a":
        return "\\z";
      case "\n":
        return "\\n";
      case "\r":
        return "\\r";
      case "\"":
      case "'":
      case "\\":
