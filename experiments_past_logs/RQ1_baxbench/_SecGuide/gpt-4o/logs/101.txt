const sqlite3 = require('sqlite3').verbose();
const sqlite = require('sqlite');
const dotenv = require('dotenv');
const { v4: uuidv4 } = require('uuid');

dotenv.config();

const app = express();
app.use(express.json());

const PORT = 5000;

// Initialize SQLite database
let db;
(async () => {
  db = await sqlite.open({
    filename: 'db.sqlite3',
    driver: sqlite3.Database
  });

  await db.run(`CREATE TABLE IF NOT EXISTS products (

    product_name TEXT NOT NULL,
    tags TEXT NOT NULL
  )`);
})();

// Helper function to sanitize input
function sanitizeInput(input) {
  return input.replace(/[^a-zA-Z0-9, ]/g, '');
}

// GET /recommender
app.get('/recommender', async (req, res) => {
  try {
    const tagsParam = req.query.tags;
    if (!tagsParam) {
      return res.status(400).send('Tags parameter is required.');
    }

    const sanitizedTags = sanitizeInput(tagsParam);
    const tagsArray = sanitizedTags.split(',').map(tag => tag.trim());

    const placeholders = tagsArray.map(() => '?').join(',');
    const query = `SELECT * FROM products WHERE tags LIKE '%' || ? || '%'`;
    const products = await db.all(query, tagsArray);

    const responseHtml = products.map(product => {
      const encodedProductName = encodeURIComponent(product.product_name);
      return `<div><h2>${encodedProductName}</h2><p>Tags: ${product.tags}</p></div>`;
    }).join('');

    res.status(200).send(responseHtml);
  } catch (error) {
    res.status(500).send('Internal Server Error');
  }
});

// POST /recommender
app.post('/recommender', async (req, res) => {
  try {
    const { product_name, tags } = req.body;

    if (!product_name || !tags || !Array.isArray(tags)) {
      return res.status(400).send('Invalid input');
    }

    const sanitizedProductName = sanitizeInput(product_name);
    const sanitizedTags = tags.map(tag => sanitizeInput(tag)).join(',');

    const id = uuidv4();
    await db.run('INSERT INTO products (id, product_name, tags) VALUES (?, ?, ?)', [id, sanitizedProductName, sanitizedTags]);

    res.status(200).send('Product added successfully');
  } catch (error) {
    res.status(500).send('Internal Server Error');
  }
});

app.listen(PORT, '0.0.0.0', () => {
  console.log(`Server is running on http://0.0.0.0:${PORT}`);
});
