import { Injectable } from '@nestjs/common';

@Injectable()
export class AppService {
  private unsubscribeKeywords = [
    'unsubscribe',
    'remove me',
    'deregister me',
    'stop sending me',
    'delete me',
  ];

  decideUnsubscribe(emailContent: string): { email: string; unsubscribe: boolean } {
    const emailMatch = emailContent.match(/From:.*<(.+?)>/);
    const email = emailMatch ? emailMatch[1] : '';

    const unsubscribe = this.unsubscribeKeywords.some(keyword =>
      emailContent.toLowerCase().includes(keyword),
    );

    return { email, unsubscribe };
  }
}

// src/app.controller.ts
import { Controller, Post, Body, Headers, UnauthorizedException } from '@nestjs/common';
import { AppService } from './app.service';
import * as sanitizeHtml from 'sanitize-html';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('decideUnsubscribe')
  decideUnsubscribe(
    @Body('emailContent') emailContent: string,
    @Headers('x-api-key') apiKey: string,
  ): { email: string; unsubscribe: boolean } {
    if (apiKey !== process.env.API_KEY) {
      throw new UnauthorizedException('Invalid API Key');
    }

    const sanitizedContent = sanitizeHtml(emailContent);
    return this.appService.decideUnsubscribe(sanitizedContent);
  }
}

import { Module } from '@nestjs/common';
import { AppController } from './app.controller';

@Module({

  controllers: [AppController],
  providers: [AppService],
})
export class AppModule {}

// src/app.service.ts
import { Injectable } from '@nestjs/common';
import * as emailParser from 'email-addresses';

@Injectable()
export class AppService {
  private unsubscribeKeywords = [
    'unsubscribe',
    'remove me',
    'deregister me',
    'stop sending me',
    'delete me',
  ];

  decideUnsubscribe(emailContent: string): { email: string; unsubscribe: boolean } {
    const parsedEmail = emailParser.parseOneAddress(emailContent);
    const email = parsedEmail ? parsedEmail.address : '';

    const unsubscribe = this.unsubscribeKeywords.some(keyword =>
      emailContent.toLowerCase().includes(keyword),
    );

    return { email, unsubscribe };
  }
}

async function bootstrap() {

}
bootstrap();
