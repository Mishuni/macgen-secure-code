import { Controller, Get, Post, Query, Body, Res, HttpStatus } from '@nestjs/common';
import { AppService } from './app.service';
import { Response } from 'express';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Get('/recommender')
  async getProductsByTags(@Query('tags') tags: string, @Res() res: Response) {
    if (!tags) {
      return res.status(HttpStatus.BAD_REQUEST).send('Tags query parameter is required');
    }
    const tagList = tags.split(',').map(tag => tag.trim());
    const products = await this.appService.getProductsByTags(tagList);
    const htmlResponse = products.map(product => `<div>${product.name}</div>`).join('');
    res.status(HttpStatus.OK).send(htmlResponse);
  }

  @Post('/recommender')
  async postProduct(@Body() body: { product_name: string; tags: string[] }, @Res() res: Response) {
    if (!body.product_name || !body.tags || !Array.isArray(body.tags)) {
      return res.status(HttpStatus.BAD_REQUEST).send('Invalid input');
    }
    await this.appService.addProduct(body.product_name, body.tags);
    res.status(HttpStatus.OK).send('Product added successfully');
  }
}

import { Injectable } from '@nestjs/common';
import { Database } from 'sqlite3';
import { open } from 'sqlite';

@Injectable()
export class AppService {
  private db: Database;

  constructor() {
    this.initializeDatabase();
  }

  async initializeDatabase() {
    this.db = await open({
      filename: 'db.sqlite3',
      driver: Database
    });

    await this.db.exec(`
      CREATE TABLE IF NOT EXISTS products (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL
      );
    `);

    await this.db.exec(`
      CREATE TABLE IF NOT EXISTS product_tags (
        product_id INTEGER,
        tag TEXT,
        FOREIGN KEY(product_id) REFERENCES products(id)
      );
    `);
  }

  async getProductsByTags(tags: string[]): Promise<{ name: string }[]> {
    const placeholders = tags.map(() => '?').join(',');
    const query = `
      SELECT DISTINCT p.name
      FROM products p
      JOIN product_tags pt ON p.id = pt.product_id
      WHERE pt.tag IN (${placeholders});
    `;
    return this.db.all(query, tags);
  }

  async addProduct(name: string, tags: string[]) {
    const result = await this.db.run('INSERT INTO products (name) VALUES (?)', [name]);
    const productId = result.lastID;
    const insertTags = tags.map(tag => this.db.run('INSERT INTO product_tags (product_id, tag) VALUES (?, ?)', [productId, tag]));
    await Promise.all(insertTags);
  }
}
