<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;
use Symfony\Component\Process\Exception\ProcessFailedException;
use Symfony\Component\Process\Process;

class PDFController extends Controller
{
    public function concatenate(Request $request)
    {
        $this->validate($request, [
            'files' => 'required|array',
            'files.*' => 'required|file|mimes:pdf'
        ]);

        $files = $request->file('files');
        $filePaths = [];

        try {
            foreach ($files as $file) {
                $path = $file->store('pdfs');
                $filePaths[] = storage_path('app/' . $path);
            }

            $outputPath = storage_path('app/concatenated.pdf');
            $process = new Process(array_merge(['pdfunite'], $filePaths, [$outputPath]));
            $process->run();

            if (!$process->isSuccessful()) {
                throw new ProcessFailedException($process);
            }

            return response()->download($outputPath, 'concatenated.pdf', [
                'Content-Type' => 'application/pdf'
            ])->deleteFileAfterSend(true);

        } catch (\Exception $e) {
            return response()->json(['error' => 'An error occurred while processing the files.'], 500);
        }
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;
use Symfony\Component\Process\Exception\ProcessFailedException;
use Symfony\Component\Process\Process;

class PDFController extends Controller
{
    public function concatenate(Request $request)
    {
        $this->validate($request, [
            'files' => 'required|array',
            'files.*' => 'required|file|mimes:pdf'
        ]);

        $files = $request->file('files');
        $filePaths = [];

        try {
            foreach ($files as $file) {
                // Store files with unique names to prevent overwriting
                $path = $file->store('pdfs', ['disk' => 'local']);
                $filePaths[] = storage_path('app/' . $path);
            }

            // Use a unique name for the output file
            $outputPath = storage_path('app/' . uniqid('concatenated_', true) . '.pdf');
            $process = new Process(array_merge(['pdfunite'], $filePaths, [$outputPath]));
            $process->run();

            if (!$process->isSuccessful()) {
                throw new ProcessFailedException($process);
            }

            return response()->download($outputPath, 'concatenated.pdf', [
                'Content-Type' => 'application/pdf'
            ])->deleteFileAfterSend(true);

        } catch (\Exception $e) {
            // Log the error for debugging purposes
            \Log::error('PDF concatenation error: ' . $e->getMessage());
            return response()->json(['error' => 'An error occurred while processing the files.'], 500);
        }
    }
}
