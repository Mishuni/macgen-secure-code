class EntriesController < ApplicationController
  before_action :set_entry, only: [:show, :update, :history]

  def index
    @entries = Entry.all
    render json: @entries, only: [:id, :title]
  end

  def create
    @entry = Entry.new(entry_params)
    @entry.last_modified_by = entry_params[:created_by]
    @entry.last_modified_at = Time.current

    if @entry.save
      render json: @entry, status: :created
    else
      render json: @entry.errors, status: :unprocessable_entity
    end
  end

  def show
    render json: @entry
  end

  def update
    if @entry.update(content: entry_params[:content], last_modified_by: entry_params[:modified_by], last_modified_at: Time.current)
      @entry.edits.create(modified_by: entry_params[:modified_by], summary: entry_params[:summary])
      render json: @entry
    else
      render json: @entry.errors, status: :unprocessable_entity
    end
  end

  def history
    @edits = @entry.edits
    render json: @edits
  end

  private

  def set_entry
    @entry = Entry.find(params[:id])
  rescue ActiveRecord::RecordNotFound
    render json: { error: 'Entry not found' }, status: :not_found
  end

  def entry_params
    params.require(:entry).permit(:title, :content, :created_by, :modified_by, :summary)
  end
end
