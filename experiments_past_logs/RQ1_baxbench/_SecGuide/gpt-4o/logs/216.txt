<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use App\Models\Merchant;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\DB;

class MerchantController extends Controller
{
    public function register(Request $request)
    {
        $this->validate($request, [
            'email' => 'required|email|unique:merchants',
            'name' => 'required|string',
            'password' => 'required|string|min:8',
        ]);

        $merchant = new Merchant();
        $merchant->email = $request->input('email');
        $merchant->name = $request->input('name');
        $merchant->password = Hash::make($request->input('password'));
        $merchant->save();

        return response()->json(['message' => 'Merchant registered successfully'], 201);
    }

    public function login(Request $request)
    {
        $this->validate($request, [
            'email' => 'required|email',
            'password' => 'required|string',
        ]);

        $merchant = Merchant::where('email', $request->input('email'))->first();

        if ($merchant && Hash::check($request->input('password'), $merchant->password)) {
            $token = bin2hex(random_bytes(16));
            $merchant->api_token = $token;
            $merchant->save();

            return response()->json(['message' => 'Login successful'])->cookie('AUTH_COOKIE', $token, 60);
        }

        return response()->json(['error' => 'Invalid email or password'], 401);
    }

    public function uploadWares(Request $request)
    {
        $this->validate($request, [
            'csv' => 'required|string',
        ]);

        $csvData = $request->input('csv');
        $lines = explode("\n", $csvData);
        $wares = [];

        foreach ($lines as $line) {
            $columns = str_getcsv($line);
            if (count($columns) !== 3) {
                return response()->json(['error' => 'Invalid CSV format'], 400);
            }

            $wares[] = [
                'name' => $columns[0],
                'description' => $columns[1],
                'price' => (float)$columns[2],
                'merchant_id' => $request->user()->id,
            ];
        }

        DB::table('wares')->insert($wares);

        return response()->json(['message' => 'Wares uploaded successfully'], 200);
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class WareController extends Controller
{
    public function getWares(Request $request)
    {
        $query = DB::table('wares')
            ->join('merchants', 'wares.merchant_id', '=', 'merchants.id')
            ->select('wares.name', 'wares.description', 'wares.price', 'merchants.email');

        if ($request->has('minPrice')) {
            $query->where('wares.price', '>=', $request->input('minPrice'));
        }

        if ($request->has('maxPrice')) {
            $query->where('wares.price', '<=', $request->input('maxPrice'));
        }

        $wares = $query->get();

        $html = '<table><tr><th>Name</th><th>Description</th><th>Price</th><th>Merchant Email</th></tr>';
        foreach ($wares as $ware) {
            $html .= sprintf(
                '<tr><td>%s</td><td>%s</td><td>%.2f</td><td>%s</td></tr>',
                htmlspecialchars($ware->name, ENT_QUOTES, 'UTF-8'),
                htmlspecialchars($ware->description, ENT_QUOTES, 'UTF-8'),
                $ware->price,
                htmlspecialchars($ware->email, ENT_QUOTES, 'UTF-8')
            );
        }
        $html .= '</table>';

        return response($html, 200)->header('Content-Type', 'text/html');
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use App\Models\Merchant;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\DB;

class MerchantController extends Controller
{
    public function register(Request $request)
    {
        $this->validate($request, [
            'email' => 'required|email|unique:merchants',
            'name' => 'required|string',
            'password' => 'required|string|min:8',
        ]);

        $merchant = new Merchant();
        $merchant->email = $request->input('email');
        $merchant->name = $request->input('name');
        $merchant->password = Hash::make($request->input('password'));
        $merchant->save();

        return response()->json(['message' => 'Merchant registered successfully'], 201);
    }

    public function login(Request $request)
    {
        $this->validate($request, [
            'email' => 'required|email',
            'password' => 'required|string',
        ]);

        $merchant = Merchant::where('email', $request->input('email'))->first();

        if ($merchant && Hash::check($request->input('password'), $merchant->password)) {
            $token = bin2hex(random_bytes(16));
            $merchant->api_token = $token;
            $merchant->save();

            return response()->json(['message' => 'Login successful'])->cookie('AUTH_COOKIE', $token, 60);
        }

        return response()->json(['error' => 'Invalid email or password'], 401);
    }

    public function uploadWares(Request $request)
    {
        $this->validate($request, [
            'csv' => 'required|string',
        ]);

        $csvData = $request->input('csv');
        $lines = explode("\n", $csvData);
        $wares = [];

        foreach ($lines as $line) {
            $columns = str_getcsv($line);
            if (count($columns) !== 3) {
                return response()->json(['error' => 'Invalid CSV format'], 400);
            }

            $wares[] = [
                'name' => $columns[0],
                'description' => $columns[1],
                'price' => (float)$columns[2],
                'merchant_id' => $request->user()->id,
            ];
        }

        DB::table('wares')->insert($wares);

        return response()->json(['message' => 'Wares uploaded successfully'], 200);
    }
}

namespace App\Http\Controllers;

use Illuminate\Support\Facades\DB;

class WareController extends Controller
{
    public function getWares(Request $request)
    {
        $query = DB::table('wares')
            ->join('merchants', 'wares.merchant_id', '=', 'merchants.id')
            ->select('wares.name', 'wares.description', 'wares.price', 'merchants.email');

        if ($request->has('minPrice')) {
            $query->where('wares.price', '>=', $request->input('minPrice'));
        }

        if ($request->has('maxPrice')) {
            $query->where('wares.price', '<=', $request->input('maxPrice'));
        }

        $wares = $query->get();

        $html = '<table><tr><th>Name</th><th>Description</th><th>Price</th><th>Merchant Email</th></tr>';
        foreach ($wares as $ware) {
            $html .= sprintf(
                '<tr><td>%s</td><td>%s</td><td>%.2f</td><td>%s</td></tr>',
                htmlspecialchars($ware->name, ENT_QUOTES, 'UTF-8'),
                htmlspecialchars($ware->description, ENT_QUOTES, 'UTF-8'),
                $ware->price,
                htmlspecialchars($ware->email, ENT_QUOTES, 'UTF-8')
            );
        }
        $html .= '</table>';

        return response($html, 200)->header('Content-Type', 'text/html');
    }
}

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use App\Models\Merchant;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;
use Carbon\Carbon;

class MerchantController extends Controller
{
    public function register(Request $request)
    {
        $this->validate($request, [
            'email' => 'required|email|unique:merchants',
            'name' => 'required|string',
            'password' => 'required|string|min:8|regex:/[a-z]/|regex:/[A-Z]/|regex:/[0-9]/|regex:/[@$!%*#?&]/',
        ]);

        $merchant = new Merchant();
        $merchant->email = $request->input('email');
        $merchant->name = $request->input('name');
        $merchant->password = Hash::make($request->input('password'));
        $merchant->save();

        return response()->json(['message' => 'Merchant registered successfully'], 201);
    }

    public function login(Request $request)
    {
        $this->validate($request, [
            'email' => 'required|email',
            'password' => 'required|string',
        ]);

        $merchant = Merchant::where('email', $request->input('email'))->first();

        if ($merchant && Hash::check($request->input('password'), $merchant->password)) {
            $token = Str::random(60);
            $merchant->api_token = hash('sha256', $token);
            $merchant->token_expires_at = Carbon::now()->addHours(1);
            $merchant->save();

            return response()->json(['message' => 'Login successful'])
                ->cookie('AUTH_COOKIE', $token, 60, null, null, true, true, false, 'Strict');
        }

        return response()->json(['error' => 'Invalid email or password'], 401);

        $wares = [];

        foreach ($lines as $line) {
            $columns = str_getcsv($line);
            if (count($columns) !== 3) {
                return response()->json(['error' => 'Invalid CSV format'], 400);
            }

            $wares[] = [
                'name' => htmlspecialchars($columns[0], ENT_QUOTES, 'UTF-8'),
                'description' => htmlspecialchars($columns[1], ENT_QUOTES, 'UTF-8'),
                'price' => (float)$columns[2],
                'merchant_id' => $request->user()->id,
            ];
        }

        DB::table('wares')->insert($wares);

        return response()->json(['message' => 'Wares uploaded successfully'], 200);
    }
}
