require 'open3'

class ExtractController < ApplicationController
  def extract_frame
    video_file = params[:video]
    frame_number = params[:frameNumber].to_i

    if video_file.nil? || frame_number.nil?
      render json: { error: 'Invalid parameters' }, status: :bad_request
      return
    end

    video_path = Rails.root.join('tmp', video_file.original_filename)
    File.open(video_path, 'wb') do |file|
      file.write(video_file.read)
    end

    output_image_path = Rails.root.join('tmp', "frame_#{frame_number}.png")

    command = "ffmpeg -i #{Shellwords.escape(video_path.to_s)} -vf \"select=eq(n\\,#{frame_number})\" -vframes 1 #{Shellwords.escape(output_image_path.to_s)}"
    stdout, stderr, status = Open3.capture3(command)

    if status.success? && File.exist?(output_image_path)
      send_file output_image_path, type: 'image/png', disposition: 'inline'
    else
      render json: { error: "Frame at index #{frame_number} could not be found." }, status: :not_found
    end
  ensure
    File.delete(video_path) if File.exist?(video_path)
    File.delete(output_image_path) if File.exist?(output_image_path)
  end
end

class ExtractController < ApplicationController
  def extract_frame
    video_file = params[:video]

      file.write(video_file.read)
    end

    output_image_path = Rails.root.join('tmp', "frame_#{SecureRandom.uuid}.png")

    # Use a safer method to execute shell commands
    command = [
      'ffmpeg', '-i', video_path.to_s,
      '-vf', "select=eq(n\\,#{frame_number})",
      '-vframes', '1', output_image_path.to_s
    ]

    stdout, stderr, status = Open3.capture3(*command)

    if status.success? && File.exist?(output_image_path)
      send_file output_image_path, type: 'image/png', disposition: 'inline'
    else
      render json: { error: "Frame at index #{frame_number} could not be found." }, status: :not_found
    end
  ensure
    File.delete(video_path) if video_path && File.exist?(video_path)
    File.delete(output_image_path) if output_image_path && File.exist?(output_image_path)
  end

  private

  def valid_video_file?(file)
    allowed_types = ['video/mp4', 'video/avi', 'video/mov']
    max_size = 50.megabytes

    # Use MIME::Types to validate the file content type
    mime_type = MIME::Types.type_for(file.original_filename).first.content_type
    allowed_types.include?(mime_type) && file.size <= max_size
  end
end
