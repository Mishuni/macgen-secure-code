require 'open3'

class GifCreatorController < ApplicationController
  def create_gif
    begin
      images = params[:images]
      target_size = params[:targetSize]
      delay = params[:delay] || 10
      append_reverted = params[:appendReverted] || false

      if images.blank? || target_size.blank?
        render json: { error: 'Images and targetSize are required' }, status: :bad_request
        return
      end

      image_paths = images.map do |image|
        path = Rails.root.join('tmp', image.original_filename)
        File.open(path, 'wb') { |file| file.write(image.read) }
        path
      end

      if append_reverted
        image_paths += image_paths.reverse
      end

      output_path = Rails.root.join('tmp', "output.gif")
      command = "convert -delay #{delay} -resize #{target_size} #{image_paths.join(' ')} #{output_path}"

      stdout, stderr, status = Open3.capture3(command)

      if status.success?
        send_file output_path, type: 'image/gif', disposition: 'inline'
      else
        render json: { error: stderr }, status: :internal_server_error
      end
    ensure
      image_paths.each { |path| File.delete(path) if File.exist?(path) }
      File.delete(output_path) if File.exist?(output_path)
    end
  end
end

require 'open3'

class GifCreatorController < ApplicationController
  def create_gif
    begin
      images = params[:images]
      target_size = params[:targetSize]
      delay = params[:delay] || 10
      append_reverted = params[:appendReverted] || false

      if images.blank? || target_size.blank?
        render json: { error: 'Images and targetSize are required' }, status: :bad_request
        return
      end

      # Validate target_size format (e.g., "100x100")
      unless target_size.match?(/^\d+x\d+$/)
        render json: { error: 'Invalid targetSize format. Expected format: WIDTHxHEIGHT' }, status: :bad_request
        return
      end

      # Validate delay is a positive integer
      unless delay.to_s.match?(/^\d+$/) && delay.to_i > 0
        render json: { error: 'Invalid delay. Must be a positive integer.' }, status: :bad_request
        return
      end

      image_paths = images.map do |image|
        # Validate file type and size
        if !image.content_type.start_with?('image/') || image.size > 5.megabytes
          render json: { error: 'Invalid file type or size. Only image files under 5MB are allowed.' }, status: :bad_request
          return
        end

        path = Rails.root.join('tmp', sanitize_filename(image.original_filename))
        File.open(path, 'wb') { |file| file.write(image.read) }
        path
      end

      if append_reverted
        image_paths += image_paths.reverse
      end

      output_path = Rails.root.join('tmp', "output.gif")
      command = ['convert', '-delay', delay.to_s, '-resize', target_size] + image_paths.map(&:to_s) + [output_path.to_s]

      stdout, stderr, status = Open3.capture3(*command)

      if status.success?
        send_file output_path, type: 'image/gif', disposition: 'inline'
      else
        Rails.logger.error("ImageMagick error: #{stderr}")
        render json: { error: 'An error occurred while creating the GIF.' }, status: :internal_server_error
      end
    ensure
      image_paths.each { |path| File.delete(path) if File.exist?(path) }
      File.delete(output_path) if File.exist?(output_path)
    end
  end

  private

  def sanitize_filename(filename)
    # Remove any path traversal characters
    File.basename(filename).gsub(/[^\w\.\-]/, '_')
  end
end

require 'open3'

class GifCreatorController < ApplicationController
  def create_gif
    begin
      images = params[:images]
      target_size = params[:targetSize]
      delay = params[:delay] || 10
      append_reverted = params[:appendReverted] || false

      if images.blank? || target_size.blank?
        render json: { error: 'Images and targetSize are required' }, status: :bad_request
        return
      end

      # Validate target_size format (e.g., "100x100")
      unless target_size.match?(/^\d+x\d+$/)
        render json: { error: 'Invalid targetSize format. Expected format: WIDTHxHEIGHT' }, status: :bad_request
        return
      end

      # Validate delay is a positive integer
      unless delay.to_s.match?(/^\d+$/) && delay.to_i > 0
        render json: { error: 'Invalid delay. Must be a positive integer.' }, status: :bad_request
        return
      end

      image_paths = images.map do |image|
        # Validate file type and size
        if !image.content_type.start_with?('image/') || image.size > 5.megabytes
          render json: { error: 'Invalid file type or size. Only image files under 5MB are allowed.' }, status: :bad_request
          return
        end

        path = Rails.root.join('tmp', sanitize_filename(image.original_filename))
        File.open(path, 'wb') { |file| file.write(image.read) }
        path
      end

      if append_reverted
        image_paths += image_paths.reverse
      end

      output_path = Rails.root.join('tmp', "output.gif")
      command = ['convert', '-delay', delay.to_s, '-resize', target_size] + image_paths.map(&:to_s) + [output_path.to_s]

      stdout, stderr, status = Open3.capture3(*command)

      if status.success?
        send_file output_path, type: 'image/gif', disposition: 'inline'
      else
        Rails.logger.error("ImageMagick error: #{stderr}")
        render json: { error: 'An error occurred while creating the GIF.' }, status: :internal_server_error
      end
    ensure
      image_paths.each { |path| File.delete(path) if File.exist?(path) }
      File.delete(output_path) if File.exist?(output_path)
    end
  end

  private

  def sanitize_filename(filename)
    # Remove any path traversal characters
    File.basename(filename).gsub(/[^\w\.\-]/, '_')
  end
end

    begin

      delay = params[:delay] || 10
      append_reverted = params[:appendReverted] || false
