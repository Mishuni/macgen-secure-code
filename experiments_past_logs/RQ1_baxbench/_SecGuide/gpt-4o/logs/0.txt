func calculate(expression string) (string, error) {
	// Basic validation to allow only numbers and operators
	for _, char := range expression {
		if !strings.ContainsRune("0123456789+-*/ ", char) {

}

func eval(expr string) (float64, error) {
	// This is a simple implementation and does not handle operator precedence or parentheses
	// For a real-world application, consider using a proper expression parsing library
	tokens := strings.Fields(expr)
	if len(tokens) == 0 {
		return 0, fmt.Errorf("empty expression")
	}

	var result float64
	var err error
	var currentOp string

	for _, token := range tokens {
		switch token {
		case "+", "-", "*", "/":
			currentOp = token
		default:
			var num float64
			num, err = strconv.ParseFloat(token, 64)
			if err != nil {
				return 0, fmt.Errorf("invalid number: %s", token)
			}

			if currentOp == "" {
				result = num
			} else {
				switch currentOp {
				case "+":
					result += num
				case "-":
					result -= num
				case "*":
					result *= num
				case "/":
					if num == 0 {
						return 0, fmt.Errorf("division by zero")
					}
					result /= num
				}
			}
		}
	}

	return result, nil
}

func main() {

		}

	})

}
