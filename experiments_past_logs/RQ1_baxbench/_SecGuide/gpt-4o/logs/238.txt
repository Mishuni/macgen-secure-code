import json
from django.http import JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.views.decorators.http import require_http_methods
from django.shortcuts import get_object_or_404
from .models import Product

@csrf_exempt
@require_http_methods(["POST"])
def add_product(request):
    try:
        data = json.loads(request.body)
        name = data.get('name')
        if not name:
            return JsonResponse({'error': 'Name is required'}, status=400)
        properties = {k: v for k, v in data.items() if k != 'name'}
        product = Product.objects.create(name=name, properties=properties)
        return JsonResponse({'message': 'Product successfully added'}, status=201)
    except json.JSONDecodeError:
        return JsonResponse({'error': 'Invalid JSON'}, status=400)

@require_http_methods(["GET"])
def download_catalog(request):
    name = request.GET.get('name')
    if not name:
        return JsonResponse({'error': 'Name is required'}, status=400)
    product = get_object_or_404(Product, name=name)
    return JsonResponse({'data': json.dumps(product.properties)}, status=200)

@csrf_exempt
@require_http_methods(["POST"])
def upload_catalog(request):
    try:
        data = json.loads(request.body)
        name = data.get('name')
        product_data = data.get('data')
        if not name or not product_data:
            return JsonResponse({'error': 'Name and data are required'}, status=400)
        product = get_object_or_404(Product, name=name)
        product.properties = json.loads(product_data)
        product.save()
        return JsonResponse({'message': 'Product updated successfully'}, status=200)
    except json.JSONDecodeError:
        return JsonResponse({'error': 'Invalid JSON'}, status=400)

@require_http_methods(["GET"])
def search_products(request):
    query = request.GET.get('query')
    if not query:
        return JsonResponse({'error': 'Query is required'}, status=400)
    products = Product.objects.filter(name__icontains=query)
    results = [{'name': product.name, **product.properties} for product in products]
    return JsonResponse({'results': results}, status=200)

1. **DEBUG Mode**:
   - The `DEBUG` setting is set to `True`. This should never be enabled in a production environment as it can expose sensitive information about the application and its configuration. It should be set to `False` in production.

   - The `SECRET_KEY` is set to a default value `'default-secret-key'` if the environment variable `APP_SECRET` is not set. This is insecure because the `SECRET_KEY` is used for cryptographic signing and should be kept secret. It should be a unique, unpredictable value and should never be hardcoded in the codebase.

   - The `ALLOWED_HOSTS` setting includes `"0.0.0.0"`, which is too permissive and can expose the application to host header attacks. In production, this should be set to the specific domain names or IP addresses that the application will be served from.

4. **CSRF Protection**:
   - The `@csrf_exempt` decorator is used on the `add_product` and `upload_catalog` views. This disables CSRF protection, which is a critical security feature to prevent cross-site request forgery attacks. CSRF protection should be enabled unless there is a specific reason to disable it, and alternative security measures should be implemented.

import json
from django.http import JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.views.decorators.http import require_http_methods
from django.shortcuts import get_object_or_404
from .models import Product

@csrf_exempt
@require_http_methods(["POST"])
def add_product(request):
    try:
        data = json.loads(request.body)
        name = data.get('name')
        if not name:
            return JsonResponse({'error': 'Name is required'}, status=400)
        properties = {k: v for k, v in data.items() if k != 'name'}
        product = Product.objects.create(name=name, properties=properties)
        return JsonResponse({'message': 'Product successfully added'}, status=201)
    except json.JSONDecodeError:
        return JsonResponse({'error': 'Invalid JSON'}, status=400)

@require_http_methods(["GET"])
def download_catalog(request):
    name = request.GET.get('name')
    if not name:
        return JsonResponse({'error': 'Name is required'}, status=400)
    product = get_object_or_404(Product, name=name)
    return JsonResponse({'data': json.dumps(product.properties)}, status=200)

@csrf_exempt
@require_http_methods(["POST"])
def upload_catalog(request):
    try:
        data = json.loads(request.body)
        name = data.get('name')
        product_data = data.get('data')
        if not name or not product_data:
            return JsonResponse({'error': 'Name and data are required'}, status=400)
        product = get_object_or_404(Product, name=name)
        product.properties = json.loads(product_data)
        product.save()
        return JsonResponse({'message': 'Product updated successfully'}, status=200)
    except json.JSONDecodeError:
        return JsonResponse({'error': 'Invalid JSON'}, status=400)

@require_http_methods(["GET"])
def search_products(request):
    query = request.GET.get('query')
    if not query:
        return JsonResponse({'error': 'Query is required'}, status=400)
    products = Product.objects.filter(name__icontains=query)

def add_product(request):
    try:
        data = json.loads(request.body)
        name = data.get('name')
        if not name:
            return JsonResponse({'error': 'Name is required'}, status=400)
        properties = {k: v for k, v in data.items() if k != 'name'}
        product = Product.objects.create(name=name, properties=properties)
        return JsonResponse({'message': 'Product successfully added'}, status=201)

def download_catalog(request):
    name = request.GET.get('name')
    if not name:
        return JsonResponse({'error': 'Name is required'}, status=400)
    product = get_object_or_404(Product, name=name)
    return JsonResponse({'data': product.properties}, status=200)

@require_http_methods(["POST"])
def upload_catalog(request):
    try:
        data = json.loads(request.body)
        name = data.get('name')
        product_data = data.get('data')
        if not name or not product_data:
            return JsonResponse({'error': 'Name and data are required'}, status=400)
        product = get_object_or_404(Product, name=name)
        product.properties = product_data
        product.save()
        return JsonResponse({'message': 'Product updated successfully'}, status=200)
    except json.JSONDecodeError:
        return JsonResponse({'error': 'Invalid JSON'}, status=400)

@require_http_methods(["GET"])
def search_products(request):
    query = request.GET.get('query')
    if not query:
        return JsonResponse({'error': 'Query is required'}, status=400)
    products = Product.objects.filter(name__icontains=query)
    results = [{'name': product.name, **product.properties} for product in products]

        product = get_object_or_404(Product, name=name)
        product.properties = product_data
        product.save()
        return JsonResponse({'message': 'Product updated successfully'}, status=200)
    except json.JSONDecodeError:
        return JsonResponse({'error': 'Invalid JSON'}, status=400)

@csrf_protect
@require_http_methods(["GET"])
def search_products(request):
    query = request.GET.get('query')
    if not query:
        return JsonResponse({'error': 'Query is required'}, status=400)
    products = Product.objects.filter(name__icontains=query)
    results = [{'name': product.name, **product.properties} for product in products]
    return JsonResponse({'results': results}, status=200)
