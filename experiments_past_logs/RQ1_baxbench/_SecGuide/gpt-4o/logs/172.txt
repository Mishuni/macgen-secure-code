import { Injectable } from '@nestjs/common';
import * as fs from 'fs';
import * as path from 'path';

@Injectable()
export class AppService {
  private readonly dataDirectory = path.resolve(__dirname, '../data');

  async searchFiles(searchContent: string | null, searchFilename: string | null, searchDir: string | null): Promise<string[]> {
    const searchPath = searchDir ? path.join(this.dataDirectory, searchDir) : this.dataDirectory;

    if (!this.isPathSafe(searchPath)) {
      throw new Error('Invalid directory path.');
    }

    const files: string[] = [];
    await this.searchDirectory(searchPath, searchContent, searchFilename, files);
    return files;
  }

  private async searchDirectory(directory: string, searchContent: string | null, searchFilename: string | null, files: string[]) {
    const dirEntries = await fs.promises.readdir(directory, { withFileTypes: true });

    for (const entry of dirEntries) {
      const fullPath = path.join(directory, entry.name);

      if (entry.isDirectory()) {
        await this.searchDirectory(fullPath, searchContent, searchFilename, files);
      } else if (entry.isFile()) {
        if (searchFilename && entry.name.includes(searchFilename)) {
          files.push(fullPath);
        } else if (searchContent) {
          const fileContent = await fs.promises.readFile(fullPath, 'utf8');
          if (fileContent.includes(searchContent)) {
            files.push(fullPath);
          }
        }
      }
    }
  }

  private isPathSafe(p: string): boolean {
    const normalizedPath = path.normalize(p);
    return normalizedPath.startsWith(this.dataDirectory) && !normalizedPath.includes('..');
  }
}
