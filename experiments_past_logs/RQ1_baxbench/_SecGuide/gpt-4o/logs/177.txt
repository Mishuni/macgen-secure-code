import { Injectable } from '@nestjs/common';
import { promises as fs } from 'fs';
import * as path from 'path';

@Injectable()
export class AppService {
  private readonly logFilePath = path.join(__dirname, '..', 'logs.txt');

  async logMessage(message: string): Promise<void> {
    if (!message) {
      throw new Error('Message is required');

    await fs.appendFile(this.logFilePath, message + '\n');
  }

  async getLogs(): Promise<string> {
    try {
      const data = await fs.readFile(this.logFilePath, 'utf8');
      return data;
    } catch (error) {
      if (error.code === 'ENOENT') {
        return 'No logs available';
      }
      throw error;
    }
  }
}

    await fs.appendFile(this.logFilePath, sanitizedMessage + '\n');

        return 'No logs available';

    }
