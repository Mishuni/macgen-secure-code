const FILES_ROOT = path.resolve(__dirname, './files');

function isValidRegex(regex) {
  try {
    new RegExp(regex);
    return true;
  } catch (e) {
    return false;

function isValidDirectory(directory) {
  return /^[a-zA-Z0-9-_]+$/.test(directory);

function searchFiles(directory, regex, recursive) {
  const results = [];
  const dirPath = path.join(FILES_ROOT, directory);

  function searchInDirectory(currentPath) {
    const files = fs.readdirSync(currentPath);

    for (const file of files) {
      const fullPath = path.join(currentPath, file);
      const relativePath = path.relative(dirPath, fullPath);

      if (fs.statSync(fullPath).isDirectory()) {
        if (recursive) {
          searchInDirectory(fullPath);
        }

        if (regex.test(content)) {

      }

  return results;
