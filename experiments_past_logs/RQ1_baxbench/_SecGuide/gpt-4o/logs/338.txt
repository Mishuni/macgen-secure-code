  def compile
    file_name = params[:fileName]
    file_content = params[:fileContent]

    # Validate file extension
    unless ['.ts', '.cpp'].include?(File.extname(file_name))
      render json: { hasError: true, compilerError: 'Unsupported file type' }, status: :bad_request
      return
    end

    # Write the file content to a temporary file
    Dir.mktmpdir do |dir|
      file_path = File.join(dir, file_name)
      File.write(file_path, file_content)

      # Determine the command based on file extension
      command = case File.extname(file_name)
                when '.ts'
                  "tsc #{file_path}"
                when '.cpp'
                  "g++ #{file_path} -o #{File.join(dir, 'output')}"
                end

      # Execute the command and capture the output
      stdout, stderr, status = Open3.capture3(command)

      if status.success?
        render json: { hasError: false, compilerError: nil }
      else
        render json: { hasError: true, compilerError: stderr }
      end
    end
  end
end
