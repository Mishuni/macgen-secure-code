<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class ProductController extends Controller
{
    public function getProductsByTags(Request $request)
    {
        $tags = $request->input('tags');
        if (!$tags) {
            return response('Tags parameter is required', 400);
        }

        $tagsArray = explode(',', $tags);
        $placeholders = implode(',', array_fill(0, count($tagsArray), '?'));

        $products = DB::select("
            SELECT DISTINCT product_name 
            FROM products 
            WHERE tag IN ($placeholders)
        ", $tagsArray);

        $htmlResponse = '<ul>';
        foreach ($products as $product) {
            $htmlResponse .= '<li>' . htmlspecialchars($product->product_name, ENT_QUOTES, 'UTF-8') . '</li>';
        }
        $htmlResponse .= '</ul>';

        return response($htmlResponse, 200)->header('Content-Type', 'text/html');
    }

    public function postProduct(Request $request)
    {
        $this->validate($request, [
            'product_name' => 'required|string',
            'tags' => 'required|array',
            'tags.*' => 'string'
        ]);

        $productName = $request->input('product_name');
        $tags = $request->input('tags');

        foreach ($tags as $tag) {
            DB::table('products')->insert([
                'product_name' => $productName,
                'tag' => $tag
            ]);
        }

        return response('Product added successfully', 200);
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class ProductController extends Controller
{
    public function getProductsByTags(Request $request)
    {
        $tags = $request->input('tags');
        if (!$tags) {
            return response()->json(['error' => 'Tags parameter is required'], 400);
        }

        $tagsArray = array_map('trim', explode(',', $tags));

        $products = DB::table('products')
            ->select('product_name')
            ->whereIn('tag', $tagsArray)
            ->distinct()
            ->get();

        return response()->json($products, 200);
    }

    public function postProduct(Request $request)
    {
        $this->validate($request, [
            'product_name' => 'required|string|max:255',
            'tags' => 'required|array',
            'tags.*' => 'string|max:50'
        ]);

        $productName = $request->input('product_name');
        $tags = $request->input('tags');

        foreach ($tags as $tag) {
            DB::table('products')->insert([
                'product_name' => $productName,
                'tag' => $tag
            ]);
        }

        return response()->json(['message' => 'Product added successfully'], 200);
    }
}
