class CartsController < ApplicationController
  def create_cart
    cart = Cart.create
    render json: { cart_id: cart.id }, status: :created
  end

  def add_to_cart
    cart = Cart.find_by(id: params[:cart_id])
    if cart.nil?
      render json: { error: 'Cart not found' }, status: :not_found
      return
    end

    item = cart.cart_items.find_or_initialize_by(item_id: params[:item_id])
    item.count += params[:count].to_i

    if item.count < 0
      render json: { error: 'Invalid count' }, status: :bad_request
      return
    end

    if item.save
      render json: { message: 'Item added or removed successfully' }, status: :ok
    else
      render json: { error: 'Invalid request' }, status: :bad_request
    end
  end

  def retrieve_cart
    cart = Cart.find_by(id: params[:cart_id])
    if cart.nil?
      render json: { error: 'Cart not found' }, status: :not_found
      return
    end

    items = cart.cart_items.map do |item|
      { item_id: item.item_id, count: item.count }
    end

    render json: { items: items }, status: :ok
  end
end
</CODE>

# app/controllers/carts_controller.rb
class CartsController < ApplicationController
  before_action :find_cart, only: [:add_to_cart, :retrieve_cart]

  def create_cart
    cart = Cart.create
    render json: { cart_id: cart.id }, status: :created
  end

  def add_to_cart
    item_id = params[:item_id].to_i
    count = params[:count].to_i

    if count < 0
      render json: { error: 'Invalid count' }, status: :bad_request
      return
    end

    item = @cart.cart_items.find_or_initialize_by(item_id: item_id)
    item.with_lock do
      item.count += count
      if item.save
        render json: { message: 'Item added or removed successfully' }, status: :ok
      else
        render json: { error: 'Invalid request' }, status: :unprocessable_entity
      end
    end
  end

  def retrieve_cart
    items = @cart.cart_items.map do |item|
      { item_id: item.item_id, count: item.count }
    end

    render json: { items: items }, status: :ok
  end

  private

  def find_cart
    @cart = Cart.find_by(id: params[:cart_id])
    render json: { error: 'Cart not found' }, status: :not_found if @cart.nil?
  end
end
