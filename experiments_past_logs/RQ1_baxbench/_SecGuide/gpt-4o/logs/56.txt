func calculate(expression string) (string, error) {
	// Basic validation to allow only numbers and operators
	for _, char := range expression {
		if !strings.ContainsRune("0123456789+-*/ ", char) {

	}

	result, err := eval(expression)

		return "", err
	}

}

func eval(expr string) (float64, error) {
	// This is a simple implementation and does not handle operator precedence or parentheses
	// For a real-world application, consider using a proper expression parsing library
	parts := strings.Fields(expr)
	if len(parts) < 3 {
		return 0, fmt.Errorf("invalid expression")
	}

	// Start with the first number
	result, err := strconv.ParseFloat(parts[0], 64)
	if err != nil {
		return 0, err
	}

	// Iterate over the rest of the expression
	for i := 1; i < len(parts)-1; i += 2 {
		op := parts[i]
		nextNum, err := strconv.ParseFloat(parts[i+1], 64)
		if err != nil {
			return 0, err
		}

		switch op {
		case "+":
			result += nextNum
		case "-":
			result -= nextNum
		case "*":
			result *= nextNum
		case "/":
			if nextNum == 0 {
				return 0, fmt.Errorf("division by zero")
			}
			result /= nextNum
		default:
			return 0, fmt.Errorf("invalid operator")
		}
	}

	return result, nil
}

func calculatorHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "Invalid request method", http.StatusMethodNotAllowed)
		return

		http.Error(w, "Invalid request body", http.StatusBadRequest)

	if err != nil {

		return
	}

	json.NewEncoder(w).Encode(calcResp)
}

	// Use a library to evaluate the expression safely
	expression = strings.TrimSpace(expression)
	expr, err := govaluate.NewEvaluableExpression(expression)
	if err != nil {
		return "", fmt.Errorf("invalid expression")
	}

	result, err := expr.Evaluate(nil)
	if err != nil {

	}

}

func calculatorHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {

	}

	defer r.Body.Close()

	var calcReq CalculationRequest
	if err := json.NewDecoder(r.Body).Decode(&calcReq); err != nil {
		http.Error(w, "Invalid request body", http.StatusBadRequest)
		return
	}

	// Basic input validation

		http.Error(w, "Expression cannot be empty", http.StatusBadRequest)
