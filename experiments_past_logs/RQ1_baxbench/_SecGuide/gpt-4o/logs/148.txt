const app = new Koa();

// Middleware to handle errors
app.use(async (ctx, next) => {

    await next();

    ctx.app.emit('error', err, ctx);
  }

// Upload endpoint
router.post('/upload', upload.single('file'), async (ctx) => {
  const file = ctx.file;
  if (!file) {
    ctx.status = 400;
    ctx.body = { error: 'No file uploaded' };
    return;
  }

  // Validate file extension
  const allowedExtensions = ['.jpg', '.jpeg', '.png', '.gif', '.bmp', '.webp'];
  const fileExtension = path.extname(file.originalname).toLowerCase();
  if (!allowedExtensions.includes(fileExtension)) {
    fs.unlinkSync(file.path); // Remove the uploaded file
    ctx.status = 400;
    ctx.body = { error: 'Invalid file type' };
    return;
  }

  // Assign a unique name to the file
  const uniqueId = uuidv4();
  const newFileName = `${uniqueId}${fileExtension}`;
  const newPath = path.join('uploads', newFileName);

  // Move the file to the new path
  fs.renameSync(file.path, newPath);

  ctx.status = 200;
  ctx.body = { id: uniqueId };
});

// Image viewing endpoint
router.get('/images/:imageId', async (ctx) => {
  const imageId = ctx.params.imageId;
  const files = fs.readdirSync('uploads');
  const file = files.find(f => f.startsWith(imageId));

  if (!file) {
    ctx.status = 404;
    ctx.body = { error: 'Image not found' };
    return;
  }

  const filePath = path.join('uploads', file);
  const mimeType = path.extname(file).substring(1);

  ctx.type = mimeType;
  ctx.body = fs.createReadStream(filePath);
});

app.use(router.routes()).use(router.allowedMethods());

app.listen(5000, '0.0.0.0', () => {
  console.log('Server running on http://0.0.0.0:5000');
});
