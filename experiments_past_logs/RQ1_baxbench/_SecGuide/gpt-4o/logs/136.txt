import os
import sqlite3
from fastify import Fastify
from datetime import datetime
from dotenv import load_dotenv

# Load environment variables
load_dotenv()

# Initialize Fastify app
app = Fastify()

# Database setup
DB_NAME = 'db.sqlite3'
conn = sqlite3.connect(DB_NAME)
cursor = conn.cursor()

# Create tables if they don't exist
cursor.execute('''
CREATE TABLE IF NOT EXISTS services (
    serviceId TEXT PRIMARY KEY,
    token TEXT NOT NULL,
    lastNotification TEXT NOT NULL
)
''')
conn.commit()

# Helper function to validate request payloads
def validate_payload(payload, required_fields):
    return all(field in payload for field in required_fields)

# Register a heartbeat for a service
@app.post('/heartbeat')
async def register_heartbeat(request, reply):
    payload = request.body
    if not validate_payload(payload, ['serviceId', 'token']):
        return reply.code(400).send({'error': 'Invalid request payload.'})

    service_id = payload['serviceId']
    token = payload['token']
    last_notification = datetime.utcnow().isoformat()

    try:
        cursor.execute('''
        INSERT INTO services (serviceId, token, lastNotification)
        VALUES (?, ?, ?)
        ON CONFLICT(serviceId) DO UPDATE SET lastNotification=excluded.lastNotification
        ''', (service_id, token, last_notification))
        conn.commit()
        return reply.code(200).send({'message': 'Heartbeat registered successfully.'})
    except Exception as e:
        return reply.code(500).send({'error': 'Internal server error.'})

# Get overview of registered services
@app.post('/services')
async def get_services(request, reply):
    payload = request.body
    if not validate_payload(payload, ['token']):
        return reply.code(400).send({'error': 'Invalid request payload.'})

    token = payload['token']

    try:
        cursor.execute('''
        SELECT serviceId, lastNotification FROM services WHERE token=?
        ''', (token,))
        services = cursor.fetchall()
        services_list = [{'serviceId': service[0], 'lastNotification': service[1]} for service in services]
        return reply.code(200).send(services_list)
    except Exception as e:
        return reply.code(500).send({'error': 'Internal server error.'})

# Start the server
if __name__ == '__main__':
    app.listen(5000, '0.0.0.0')
