<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use ZipArchive;
use Illuminate\Support\Facades\Storage;

class ZipToTxtController extends Controller
{
    public function convert(Request $request)
    {
        $this->validate($request, [
            'file' => 'required|file|mimes:zip|max:10240', // max 10MB
        ]);

        $file = $request->file('file');
        $zip = new ZipArchive;

        if ($zip->open($file->getRealPath()) === TRUE) {
            $textContent = '';

            for ($i = 0; $i < $zip->numFiles; $i++) {
                $filename = $zip->getNameIndex($i);
                if (pathinfo($filename, PATHINFO_EXTENSION) === 'txt') {
                    $textContent .= $zip->getFromIndex($i) . "\n";
                }
            }

            $zip->close();

            return response($textContent, 200)
                ->header('Content-Type', 'text/plain');
        } else {
            return response('Invalid input', 400);
        }
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use ZipArchive;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Facades\Log;

class ZipToTxtController extends Controller
{
    public function convert(Request $request)
    {
        $this->validate($request, [
            'file' => 'required|file|mimes:zip|max:10240', // max 10MB
        ]);

        $file = $request->file('file');
        $zip = new ZipArchive;

        if ($zip->open($file->getRealPath()) === TRUE) {
            $textContent = '';

            for ($i = 0; $i < $zip->numFiles; $i++) {
                $filename = $zip->getNameIndex($i);

                // Prevent Zip Slip vulnerability
                if ($this->isValidFilename($filename) && pathinfo($filename, PATHINFO_EXTENSION) === 'txt') {
                    $textContent .= $zip->getFromIndex($i) . "\n";
                }
            }

            $zip->close();

            return response($textContent, 200)
                ->header('Content-Type', 'text/plain');
        } else {
            Log::error('Failed to open zip file.');
            return response('Invalid input', 400);
        }
    }

    private function isValidFilename($filename)
    {
        // Ensure the filename does not contain directory traversal characters
        return !preg_match('/\.\.[\/\\\\]/', $filename);
    }
}
