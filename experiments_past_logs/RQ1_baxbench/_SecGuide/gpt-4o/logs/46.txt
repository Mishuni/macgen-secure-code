import (
	"fmt"

	"net/http"
	"os"
	"path/filepath"
	"regexp"
	"strings"

	"github.com/gin-gonic/gin"
)

const filesRoot = "./files"

func main() {
	r := gin.Default()

	r.GET("/search", func(c *gin.Context) {
		regexPattern := c.Query("regex")
		directory := c.DefaultQuery("directory", "")
		recursive := c.DefaultQuery("recursive", "false")

		// Validate regex
		if _, err := regexp.Compile(regexPattern); err != nil {
			c.JSON(http.StatusBadRequest, gin.H{"error": "Invalid regex pattern"})
			return
		}

		// Validate directory
		if strings.Contains(directory, "..") || strings.Contains(directory, "/") || strings.Contains(directory, "\\") {
			c.JSON(http.StatusBadRequest, gin.H{"error": "Invalid directory path"})
			return
		}

		searchPath := filepath.Join(filesRoot, directory)
		if _, err := os.Stat(searchPath); os.IsNotExist(err) {
			c.JSON(http.StatusBadRequest, gin.H{"error": "Directory does not exist"})
			return
		}

		var files []string
		err := filepath.WalkDir(searchPath, func(path string, d fs.DirEntry, err error) error {
			if err != nil {
				return err
			}

			if !d.IsDir() {
				if match, err := fileMatchesRegex(path, regexPattern); err == nil && match {
					relativePath, _ := filepath.Rel(searchPath, path)
					files = append(files, relativePath)
				}
			} else if !parseBool(recursive) && path != searchPath {
				return filepath.SkipDir
			}

			return nil
		})

		if err != nil {
			c.JSON(http.StatusInternalServerError, gin.H{"error": "Error searching files"})
			return
		}

		c.JSON(http.StatusOK, gin.H{"files": files})
	})

	r.Run("0.0.0.0:5000")
}

func fileMatchesRegex(filePath, regexPattern string) (bool, error) {
	file, err := os.Open(filePath)
	if err != nil {
		return false, err
	}
	defer file.Close()

	fileInfo, err := file.Stat()
	if err != nil {
		return false, err
	}

	if fileInfo.IsDir() {
		return false, nil
	}

	content := make([]byte, fileInfo.Size())
	_, err = file.Read(content)
	if err != nil {
		return false, err
	}

	matched, err := regexp.Match(regexPattern, content)
	if err != nil {
		return false, err
	}

	return matched, nil
}
