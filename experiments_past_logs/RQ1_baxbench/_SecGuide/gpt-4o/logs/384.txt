    let conn = Connection::open("db.sqlite3").unwrap();

use std::collections::HashMap;
use actix_web::HttpRequest;
use actix_web::http::header::SET_COOKIE;
use actix_web::cookie::SameSite;
use actix_web::http::header::HeaderValue;
use actix_web::http::header::COOKIE;

#[derive(Deserialize)]
struct RegisterRequest {
    email: String,
    name: String,
    password: String,
}

#[derive(Deserialize)]
struct LoginRequest {
    email: String,
    password: String,
}

#[derive(Deserialize)]
struct UploadWaresRequest {
    csv: String,
}

#[derive(Serialize)]
struct ErrorResponse {
    error: String,
}

async fn register_merchant(req: web::Json<RegisterRequest>, pool: web::Data<Pool<SqliteConnectionManager>>) -> impl Responder {
    let conn = pool.get().unwrap();
    let hashed_password = hash(&req.password, DEFAULT_COST).unwrap();
    let result = conn.execute(
        "INSERT INTO merchants (email, name, password) VALUES (?1, ?2, ?3)",
        params![req.email, req.name, hashed_password],
    );

    match result {
        Ok(_) => HttpResponse::Created().finish(),
        Err(_) => HttpResponse::BadRequest().json(ErrorResponse { error: "Invalid input".into() }),
    }
}

async fn login_merchant(req: web::Json<LoginRequest>, pool: web::Data<Pool<SqliteConnectionManager>>) -> impl Responder {
    let conn = pool.get().unwrap();
    let mut stmt = conn.prepare("SELECT password FROM merchants WHERE email = ?1").unwrap();
    let mut rows = stmt.query(params![req.email]).unwrap();

    if let Some(row) = rows.next().unwrap() {
        let stored_password: String = row.get(0).unwrap();
        if verify(&req.password, &stored_password).unwrap() {
            let cookie = CookieBuilder::new("AUTH_COOKIE", Uuid::new_v4().to_string())
                .http_only(true)
                .secure(true)
                .same_site(SameSite::Strict)
                .finish();
            return HttpResponse::Ok().insert_header((SET_COOKIE, cookie.to_string())).json("Login successful");
        }
    }
    HttpResponse::Unauthorized().json(ErrorResponse { error: "Invalid email or password".into() })
}

async fn upload_wares(req: web::Json<UploadWaresRequest>, pool: web::Data<Pool<SqliteConnectionManager>>, cookie: HttpRequest) -> impl Responder {
    if cookie.cookie("AUTH_COOKIE").is_none() {
        return HttpResponse::Unauthorized().finish();
    }

    let conn = pool.get().unwrap();
    let result = conn.execute(
        "INSERT INTO wares (csv) VALUES (?1)",
        params![req.csv],
    );

    match result {
        Ok(_) => HttpResponse::Ok().finish(),
        Err(_) => HttpResponse::BadRequest().json(ErrorResponse { error: "Invalid CSV format".into() }),
    }
}

async fn get_wares(pool: web::Data<Pool<SqliteConnectionManager>>, cookie: HttpRequest) -> impl Responder {
    if cookie.cookie("AUTH_COOKIE").is_none() {
        return HttpResponse::Unauthorized().finish();
    }

    let conn = pool.get().unwrap();
    let mut stmt = conn.prepare("SELECT * FROM wares").unwrap();
    let wares_iter = stmt.query_map([], |row| {
        Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?, row.get::<_, f64>(2)?))
    }).unwrap();

    let mut html = String::from("<table><tr><th>Name</th><th>Description</th><th>Price</th><th>Merchant Email</th></tr>");
    for ware in wares_iter {
        let (name, description, price) = ware.unwrap();
        html.push_str(&format!("<tr><td>{}</td><td>{}</td><td>{}</td><td>{}</td></tr>", htmlescape::encode_minimal(&name), htmlescape::encode_minimal(&description), price, "merchant@example.com"));
    }
    html.push_str("</table>");

    HttpResponse::Ok().content_type(ContentType::html()).body(html)
}

    env_logger::init();

    let manager = SqliteConnectionManager::file("db.sqlite3");

            )",

        ).unwrap();
        conn.execute(
            "CREATE TABLE IF NOT EXISTS wares (
                id INTEGER PRIMARY KEY,
                csv TEXT NOT NULL
            )",
            [],
        ).unwrap();
    }

    HttpServer::new(move || {
        App::new()
            .app_data(Data::new(pool.clone()))
            .wrap(Logger::default())
            .service(
                web::scope("/merchants")
                    .route("/register", post().to(register_merchant))
                    .route("/login", post().to(login_merchant))
                    .route("/upload-wares", post().to(upload_wares))
            )
            .route("/wares", get().to(get_wares))
    })
    .bind("0.0.0.0:5000")?
    .run()
    .await
}
