                       (cart_id, item_id, count))

async def retrieve_cart(request):
    data = await request.json()
    cart_id = data.get('cart_id')

        return web.json_response({'error': 'Invalid request'}, status=400)

    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()

# API Handlers

    conn.commit()
    conn.close()

    conn = sqlite3.connect(DB_NAME)
