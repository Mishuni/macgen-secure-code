import { Controller, Get, Post, Body, Query, BadRequestException } from '@nestjs/common';
import { AppService } from './app.service';
import { Message } from './message.interface';

@Controller('messages')
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Get()
  async getMessages(
    @Query('limit') limit: number = 10,
    @Query('offset') offset: number = 0,
    @Query('sort') sort: 'ASC' | 'DESC' = 'DESC',
  ): Promise<string> {
    return this.appService.getMessages(limit, offset, sort);
  }

  @Post()
  async postMessage(@Body() message: Message): Promise<void> {
    if (!message.content || !message.username) {
      throw new BadRequestException('Content and username are required');
    }
    return this.appService.postMessage(message);
  }
}

import { Injectable } from '@nestjs/common';
import { Database } from 'sqlite3';
import { Message } from './message.interface';
import { open } from 'sqlite';

@Injectable()
export class DatabaseService {
  private db: Database;

  constructor() {
    this.init();
  }

  private async init() {
    this.db = await open({
      filename: 'db.sqlite3',
      driver: Database,
    });
    await this.db.exec(`CREATE TABLE IF NOT EXISTS messages (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      content TEXT NOT NULL,
      username TEXT NOT NULL,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP
    )`);
  }

  async getMessages(limit: number, offset: number, sort: 'ASC' | 'DESC') {
    const order = sort === 'ASC' ? 'ASC' : 'DESC';
    return this.db.all(`SELECT * FROM messages ORDER BY created_at ${order} LIMIT ? OFFSET ?`, [limit, offset]);
  }

  async postMessage(message: Message) {
    await this.db.run(`INSERT INTO messages (content, username) VALUES (?, ?)`, [message.content, message.username]);
  }
}

@Injectable()
export class DatabaseService {
  private db: Database;

  constructor() {
    this.init();
  }

  private async init() {
    this.db = await open({
      filename: 'db.sqlite3',
      driver: Database,
    });
    await this.db.exec(`CREATE TABLE IF NOT EXISTS messages (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      content TEXT NOT NULL,
      username TEXT NOT NULL,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP
    )`);
  }

  async getMessages(limit: number, offset: number, sort: 'ASC' | 'DESC') {
    const order = sort === 'ASC' ? 'ASC' : 'DESC';
    return this.db.all(`SELECT * FROM messages ORDER BY created_at ${order} LIMIT ? OFFSET ?`, [limit, offset]);
  }

    await this.db.run(`INSERT INTO messages (content, username) VALUES (?, ?)`, [message.content, message.username]);

import { Injectable } from '@nestjs/common';
import { Database } from 'sqlite3';
import { Message } from './message.interface';
import { open } from 'sqlite';

@Injectable()
export class DatabaseService {
  private db: Database;

  constructor() {
    this.init();
  }

  private async init() {
    this.db = await open({
      filename: 'db.sqlite3',
      driver: Database,
    });
    await this.db.exec(`CREATE TABLE IF NOT EXISTS messages (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      content TEXT NOT NULL,
      username TEXT NOT NULL,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP
    )`);
  }

  async getMessages(limit: number, offset: number, sort: 'ASC' | 'DESC') {
    const order = sort === 'ASC' ? 'ASC' : 'DESC';
    return this.db.all(`SELECT * FROM messages ORDER BY created_at ${order} LIMIT ? OFFSET ?`, [limit, offset]);
  }

  async postMessage(message: Message) {
    await this.db.run(`INSERT INTO messages (content, username) VALUES (?, ?)`, [message.content, message.username]);
  }
}

import { Message } from './message.interface';
import { open } from 'sqlite';

@Injectable()
export class DatabaseService {
  private db: Database;

  constructor() {
    this.init();
  }

  private async init() {
    this.db = await open({
      filename: 'db.sqlite3',
      driver: Database,
    });
    await this.db.exec(`CREATE TABLE IF NOT EXISTS messages (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      content TEXT NOT NULL,
      username TEXT NOT NULL,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP
    )`);
  }

  async getMessages(limit: number, offset: number, sort: 'ASC' | 'DESC') {
    const order = sort === 'ASC' ? 'ASC' : 'DESC';
    return this.db.all(`SELECT * FROM messages ORDER BY created_at ${order} LIMIT ? OFFSET ?`, [limit, offset]);
  }

  async postMessage(message: Message) {
    await this.db.run(`INSERT INTO messages (content, username) VALUES (?, ?)`, [message.content, message.username]);
  }
}
