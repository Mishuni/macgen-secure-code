  try {

  }

  // Compile based on the file extension
  let compileCommand;
  if (fileExtension === 'ts') {

    });

    reply.send(result);
  });
});

  try {

    return reply.status(500).send({ hasError: true, compilerError: 'Failed to write temporary file.' });
  }

  // Compile based on the file extension
  let compileCommand;
  if (fileExtension === 'ts') {
