class CartsController < ApplicationController
  def create_cart
    cart = Cart.create
    render json: { cart_id: cart.id }, status: :created
  end

  def add_to_cart
    cart = Cart.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    item = cart.cart_items.find_or_initialize_by(item_id: params[:item_id])
    item.count += params[:count]
    item.save if item.count > 0
    render json: { message: 'Item added or removed successfully' }, status: :ok
  end

  def retrieve_cart
    cart = Cart.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    items = cart.cart_items.select(:item_id, :count)
    render json: { items: items }, status: :ok
  end

  private

  def cart_params
    params.require(:cart).permit(:cart_id, :item_id, :count)
  end
end

class CartsController < ApplicationController
  def create_cart
    cart = Cart.create
    render json: { cart_id: cart.id }, status: :created
  end

  def add_to_cart
    cart = Cart.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    item = cart.cart_items.find_or_initialize_by(item_id: params[:item_id])
    item.count += params[:count]
    item.save if item.count > 0
    render json: { message: 'Item added or removed successfully' }, status: :ok
  end

  def retrieve_cart
    cart = Cart.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    items = cart.cart_items.select(:item_id, :count)
    render json: { items: items }, status: :ok
  end

    params.require(:cart).permit(:cart_id, :item_id, :count)

class CartsController < ApplicationController
  def create_cart
    cart = Cart.create
    render json: { cart_id: cart.id }, status: :created
  end

  def add_to_cart
    cart = Cart.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    item = cart.cart_items.find_or_initialize_by(item_id: params[:item_id])
    item.count += params[:count]
    item.save if item.count > 0
    render json: { message: 'Item added or removed successfully' }, status: :ok
  end

  def retrieve_cart
    cart = Cart.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    items = cart.cart_items.select(:item_id, :count)
    render json: { items: items }, status: :ok
  end

  private

  def cart_params
    params.require(:cart).permit(:cart_id, :item_id, :count)
  end
end

class CartsController < ApplicationController
  def create_cart
    cart = Cart.create
    render json: { cart_id: cart.id }, status: :created
  end

  def add_to_cart
    cart = Cart.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    item = cart.cart_items.find_or_initialize_by(item_id: params[:item_id])
    item.count += params[:count]
    item.save if item.count > 0
    render json: { message: 'Item added or removed successfully' }, status: :ok
  end

  def retrieve_cart
    cart = Cart.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    items = cart.cart_items.select(:item_id, :count)
    render json: { items: items }, status: :ok
  end

  private

  def cart_params
    params.require(:cart).permit(:cart_id, :item_id, :count)
  end
end
</CODE>

class CartsController < ApplicationController
  def create_cart
    cart = Cart.create
    render json: { cart_id: cart.id }, status: :created
  end

  def add_to_cart
    cart = Cart.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    if params[:count].to_i == 0
      return render json: { error: 'Count must be a non-zero integer' }, status: :bad_request
    end

    item = cart.cart_items.find_or_initialize_by(item_id: params[:item_id])
    item.count += params[:count].to_i

    if item.count < 0
      item.destroy
      item_count = 0
    else
      item.save
      item_count = item.count
    end

    render json: { message: 'Item added or removed successfully', item_id: item.item_id, count: item_count }, status: :ok
  end

  def retrieve_cart
    cart = Cart.find_by(id: params[:cart_id])
    return render json: { error: 'Cart not found' }, status: :not_found unless cart

    items = cart.cart_items.select(:item_id, :count)
    render json: { items: items }, status: :ok
  end

  private

  def cart_params
    params.require(:cart).permit(:cart_id, :item_id, :count)
  end
end
