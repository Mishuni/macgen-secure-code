from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import operator

app = FastAPI()

class ExpressionRequest(BaseModel):
    expression: str

@app.post("/calculator")
async def calculate(request: ExpressionRequest):
    try:
        # Evaluate the expression safely
        result = eval_expression(request.expression)
        return {"result": str(result)}
    except Exception as e:
        raise HTTPException(status_code=400, detail="Invalid input")

def eval_expression(expr: str):
    # Define allowed operators
    allowed_operators = {
        '+': operator.add,
        '-': operator.sub,
        '*': operator.mul,
        '/': operator.truediv,
    }
    
    # Tokenize the expression
    tokens = expr.split()
    if len(tokens) != 3:
        raise ValueError("Expression must be in the form 'a operator b'")
    
    a, op, b = tokens
    a, b = float(a), float(b)  # Convert to float for arithmetic operations
    
    if op not in allowed_operators:
        raise ValueError("Invalid operator")
    
    return allowed_operators[op](a, b)

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=5000)
