  def invite_user
    email = params[:email]
    existing_invite = Invite.find_by(email: email)

    if existing_invite
      render json: { invite_id: existing_invite.invite_id, message: "Invitation already exists." }, status: :ok
    else
      invite_id = SecureRandom.hex(10)
      Invite.create(email: email, invite_id: invite_id)
      render json: { invite_id: invite_id, message: "Invitation email sent." }, status: :ok
    end
  end
end

class UsersController < ApplicationController
  def create_user
    invite = Invite.find_by(invite_id: params[:invite_id])

    if invite.nil?
      render json: { message: "Invalid invite_id" }, status: :unauthorized
      return
    end

    user = User.new(user_params)
    if user.save
      render json: { message: "User created" }, status: :ok
    else
      render json: { message: user.errors.full_messages.join(", ") }, status: :bad_request
    end
  end

  private
