use actix_web::{post, web, App, HttpResponse, HttpServer, Responder};
use serde_json::json;
use std::collections::HashSet;

#[post("/decideUnsubscribe")]
async fn decide_unsubscribe(body: String) -> impl Responder {
    let unsubscribe_keywords: HashSet<&str> = [
        "unsubscribe",
        "remove me",
        "deregister me",
        "stop sending me",
        "delete me",
    ]
    .iter()
    .cloned()
    .collect();

    let email = extract_email(&body);
    let unsubscribe = unsubscribe_keywords.iter().any(|&keyword| body.to_lowercase().contains(keyword));

    HttpResponse::Ok().json(json!({
        "email": email,
        "unsubscribe": unsubscribe,
    }))
}

fn extract_email(body: &str) -> String {
    // A simple regex to extract email from the "From" field in the email body
    if let Some(from_line) = body.lines().find(|line| line.to_lowercase().starts_with("from:")) {
        if let Some(email_start) = from_line.find('<') {
            if let Some(email_end) = from_line.find('>') {
                return from_line[email_start + 1..email_end].to_string();
            }
        }
    }
    String::new() // Return an empty string if no email is found
}

#[actix_web::main]
async fn main() -> std::io::Result<()> {
    HttpServer::new(|| {
        App::new()
            .service(decide_unsubscribe)
    })
    .bind("0.0.0.0:5000")?
    .run()
    .await
}
