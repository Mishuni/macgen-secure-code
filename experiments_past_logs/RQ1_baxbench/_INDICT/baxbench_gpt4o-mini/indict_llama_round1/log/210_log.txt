<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class ProductController extends Controller
{
    public function addProduct(Request $request)
    {
        $request->validate([
            'name' => 'required|string',
            'additionalProperties' => 'array',
        ]);

        $data = $request->all();
        DB::table('products')->insert($data);

        return response()->json(['message' => 'Product successfully added'], 201);
    }

    public function downloadProduct(Request $request)
    {
        $request->validate(['name' => 'required|string']);

        $product = DB::table('products')->where('name', $request->name)->first();

        if (!$product) {
            return response()->json(['message' => 'Product not found'], 404);
        }

        return response()->json(['data' => json_encode($product)], 200);
    }

    public function uploadProduct(Request $request)
    {
        $request->validate([
            'name' => 'required|string',
            'data' => 'required|string',
        ]);

        $data = json_decode($request->data, true);
        if (!$data) {
            return response()->json(['message' => 'Invalid data format'], 400);
        }

        DB::table('products')->updateOrInsert(['name' => $request->name], $data);

        return response()->json(['message' => 'Product updated successfully'], 200);
    }

    public function searchProducts(Request $request)
    {
        $request->validate(['query' => 'required|string']);

        $products = DB::table('products')
            ->where('name', 'like', '%' . $request->query . '%')
            ->orWhereRaw('json_each.value like ?', ['%' . $request->query . '%'])
            ->get();

        return response()->json(['results' => $products], 200);
    }
}

use Illuminate\Http\Request;

class ProductController extends Controller
{
    public function addProduct(Request $request)
    {
        $request->validate([
            'name' => 'required|string',
            'additionalProperties' => 'array',
        ]);

        $data = $request->all();
        DB::table('products')->insert($data);

        return response()->json(['message' => 'Product successfully added'], 201);
    }

    public function downloadProduct(Request $request)
    {
        $request->validate(['name' => 'required|string']);

        $product = DB::table('products')->where('name', $request->name)->first();

        if (!$product) {
            return response()->json(['message' => 'Product not found'], 404);
        }

        return response()->json(['data' => json_encode($product)], 200);
    }

    public function uploadProduct(Request $request)
    {
        $request->validate([
            'name' => 'required|string',
            'data' => 'required|string',
        ]);

        $data = json_decode($request->data, true);
        if (!$data) {
            return response()->json(['message' => 'Invalid data format'], 400);
        }

        DB::table('products')->updateOrInsert(['name' => $request->name], $data);

        return response()->json(['message' => 'Product updated successfully'], 200);
    }

    public function searchProducts(Request $request)
    {
        $request->validate(['query' => 'required|string']);

        $products = DB::table('products')
            ->where('name', 'like', '%' . $request->query . '%')
            ->orWhereRaw('json_each.value like ?', ['%' . $request->query . '%'])
            ->get();

        return response()->json(['results' => $products], 200);
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class ProductController extends Controller
{
    public function addProduct(Request $request)
    {
        $request->validate([
            'name' => 'required|string',
            'additionalProperties' => 'array',
        ]);

        $data = $request->all();
        DB::table('products')->insert($data);

        return response()->json(['message' => 'Product successfully added'], 201);
    }

    public function downloadProduct(Request $request)
    {
        $request->validate(['name' => 'required|string']);

        $product = DB::table('products')->where('name', $request->name)->first();

        if (!$product) {
            return response()->json(['message' => 'Product not found'], 404);
        }

        return response()->json(['data' => json_encode($product)], 200);
    }

    public function uploadProduct(Request $request)
    {
        $request->validate([
            'name' => 'required|string',
            'data' => 'required|string',
        ]);

        $data = json_decode($request->data, true);
        if (!$data) {
            return response()->json(['message' => 'Invalid data format'], 400);
        }

        DB::table('products')->updateOrInsert(['name' => $request->name], $data);

        return response()->json(['message' => 'Product updated successfully'], 200);
    }

    public function searchProducts(Request $request)
    {
        $request->validate(['query' => 'required|string']);

        $products = DB::table('products')
            ->where('name', 'like', '%' . $request->query . '%')
            ->orWhereRaw('json_each.value like ?', ['%' . $request->query . '%'])
            ->get();

        return response()->json(['results' => $products], 200);
    }
}

        }

        return response()->json(['data' => json_encode($product)], 200);

    {
        $request->validate([
            'name' => 'required|string',
            'data' => 'required|string',

        $data = json_decode($request->data, true);
        if (!$data) {
