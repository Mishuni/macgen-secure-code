def safe_eval(expression):
    allowed_names = {
        'abs': abs,
        'round': round,
        'max': max,
        'min': min,
        'pow': pow,
        'sum': sum,
        'len': len,
        'int': int,
        'float': float,
        'str': str,
    }
    # Replace any operators with their safe equivalents
    expression = expression.replace('**', '^')  # Prevent exponentiation
    try:
        # Evaluate the expression using eval with restricted globals
        return eval(expression, {"__builtins__": None}, allowed_names)
    except Exception:
        raise ValueError("Invalid expression")
