from aiohttp import web
import sqlite3
import json
import os

# Initialize SQLite database

db_cursor = db_connection.cursor()

async def add_product(request):
    try:
        product = await request.json()
        name = product['name']
        data = json.dumps(product)  # Store the entire product as JSON

        db_cursor.execute('INSERT INTO products (name, data) VALUES (?, ?)', (name, data))
        db_connection.commit()
        return web.Response(status=201)
    except (KeyError, json.JSONDecodeError):
        return web.Response(status=400)

async def download_product(request):
    name = request.query.get('name')
    db_cursor.execute('SELECT data FROM products WHERE name = ?', (name,))
    row = db_cursor.fetchone()
    
    if row:
        return web.json_response({'data': row[0]})
    else:
        return web.Response(status=404)

async def upload_product(request):
    try:
        product = await request.json()
        name = product['name']
        data = product['data']

        db_cursor.execute('REPLACE INTO products (name, data) VALUES (?, ?)', (name, data))
        db_connection.commit()
        return web.Response(status=200)
    except (KeyError, json.JSONDecodeError):
        return web.Response(status=400)

async def search_products(request):
    query = request.query.get('query')
    db_cursor.execute('SELECT name, data FROM products')
    rows = db_cursor.fetchall()

    results = []
    for name, data in rows:
        product = json.loads(data)
        if query.lower() in name.lower() or any(query.lower() in str(value).lower() for value in product.values()):
            results.append(product)

    return web.json_response({'results': results})
