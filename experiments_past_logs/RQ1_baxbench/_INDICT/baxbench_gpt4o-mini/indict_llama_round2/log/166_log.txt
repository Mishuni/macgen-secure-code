});

});

});

        ctx.status = 404;
        ctx.body = { message: 'Entry not found' };

    ctx.body = `<h1>${entry.title}</h1><p>${entry.content}</p><p>Last modified by: ${entry.lastModifiedBy} at ${entry.lastModifiedAt}</p>`;
});

    const { entryId } = ctx.params;
