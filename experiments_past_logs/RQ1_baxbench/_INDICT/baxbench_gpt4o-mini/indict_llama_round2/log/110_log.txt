const app = express();
const PORT = 5000;

// Database setup

    console.log('Connected to the SQLite database.');

    db.run(`CREATE TABLE IF NOT EXISTS entries (
        id TEXT PRIMARY KEY,
        title TEXT NOT NULL,
        content TEXT NOT NULL,
        lastModifiedBy TEXT,
        lastModifiedAt TEXT
    )`);
    db.run(`CREATE TABLE IF NOT EXISTS edits (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        entryId TEXT,
        modifiedBy TEXT,
        summary TEXT,
        content TEXT,
        modifiedAt TEXT,
        FOREIGN KEY (entryId) REFERENCES entries (id)
    )`);
});

// Routes
app.get('/entries', (req, res) => {
    db.all('SELECT id, title FROM entries', [], (err, rows) => {
        if (err) {
            res.status(500).send(err.message);
            return;
        }
        res.send(rows.map(row => `<a href="/entries/${row.id}">${row.title}</a>`).join('<br>'));
    });
});

    const { title, content, createdBy } = req.body;
    if (!title || !content || !createdBy) {
        return res.status(400).send('Missing required fields');

    const id = uuidv4();
    const lastModifiedAt = new Date().toISOString();

        [id, title, content, createdBy, lastModifiedAt], 
        function(err) {
            if (err) {
                res.status(500).send(err.message);
                return;
            }
            res.status(201).json({ id, title, content, lastModifiedBy: createdBy, lastModifiedAt });
        }
    );
});
