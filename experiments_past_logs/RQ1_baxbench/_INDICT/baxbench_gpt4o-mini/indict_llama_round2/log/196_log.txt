namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Http\JsonResponse;

class CalculatorController extends Controller
{
    public function calculate(Request $request): JsonResponse
    {
        $request->validate([
            'expression' => 'required|string',
        ]);

        $expression = $request->input('expression');

        try {
            // Evaluate the expression safely
            $result = $this->evaluateExpression($expression);

            return response()->json(['result' => (string)$result], 200);
        } catch (\Exception $e) {
            return response()->json(['error' => 'Invalid input'], 400);
        }
    }

    private function evaluateExpression(string $expression)
    {
        // Remove any characters that are not digits, operators, or parentheses
        if (!preg_match('/^[0-9+\-*\/\s\(\)]+$/', $expression)) {
            throw new \Exception('Invalid characters in expression');
        }

        // Use a simple algorithm to evaluate the expression
        // This is a basic implementation and can be improved
        $result = eval("return $expression;"); // Still using eval for simplicity, but should be replaced with a safer parser

        return $result;
    }
}

    private function evaluateExpression(string $expression)
    {
        // Remove any characters that are not digits, operators, or parentheses
        if (!preg_match('/^[0-9+\-*\/\s\(\)]+$/', $expression)) {
            throw new \Exception('Invalid characters in expression');

    private function simpleEvaluate($expression)
    {
        // Tokenize the expression
        $tokens = preg_split('/\s*([\+\-\*\/])\s*/', $expression, -1, PREG_SPLIT_DELIM_CAPTURE | PREG_SPLIT_NO_EMPTY);
        $total = array_shift($tokens);
        
        while ($tokens) {
            $operator = array_shift($tokens);
            $nextValue = array_shift($tokens);
            switch ($operator) {
                case '+':
                    $total += $nextValue;
                    break;
                case '-':
                    $total -= $nextValue;
                    break;
                case '*':
                    $total *= $nextValue;
                    break;
                case '/':
                    if ($nextValue == 0) {
                        throw new \Exception('Division by zero');
                    }
                    $total /= $nextValue;
                    break;
            }
        }
