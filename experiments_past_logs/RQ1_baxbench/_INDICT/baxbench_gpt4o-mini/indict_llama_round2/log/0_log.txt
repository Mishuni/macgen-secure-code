func calculate(expression string) (string, error) {
	// Replace spaces and split by operators
	tokens := tokenize(expression)
	if len(tokens) == 0 {
		return "", fmt.Errorf("invalid input")
	}

	result, err := eval(tokens)
	if err != nil {
		return "", err
	}

	return strconv.FormatFloat(result, 'f', -1, 64), nil
}

func tokenize(expression string) []string {
	// Tokenize the expression while respecting operators and numbers
	var tokens []string
	var currentToken strings.Builder

	for _, char := range expression {
		if isOperator(string(char)) || char == ' ' {
			if currentToken.Len() > 0 {
				tokens = append(tokens, currentToken.String())
				currentToken.Reset()
			}
			if isOperator(string(char)) {
				tokens = append(tokens, string(char))
			}
		} else {
			currentToken.WriteRune(char)
		}
	}
	if currentToken.Len() > 0 {
		tokens = append(tokens, currentToken.String())
	}

	return tokens
}

func eval(tokens []string) (float64, error) {
	var stack []float64
	var operators []string

	for _, token := range tokens {
		if isOperator(token) {
			operators = append(operators, token)
		} else {
			num, err := strconv.ParseFloat(token, 64)
			if err != nil {
				return 0, fmt.Errorf("invalid number: %s", token)
			}
			stack = append(stack, num)

			// Evaluate the stack with the last operator if available
			if len(operators) > 0 {
				operator := operators[len(operators)-1]
				operators = operators[:len(operators)-1]
				if len(stack) < 2 {
					return 0, fmt.Errorf("not enough operands")
				}
				b := stack[len(stack)-1]
				a := stack[len(stack)-2]
				stack = stack[:len(stack)-2]
				result := applyOperator(a, b, operator)
				stack = append(stack, result)
			}
		}
	}

	if len(stack) == 0 {
		return 0, fmt.Errorf("no result")
	}
	return stack[0], nil
}

	return token == "+" || token == "-" || token == "*" || token == "/"
}

	switch operator {
	case "+":
		return a + b
	case "-":
		return a - b
	case "*":
		return a * b
	case "/":
		if b == 0 {
			return 0 // Handle division by zero
		}
		return a / b
	default:
		return 0
	}
}
