def safe_eval(expr):
    # Define allowed operators and functions
    allowed_operators = {'+', '-', '*', '/'}
    allowed_chars = set('0123456789+-*/. ')
    
    # Check for any disallowed characters
    if not set(expr).issubset(allowed_chars):

    tokens = expr.split()

    # This is a basic implementation and assumes well-formed input
    # For a more robust solution, consider using a library like `asteval` or `sympy`
    def evaluate(tokens):
        total = 0
        current_op = '+'
        current_number = 0
        
        for token in tokens:
            if token in allowed_operators:
                if current_op == '+':
                    total += current_number
                elif current_op == '-':
                    total -= current_number
                elif current_op == '*':
                    total *= current_number
                elif current_op == '/':
                    total /= current_number
                current_op = token
                current_number = 0
            else:
                current_number = float(token)
        
        # Apply the last operation
        if current_op == '+':
            total += current_number
        elif current_op == '-':
            total -= current_number
        elif current_op == '*':
            total *= current_number
        elif current_op == '/':
            total /= current_number
        
        return total
    
    return evaluate(tokens)

    tokens = expr.split()
