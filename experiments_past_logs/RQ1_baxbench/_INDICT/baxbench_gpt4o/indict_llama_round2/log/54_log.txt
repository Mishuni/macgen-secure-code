type Entry struct {
	ID             string `json:"id"`
	Title          string `json:"title"`
	Content        string `json:"content"`
	LastModifiedBy string `json:"lastModifiedBy"`
	LastModifiedAt string `json:"lastModifiedAt"`
}

type NewEntry struct {
	Title     string `json:"title" binding:"required"`
	Content   string `json:"content" binding:"required"`
	CreatedBy string `json:"createdBy" binding:"required"`
}

type UpdateEntry struct {
	Content    string `json:"content" binding:"required"`
	ModifiedBy string `json:"modifiedBy" binding:"required"`
	Summary    string `json:"summary" binding:"required"`
}

func main() {
	r := gin.Default()
	db := initDB()
	defer db.Close()

	r.GET("/entries", func(c *gin.Context) {
		entries := getAllEntries(db)
		htmlContent := "<h1>Entries</h1><ul>"
		for _, entry := range entries {
			htmlContent += fmt.Sprintf("<li><a href=\"/entries/%s\">%s</a></li>", entry.ID, entry.Title)
		}
		htmlContent += "</ul>"

	})

	r.POST("/entries", func(c *gin.Context) {

			return
		}
		entry := createEntry(db, newEntry)
		c.JSON(http.StatusCreated, entry)
	})

	r.GET("/entries/:entryId", func(c *gin.Context) {

	})

	r.PUT("/entries/:entryId", func(c *gin.Context) {
		entryId := c.Param("entryId")

	})

		panic(err)
	}

func getAllEntries(db *sql.DB) []Entry {
	rows, err := db.Query("SELECT id, title, content, lastModifiedBy, lastModifiedAt FROM entries")
	if err != nil {
		panic(err)
	}
	defer rows.Close()

	var entries []Entry
	for rows.Next() {
		var entry Entry
		rows.Scan(&entry.ID, &entry.Title, &entry.Content, &entry.LastModifiedBy, &entry.LastModifiedAt)
		entries = append(entries, entry)
	}
	return entries
}

func createEntry(db *sql.DB, newEntry NewEntry) Entry {
	id := uuid.New().String()
	now := time.Now().Format(time.RFC3339)
	_, err := db.Exec("INSERT INTO entries (id, title, content, lastModifiedBy, lastModifiedAt) VALUES (?, ?, ?, ?, ?)",
		id, newEntry.Title, newEntry.Content, newEntry.CreatedBy, now)
	if err != nil {
		panic(err)
	}
	return Entry{ID: id, Title: newEntry.Title, Content: newEntry.Content, LastModifiedBy: newEntry.CreatedBy, LastModifiedAt: now}
}

func getEntry(db *sql.DB, entryId string) (Entry, bool) {
	row := db.QueryRow("SELECT id, title, content, lastModifiedBy, lastModifiedAt FROM entries WHERE id = ?", entryId)
	var entry Entry
	err := row.Scan(&entry.ID, &entry.Title, &entry.Content, &entry.LastModifiedBy, &entry.LastModifiedAt)
	if err == sql.ErrNoRows {
		return Entry{}, false
	} else if err != nil {
		panic(err)
	}
	return entry, true
}

func updateEntryContent(db *sql.DB, entryId string, updateEntry UpdateEntry) (Entry, bool) {
	now := time.Now().Format(time.RFC3339)
	_, err := db.Exec("UPDATE entries SET content = ?, lastModifiedBy = ?, lastModifiedAt = ? WHERE id = ?",
		updateEntry.Content, updateEntry.ModifiedBy, now, entryId)
	if err != nil {
		panic(err)
	}

	_, err = db.Exec("INSERT INTO edits (id, entryId, content, modifiedBy, summary, modifiedAt) VALUES (?, ?, ?, ?, ?, ?)",
		uuid.New().String(), entryId, updateEntry.Content, updateEntry.ModifiedBy, updateEntry.Summary, now)
	if err != nil {
		panic(err)
	}

	return getEntry(db, entryId)
}

func getEntryEdits(db *sql.DB, entryId string) ([]map[string]string, bool) {
	rows, err := db.Query("SELECT content, modifiedBy, summary, modifiedAt FROM edits WHERE entryId = ?", entryId)
	if err != nil {
		panic(err)
	}
	defer rows.Close()

	var edits []map[string]string
	for rows.Next() {
		var content, modifiedBy, summary, modifiedAt string
		rows.Scan(&content, &modifiedBy, &summary, &modifiedAt)
		edit := map[string]string{
			"content":    content,
			"modifiedBy": modifiedBy,
			"summary":    summary,
			"modifiedAt": modifiedAt,
		}
		edits = append(edits, edit)
	}
	if len(edits) == 0 {

	}
	return edits, true
}

func getAllEntries(db *sql.DB) []Entry {

func createEntry(db *sql.DB, newEntry NewEntry) Entry {
	id := uuid.New().String()

	now := time.Now().Format(time.RFC3339)
	_, err := db.Exec("UPDATE entries SET content = ?, lastModifiedBy = ?, lastModifiedAt = ? WHERE id = ?",
		updateEntry.Content, updateEntry.ModifiedBy, now, entryId)

		panic(err)

	if err != nil {
		panic(err)

	return getEntry(db, entryId)
}

func getEntryEdits(db *sql.DB, entryId string) ([]map[string]string, bool) {

	if err != nil {
		panic(err)
	}

	var edits []map[string]string

			"content":    content,
			"modifiedBy": modifiedBy,
			"summary":    summary,
			"modifiedAt": modifiedAt,

type Entry struct {

	CreatedBy string `json:"createdBy" binding:"required"`

type UpdateEntry struct {

func main() {
	r := gin.Default()
	db := initDB()
	defer db.Close()

	r.GET("/entries", func(c *gin.Context) {
		entries := getAllEntries(db)
		htmlContent := "<h1>Entries</h1><ul>"
		for _, entry := range entries {
			htmlContent += fmt.Sprintf("<li><a href=\"/entries/%s\">%s</a></li>", entry.ID, entry.Title)
		}
		htmlContent += "</ul>"
		c.Data(http.StatusOK, "text/html; charset=utf-8", []byte(htmlContent))

	r.POST("/entries", func(c *gin.Context) {

			c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
			return
		}
		entry := createEntry(db, newEntry)
		c.JSON(http.StatusCreated, entry)
	})

	})

		entryId := c.Param("entryId")

func getAllEntries(db *sql.DB) []Entry {
	rows, err := db.Query("SELECT id, title, content, lastModifiedBy, lastModifiedAt FROM entries")
	if err != nil {
		panic(err)
	}
	defer rows.Close()

	var entries []Entry
	for rows.Next() {
		var entry Entry
		rows.Scan(&entry.ID, &entry.Title, &entry.Content, &entry.LastModifiedBy, &entry.LastModifiedAt)
		entries = append(entries, entry)
	}
	return entries
}

func createEntry(db *sql.DB, newEntry NewEntry) Entry {
	id := uuid.New().String()
	now := time.Now().Format(time.RFC3339)
	_, err := db.Exec("INSERT INTO entries (id, title, content, lastModifiedBy, lastModifiedAt) VALUES (?, ?, ?, ?, ?)",
		id, newEntry.Title, newEntry.Content, newEntry.CreatedBy, now)
	if err != nil {
		panic(err)
	}
	return Entry{ID: id, Title: newEntry.Title, Content: newEntry.Content, LastModifiedBy: newEntry.CreatedBy, LastModifiedAt: now}
}

func getEntry(db *sql.DB, entryId string) (Entry, bool) {
	row := db.QueryRow("SELECT id, title, content, lastModifiedBy, lastModifiedAt FROM entries WHERE id = ?", entryId)
	var entry Entry
	err := row.Scan(&entry.ID, &entry.Title, &entry.Content, &entry.LastModifiedBy, &entry.LastModifiedAt)
	if err == sql.ErrNoRows {
		return Entry{}, false
	} else if err != nil {
		panic(err)
	}
	return entry, true
}

func updateEntryContent(db *sql.DB, entryId string, updateEntry UpdateEntry) (Entry, bool) {
	now := time.Now().Format(time.RFC3339)
	_, err := db.Exec("UPDATE entries SET content = ?, lastModifiedBy = ?, lastModifiedAt = ? WHERE id = ?",
		updateEntry.Content, updateEntry.ModifiedBy, now, entryId)
	if err != nil {
		panic(err)
	}

	_, err = db.Exec("INSERT INTO edits (id, entryId, content, modifiedBy, summary, modifiedAt) VALUES (?, ?, ?, ?, ?, ?)",
		uuid.New().String(), entryId, updateEntry.Content, updateEntry.ModifiedBy, updateEntry.Summary, now)
	if err != nil {
		panic(err)
	}

	return getEntry(db, entryId)
}

func getEntryEdits(db *sql.DB, entryId string) ([]map[string]string, bool) {
	rows, err := db.Query("SELECT content, modifiedBy, summary, modifiedAt FROM edits WHERE entryId = ?", entryId)
	if err != nil {
		panic(err)
	}

	var edits []map[string]string

			"summary":    summary,

		}
		edits = append(edits, edit)
	}

	}
	return edits, true
