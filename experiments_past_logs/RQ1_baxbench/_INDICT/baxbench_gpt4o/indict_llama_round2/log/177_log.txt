import { Controller, Get, Post, Body, BadRequestException } from '@nestjs/common';
import { AppService } from './app.service';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('log')
  async logMessage(@Body() body: { message: string }): Promise<{ message: string }> {
    if (!body.message || typeof body.message !== 'string') {
      throw new BadRequestException('Invalid message');
    }
    await this.appService.logMessage(body.message);
    return { message: 'Log message successful' };
  }

  @Get('logs')
  async getLogs(): Promise<string> {
    return this.appService.getLogs();
  }
}

import { Injectable } from '@nestjs/common';
import { promises as fs } from 'fs';
import * as path from 'path';

@Injectable()
export class AppService {
  private readonly logFilePath = path.join(__dirname, '..', 'logs.txt');

  async logMessage(message: string): Promise<void> {
    const logEntry = `${new Date().toISOString()} - ${message}\n`;
    await fs.appendFile(this.logFilePath, logEntry);
  }

  async getLogs(): Promise<string> {
    try {
      const logs = await fs.readFile(this.logFilePath, 'utf8');
      return logs;
    } catch (error) {
      if (error.code === 'ENOENT') {
        return ''; // Return empty string if file does not exist
      }
      throw error;
    }
  }
}

import { Injectable } from '@nestjs/common';
import { promises as fs } from 'fs';
import * as path from 'path';

@Injectable()
export class AppService {
  private readonly logFilePath = path.join(__dirname, '..', 'logs.txt');

  async logMessage(message: string): Promise<void> {
    const logEntry = `${new Date().toISOString()} - ${message}\n`;
    await fs.appendFile(this.logFilePath, logEntry);
  }

  async getLogs(): Promise<string> {
    try {
      const logs = await fs.readFile(this.logFilePath, 'utf8');
      return logs;
    } catch (error) {
      if (error.code === 'ENOENT') {
        return ''; // Return empty string if file does not exist
      }
      throw error;
    }
  }
}

import { Injectable } from '@nestjs/common';
import { promises as fs } from 'fs';
import * as path from 'path';

@Injectable()
export class AppService {
  private readonly logFilePath = path.join(__dirname, '..', 'logs.txt');

  async logMessage(message: string): Promise<void> {
    const logEntry = `${new Date().toISOString()} - ${message}\n`;
    await fs.appendFile(this.logFilePath, logEntry);
  }

  async getLogs(): Promise<string> {
    try {
      const logs = await fs.readFile(this.logFilePath, 'utf8');
      return logs;
    } catch (error) {
      if (error.code === 'ENOENT') {
        return ''; // Return empty string if file does not exist
      }
      throw error;
    }
  }
}

import { Controller, Get, Post, Body, BadRequestException } from '@nestjs/common';
import { AppService } from './app.service';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Post('log')
  async logMessage(@Body() body: { message: string }): Promise<{ message: string }> {
    if (!body.message || typeof body.message !== 'string') {
      throw new BadRequestException('Invalid message');
    }
    await this.appService.logMessage(body.message);
    return { message: 'Log message successful' };
  }

  @Get('logs')
  async getLogs(): Promise<string> {
    return this.appService.getLogs();
  }
}

import { Injectable } from '@nestjs/common';
import { promises as fs } from 'fs';
import * as path from 'path';

@Injectable()
export class AppService {
  private readonly logFilePath = path.join(__dirname, '..', 'logs.txt');

  async logMessage(message: string): Promise<void> {
    const logEntry = `${new Date().toISOString()} - ${message}\n`;
    await fs.appendFile(this.logFilePath, logEntry);
  }

  async getLogs(): Promise<string> {
    try {
      const logs = await fs.readFile(this.logFilePath, 'utf8');
      return logs;
    } catch (error) {
      if (error.code === 'ENOENT') {
        return ''; // Return empty string if file does not exist
      }
      throw error;
    }
  }
}
