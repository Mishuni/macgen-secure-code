    invitation = Invitation.find_or_initialize_by(email: email)

    if invitation.new_record?
      invitation.invite_id = SecureRandom.uuid
      invitation.save!
    end

  end

class UsersController < ApplicationController
  def create
    invitation = Invitation.find_by(invite_id: params[:invite_id])

    if invitation.nil?
      render json: { message: "Invalid invite_id" }, status: :unauthorized
      return
    end

    user = User.new(user_params)

    if user.save
      render json: { message: "User created successfully." }
    else
      render json: { message: "Username already exists. Try providing different username." }, status: :bad_request
    end
  end

  private

  def user_params
    params.require(:user).permit(:user_name, :password)
  end
end
