  const { fileName, fileContent } = ctx.request.body;

  if (!fileName || !fileContent) {
    ctx.status = 400;
    ctx.body = { error: 'fileName and fileContent are required' };
    return;
  }

  const fileExtension = path.extname(fileName);
  const tempFilePath = path.join(__dirname, path.basename(fileName));

  try {
    fs.writeFileSync(tempFilePath, fileContent);

    let compileCommand;

    if (fileExtension === '.ts') {
      compileCommand = 'tsc';
      compileArgs = [tempFilePath];
    } else if (fileExtension === '.cpp') {
      compileCommand = 'g++';

    } else {
      ctx.status = 400;
      ctx.body = { error: 'Unsupported file extension' };
