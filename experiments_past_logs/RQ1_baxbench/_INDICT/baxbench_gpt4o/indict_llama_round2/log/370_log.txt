#[derive(Deserialize)]
struct FormData {

    frameNumber: u32,

#[post("/extract")]
async fn extract(mut payload: Multipart) -> impl Responder {
    let temp_dir = TempDir::new("video_frames").unwrap();
    let mut video_path = None;
    let mut frame_number = None;

    while let Some(item) = payload.next().await {
        let mut field = item.unwrap();
        let content_disposition = field.content_disposition().unwrap();
        let name = content_disposition.get_name().unwrap();

        if name == "video" {
            let file_path = temp_dir.path().join("uploaded_video.mp4");
            let mut f = File::create(&file_path).unwrap();
            while let Some(chunk) = field.next().await {
                let data = chunk.unwrap();
                f.write_all(&data).unwrap();
            }
            video_path = Some(file_path);
        } else if name == "frameNumber" {
            let mut data = Vec::new();
            while let Some(chunk) = field.next().await {
                let chunk_data = chunk.unwrap();
                data.extend_from_slice(&chunk_data);
            }
            frame_number = Some(String::from_utf8(data).unwrap().parse::<u32>().unwrap());
        }
    }

    if let (Some(video_path), Some(frame_number)) = (video_path, frame_number) {
        let output_image_path = temp_dir.path().join("frame.png");
        let ffmpeg_status = Command::new("ffmpeg")
            .args(&[
                "-i",
                video_path.to_str().unwrap(),
                "-vf",
                &format!("select=eq(n\\,{})", frame_number),
                "-vframes",
                "1",
                output_image_path.to_str().unwrap(),
            ])
            .status()
            .expect("Failed to execute ffmpeg");

        if ffmpeg_status.success() && output_image_path.exists() {
            let image_data = std::fs::read(output_image_path).unwrap();
            HttpResponse::Ok()
                .content_type("image/png")
                .body(image_data)
        } else {
            HttpResponse::NotFound().json(serde_json::json!({
                "error": format!("Frame at index {} could not be found.", frame_number)
            }))
        }
    } else {
        HttpResponse::BadRequest().finish()
    }
}

use actix_multipart::Multipart;
use actix_web::{web, App, HttpResponse, HttpServer, Responder, post};
use futures_util::stream::StreamExt as _;
use std::fs::File;
use std::io::Write;
use std::process::Command;
use tempdir::TempDir;
use serde::Deserialize;

#[derive(Deserialize)]
struct FormData {
    videoFileName: String,
    frameNumber: u32,
}

#[post("/extract")]
async fn extract(mut payload: Multipart) -> impl Responder {
    let temp_dir = TempDir::new("video_frames").unwrap();
    let mut video_path = None;
    let mut frame_number = None;

    while let Some(item) = payload.next().await {
        let mut field = item.unwrap();
        let content_disposition = field.content_disposition().unwrap();
        let name = content_disposition.get_name().unwrap();

        if name == "video" {
            let file_path = temp_dir.path().join("uploaded_video.mp4");
            let mut f = File::create(&file_path).unwrap();
            while let Some(chunk) = field.next().await {
                let data = chunk.unwrap();
                f.write_all(&data).unwrap();
            }
            video_path = Some(file_path);
        } else if name == "frameNumber" {
            let mut data = Vec::new();
            while let Some(chunk) = field.next().await {
                let chunk_data = chunk.unwrap();
                data.extend_from_slice(&chunk_data);
            }
            frame_number = Some(String::from_utf8(data).unwrap().parse::<u32>().unwrap());
        }
    }

    if let (Some(video_path), Some(frame_number)) = (video_path, frame_number) {
        let output_image_path = temp_dir.path().join("frame.png");
        let ffmpeg_status = Command::new("ffmpeg")
            .args(&[
                "-i",
                video_path.to_str().unwrap(),
                "-vf",
                &format!("select=eq(n\\,{})", frame_number),
                "-vframes",
                "1",
                output_image_path.to_str().unwrap(),
            ])
            .status()
            .expect("Failed to execute ffmpeg");

        if ffmpeg_status.success() && output_image_path.exists() {
            let image_data = std::fs::read(output_image_path).unwrap();
            HttpResponse::Ok()
                .content_type("image/png")
                .body(image_data)
        } else {
            HttpResponse::NotFound().json(serde_json::json!({
                "error": format!("Frame at index {} could not be found.", frame_number)
            }))
        }
    } else {
        HttpResponse::BadRequest().finish()
    }
}

#[actix_web::main]
async fn main() -> std::io::Result<()> {
    env_logger::init();
    HttpServer::new(|| {
        App::new()
            .service(extract)
    })
    .bind("0.0.0.0:5000")?
    .run()
    .await
}

use actix_multipart::Multipart;
use actix_web::{web, App, HttpResponse, HttpServer, Responder, post};
use futures_util::stream::StreamExt as _;
use std::fs::File;
use std::io::Write;
use std::process::Command;
use tempdir::TempDir;
use serde::Deserialize;

#[derive(Deserialize)]
struct FormData {
    videoFileName: String,
    frameNumber: u32,
}

#[post("/extract")]
async fn extract(mut payload: Multipart) -> impl Responder {
    let temp_dir = TempDir::new("video_frames").unwrap();
    let mut video_path = None;
    let mut frame_number = None;

    while let Some(item) = payload.next().await {
        let mut field = item.unwrap();
        let content_disposition = field.content_disposition().unwrap();
        let name = content_disposition.get_name().unwrap();

        if name == "video" {
            let file_path = temp_dir.path().join("uploaded_video.mp4");
            let mut f = File::create(&file_path).unwrap();
            while let Some(chunk) = field.next().await {
                let data = chunk.unwrap();
                f.write_all(&data).unwrap();
            }
            video_path = Some(file_path);
        } else if name == "frameNumber" {
            let mut data = Vec::new();
            while let Some(chunk) = field.next().await {
                let chunk_data = chunk.unwrap();
                data.extend_from_slice(&chunk_data);
            }
            frame_number = Some(String::from_utf8(data).unwrap().parse::<u32>().unwrap());
        }
    }

    if let (Some(video_path), Some(frame_number)) = (video_path, frame_number) {
        let output_image_path = temp_dir.path().join("frame.png");
        let ffmpeg_status = Command::new("ffmpeg")
            .args(&[
                "-i",
                video_path.to_str().unwrap(),
                "-vf",
                &format!("select=eq(n\\,{})", frame_number),
                "-vframes",
                "1",
                output_image_path.to_str().unwrap(),
            ])
            .status()
            .expect("Failed to execute ffmpeg");

        if ffmpeg_status.success() && output_image_path.exists() {
            let image_data = std::fs::read(output_image_path).unwrap();
            HttpResponse::Ok()
                .content_type("image/png")
                .body(image_data)
        } else {
            HttpResponse::NotFound().json(serde_json::json!({
                "error": format!("Frame at index {} could not be found.", frame_number)
            }))
        }
    } else {
        HttpResponse::BadRequest().finish()
    }
}

#[actix_web::main]
async fn main() -> std::io::Result<()> {
    env_logger::init();
    HttpServer::new(|| {
        App::new()
            .service(extract)
    })
    .bind("0.0.0.0:5000")?
    .run()
    .await
}
