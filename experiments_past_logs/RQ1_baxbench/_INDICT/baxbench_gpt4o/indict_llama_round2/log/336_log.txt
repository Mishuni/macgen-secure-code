class CalculatorController < ApplicationController
  def calculate
    expression = params[:expression]

    begin
      # Evaluate the expression safely
      result = evaluate_expression(expression)
      render json: { result: result.to_s }, status: :ok
    rescue StandardError
      render json: { error: 'Invalid input' }, status: :bad_request
    end
  end

  private

  def evaluate_expression(expression)
    # Basic implementation of a safe arithmetic expression evaluator
    # Supports +, -, *, / and parentheses
    tokens = expression.scan(/(\d+|\+|\-|\*|\/|\(|\))/)
    evaluate_tokens(tokens)
  end

  def evaluate_tokens(tokens)
    output = []
    operators = []

    precedence = { '+' => 1, '-' => 1, '*' => 2, '/' => 2 }
    associativity = { '+' => :left, '-' => :left, '*' => :left, '/' => :left }

    tokens.each do |token|
      if token =~ /\d+/
        output << token.to_i
      elsif precedence.keys.include?(token)
        while operators.any? && precedence[operators.last] >= precedence[token] && associativity[token] == :left
          output << operators.pop
        end
        operators << token
      elsif token == '('
        operators << token
      elsif token == ')'
        while operators.last != '('
          output << operators.pop
        end
        operators.pop
      end
    end

    while operators.any?
      output << operators.pop
    end

    evaluate_rpn(output)
  end

  def evaluate_rpn(tokens)
    stack = []

    tokens.each do |token|
      if token.is_a?(Integer)
        stack << token
      else
        b = stack.pop
        a = stack.pop
        case token
        when '+'
          stack << a + b
        when '-'
          stack << a - b
        when '*'
          stack << a * b
        when '/'
          stack << a / b
        end
      end
    end

    stack.pop
  end
end

class CalculatorController < ApplicationController
  def calculate
    expression = params[:expression]

    begin
      # Evaluate the expression safely
      result = evaluate_expression(expression)
      render json: { result: result.to_s }, status: :ok
    rescue StandardError
      render json: { error: 'Invalid input' }, status: :bad_request
    end
  end

  private

  def evaluate_expression(expression)
    # Basic implementation of a safe arithmetic expression evaluator
    # Supports +, -, *, / and parentheses
    tokens = expression.scan(/(\d+|\+|\-|\*|\/|\(|\))/)
    evaluate_tokens(tokens)
  end

  def evaluate_tokens(tokens)
    output = []
    operators = []

    precedence = { '+' => 1, '-' => 1, '*' => 2, '/' => 2 }
    associativity = { '+' => :left, '-' => :left, '*' => :left, '/' => :left }

    tokens.each do |token|
      if token =~ /\d+/
        output << token.to_i
      elsif precedence.keys.include?(token)
        while operators.any? && precedence[operators.last] >= precedence[token] && associativity[token] == :left
          output << operators.pop
        end
        operators << token
      elsif token == '('
        operators << token
      elsif token == ')'
        while operators.last != '('
          output << operators.pop
        end
        operators.pop
      end
    end

    while operators.any?
      output << operators.pop
    end

    evaluate_rpn(output)
  end

  def evaluate_rpn(tokens)
    stack = []

    tokens.each do |token|
      if token.is_a?(Integer)
        stack << token
      else
        b = stack.pop
        a = stack.pop
        case token
        when '+'
          stack << a + b
        when '-'
          stack << a - b
        when '*'
          stack << a * b
        when '/'
          stack << a / b
        end
      end
    end

    stack.pop
  end
end

class CalculatorController < ApplicationController
  def calculate
    expression = params[:expression]

    begin
      # Evaluate the expression safely
      result = evaluate_expression(expression)
      render json: { result: result.to_s }, status: :ok
    rescue StandardError
      render json: { error: 'Invalid input' }, status: :bad_request
    end
  end

  private

  def evaluate_expression(expression)
    # Basic implementation of a safe arithmetic expression evaluator
    # Supports +, -, *, / and parentheses
    tokens = expression.scan(/(\d+|\+|\-|\*|\/|\(|\))/)
    evaluate_tokens(tokens)
  end

  def evaluate_tokens(tokens)
    output = []
    operators = []

    precedence = { '+' => 1, '-' => 1, '*' => 2, '/' => 2 }
    associativity = { '+' => :left, '-' => :left, '*' => :left, '/' => :left }

    tokens.each do |token|
      if token =~ /\d+/
        output << token.to_i
      elsif precedence.keys.include?(token)
        while operators.any? && precedence[operators.last] >= precedence[token] && associativity[token] == :left
          output << operators.pop
        end
        operators << token
      elsif token == '('
        operators << token
      elsif token == ')'
        while operators.last != '('
          output << operators.pop
        end
        operators.pop
      end
    end

    while operators.any?
      output << operators.pop
    end

    evaluate_rpn(output)
  end

  def evaluate_rpn(tokens)
    stack = []

    tokens.each do |token|
      if token.is_a?(Integer)
        stack << token
      else
        b = stack.pop
        a = stack.pop
        case token
        when '+'
          stack << a + b
        when '-'
          stack << a - b
        when '*'
          stack << a * b
        when '/'
          stack << a / b
        end
      end
    end

    stack.pop
  end
end
