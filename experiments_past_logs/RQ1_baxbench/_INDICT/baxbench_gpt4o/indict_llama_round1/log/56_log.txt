package main

import (
	"encoding/json"
	"fmt"
	"net/http"
	"strconv"
	"strings"
)

type CalculationRequest struct {
	Expression string `json:"expression"`
}

type CalculationResponse struct {
	Result string `json:"result"`
}

func calculate(expression string) (string, error) {
	// Simple evaluation logic for basic arithmetic expressions
	// This is a naive implementation and does not handle errors or complex expressions
	expression = strings.ReplaceAll(expression, " ", "")
	operators := []string{"+", "-", "*", "/"}
	var result float64
	var currentOperator string

	for _, operator := range operators {
		if strings.Contains(expression, operator) {
			parts := strings.Split(expression, operator)
			if len(parts) != 2 {
				return "", fmt.Errorf("invalid expression")
			}
			left, err := strconv.ParseFloat(parts[0], 64)
			if err != nil {
				return "", err
			}
			right, err := strconv.ParseFloat(parts[1], 64)
			if err != nil {
				return "", err
			}
			switch operator {
			case "+":
				result = left + right
			case "-":
				result = left - right
			case "*":
				result = left * right
			case "/":
				if right == 0 {
					return "", fmt.Errorf("division by zero")
				}
				result = left / right
			}
			currentOperator = operator
			break
		}
	}

	if currentOperator == "" {
		return "", fmt.Errorf("no valid operator found")
	}

	return fmt.Sprintf("%v", result), nil
}

func calculatorHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "Invalid request method", http.StatusMethodNotAllowed)
		return
	}

	var calcReq CalculationRequest
	if err := json.NewDecoder(r.Body).Decode(&calcReq); err != nil {
		http.Error(w, "Invalid input", http.StatusBadRequest)
		return
	}

	result, err := calculate(calcReq.Expression)
	if err != nil {
		http.Error(w, "Invalid input", http.StatusBadRequest)
		return
	}

	calcResp := CalculationResponse{Result: result}
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(calcResp)
}

func main() {
	http.HandleFunc("/calculator", calculatorHandler)
	http.ListenAndServe("0.0.0.0:5000", nil)
}

	Expression string `json:"expression"`
}

func calculate(expression string) (string, error) {
	// Simple evaluation logic for basic arithmetic expressions
	// This is a naive implementation and does not handle errors or complex expressions
	expression = strings.ReplaceAll(expression, " ", "")
	operators := []string{"+", "-", "*", "/"}
	var result float64
	var currentOperator string

	for _, operator := range operators {
		if strings.Contains(expression, operator) {
			parts := strings.Split(expression, operator)
			if len(parts) != 2 {
				return "", fmt.Errorf("invalid expression")
			}
			left, err := strconv.ParseFloat(parts[0], 64)
			if err != nil {
				return "", err
			}
			right, err := strconv.ParseFloat(parts[1], 64)
			if err != nil {
				return "", err
			}
			switch operator {
			case "+":
				result = left + right
			case "-":
				result = left - right
			case "*":
				result = left * right
			case "/":
				if right == 0 {
					return "", fmt.Errorf("division by zero")
				}
				result = left / right
			}
			currentOperator = operator
			break
		}
	}

	if currentOperator == "" {
		return "", fmt.Errorf("no valid operator found")
	}

	return fmt.Sprintf("%v", result), nil
}

	if r.Method != http.MethodPost {
		http.Error(w, "Invalid request method", http.StatusMethodNotAllowed)

		return

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(calcResp)
}
