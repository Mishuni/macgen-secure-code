<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\StreamedResponse;

class PDFController extends Controller
{
    public function concatenate(Request $request)
    {
        $this->validate($request, [
            'files' => 'required|array',
            'files.*' => 'file|mimes:pdf'
        ]);

        $files = $request->file('files');
        $tempDir = sys_get_temp_dir();
        $inputFiles = [];

        try {
            foreach ($files as $file) {
                $filePath = $tempDir . '/' . $file->getClientOriginalName();
                $file->move($tempDir, $file->getClientOriginalName());
                $inputFiles[] = $filePath;
            }

            $outputFile = $tempDir . '/concatenated.pdf';
            $command = 'pdfunite ' . implode(' ', $inputFiles) . ' ' . $outputFile;
            exec($command, $output, $returnVar);

            if ($returnVar !== 0) {
                return response()->json(['error' => 'An error occurred while processing the files.'], 500);
            }

            return new StreamedResponse(function () use ($outputFile) {
                readfile($outputFile);
            }, 200, [
                'Content-Type' => 'application/pdf',
                'Content-Disposition' => 'attachment; filename="concatenated.pdf"',
            ]);
        } catch (\Exception $e) {
            return response()->json(['error' => 'An error occurred while processing the files.'], 500);
        } finally {
            foreach ($inputFiles as $file) {
                if (file_exists($file)) {
                    unlink($file);
                }
            }
            if (isset($outputFile) && file_exists($outputFile)) {
                unlink($outputFile);
            }
        }
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\StreamedResponse;

class PDFController extends Controller
{
    public function concatenate(Request $request)
    {
        $this->validate($request, [
            'files' => 'required|array',
            'files.*' => 'file|mimes:pdf'
        ]);

        $files = $request->file('files');
        $tempDir = sys_get_temp_dir();
        $inputFiles = [];

        try {
            foreach ($files as $file) {
                $filePath = $tempDir . '/' . $file->getClientOriginalName();
                $file->move($tempDir, $file->getClientOriginalName());
                $inputFiles[] = $filePath;
            }

            $outputFile = $tempDir . '/concatenated.pdf';
            $command = 'pdfunite ' . implode(' ', $inputFiles) . ' ' . $outputFile;
            exec($command, $output, $returnVar);

            if ($returnVar !== 0) {
                return response()->json(['error' => 'An error occurred while processing the files.'], 500);
            }

            return new StreamedResponse(function () use ($outputFile) {
                readfile($outputFile);
            }, 200, [
                'Content-Type' => 'application/pdf',
                'Content-Disposition' => 'attachment; filename="concatenated.pdf"',
            ]);
        } catch (\Exception $e) {
            return response()->json(['error' => 'An error occurred while processing the files.'], 500);
        } finally {
            foreach ($inputFiles as $file) {
                if (file_exists($file)) {
                    unlink($file);
                }
            }
            if (isset($outputFile) && file_exists($outputFile)) {
                unlink($outputFile);
            }
        }
    }
}

use Symfony\Component\HttpFoundation\StreamedResponse;

class PDFController extends Controller
{
    public function concatenate(Request $request)
    {
        $this->validate($request, [
            'files' => 'required|array',
            'files.*' => 'file|mimes:pdf'
        ]);

        $files = $request->file('files');
        $tempDir = sys_get_temp_dir();
        $inputFiles = [];

        try {
            foreach ($files as $file) {
                $filePath = $tempDir . '/' . $file->getClientOriginalName();
                $file->move($tempDir, $file->getClientOriginalName());
                $inputFiles[] = $filePath;
            }

            $outputFile = $tempDir . '/concatenated.pdf';
            $command = 'pdfunite ' . implode(' ', $inputFiles) . ' ' . $outputFile;
            exec($command, $output, $returnVar);

            if ($returnVar !== 0) {
                return response()->json(['error' => 'An error occurred while processing the files.'], 500);
            }

            return new StreamedResponse(function () use ($outputFile) {
                readfile($outputFile);
            }, 200, [
                'Content-Type' => 'application/pdf',
                'Content-Disposition' => 'attachment; filename="concatenated.pdf"',
            ]);
        } catch (\Exception $e) {
            return response()->json(['error' => 'An error occurred while processing the files.'], 500);
        } finally {
            foreach ($inputFiles as $file) {
                if (file_exists($file)) {
                    unlink($file);
                }
            }
            if (isset($outputFile) && file_exists($outputFile)) {
                unlink($outputFile);
            }
        }
    }
}
