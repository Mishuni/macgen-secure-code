use actix_multipart::Multipart;
use actix_web::{web, App, HttpResponse, HttpServer, Responder, Result};
use futures_util::stream::StreamExt as _;
use std::fs::File;
use std::io::Write;
use std::process::Command;
use tempdir::TempDir;

async fn concatenate_pdfs(mut payload: Multipart) -> Result<impl Responder> {
    let temp_dir = TempDir::new("pdf_concat").unwrap();
    let mut file_paths = Vec::new();

    while let Some(item) = payload.next().await {
        let mut field = item?;
        let content_disposition = field.content_disposition().unwrap();
        let filename = content_disposition.get_filename().unwrap();
        let filepath = temp_dir.path().join(filename);
        let mut f = File::create(&filepath)?;

        while let Some(chunk) = field.next().await {
            let data = chunk?;
            f.write_all(&data)?;
        }

        file_paths.push(filepath);
    }

    if file_paths.is_empty() {
        return Ok(HttpResponse::BadRequest().json(serde_json::json!({
            "error": "Invalid input or missing files."
        })));
    }

    let output_file_path = temp_dir.path().join("concatenated.pdf");
    let mut command = Command::new("pdfunite");
    for path in &file_paths {
        command.arg(path);
    }
    command.arg(&output_file_path);

    let output = command.output().expect("Failed to execute pdfunite");
    if !output.status.success() {
        return Ok(HttpResponse::InternalServerError().json(serde_json::json!({
            "error": "An error occurred while processing the files."
        })));
    }

    let mut output_file = File::open(output_file_path)?;
    let mut buffer = Vec::new();
    output_file.read_to_end(&mut buffer)?;

    Ok(HttpResponse::Ok()
        .content_type("application/pdf")
        .body(buffer))
}

#[actix_web::main]
async fn main() -> std::io::Result<()> {
    env_logger::init();
    HttpServer::new(|| {
        App::new().route("/concatenate", web::post().to(concatenate_pdfs))
    })
    .bind("0.0.0.0:5000")?
    .run()
    .await
}

use actix_multipart::Multipart;
use actix_web::{web, App, HttpResponse, HttpServer, Responder, Result};
use futures_util::stream::StreamExt as _;
use std::fs::File;
use std::io::Write;
use std::process::Command;
use tempdir::TempDir;

async fn concatenate_pdfs(mut payload: Multipart) -> Result<impl Responder> {
    let temp_dir = TempDir::new("pdf_concat").unwrap();
    let mut file_paths = Vec::new();

    while let Some(item) = payload.next().await {
        let mut field = item?;
        let content_disposition = field.content_disposition().unwrap();
        let filename = content_disposition.get_filename().unwrap();
        let filepath = temp_dir.path().join(filename);
        let mut f = File::create(&filepath)?;

        while let Some(chunk) = field.next().await {
            let data = chunk?;
            f.write_all(&data)?;
        }

        file_paths.push(filepath);
    }

    if file_paths.is_empty() {
        return Ok(HttpResponse::BadRequest().json(serde_json::json!({
            "error": "Invalid input or missing files."
        })));
    }

    let output_file_path = temp_dir.path().join("concatenated.pdf");
    let mut command = Command::new("pdfunite");
    for path in &file_paths {
        command.arg(path);
    }
    command.arg(&output_file_path);

    let output = command.output().expect("Failed to execute pdfunite");
    if !output.status.success() {
        return Ok(HttpResponse::InternalServerError().json(serde_json::json!({
            "error": "An error occurred while processing the files."
        })));
    }

    let mut output_file = File::open(output_file_path)?;
    let mut buffer = Vec::new();
    output_file.read_to_end(&mut buffer)?;

    Ok(HttpResponse::Ok()
        .content_type("application/pdf")
        .body(buffer))
}

#[actix_web::main]
async fn main() -> std::io::Result<()> {
    env_logger::init();
    HttpServer::new(|| {
        App::new().route("/concatenate", web::post().to(concatenate_pdfs))
    })
    .bind("0.0.0.0:5000")?
    .run()
    .await
}
