require 'open3'

class GifCreatorController < ApplicationController
  def create_gif
    images = params[:images]
    target_size = params[:targetSize]
    delay = params[:delay] || 10
    append_reverted = params[:appendReverted] || false

    if images.nil? || target_size.nil?
      render json: { error: 'images and targetSize are required' }, status: :bad_request
      return
    end

    begin
      image_paths = images.map { |img| img.tempfile.path }
      gif_path = Rails.root.join('tmp', "output_#{SecureRandom.hex}.gif")

      command = build_convert_command(image_paths, target_size, delay, append_reverted, gif_path)
      stdout, stderr, status = Open3.capture3(command)

      if status.success?
        send_file gif_path, type: 'image/gif', disposition: 'inline'
      else
        render json: { error: stderr }, status: :internal_server_error
      end
    ensure
      File.delete(gif_path) if File.exist?(gif_path)
    end
  end

  private

  def build_convert_command(image_paths, target_size, delay, append_reverted, output_path)
    command = "convert -delay #{delay} -resize #{target_size} #{image_paths.join(' ')}"
    command += " -reverse #{image_paths.join(' ')}" if append_reverted
    command += " #{output_path}"
    command
  end
end

class GifCreatorController < ApplicationController
  def create_gif
    images = params[:images]
    target_size = params[:targetSize]
    delay = params[:delay] || 10
    append_reverted = params[:appendReverted] || false

    if images.nil? || target_size.nil?
      render json: { error: 'images and targetSize are required' }, status: :bad_request
      return
    end

    begin
      image_paths = images.map { |img| img.tempfile.path }
      gif_path = Rails.root.join('tmp', "output_#{SecureRandom.hex}.gif")

      command = build_convert_command(image_paths, target_size, delay, append_reverted, gif_path)
      stdout, stderr, status = Open3.capture3(command)

      if status.success?
        send_file gif_path, type: 'image/gif', disposition: 'inline'
      else
        render json: { error: stderr }, status: :internal_server_error
      end
    ensure
      File.delete(gif_path) if File.exist?(gif_path)
    end
  end

  private

  def build_convert_command(image_paths, target_size, delay, append_reverted, output_path)
    command = "convert -delay #{delay} -resize #{target_size} #{image_paths.join(' ')}"
    command += " -reverse #{image_paths.join(' ')}" if append_reverted
    command += " #{output_path}"
    command
  end

require 'open3'

class GifCreatorController < ApplicationController
  def create_gif
    images = params[:images]
    target_size = params[:targetSize]
    delay = params[:delay] || 10
    append_reverted = params[:appendReverted] || false

    if images.nil? || target_size.nil?
      render json: { error: 'images and targetSize are required' }, status: :bad_request
      return
    end

    begin
      image_paths = images.map { |img| img.tempfile.path }
      gif_path = Rails.root.join('tmp', "output_#{SecureRandom.hex}.gif")

      command = build_convert_command(image_paths, target_size, delay, append_reverted, gif_path)
      stdout, stderr, status = Open3.capture3(command)

      if status.success?
        send_file gif_path, type: 'image/gif', disposition: 'inline'
      else
        render json: { error: stderr }, status: :internal_server_error
      end
    ensure
      File.delete(gif_path) if File.exist?(gif_path)
    end
  end

  private

  def build_convert_command(image_paths, target_size, delay, append_reverted, output_path)
    command = "convert -delay #{delay} -resize #{target_size} #{image_paths.join(' ')}"
    command += " -reverse #{image_paths.join(' ')}" if append_reverted
    command += " #{output_path}"
    command
  end
end
