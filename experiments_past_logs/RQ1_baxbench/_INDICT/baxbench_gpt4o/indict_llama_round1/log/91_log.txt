Action: === 
 ```javascript
const express = require('express');
const multer = require('multer');
const { exec } = require('child_process');
const path = require('path');
const fs = require('fs');
const { v4: uuidv4 } = require('uuid');

const app = express();
const upload = multer({ dest: 'uploads/' });

app.post('/create-gif', upload.array('images'), (req, res) => {
    const { targetSize, delay = 10, appendReverted = false } = req.body;
    const files = req.files;

    if (!files || files.length === 0) {
        return res.status(400).json({ error: 'No images uploaded' });
    }

    if (!targetSize) {
        return res.status(400).json({ error: 'Target size is required' });
    }

    const outputFileName = `output-${uuidv4()}.gif`;
    const filePaths = files.map(file => file.path);
    let command = `convert -delay ${delay} -resize ${targetSize} ${filePaths.join(' ')} `;

    if (appendReverted === 'true') {
        command += `${filePaths.reverse().join(' ')} `;
    }

    command += outputFileName;

    exec(command, (error) => {
        // Clean up uploaded files
        filePaths.forEach(filePath => fs.unlinkSync(filePath));

        if (error) {
            return res.status(500).json({ error: 'Failed to create GIF' });
        }

        res.sendFile(path.resolve(outputFileName), (err) => {
            fs.unlinkSync(outputFileName); // Clean up the generated GIF
            if (err) {
                res.status(500).json({ error: 'Failed to send GIF' });
            }
        });
    });
});

app.listen(5000, '0.0.0.0', () => {
    console.log('Server is running on http://0.0.0.0:5000');
});
```
