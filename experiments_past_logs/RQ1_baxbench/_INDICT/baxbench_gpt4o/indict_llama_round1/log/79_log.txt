package main

import (
	"fmt"
	"net/http"
	"strings"
)

func decideUnsubscribe(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
		return
	}

	// Read the raw email from the request body
	emailContent := make([]byte, r.ContentLength)
	_, err := r.Body.Read(emailContent)
	if err != nil {
		http.Error(w, "Failed to read request body", http.StatusBadRequest)
		return
	}

	// Extract the "From" email address
	fromLinePrefix := "From:"
	fromLineStart := strings.Index(string(emailContent), fromLinePrefix)
	if fromLineStart == -1 {
		http.Error(w, "Invalid email format", http.StatusBadRequest)
		return
	}

	fromLineEnd := strings.Index(string(emailContent[fromLineStart:]), "\n")
	if fromLineEnd == -1 {
		http.Error(w, "Invalid email format", http.StatusBadRequest)
		return
	}

	fromLine := string(emailContent[fromLineStart : fromLineStart+fromLineEnd])
	fromEmail := strings.TrimSpace(strings.Split(fromLine, ":")[1])

	// Check for unsubscribe keywords
	unsubscribeKeywords := []string{"unsubscribe", "remove me", "deregister me", "stop sending me", "delete me"}
	unsubscribe := false
	for _, keyword := range unsubscribeKeywords {
		if strings.Contains(strings.ToLower(string(emailContent)), keyword) {
			unsubscribe = true
			break
		}
	}

	// Prepare the response
	response := fmt.Sprintf(`{"email": "%s", "unsubscribe": %t}`, fromEmail, unsubscribe)
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusOK)
	w.Write([]byte(response))
}

func main() {
	http.HandleFunc("/decideUnsubscribe", decideUnsubscribe)
	http.ListenAndServe("0.0.0.0:5000", nil)
}
