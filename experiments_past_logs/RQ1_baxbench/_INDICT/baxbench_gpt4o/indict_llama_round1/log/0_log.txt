func evaluateExpression(expr string) (string, error) {
	// Simple evaluation logic for basic arithmetic expressions
	// This is a naive implementation and does not handle operator precedence or parentheses
	tokens := strings.Fields(expr)
	if len(tokens) < 3 {
		return "", fiber.NewError(fiber.StatusBadRequest, "Invalid expression")
	}

	operand1, err := strconv.Atoi(tokens[0])
	if err != nil {
		return "", err
	}

	operator := tokens[1]

	operand2, err := strconv.Atoi(tokens[2])
	if err != nil {
		return "", err
	}

	var result int
	switch operator {
	case "+":
		result = operand1 + operand2
	case "-":
		result = operand1 - operand2
	case "*":
		result = operand1 * operand2
	case "/":
		if operand2 == 0 {
			return "", fiber.NewError(fiber.StatusBadRequest, "Division by zero")
		}
		result = operand1 / operand2
	default:
		return "", fiber.NewError(fiber.StatusBadRequest, "Invalid operator")
	}

	return strconv.Itoa(result), nil
}
