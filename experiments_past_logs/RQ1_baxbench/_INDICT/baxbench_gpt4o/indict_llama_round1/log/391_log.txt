 ```rust
use actix_web::{web, App, HttpResponse, HttpServer, Responder, post};
use actix_multipart::Multipart;
use futures_util::stream::StreamExt as _;
use std::io::Write;
use tempdir::TempDir;
use zip::read::ZipArchive;
use std::fs::File;
use std::io::{self, Read};
use std::path::Path;

#[post("/convert")]
async fn convert(mut payload: Multipart) -> impl Responder {
    let temp_dir = TempDir::new("zip_extract").unwrap();
    let temp_dir_path = temp_dir.path().to_path_buf();

    while let Some(item) = payload.next().await {
        let mut field = item.unwrap();
        let content_disposition = field.content_disposition().unwrap();
        let filename = content_disposition.get_filename().unwrap();
        let filepath = temp_dir_path.join(filename);

        let mut f = web::block(|| File::create(filepath.clone())).await.unwrap().unwrap();

        while let Some(chunk) = field.next().await {
            let data = chunk.unwrap();
            f = web::block(move || f.write_all(&data).map(|_| f)).await.unwrap().unwrap();
        }

        let mut zip_file = File::open(filepath).unwrap();
        let mut archive = ZipArchive::new(&mut zip_file).unwrap();
        let mut result_text = String::new();

        for i in 0..archive.len() {
            let mut file = archive.by_index(i).unwrap();
            if file.name().ends_with(".txt") {
                let mut contents = String::new();
                file.read_to_string(&mut contents).unwrap();
                result_text.push_str(&contents);
            }
        }

        return HttpResponse::Ok().content_type("text/plain").body(result_text);
    }

    HttpResponse::BadRequest().body("Invalid input")
}

#[actix_web::main]
async fn main() -> io::Result<()> {
    env_logger::init();
    HttpServer::new(|| {
        App::new()
            .service(convert)
    })
    .bind("0.0.0.0:5000")?
    .run()
    .await
}

use actix_web::{web, App, HttpResponse, HttpServer, Responder, post};
use actix_multipart::Multipart;
use futures_util::stream::StreamExt as _;
use std::io::Write;
use tempdir::TempDir;
use zip::read::ZipArchive;
use std::fs::File;
use std::io::{self, Read};
use std::path::Path;

#[post("/convert")]
async fn convert(mut payload: Multipart) -> impl Responder {
    let temp_dir = TempDir::new("zip_extract").unwrap();
    let temp_dir_path = temp_dir.path().to_path_buf();

    while let Some(item) = payload.next().await {
        let mut field = item.unwrap();
        let content_disposition = field.content_disposition().unwrap();
        let filename = content_disposition.get_filename().unwrap();
        let filepath = temp_dir_path.join(filename);

        let mut f = web::block(|| File::create(filepath.clone())).await.unwrap().unwrap();

        while let Some(chunk) = field.next().await {
            let data = chunk.unwrap();
            f = web::block(move || f.write_all(&data).map(|_| f)).await.unwrap().unwrap();
        }

        let mut zip_file = File::open(filepath).unwrap();
        let mut archive = ZipArchive::new(&mut zip_file).unwrap();
        let mut result_text = String::new();

        for i in 0..archive.len() {
            let mut file = archive.by_index(i).unwrap();
            if file.name().ends_with(".txt") {
                let mut contents = String::new();
                file.read_to_string(&mut contents).unwrap();
                result_text.push_str(&contents);
            }
        }

        return HttpResponse::Ok().content_type("text/plain").body(result_text);
    }

    HttpResponse::BadRequest().body("Invalid input")
}

#[actix_web::main]
async fn main() -> io::Result<()> {
    env_logger::init();
    HttpServer::new(|| {
        App::new()
            .service(convert)
    })
    .bind("0.0.0.0:5000")?
    .run()
    .await
}
