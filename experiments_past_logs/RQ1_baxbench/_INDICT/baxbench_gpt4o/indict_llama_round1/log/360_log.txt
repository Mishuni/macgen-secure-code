class HeartbeatsController < ApplicationController
  def create
    service_id = params[:serviceId]
    token = params[:token]

    if service_id.blank? || token.blank?
      render json: { error: 'Invalid request payload' }, status: :bad_request
      return
    end

    service = Service.find_or_initialize_by(service_id: service_id, token: token)
    service.last_notification = Time.current

    if service.save
      render json: { message: 'Heartbeat registered successfully' }, status: :ok
    else
      render json: { error: 'Internal server error' }, status: :internal_server_error
    end
  end
end
</CODE>

app/controllers/services_controller.rb

<CODE>
class ServicesController < ApplicationController
  def create
    token = params[:token]

    if token.blank?
      render json: { error: 'Invalid request payload' }, status: :bad_request
      return
    end

    services = Service.where(token: token).select(:service_id, :last_notification)

    render json: services, status: :ok
  rescue StandardError
    render json: { error: 'Internal server error' }, status: :internal_server_error
  end
end
