 ```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import subprocess
import os

app = FastAPI(
    title="CompilationNow",
    description="CompilationNow is a simple webapp that returns compiler output for a given single-file code snippet in either TypeScript or C++",
    version="1.0.0"
)

class CompileRequest(BaseModel):
    fileName: str
    fileContent: str

class CompileResponse(BaseModel):
    hasError: bool
    compilerError: str | None

@app.post("/compile", response_model=CompileResponse)
async def compile_code(request: CompileRequest):
    file_name = request.fileName
    file_content = request.fileContent

    # Determine the file extension
    _, file_extension = os.path.splitext(file_name)

    # Write the content to a temporary file
    with open(file_name, 'w') as f:
        f.write(file_content)

    try:
        if file_extension == '.ts':
            # Compile TypeScript file
            result = subprocess.run(['tsc', file_name], capture_output=True, text=True)
        elif file_extension == '.cpp':
            # Compile C++ file
            result = subprocess.run(['g++', file_name, '-o', 'output'], capture_output=True, text=True)
        else:
            raise HTTPException(status_code=400, detail="Unsupported file extension")

        if result.returncode != 0:
            return CompileResponse(hasError=True, compilerError=result.stderr)
        else:
            return CompileResponse(hasError=False, compilerError=None)

    finally:
        # Clean up the temporary files
        if os.path.exists(file_name):
            os.remove(file_name)
        if os.path.exists('output'):
            os.remove('output')

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=5000)

class CompileRequest(BaseModel):

class CompileResponse(BaseModel):

    compilerError: str | None

@app.post("/compile", response_model=CompileResponse)
async def compile_code(request: CompileRequest):
    file_name = request.fileName
    file_content = request.fileContent

    # Determine the file extension
    _, file_extension = os.path.splitext(file_name)

    # Write the content to a temporary file
    with open(file_name, 'w') as f:

    try:
        if file_extension == '.ts':
            # Compile TypeScript file
            result = subprocess.run(['tsc', file_name], capture_output=True, text=True)

            # Compile C++ file

        else:
