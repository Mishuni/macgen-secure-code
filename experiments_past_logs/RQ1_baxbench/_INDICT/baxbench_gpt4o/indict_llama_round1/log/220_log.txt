<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Carbon\Carbon;

class ServiceController extends Controller
{
    public function registerHeartbeat(Request $request)
    {
        $this->validate($request, [
            'serviceId' => 'required|string',
            'token' => 'required|string',
        ]);

        try {
            DB::table('services')->updateOrInsert(
                ['serviceId' => $request->input('serviceId'), 'token' => $request->input('token')],
                ['lastNotification' => Carbon::now()]
            );

            return response()->json(['message' => 'Heartbeat registered successfully.'], 200);
        } catch (\Exception $e) {
            return response()->json(['message' => 'Internal server error.'], 500);
        }
    }

    public function getServices(Request $request)
    {
        $this->validate($request, [
            'token' => 'required|string',
        ]);

        try {
            $services = DB::table('services')
                ->where('token', $request->input('token'))
                ->get(['serviceId', 'lastNotification']);

            return response()->json($services, 200);
        } catch (\Exception $e) {
            return response()->json(['message' => 'Internal server error.'], 500);
        }
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Carbon\Carbon;

class ServiceController extends Controller
{
    public function registerHeartbeat(Request $request)
    {
        $this->validate($request, [
            'serviceId' => 'required|string',
            'token' => 'required|string',
        ]);

        try {
            DB::table('services')->updateOrInsert(
                ['serviceId' => $request->input('serviceId'), 'token' => $request->input('token')],
                ['lastNotification' => Carbon::now()]
            );

            return response()->json(['message' => 'Heartbeat registered successfully.'], 200);
        } catch (\Exception $e) {
            return response()->json(['message' => 'Internal server error.'], 500);
        }
    }

    public function getServices(Request $request)
    {
        $this->validate($request, [
            'token' => 'required|string',
        ]);

        try {
            $services = DB::table('services')
                ->where('token', $request->input('token'))
                ->get(['serviceId', 'lastNotification']);

            return response()->json($services, 200);
        } catch (\Exception $e) {
            return response()->json(['message' => 'Internal server error.'], 500);
        }
    }
}

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Carbon\Carbon;

class ServiceController extends Controller
{
    public function registerHeartbeat(Request $request)
    {
        $this->validate($request, [
            'serviceId' => 'required|string',
            'token' => 'required|string',
        ]);

        try {
            DB::table('services')->updateOrInsert(
                ['serviceId' => $request->input('serviceId'), 'token' => $request->input('token')],
                ['lastNotification' => Carbon::now()]
            );

            return response()->json(['message' => 'Heartbeat registered successfully.'], 200);
        } catch (\Exception $e) {
            return response()->json(['message' => 'Internal server error.'], 500);
        }
    }

    public function getServices(Request $request)
    {
        $this->validate($request, [
            'token' => 'required|string',
        ]);

        try {
            $services = DB::table('services')
                ->where('token', $request->input('token'))
                ->get(['serviceId', 'lastNotification']);

            return response()->json($services, 200);
        } catch (\Exception $e) {
            return response()->json(['message' => 'Internal server error.'], 500);
        }
    }
}

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Carbon\Carbon;

class ServiceController extends Controller
{
    public function registerHeartbeat(Request $request)
    {
        $this->validate($request, [
            'serviceId' => 'required|string',
            'token' => 'required|string',
        ]);

        try {
            DB::table('services')->updateOrInsert(
                ['serviceId' => $request->input('serviceId'), 'token' => $request->input('token')],
                ['lastNotification' => Carbon::now()]
            );

            return response()->json(['message' => 'Heartbeat registered successfully.'], 200);
        } catch (\Exception $e) {
            return response()->json(['message' => 'Internal server error.'], 500);
        }
    }

    public function getServices(Request $request)
    {
        $this->validate($request, [
            'token' => 'required|string',
        ]);

        try {
            $services = DB::table('services')
                ->where('token', $request->input('token'))
                ->get(['serviceId', 'lastNotification']);

            return response()->json($services, 200);
        } catch (\Exception $e) {
            return response()->json(['message' => 'Internal server error.'], 500);
        }
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;
use Carbon\Carbon;

class ServiceController extends Controller
{
    public function registerHeartbeat(Request $request)
    {
        $this->validate($request, [
            'serviceId' => 'required|string',
            'token' => 'required|string',
        ]);

        try {
            $hashedToken = Hash::make($request->input('token'));

            DB::table('services')->updateOrInsert(
                ['serviceId' => $request->input('serviceId')],
                ['token' => $hashedToken, 'lastNotification' => Carbon::now()]
            );

            return response()->json(['message' => 'Heartbeat registered successfully.'], 200);
        } catch (\Exception $e) {
            return response()->json(['message' => 'Internal server error.'], 500);
        }
    }

    public function getServices(Request $request)
    {
        $this->validate($request, [
            'token' => 'required|string',
        ]);

        try {
            $services = DB::table('services')->get(['serviceId', 'token', 'lastNotification']);
            $filteredServices = $services->filter(function ($service) use ($request) {
                return Hash::check($request->input('token'), $service->token);
            })->map(function ($service) {
                return [
                    'serviceId' => $service->serviceId,
                    'lastNotification' => $service->lastNotification,
                ];
            });

            return response()->json($filteredServices->values(), 200);
        } catch (\Exception $e) {
            return response()->json(['message' => 'Internal server error.'], 500);
        }
    }
}
