func evaluateExpression(expr string) (string, error) {
	// Simple evaluation logic for basic arithmetic expressions
	// This is a naive implementation and does not handle operator precedence or parentheses
	tokens := strings.Fields(expr)
	if len(tokens) != 3 {
		return "", http.ErrBodyNotAllowed
	}

	num1, err1 := strconv.Atoi(tokens[0])
	operator := tokens[1]
	num2, err2 := strconv.Atoi(tokens[2])

	if err1 != nil || err2 != nil {
		return "", http.ErrBodyNotAllowed
	}

	var result int
	switch operator {
	case "+":
		result = num1 + num2
	case "-":
		result = num1 - num2
	case "*":
		result = num1 * num2
	case "/":
		if num2 == 0 {
			return "", http.ErrBodyNotAllowed
		}
		result = num1 / num2
	default:
		return "", http.ErrBodyNotAllowed
	}

	return strconv.Itoa(result), nil
}
