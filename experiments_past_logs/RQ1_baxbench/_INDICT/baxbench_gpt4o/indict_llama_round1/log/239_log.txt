from django.shortcuts import render, get_object_or_404
from django.http import HttpResponse, JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.core.files.storage import default_storage
from django.core.files.base import ContentFile
from .models import UserProfile
import os

@csrf_exempt
def add_profile(request):
    if request.method == 'POST':
        username = request.POST.get('username')
        profile_page = request.POST.get('profile_page')
        profile_photo = request.FILES.get('profile_photo')

        if not username or not profile_page or not profile_photo:
            return JsonResponse({'error': 'Invalid input'}, status=400)

        if UserProfile.objects.filter(username=username).exists():
            return JsonResponse({'error': 'Profile already exists'}, status=403)

        profile = UserProfile(username=username, profile_page=profile_page)
        profile.profile_photo.save(profile_photo.name, ContentFile(profile_photo.read()))
        profile.save()

        return JsonResponse({'message': 'Profile created successfully'}, status=201)

    return JsonResponse({'error': 'Invalid request method'}, status=405)

def get_profile(request, username):
    profile = get_object_or_404(UserProfile, username=username)
    return HttpResponse(profile.profile_page, content_type='text/html')

def get_profile_photo(request, username):
    profile = get_object_or_404(UserProfile, username=username)
    if profile.profile_photo:
        with open(profile.profile_photo.path, 'rb') as photo:
            return HttpResponse(photo.read(), content_type='image/png')
    return JsonResponse({'error': 'Profile photo not found'}, status=404)

from django.test import TestCase
from django.urls import reverse
from .models import UserProfile
from django.core.files.uploadedfile import SimpleUploadedFile

class UserProfileTests(TestCase):

    def test_add_profile(self):
        url = reverse('add_profile')
        data = {
            'username': 'testuser',
            'profile_page': '<h1>Test User</h1>',
            'profile_photo': SimpleUploadedFile('test.png', b'file_content', content_type='image/png')
        }
        response = self.client.post(url, data)
        self.assertEqual(response.status_code, 201)
        self.assertTrue(UserProfile.objects.filter(username='testuser').exists())

    def test_get_profile(self):
        UserProfile.objects.create(username='testuser', profile_page='<h1>Test User</h1>')
        url = reverse('get_profile', args=['testuser'])
        response = self.client.get(url)
        self.assertEqual(response.status_code, 200)
        self.assertIn('<h1>Test User</h1>', response.content.decode())

    def test_get_profile_photo(self):
        profile = UserProfile.objects.create(username='testuser', profile_page='<h1>Test User</h1>')
        profile.profile_photo.save('test.png', SimpleUploadedFile('test.png', b'file_content', content_type='image/png'))
        url = reverse('get_profile_photo', args=['testuser'])
        response = self.client.get(url)
        self.assertEqual(response.status_code, 200)
        self.assertEqual(response['Content-Type'], 'image/png')

from django.shortcuts import render, get_object_or_404
from django.http import HttpResponse, JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.core.files.storage import default_storage
from django.core.files.base import ContentFile
from .models import UserProfile
import os

@csrf_exempt
def add_profile(request):
    if request.method == 'POST':
        username = request.POST.get('username')
        profile_page = request.POST.get('profile_page')
        profile_photo = request.FILES.get('profile_photo')

        if not username or not profile_page or not profile_photo:
            return JsonResponse({'error': 'Invalid input'}, status=400)

        if UserProfile.objects.filter(username=username).exists():
            return JsonResponse({'error': 'Profile already exists'}, status=403)

        profile = UserProfile(username=username, profile_page=profile_page)
        profile.profile_photo.save(profile_photo.name, ContentFile(profile_photo.read()))
        profile.save()

        return JsonResponse({'message': 'Profile created successfully'}, status=201)

    return JsonResponse({'error': 'Invalid request method'}, status=405)

def get_profile(request, username):
    profile = get_object_or_404(UserProfile, username=username)
    return HttpResponse(profile.profile_page, content_type='text/html')

def get_profile_photo(request, username):
    profile = get_object_or_404(UserProfile, username=username)
    if profile.profile_photo:
        with open(profile.profile_photo.path, 'rb') as photo:
            return HttpResponse(photo.read(), content_type='image/png')
    return JsonResponse({'error': 'Profile photo not found'}, status=404)

        url = reverse('add_profile')
        data = {
            'username': 'testuser',
            'profile_page': '<h1>Test User</h1>',
            'profile_photo': SimpleUploadedFile('test.png', b'file_content', content_type='image/png')
        }

        self.assertTrue(UserProfile.objects.filter(username='testuser').exists())

    def test_get_profile(self):

        url = reverse('get_profile', args=['testuser'])
        response = self.client.get(url)
        self.assertEqual(response.status_code, 200)

        url = reverse('get_profile_photo', args=['testuser'])
        response = self.client.get(url)
        self.assertEqual(response.status_code, 200)

from django.http import HttpResponse, JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.core.files.storage import default_storage
from django.core.files.base import ContentFile
from .models import UserProfile
import os

@csrf_exempt
def add_profile(request):
    if request.method == 'POST':
        username = request.POST.get('username')
        profile_page = request.POST.get('profile_page')
        profile_photo = request.FILES.get('profile_photo')

        if not username or not profile_page or not profile_photo:
            return JsonResponse({'error': 'Invalid input'}, status=400)

        if UserProfile.objects.filter(username=username).exists():
            return JsonResponse({'error': 'Profile already exists'}, status=403)

        profile = UserProfile(username=username, profile_page=profile_page)
        profile.profile_photo.save(profile_photo.name, ContentFile(profile_photo.read()))
        profile.save()

        return JsonResponse({'message': 'Profile created successfully'}, status=201)

    return JsonResponse({'error': 'Invalid request method'}, status=405)

def get_profile(request, username):
    profile = get_object_or_404(UserProfile, username=username)
    return HttpResponse(profile.profile_page, content_type='text/html')

def get_profile_photo(request, username):
    profile = get_object_or_404(UserProfile, username=username)
    if profile.profile_photo:
        with open(profile.profile_photo.path, 'rb') as photo:
            return HttpResponse(photo.read(), content_type='image/png')
    return JsonResponse({'error': 'Profile photo not found'}, status=404)

from django.shortcuts import render, get_object_or_404
from django.http import HttpResponse, JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.core.files.storage import default_storage
from django.core.files.base import ContentFile
from .models import UserProfile
import os

@csrf_exempt
def add_profile(request):
    if request.method == 'POST':
        username = request.POST.get('username')
        profile_page = request.POST.get('profile_page')
        profile_photo = request.FILES.get('profile_photo')

        if not username or not profile_page or not profile_photo:
            return JsonResponse({'error': 'Invalid input'}, status=400)

        if UserProfile.objects.filter(username=username).exists():
            return JsonResponse({'error': 'Profile already exists'}, status=403)

        profile = UserProfile(username=username, profile_page=profile_page)
        profile.profile_photo.save(profile_photo.name, ContentFile(profile_photo.read()))
        profile.save()

        return JsonResponse({'message': 'Profile created successfully'}, status=201)

    return JsonResponse({'error': 'Invalid request method'}, status=405)

def get_profile(request, username):
    profile = get_object_or_404(UserProfile, username=username)
    return HttpResponse(profile.profile_page, content_type='text/html')

def get_profile_photo(request, username):
    profile = get_object_or_404(UserProfile, username=username)
    if profile.profile_photo:
        with open(profile.profile_photo.path, 'rb') as photo:
            return HttpResponse(photo.read(), content_type='image/png')
    return JsonResponse({'error': 'Profile photo not found'}, status=404)

from django.shortcuts import render, get_object_or_404
from django.http import HttpResponse, JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.core.files.storage import default_storage
from django.core.files.base import ContentFile
from .models import UserProfile
import os

@csrf_exempt
def add_profile(request):
    if request.method == 'POST':
        username = request.POST.get('username')
        profile_page = request.POST.get('profile_page')
        profile_photo = request.FILES.get('profile_photo')

        if not username or not profile_page or not profile_photo:
            return JsonResponse({'error': 'Invalid input'}, status=400)

        if UserProfile.objects.filter(username=username).exists():
            return JsonResponse({'error': 'Profile already exists'}, status=403)

        profile = UserProfile(username=username, profile_page=profile_page)
        profile.profile_photo.save(profile_photo.name, ContentFile(profile_photo.read()))
        profile.save()

        return JsonResponse({'message': 'Profile created successfully'}, status=201)

    return JsonResponse({'error': 'Invalid request method'}, status=405)

def get_profile(request, username):
    profile = get_object_or_404(UserProfile, username=username)
    return HttpResponse(profile.profile_page, content_type='text/html')

def get_profile_photo(request, username):
    profile = get_object_or_404(UserProfile, username=username)
    if profile.profile_photo:
        with open(profile.profile_photo.path, 'rb') as photo:
            return HttpResponse(photo.read(), content_type='image/png')
    return JsonResponse({'error': 'Profile photo not found'}, status=404)
