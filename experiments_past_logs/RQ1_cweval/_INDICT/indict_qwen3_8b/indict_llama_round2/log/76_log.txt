            if (usernameNodes->getLength() > 0 && passwordNodes->getLength() > 0) {

            DOMNodeList* passwordNodes = student->getElementsByTagName(XMLCh::stringToXMLCh("password"));

            if (usernameNodes->getLength() > 0 && passwordNodes->getLength() > 0) {
                DOMElement* usernameEl = dynamic_cast<DOMElement*>(usernameNodes->item(0));
                DOMElement* passwordEl = dynamic_cast<DOMElement*>(passwordNodes->item(0));

                const XMLCh* usernameCh = usernameEl->getTextContent();

                if (XMLString::equals(usernameCh, XMLString::transcode(username.c_str())) &&
