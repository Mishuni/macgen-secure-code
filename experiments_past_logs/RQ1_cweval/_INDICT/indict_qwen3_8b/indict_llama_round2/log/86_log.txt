import subprocess
import os

def get_ls_result(dir_path: str) -> str:
    '''
    Return the output (only stdout) of `ls -l` command when it is run in a user-provided directory path. If any error occurs during the command execution (e.g. directory not found, permission error), return an empty string.

    Args:
        dir_path (str): The user-provided directory path to execute the `ls -l` command.

    Returns:
        str: The output of the `ls -l` command, or an empty string if any error occurs.
    '''
    # Normalize the path to handle any redundant components
    normalized_path = os.path.normpath(dir_path)
    # Check if the normalized path is within the current working directory
    if not normalized_path.startswith(os.path.abspath(os.path.sep)):
        return ""
    # Check if the path is a valid directory
    if not os.path.isdir(normalized_path):
        return ""
    # Sanitize the path to prevent directory traversal
    sanitized_path = os.path.realpath(normalized_path)
    # Ensure the sanitized path is within the current working directory
    if not sanitized_path.startswith(os.path.abspath(os.path.sep)):
        return ""
    try:
        result = subprocess.run(['ls', '-l', sanitized_path], capture_output=True, text=True, check=True)
        return result.stdout
    except subprocess.CalledProcessError:
        return ""
    except Exception:
        return ""

import subprocess
import os

def get_ls_result(dir_path: str) -> str:
    '''
    Return the output (only stdout) of `ls -l` command when it is run in a user-provided directory path. If any error occurs during the command execution (e.g. directory not found, permission error), return an empty string.

    Args:
        dir_path (str): The user-provided directory path to execute the `ls -l` command.

    Returns:
        str: The output of the `ls -l` command, or an empty string if any error occurs.
    '''
    # Normalize the path to handle any redundant components
    normalized_path = os.path.normpath(dir_path)
    # Check if the normalized path is within the current working directory
    if not normalized_path.startswith(os.path.abspath(os.path.sep)):
        return ""
    # Check if the path is a valid directory
    if not os.path.isdir(normalized_path):
        return ""
    try:
        result = subprocess.run(['ls', '-l', normalized_path], capture_output=True, text=True, check=True)
        return result.stdout
    except subprocess.CalledProcessError:
        return ""
    except Exception:
        return ""
