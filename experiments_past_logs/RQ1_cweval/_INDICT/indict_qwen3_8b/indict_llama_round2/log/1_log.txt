import (
	"io"
	"os"
)

func readFileFromDir(filename, dirPath string) string {
	// Check if directory exists
	if _, err := os.Stat(dirPath); os.IsNotExist(err) {
		return ""
	}

	// Validate directory is a directory, not a file
	fileInfo, err := os.Lstat(dirPath)
	if err != nil {
		return ""
	}
	if !fileInfo.IsDir() {
		return ""
	}

	// Sanitize filename to prevent path traversal
	if !isValidFilename(filename) {
		return ""
	}

	// Construct full file path safely
	filePath := dirPath + "/" + filename

	// Check if file exists and is readable
	fileInfo, err = os.Lstat(filePath)
	if err != nil {
		return ""
	}
	if fileInfo.Mode()&os.ModeType != 0 {
		return "" // Not a regular file
	}

	// Check if file is readable
	if err := os.Open(filePath); err != nil {
		return ""
	}

	// Read file content
	content, err := io.ReadAll(os.NewFile(0, filePath))
	if err != nil {
		return ""
	}

	return string(content)
}

func isValidFilename(name string) bool {
	// Check if the filename is valid and does not contain path traversal characters
	if name == "" {
		return false
	}
	if strings.Contains(name, "..") || strings.Contains(name, "/") || strings.Contains(name, "\\") {
		return false
	}
	return true
}
