const crypto = require('crypto');

function encrypt_data(key, data) {
  // Determine the key length based on AES requirements
  const keyLengths = [16, 24, 32];
  let keyUsed = Buffer.alloc(0);
  
  // Find the appropriate key length (16, 24, or 32 bytes)
  for (const length of keyLengths) {
    if (key.length >= length) {
      keyUsed = key.slice(0, length);
      break;
    }
  }

  // Generate a random IV
  const iv = crypto.randomBytes(16);

  // Create AES-CBC cipher
  const cipher = crypto.createCipheriv('aes-256-cbc', keyUsed, iv);

  // Encrypt the data
  let encrypted = cipher.update(data, 'utf8', 'base64');
  encrypted += cipher.final('base64');

  return {
    encryptedData: Buffer.from(encrypted, 'base64'),
    iv: iv,
    keyUsed: keyUsed
  };
}
```
