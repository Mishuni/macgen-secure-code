function get_target_for_redirect(target, domain) {
  try {
    const targetUrl = new URL(target);
    const domainUrl = new URL('https://' + domain);

    // Check if target domain is a subdomain of the provided domain
    const targetDomain = targetUrl.hostname;
    const domainParts = domain.split('.');
    const targetParts = targetDomain.split('.');

    // Normalize the domain parts by removing any empty strings (e.g., from leading dots)
    const normalizedDomainParts = domainParts.filter(part => part !== '');
    const normalizedTargetParts = targetParts.filter(part => part !== '');

    // Check if the target domain is a subdomain of the domain
    if (normalizedTargetParts.length === normalizedDomainParts.length) {
      // Same domain
      return target;
    } else if (normalizedTargetParts.length > normalizedDomainParts.length) {
      // Check if the target is a subdomain
      const isSubdomain = normalizedTargetParts.slice(0, normalizedDomainParts.length).join('.') === domain;
      if (isSubdomain) {
        return target;
      }
    }

    // Fallback to domain URL
    return domainUrl.toString();
  } catch (e) {
    // If URL is invalid, fallback to domain
