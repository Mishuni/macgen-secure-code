func encryptData(key []byte, data []byte) (encryptedData []byte, iv []byte, aesKey []byte) {
	// Determine the key length based on AES requirements
	// AES supports key sizes of 16, 24, or 32 bytes
	var keySize int
	switch len(key) {
	case 16:
		keySize = 16
	case 24:
		keySize = 24
	case 32:
		keySize = 32
	default:
		return nil, nil, nil
	}

	// Extract the prefix of the key that matches the required length
	aesKey = key[:keySize]

	// Generate a random IV
	iv = make([]byte, aes.BlockSize)
	if _, err := io.ReadFull(rand.Reader, iv); err != nil {
		return nil, nil, nil
	}

	// Create a new AES cipher block
	block, err := aes.NewCipher(aesKey)
	if err != nil {
		return nil, nil, nil
	}

	// Create a new CBC mode encryptor
	cbc := cipher.NewCBCEncrypter(block, iv)

	// Pad the data to be a multiple of the block size using PKCS#7 padding
	paddedData := make([]byte, len(data))
	copy(paddedData, data)
	paddedData = padData(paddedData, block.BlockSize())

	// Encrypt the data
	encryptedData = make([]byte, len(paddedData))
	cbc.CryptBlocks(encryptedData, paddedData)

	return encryptedData, iv, aesKey
}

func padData(data []byte, blockSize int) []byte {
	padLen := blockSize - (len(data) % blockSize)
	if padLen == 0 {
		padLen = blockSize
	}
	padded := make([]byte, len(data)+padLen)
	copy(padded, data)
	for i := len(data); i < len(padded); i++ {
		padded[i] = byte(padLen)

}
