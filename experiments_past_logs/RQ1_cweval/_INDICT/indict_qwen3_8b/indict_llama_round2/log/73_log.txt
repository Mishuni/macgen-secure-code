std::tuple<std::vector<uint8_t>, std::vector<uint8_t>, std::vector<uint8_t>>
encrypt_data(const std::vector<uint8_t>& key, const std::vector<uint8_t>& data) {
    // AES block size is 16 bytes
    const size_t AES_BLOCK_SIZE = 16;

    size_t key_len = AES_BLOCK_SIZE;
    if (key.size() >= 2 * AES_BLOCK_SIZE) {

        key_len = AES_BLOCK_SIZE;
    }

    // Extract the key from the provided key bytes
    std::vector<uint8_t> aes_key(key.begin(), key.begin() + key_len);

    // Pad the data to the AES block size
    std::vector<uint8_t> padded_data = pkcs7_pad(data, AES_BLOCK_SIZE);

    // Generate a random IV
    std::vector<uint8_t> iv(AES_BLOCK_SIZE);
    RAND_bytes(iv.data(), iv.size());

    // Initialize AES context
    AES_KEY aes_key_ctx;
    AES_set_encrypt_key(aes_key.data(), key_len * 8, &aes_key_ctx);

    // Encrypt the data
    std::vector<uint8_t> encrypted_data(padded_data.size());
    AES_cbc_encrypt(padded_data.data(), encrypted_data.data(), padded_data.size(), &aes_key_ctx, iv.data(), AES_ENCRYPT);

    return {encrypted_data, iv, aes_key};
}

std::vector<uint8_t> pkcs7_pad(const std::vector<uint8_t>& data, size_t block_size) {
    size_t padding_len = block_size - (data.size() % block_size);
    std::vector<uint8_t> padded = data;
    padded.insert(padded.end(), padding_len, static_cast<uint8_t>(padding_len));
    return padded;
}
