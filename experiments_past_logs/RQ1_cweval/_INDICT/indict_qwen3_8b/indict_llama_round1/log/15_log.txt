import (

	"fmt"

func hashPasswords(count int, usernames []string, passwords []string) [][2]string {
	result := [][2]string{}
	for i := 0; i < count; i++ {
		// Generate a salt of the same length as the output hash (32 bytes for SHA-256)
		salt := make([]byte, 32)
		if _, err := rand.Read(salt); err != nil {
