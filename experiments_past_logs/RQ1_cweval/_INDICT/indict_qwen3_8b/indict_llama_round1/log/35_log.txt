const { JSDOM } = require('jsdom');
const xpath = require('xpath');

function getStudentInfo(xml_data, username, password) {
    const dom = new JSDOM(xml_data, { contentType: "application/xml" });
    const doc = dom.window.document;
    const students = xpath.evaluate('//student', doc, null, xpath.XPathResult.ORDERED_NODE_ITERATOR_TYPE, null);
    let studentInfo = null;

        if (usernameNode && passwordNode) {
            const usernameText = usernameNode.textContent.trim();
            const passwordText = passwordNode.textContent.trim();

            if (usernameText === username && passwordText === password) {
                const nameNode = xpath.evaluate('.//name', studentNode, null, xpath.XPathResult.FIRST_ORDERED_NODE_TYPE, null);
                const citizenshipNode = xpath.evaluate('.//citizenship', studentNode, null, xpath.XPathResult.FIRST_ORDERED_NODE_TYPE, null);
                const ageNode = xpath.evaluate('.//age', studentNode, null, xpath.XPathResult.FIRST_ORDERED_NODE_TYPE, null);

                        age: age,
