) {
    // AES key sizes: 16, 24, or 32 bytes
    size_t aes_key_size = 32; // Use 256-bit key as default
    if (key_len >= aes_key_size) {
        // Use the first 'aes_key_size' bytes of the key
        unsigned char aes_key[aes_key_size];
        memcpy(aes_key, key, aes_key_size);
        key = aes_key;
    } else {
        // If key is shorter than required, pad it with zeros (not ideal, but per the doc)
        unsigned char aes_key[aes_key_size];
        memcpy(aes_key, key, key_len);
        memset(aes_key + key_len, 0, aes_key_size - key_len);
        key = aes_key;
    }

    // AES block size is 16 bytes
    size_t block_size = AES_BLOCK_SIZE;
    size_t encrypted_len = ((data_len + block_size - 1) / block_size) * block_size;
    unsigned char* encrypted_data = (unsigned char*)malloc(encrypted_len);
    if (!encrypted_data) {
        return (encryption_result){0, 0, 0, 0};
    }

    // Generate IV (16 bytes)
    unsigned char iv[AES_BLOCK_SIZE];
    memset(iv, 0, AES_BLOCK_SIZE);

    // Initialize AES context
    AES_KEY aes_key_ctx;
    AES_set_encrypt_key(key, aes_key_size * 8, &aes_key_ctx);

    // Encrypt data in CBC mode
    AES_cbc_encrypt(data, encrypted_data, data_len, &aes_key_ctx, iv, AES_ENCRYPT);

    // Prepare result
    encryption_result result;
