 ```go
import (
	"archive/tar"
	"compress/gzip"
	"fmt"
	"io"
	"os"
	"path/filepath"
)

func extractTarToPath(tarPath string, destPath string) bool {
	// Open the tar file
	file, err := os.Open(tarPath)
	if err != nil {
		return false
	}
	defer file.Close()

	// Check if the file is a gzip compressed tar file
	reader, err := gzip.NewReader(file)
	if err != nil {
		return false
	}
	defer reader.Close()

	// Create a tar reader
	tarReader := tar.NewReader(reader)

	// Walk through the tar file entries
	for {
		header, err := tarReader.Next()
		if err == io.EOF {
			break // End of archive
		}
		if err != nil {
			return false
		}

		// Get the file name from the header
		path := header.Name

		// Validate the path to prevent directory traversal attacks
		if !filepath.IsAbs(path) {
			// Calculate the full path by joining with destPath
			fullPath := filepath.Join(destPath, path)
			if filepath.Dir(fullPath) != destPath {
				// Prevent directory traversal
				continue
			}
		} else {
			// Check if the path is within the destination directory
			if !strings.HasPrefix(path, destPath) || filepath.Dir(path) != destPath {
				continue
			}
		}

		// Create the directory if it doesn't exist
		if header.Typeflag == tar.TypeDir {
			if err := os.MkdirAll(filepath.Join(destPath, path), 0777); err != nil {
				return false
			}
			continue
		}

		// Create the file
		filePath := filepath.Join(destPath, path)
		file, err := os.OpenFile(filePath, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, os.FileMode(header.Mode))
		if err != nil {
			return false
		}
		defer file.Close()

		// Copy the file contents
		if _, err := io.Copy(file, tarReader); err != nil {
			return false
		}
	}

	return true
}

	if err != nil {
		return false
	}
	defer reader.Close()

	// Create a tar reader
	tarReader := tar.NewReader(reader)

	// Walk through the tar file entries
	for {
		header, err := tarReader.Next()
		if err == io.EOF {
			break // End of archive
		}
		if err != nil {
			return false
		}

		// Get the file name from the header
		path := header.Name

		// Validate the path to prevent directory traversal attacks
		if !filepath.IsAbs(path) {
			// Calculate the full path by joining with destPath
			fullPath := filepath.Join(destPath, path)
			if filepath.Dir(fullPath) != destPath {
				// Prevent directory traversal
				continue
			}
		} else {
			// Check if the path is within the destination directory
			if !strings.HasPrefix(path, destPath) || filepath.Dir(path) != destPath {
				continue
			}
		}

		// Create the directory if it doesn't exist
		if header.Typeflag == tar.TypeDir {
			if err := os.MkdirAll(filepath.Join(destPath, path), 0777); err != nil {
				return false
			}
			continue
