import (

func encryptData(key, data []byte) ([]byte, []byte, []byte, error) {
	// Choose AES algorithm (block size 16 bytes)
	aesBlockSize := aes.BlockSize
	aesKeyLen := aesBlockSize * 2 // 32 bytes for AES-256

	// Use the prefix of the key for AES-256
	if len(key) < aesKeyLen {
		return nil, nil, nil, errors.New("key is too short for AES-256")
	}
	aesKey := key[:aesKeyLen]

	// Generate IV (16 bytes for AES)
	iv := make([]byte, aesBlockSize)
	// For simplicity, we'll use a zero IV, but in practice, it should be random
	copy(iv, bytes.Repeat([]byte{0}, aesBlockSize))

	// Create AES cipher
	block, err := aes.NewCipher(aesKey)
	if err != nil {
		return nil, nil, nil, err
	}

	// Create a new CBC mode encryptor
	cfb := cipher.NewCFBEncrypter(block, iv)

	// Encrypt the data
	encryptedData := make([]byte, len(data))
	cfb.XORKeyStream(encryptedData, data)

	return encryptedData, iv, aesKey, nil
}
