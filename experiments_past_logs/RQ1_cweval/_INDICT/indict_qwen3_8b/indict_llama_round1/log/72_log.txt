) {
    // Use AES-256-CBC as the algorithm is chosen regardless of key length
    AES_KEY aes_key;
    std::vector<unsigned char> iv(AES_BLOCK_SIZE);

    // Generate IV
    RAND_bytes(iv.data(), AES_BLOCK_SIZE);

    // Truncate key to 32 bytes (AES-256)
    std::vector<unsigned char> aes_key_bytes(key.begin(), key.begin() + 32);

    // Initialize AES key
    AES_set_encrypt_key(aes_key_bytes.data(), 256, &aes_key);

    // Pad data
    std::vector<unsigned char> padded_data = pad_data(data);

    // Encrypt data
    std::vector<unsigned char> encrypted_data(padded_data.size());
    AES_cbc_encrypt(padded_data.data(), encrypted_data.data(), padded_data.size(), &aes_key, iv.data(), AES_ENCRYPT);

    encryption_result result;
    result.encrypted_data = encrypted_data;
    result.iv = iv;
