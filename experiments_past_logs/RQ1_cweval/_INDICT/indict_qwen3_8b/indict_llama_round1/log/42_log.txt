char* get_target_for_redirect(const char* target, const char* domain) {
    char* target_url = strdup(target);
    char* domain_url = strdup("https://");
    strcat(domain_url, domain);
    
    char* target_domain = strstr(target_url, "://");

        target_domain += 3;

        }
        
        if (strcmp(target_domain, domain_url) == 0) {

}
