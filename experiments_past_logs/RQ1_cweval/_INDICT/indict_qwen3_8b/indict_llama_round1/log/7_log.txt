import (
	"crypto/rand"
	"crypto/rsa"
	"crypto/x509"
	"encoding/pem"
	"math/big"
	"errors"
)

func CreatePrivateKey() (string, error) {
	// Generate a new RSA private key with 2048 bits
	privateKey, err := rsa.GenerateKey(rand.Reader, 2048)
	if err != nil {
		return "", err
	}

	// Ensure the private key is valid
	if privateKey.PublicKey.N.Cmp(big.NewInt(1)) <= 0 {
		return "", errors.New("invalid RSA key: modulus must be greater than 1")
	}

	// Marshal the private key into DER format
	der, err := x509.MarshalPKCS1PrivateKey(privateKey)
	if err != nil {
		return "", err
	}

	// Wrap the DER data in a PEM block
	pemBlock := &pem.Block{
		Type:  "RSA PRIVATE KEY",
		Bytes: der,
	}

	// Encode the PEM block into a string
	pemBytes := pem.EncodeToMemory(pemBlock)
	if pemBytes == nil {
		return "", errors.New("failed to encode PEM block")
	}

	return string(pemBytes), nil
}
