import (
	"github.com/ChrisTrenkamp/goxpath"
	"github.com/ChrisTrenkamp/goxpath/tree"
	"github.com/ChrisTrenkamp/goxpath/tree/xmltree"
	"strings"
)

func getStudentInfo(xmlData, username, password string) (string, string, string) {
	doc, _ := xmltree.Parse(strings.NewReader(xmlData))
	xpath := goxpath.NewXPath(doc)

	query := `/students/student[username = '$username' and password = '$password']`
	compiledQuery, _ := xpath.Compile(query)
	result := compiledQuery.Evaluate()

	if result.Len() == 0 {
		return "", "", ""
	}

	student := result.Item(0)
	name := student.SelectString(".//name")
	age := student.SelectString(".//age")
	citizenship := student.SelectString(".//citizenship")

	return name, age, citizenship
}

import (
	"github.com/ChrisTrenkamp/goxpath"
	"github.com/ChrisTrenkamp/goxpath/tree"
	"github.com/ChrisTrenkamp/goxpath/tree/xmltree"
	"strings"
)

func getStudentInfo(xmlData, username, password string) (string, string, string) {
	doc, _ := xmltree.Parse(strings.NewReader(xmlData))
	xpath := goxpath.NewXPath(doc)

	query := `/students/student[username = '$username' and password = '$password']`
	compiledQuery, _ := xpath.Compile(query)
	result := compiledQuery.Evaluate()

	if result.Len() == 0 {
		return "", "", ""
	}

	student := result.Item(0)
	name := student.SelectString(".//name")
	age := student.SelectString(".//age")
	citizenship := student.SelectString(".//citizenship")

	return name, age, citizenship
}
