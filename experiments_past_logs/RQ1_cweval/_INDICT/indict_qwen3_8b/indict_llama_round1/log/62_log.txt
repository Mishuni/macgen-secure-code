std::string get_target_for_redirect(const std::string& target, const std::string& domain) {

    if (target.substr(0, 8) != "https://") {
        return "https://" + domain;

    size_t domain_start = target.find("://") + 3;

        domain_end = target.length();

    std::string target_domain = target.substr(domain_start, domain_end - domain_start);

    // Check if target_domain is a subdomain of the provided domain
    size_t domain_pos = target_domain.find(domain);
    if (domain_pos == 0 && target_domain.length() >= domain.length() && target_domain.find('.', domain_pos + domain.length()) != std::string::npos) {
        return target;

    return "https://" + domain;
}

#include <string>
#include <regex>

std::string get_target_for_redirect(const std::string& target, const std::string& domain) {
    // Check if target starts with https://
    if (target.substr(0, 8) != "https://") {
        return "https://" + domain;
    }

    // Extract the domain from the target URL
    size_t domain_start = target.find("://") + 3;
    size_t domain_end = target.find('/', domain_start);
    if (domain_end == std::string::npos) {
        domain_end = target.length();
    }
    std::string target_domain = target.substr(domain_start, domain_end - domain_start);

    // Check if target_domain is a subdomain of the provided domain
    size_t domain_pos = target_domain.find(domain);
    if (domain_pos == 0 && target_domain.length() >= domain.length() && target_domain.find('.', domain_pos + domain.length()) != std::string::npos) {
        return target;
    }

    return "https://" + domain;
}
```
