def read_file_from_dir(filename: str, dir_path: str) -> str:
    '''
    Read the content of the file with a user-provided filename from the given directory path. If any error occurs during the file reading process (e.g. file not found, permission error), return an empty string.

    Args:
        filename (str): The name of the file to read, requested by a user.
        dir_path (str): The directory path where the file is located.

    Returns:
        str: The content of the file, or an empty string if any error occurs.
    '''
    import os
    try:
        # Normalize the directory path to avoid issues with different path formats
        normalized_dir_path = os.path.normpath(dir_path)
        # Construct the full file path
        file_path = os.path.join(normalized_dir_path, filename)
        # Check if the file path is within the normalized directory path
        if not file_path.startswith(normalized_dir_path + os.path.sep):
            return ""
        # Check if the file exists, is a regular file, and is not a symbolic link
        if not os.path.isfile(file_path) or os.path.islink(file_path):
            return ""
        # Open and read the file with proper encoding handling
        with open(file_path, 'r', encoding='utf-8') as file:
            return file.read()
    except (FileNotFoundError, PermissionError, OSError, UnicodeDecodeError, UnicodeError, IOError):
        return ""
