#include <xercesc/util/TransService.hpp>
#include <xercesc/util/Exception.hpp>

using namespace std;
using namespace xercesc;

std::map<std::string, std::string> get_student_info(const std::string& xml_data, const std::string& username, const std::string& password) {
    try {
        XMLPlatformUtils::Initialize();
        DOMParser* parser = DOMParser::createParser();
        parser->setFeature(DOM_LSSerializer::FEATURE, true);
        parser->setFeature(XMLParser::FEATURE_SECURE_PROCESSING, true);
        parser->setFeature(XMLParser::FEATURE_XINCLUDE, false);

        XMLCh* chData = XMLString::transcode(xml_data.c_str());
        DOMDocument* doc = parser->parseXML(chData);
        XMLString::release(&chData);

        DOMElement* root = doc->getDocumentElement();
        DOMNodeList* students = root->getElementsByTagName(XMLCh::stringToXMLCh("student"));

        std::map<std::string, std::string> result;
        for (XMLSize_t i = 0; i < students->getLength(); ++i) {
            DOMElement* student = dynamic_cast<DOMElement*>(students->item(i));
            DOMNodeList* usernameNodes = student->getElementsByTagName(XMLCh::stringToXMLCh("username"));
            DOMNodeList* passwordNodes = student->getElementsByTagName(XMLCh::stringToXMLCh("password"));

            if (usernameNodes->getLength() > 0 && passwordNodes->getLength() > 0) {
                DOMElement* usernameEl = dynamic_cast<DOMElement*>(usernameNodes->item(0));
                DOMElement* passwordEl = dynamic_cast<DOMElement*>(passwordNodes->item(0));

                const XMLCh* usernameCh = usernameEl->getTextContent();
                const XMLCh* passwordCh = passwordEl->getTextContent();

                if (XMLString::equals(usernameCh, XMLString::transcode(username.c_str())) &&
                    XMLString::equals(passwordCh, XMLString::transcode(password.c_str()))) {

                    DOMNodeList* nameNodes = student->getElementsByTagName(XMLCh::stringToXMLCh("name"));
                    DOMNodeList* citizenshipNodes = student->getElementsByTagName(XMLCh::stringToXMLCh("citizenship"));
                    DOMNodeList* ageNodes = student->getElementsByTagName(XMLCh::stringToXMLCh("age"));

                    if (nameNodes->getLength() > 0 && citizenshipNodes->getLength() > 0 && ageNodes->getLength() > 0) {
                        DOMElement* nameEl = dynamic_cast<DOMElement*>(nameNodes->item(0));
                        DOMElement* citizenshipEl = dynamic_cast<DOMElement*>(citizenshipNodes->item(0));
                        DOMElement* ageEl = dynamic_cast<DOMElement*>(ageNodes->item(0));

                        result["name"] = XMLString::transcode(nameEl->getTextContent());
                        result["citizenship"] = XMLString::transcode(citizenshipEl->getTextContent());
                        result["age"] = XMLString::transcode(ageEl->getTextContent());
                    }
                }
            }
        }

        delete parser;
        delete doc;
        XMLPlatformUtils::Terminate();
        return result;
    } catch (const XMLException& e) {
        return {};

        return {};
    }
}
