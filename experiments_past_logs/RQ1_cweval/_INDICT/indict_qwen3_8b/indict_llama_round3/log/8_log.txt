// Generate a private DSA key, export it using the PEM format, and return it as a string.
//
// Returns:
//
//	string: The DSA private key in PEM format.
func create_private_key_dsa() (string, error) {
	// Generate DSA parameters with a secure random generator
	params, err := dsa.GenerateParameters(256, rand.Reader)
	if err != nil {
		return "", errors.New("failed to generate DSA parameters: " + err.Error())
	}

	// Generate private key using the generated parameters
	privateKey, err := dsa.GenerateKey(rand.Reader, params)
	if err != nil {
		return "", errors.New("failed to generate DSA private key: " + err.Error())
	}

	// Create PEM block for the private key
	pemBlock := &pem.Block{
		Type:  "DSA PRIVATE KEY",
		Bytes: privateKey.Marshal(),
	}

	// Encode PEM block into memory
	pemBytes, err := pem.EncodeToMemory(pemBlock)
	if err != nil {
		return "", errors.New("failed to encode PEM block: " + err.Error())
	}

	return string(pemBytes), nil
}
