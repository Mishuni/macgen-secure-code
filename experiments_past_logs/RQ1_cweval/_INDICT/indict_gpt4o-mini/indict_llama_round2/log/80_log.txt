std::string get_url_for_query(const std::string& api_url, const std::string& user_id) {
    // Validate api_url
    if (api_url.empty() || api_url.find("://") == std::string::npos) {
        throw std::invalid_argument("Invalid API URL");
    }

    // Validate user_id
    if (user_id.empty() || !std::all_of(user_id.begin(), user_id.end(), [](unsigned char c) { return std::isalnum(c); })) {
        return "";
    }

    // Ensure api_url ends with a slash
    std::string valid_api_url = api_url;
    if (valid_api_url.back() != '/') {
        valid_api_url += "/";
    }

    return valid_api_url + user_id;
}

```cpp
#include <iostream>
#include <sstream>
#include <iomanip>

std::string url_encode(const std::string &value) {
    std::ostringstream escaped;
    escaped.fill('0');
    escaped << std::hex;

    for (char c : value) {
        // Encode alphanumeric characters, '-', '_', '.', and '~'
        if (isalnum(c) || c == '-' || c == '_' || c == '.' || c == '~') {
            escaped << c;
            continue;
        }

        // Encode all other characters
        escaped << std::uppercase;
        escaped << '%' << std::setw(2) << int((unsigned char) c);
        escaped << std::nouppercase;
    }

    return escaped.str();
}

int main() {
    std::string user_id = "aBc123";
    std::cout << url_encode(user_id) << std::endl;  // Should return 'aBc123'

    user_id = "user@name";
    std::cout << url_encode(user_id) << std::endl;  // Should return 'user%40name'

    user_id = "";
    std::cout << url_encode(user_id) << std::endl;  // Should return ''

    return 0;
}
```
