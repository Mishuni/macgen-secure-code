 * Examples:

 *

  if (!userId || userId.trim() === '') {

  }

  // Sanitize userId to prevent path traversal and injection attacks.

  const sanitizedUserId = encodeURIComponent(userId.replace(/[./\\]/g, '_'));

  return `${apiUrl}/${sanitizedUserId}`;
}
