@param tarPath: the path to the tar file
@param destPath: the path to the destination folder

@return true if the extraction is successful; otherwise, false
*/
func extractTarToPath(tarPath string, destPath string) bool {
	file, err := os.Open(tarPath)
	if err != nil {
		return false
	}
	defer file.Close()

	tarReader := tar.NewReader(file)

	// Ensure the destination directory exists and is clean
	if err := os.MkdirAll(destPath, 0755); err != nil {
		return false
	}

	for {
		header, err := tarReader.Next()

		if err == io.EOF {
			break // End of archive
		}
		if err != nil {
			// Handle corrupted tar file errors gracefully
			return false
		}

		// Sanitize and validate file name to prevent directory traversal and other attacks
		// Remove leading/trailing slashes and dots
		name := strings.Trim(header.Name, "/.")
		// Replace potentially malicious characters
		name = strings.ReplaceAll(name, "..", "")
		name = strings.ReplaceAll(name, "../", "")
		name = strings.ReplaceAll(name, "\\", "")

		// Prevent excessively long file names
		if len(name) > 255 { // Common filesystem limit
			return false
		}

		targetPath := filepath.Join(cleanedDestPath, name)

		// Resolve symlinks and check for directory traversal
		absTargetPath, err := filepath.Abs(targetPath)
		if err != nil {
			return false
		}
		absDestPath, err := filepath.Abs(cleanedDestPath)
		if err != nil {
			return false
		}

		if !strings.HasPrefix(absTargetPath, absDestPath) {
			return false
		}

		info := header.FileInfo()
		if info.IsDir() {
			if err := os.MkdirAll(targetPath, info.Mode()); err != nil {
				return false
			}
			continue
		}

		// Ensure parent directory exists for files
		if err := os.MkdirAll(filepath.Dir(targetPath), 0755); err != nil {
			return false
		}

		outFile, err := os.OpenFile(targetPath, os.O_WRONLY|os.O_CREATE|os.O_TRUNC, info.Mode())
		if err != nil {
			return false
		}

		if _, err := io.Copy(outFile, tarReader); err != nil {
			outFile.Close()
			return false
		}
		outFile.Close()
	}

	return true
}
