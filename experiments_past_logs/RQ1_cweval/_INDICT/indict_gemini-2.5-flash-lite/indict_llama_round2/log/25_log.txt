function generate_receive_log(msg) {
  const timestamp = new Date().toISOString();

  // This approach is generally safer for logs that might be displayed in a web context.
  const sanitizedMsg = msg.replace(/[<>&'"]/g, (match) => {
    switch (match) {
      case '<':
        return '&lt;';
      case '>':
        return '&gt;';
      case '&':
        return '&amp;';
      case "'":
        return '&apos;';
      case '"':
        return '&quot;';
      default:
        return match;
    }
  });
