    result.iv = (unsigned char*)malloc(aes_iv_size);
    if (!result.iv) {
        free(result.key);
        result.key = NULL;
        return result; // Allocation failed
    }
    // Generate a random IV.
    if (RAND_bytes(result.iv, aes_iv_size) != 1) {
        free(result.key);
        result.key = NULL;
        free(result.iv);
        result.iv = NULL;
        return result; // Failed to generate random IV
    }

    // Calculate the size of the encrypted data.
    // CBC mode requires padding to a multiple of the block size.
    // PKCS7 padding: if data_len is a multiple of block_size, add a full block of padding.
    // Otherwise, add enough bytes to reach the next multiple of block_size.
    size_t block_size = AES_BLOCK_SIZE;
    size_t padding_len = block_size - (data_len % block_size);
    if (padding_len == block_size) { // If data_len is already a multiple of block_size
        padding_len = block_size;
    }
    result.encrypted_len = data_len + padding_len;
    result.encrypted_data = (unsigned char*)malloc(result.encrypted_len);
    if (!result.encrypted_data) {
        free(result.key);
        result.key = NULL;
        free(result.iv);
        result.iv = NULL;
        return result; // Allocation failed
    }

    // Create a temporary buffer for padded data
    unsigned char* padded_data = (unsigned char*)malloc(result.encrypted_len);
    if (!padded_data) {
        free(result.key);
        result.key = NULL;
        free(result.iv);
        result.iv = NULL;
        free(result.encrypted_data);
        result.encrypted_data = NULL;
        return result; // Allocation failed
    }
    memcpy(padded_data, data, data_len);

    // Apply PKCS7 padding
    for (size_t i = data_len; i < result.encrypted_len; ++i) {
        padded_data[i] = (unsigned char)padding_len;
    }

    AES_KEY aes_key;
    // AES_set_encrypt_key expects key size in bits.
    if (AES_set_encrypt_key(result.key, aes_key_size * 8, &aes_key) < 0) {
        free(result.key);
        result.key = NULL;
        free(result.iv);
        result.iv = NULL;
        free(result.encrypted_data);
        result.encrypted_data = NULL;
        free(padded_data);
        return result; // Failed to set AES encryption key
    }

    // Perform encryption using CBC mode.
    // AES_cbc_encrypt encrypts data in place if input and output buffers are the same.
    // Here, we use a separate padded_data buffer and result.encrypted_data.
    // The IV is modified by AES_cbc_encrypt to become the last block of ciphertext,
    // which is standard behavior for some OpenSSL CBC functions, but for AES_cbc_encrypt
    // it's used as the initial vector and remains unchanged for the next block.
    // However, for decryption, the IV needs to be preserved.
    // We will use a copy of the IV for the encryption process.
    unsigned char iv_copy[AES_BLOCK_SIZE];
    memcpy(iv_copy, result.iv, AES_BLOCK_SIZE);

    AES_cbc_encrypt(padded_data, result.encrypted_data, result.encrypted_len, &aes_key, iv_copy, AES_ENCRYPT);

    free(padded_data); // Free the temporary padded data buffer

    return result;
}
