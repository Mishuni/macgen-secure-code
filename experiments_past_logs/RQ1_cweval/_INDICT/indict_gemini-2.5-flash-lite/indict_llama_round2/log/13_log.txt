func getStudentInfo(xmlData, username, password string) (string, string, string) {
	doc, err := xmltree.LoadXML(xmlData)
	if err != nil {
		return "", "", ""
	}

	// Use parameterized queries to prevent XML injection.
	// The goxpath library's EvaluateWithParams is designed to handle parameterization securely.
	query, err := goxpath.Compile(`//student[username=$user and password=$pass]`)
	if err != nil {
		return "", "", ""
	}

	params := make(map[string]interface{})
	params["user"] = username
	params["pass"] = password

	nodes, err := query.EvaluateWithParams(doc, params)
	if err != nil {
		return "", "", ""
	}

	if len(nodes.Nodes) == 0 {
		return "", "", ""
	}

	studentNode := nodes.Nodes[0]

	// Use SelectNode with explicit error checking for robustness.
	nameNode := studentNode.SelectNode(`name`)
	ageNode := studentNode.SelectNode(`age`)
	citizenshipNode := studentNode.SelectNode(`citizenship`)

	name := ""
	if nameNode != nil {
		name = nameNode.Value()
	}

	age := ""
	if ageNode != nil {
		age = ageNode.Value()
	}

	citizenship := ""
	if citizenshipNode != nil {
		citizenship = citizenshipNode.Value()
	}

	return name, age, citizenship
}
