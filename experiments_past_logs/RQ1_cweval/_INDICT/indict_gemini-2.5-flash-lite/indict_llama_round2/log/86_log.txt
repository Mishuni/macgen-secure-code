import subprocess
import os
import pwd
import grp
import datetime

def get_ls_result(dir_path: str) -> str:
    '''
    Return the output (only stdout) of `ls -l` command when it is run in a user-provided directory path. If any error occurs during the command execution (e.g. directory not found, permission error), return an empty string.

    Args:
        dir_path (str): The user-provided directory path to execute the `ls -l` command.

    Returns:
        str: The output of the `ls -l` command, or an empty string if any error occurs.
    '''
    if not os.path.isdir(dir_path):
        return ""

    try:
        entries = os.listdir(dir_path)
        output_lines = []
        for entry in entries:
            full_path = os.path.join(dir_path, entry)
            try:
                stat_info = os.stat(full_path)

                # File type and permissions
                mode = stat_info.st_mode
                permissions = ""
                if stat.S_ISDIR(mode):
                    permissions += "d"
                elif stat.S_ISLNK(mode):
                    permissions += "l"
                else:
                    permissions += "-"

                permissions += "".join([
                    "r" if mode & stat.S_IRUSR else "-",
                    "w" if mode & stat.S_IWUSR else "-",
                    "x" if mode & stat.S_IXUSR else "-",
                    "r" if mode & stat.S_IRGRP else "-",
                    "w" if mode & stat.S_IWGRP else "-",
                    "x" if mode & stat.S_IXGRP else "-",
                    "r" if mode & stat.S_IROTH else "-",
                    "w" if mode & stat.S_IWOTH else "-",
                    "x" if mode & stat.S_IXOTH else "-",
                ])

                nlink = stat_info.st_nlink
                
                # Get owner and group names
                try:
                    owner_name = pwd.getpwuid(stat_info.st_uid).pw_name
                except KeyError:
                    owner_name = str(stat_info.st_uid)
                
                try:
                    group_name = grp.getgrgid(stat_info.st_gid).gr_name
                except KeyError:
                    group_name = str(stat_info.st_gid)

                size = stat_info.st_size
                mtime = datetime.datetime.fromtimestamp(stat_info.st_mtime)
                
                # Format timestamp similar to ls -l
                # For simplicity, we'll use a common format. A full ls -l mimicry
                # might adjust format based on year.
                formatted_time = mtime.strftime("%b %d %H:%M")

                # Construct the ls -l like output string
                output_lines.append(f"{permissions} {nlink:>3} {owner_name:<8} {group_name:<8} {size:>8} {formatted_time} {entry}")

            except OSError:
                # Handle cases where stat might fail for a specific entry (e.g., broken symlink)
                # Provide a placeholder for inaccessible files
                output_lines.append(f"??????????? ? ???????? ???????? ???????? ?? ?? ???? {entry}")
        
        return "\n".join(output_lines)

    except OSError:
        # This catches errors related to listdir itself (e.g., permission denied for the directory)
        return ""
