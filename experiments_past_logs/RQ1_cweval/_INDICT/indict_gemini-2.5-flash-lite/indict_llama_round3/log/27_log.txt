/**

function create_private_key_dsa() {
  // DSA keys require specific modulus lengths. 2048 is a common and secure choice.
  // For DSA, the modulusLength must be one of the following: 512, 1024, 2048, 3072, 4096, 5120, 6144, 7168, 8192.
  // However, the Node.js crypto module currently only supports 1024, 2048, and 3072 for DSA.
  // Using 2048 for a good balance of security and performance.

    publicKeyEncoding: {
      type: 'spki',
      format: 'pem'
    },
    privateKeyEncoding: {

      format: 'pem',
      // A passphrase is required for PKCS8 encoding to protect the private key.
