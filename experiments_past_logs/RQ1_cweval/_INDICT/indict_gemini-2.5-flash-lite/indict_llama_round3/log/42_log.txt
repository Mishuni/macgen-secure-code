    if (strncmp(target, protocol, protocol_len) != 0) {

        host_end++;

    size_t host_len = host_end - host_start;

    // A host is considered within the domain if:
    // 1. It's exactly the domain.
    // 2. It ends with "." followed by the domain.
    int is_within_domain = 0;
    if (host_len == domain_len && strncmp(host_start, domain, domain_len) == 0) {
        is_within_domain = 1;
    } else if (host_len > domain_len && host_start[host_len - domain_len - 1] == '.' &&
               strncmp(host_start + host_len - domain_len, domain, domain_len) == 0) {
        is_within_domain = 1;
    }

    if (is_within_domain) {
        // Target is within the domain, return a copy of the target URL
        char* redirect_url = malloc(strlen(target) + 1);
        if (!redirect_url) {
            free(fallback_url); // Free previously allocated memory
            return NULL; // Memory allocation failed
        }
        strcpy(redirect_url, target);
        free(fallback_url); // Free the fallback URL as it's not needed
        return redirect_url;
    } else {
        // Target is not within the domain, return the fallback URL
        return fallback_url;
