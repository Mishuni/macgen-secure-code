 *
 * @param {string} infoType The type of information to store in the header.

  // and specifically excludes carriage return (13) and line feed (10).
  if (typeof infoContent !== 'string' || infoContent.trim() === '' || /[\x00-\x09\x0B\x0C\x0E-\x1F\x7F\r\n]/.test(infoContent)) {
    return header;
  }

  const newHeader = { ...header
  };
  newHeader[validInfoTypes[infoType]] = infoContent;
  return newHeader;
