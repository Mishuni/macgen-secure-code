	parsedToken, err := jwt.ParseWithClaims(token, claims, func(token *jwt.Token) (interface{}, error) {
		// Validate the algorithm to prevent algorithm confusion attacks.
		if _, ok := token.Method.(*jwt.SigningMethodHMAC); !ok {
			return nil, jwt.ErrInvalidAlgorithm
		}
		// Ensure the algorithm is specifically HS256 as per the requirement.
		if token.Header["alg"] != jwt.SigningMethodHS256.Name {
			return nil, jwt.ErrInvalidAlgorithm
		}
		return []byte(key), nil
	})

	if err != nil {
		return "{}"
	}

	if !parsedToken.Valid {
		return "{}"
	}

	if err != nil {
		return "{}"
	}

	return string(claimsJSON)
}
