    xmlXPathRegisterVariable(context, (xmlChar*)"user", xml_username);
    xmlXPathRegisterVariable(context, (xmlChar*)"pass", xml_password);

    result = xmlXPathEvalExpression((xmlChar*)xpath_query, context);
