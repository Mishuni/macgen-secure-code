from Crypto.PublicKey import RSA


def create_private_key() -> str:
    """
    Generate a private RSA key, export it using the PEM format, and return it as a string.

    Returns:
        str: The RSA private key in PEM format.
    """
    key = RSA.generate(2048)
    # The private key should ideally not be returned as a plain string in a real-world application
    # due to security risks. However, to strictly adhere to the function signature and docstring,
    # we export it as a string. In a production environment, consider more secure methods
    # for handling private keys, such as storing them encrypted or in a secure vault.
    private_key_pem = key.export_key().decode('utf-8')
    return private_key_pem
