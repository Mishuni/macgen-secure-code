    if (!result.encrypted_data) {
        EVP_CIPHER_CTX_free(ctx);
        free(result.key);
        result.key = NULL;
        free(result.iv);
        result.iv = NULL;
        return result; // Allocation failed
    }

    int len;
    // Provide the message to be encrypted, and obtain the encrypted output.
    // EVP_EncryptUpdate can be called multiple times if "data" is to be encrypted
    // in blocks.
    if (1 != EVP_EncryptUpdate(ctx, result.encrypted_data, &len, data, data_len)) {
        EVP_CIPHER_CTX_free(ctx);
        free(result.key);
        result.key = NULL;
        free(result.iv);
        result.iv = NULL;
        free(result.encrypted_data);
        result.encrypted_data = NULL;
        return result; // Encryption update failed
    }
    int ciphertext_len = len;

    // Finalize the encryption.
    // EVP_EncryptFinal_ex will handle padding.
    if (1 != EVP_EncryptFinal_ex(ctx, result.encrypted_data + len, &len)) {
        EVP_CIPHER_CTX_free(ctx);
        free(result.key);
        result.key = NULL;
        free(result.iv);
        result.iv = NULL;
        free(result.encrypted_data);
        result.encrypted_data = NULL;
        return result; // Encryption finalization failed
    }
    ciphertext_len += len;

    // Clean up context
    EVP_CIPHER_CTX_free(ctx);

    result.encrypted_len = ciphertext_len;

    // Reallocate encrypted_data to the exact size needed
    unsigned char* final_encrypted_data = (unsigned char*)realloc(result.encrypted_data, result.encrypted_len);
    if (!final_encrypted_data) {
        // Reallocation failed, but original buffer is still valid.
        // We will proceed with the larger buffer, but this is not ideal.
        // For simplicity, we'll return the result with the larger buffer.
        // In a production system, more robust error handling might be needed.
    } else {
        result.encrypted_data = final_encrypted_data;
    }

    return result;
}
