@param tarPath: the path to the tar file
@param destPath: the path to the destination folder

@return true if the extraction is successful; otherwise, false

func extractTarToPath(tarPath string, destPath string) bool {
	file, err := os.Open(tarPath)
	if err != nil {
		return false
	}
	defer file.Close()

	tarReader := tar.NewReader(file)

	// Ensure the destination directory exists
	if err := os.MkdirAll(destPath, 0755); err != nil {
		return false
	}
	cleanedDestPath, err := filepath.Abs(filepath.Clean(destPath))
	if err != nil {
		return false
	}

	for {
		header, err := tarReader.Next()

		if err == io.EOF {
			break // End of archive
		}
		if err != nil {
			// Handle corrupted tar file errors gracefully
			return false
		}

		// Skip empty header names
		if header.Name == "" {
			continue
		}

		// Sanitize and validate file name to prevent directory traversal and other attacks
		// Use filepath.Join to handle path separators correctly and then clean it.
		// This also helps in preventing directory traversal by ensuring the path is relative to destPath.
		relPath := filepath.Join(strings.Split(header.Name, string(filepath.Separator))...)
		targetPath := filepath.Join(cleanedDestPath, relPath)

		// Resolve symlinks and check for directory traversal
		absTargetPath, err := filepath.Abs(targetPath)
		if err != nil {
			return false
		}

		// Ensure the target path is within the destination directory
		if !strings.HasPrefix(absTargetPath, cleanedDestPath) {
			return false
		}

		info := header.FileInfo()
		if info.IsDir() {
			if err := os.MkdirAll(targetPath, info.Mode()); err != nil {
				return false
			}
			continue
		}

		// Handle symbolic links
		if header.Typeflag == tar.TypeSymlink {
			// Resolve the symlink target relative to the directory containing the symlink
			linkTarget := header.Linkname
			if !filepath.IsAbs(linkTarget) {
				linkTarget = filepath.Join(filepath.Dir(targetPath), linkTarget)
			}
			absLinkTarget, err := filepath.Abs(linkTarget)
			if err != nil {
				return false
			}

			// Ensure the symlink target is also within the destination directory
			if !strings.HasPrefix(absLinkTarget, cleanedDestPath) {
				return false
			}

			// Create the symlink
			if err := os.Symlink(header.Linkname, targetPath); err != nil {
				return false
			}
			continue
		}

		// Ensure parent directory exists for files
		if err := os.MkdirAll(filepath.Dir(targetPath), 0755); err != nil {
			return false
		}

		outFile, err := os.OpenFile(targetPath, os.O_WRONLY|os.O_CREATE|os.O_TRUNC, info.Mode())
		if err != nil {
			return false
		}

		if _, err := io.Copy(outFile, tarReader); err != nil {
			outFile.Close()
			return false
		}
		outFile.Close()
	}

	return true
}
