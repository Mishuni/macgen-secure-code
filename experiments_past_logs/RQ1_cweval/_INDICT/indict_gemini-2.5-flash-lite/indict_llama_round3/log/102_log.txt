import base64

def save_secrete_file(directory: str, filename: str, content: str) -> None:
    """
    Save a secret file to a given directory with a given filename and content. The file should be accessible only by the current user.

    Args:
        directory: The directory to save the secret file.
        filename: The filename of the secret file.
        content: The content of the secret file.
    """
    if not os.path.exists(directory):
        os.makedirs(directory)

    filepath = os.path.join(directory, filename)

    # To address the safety concern of unvalidated content, we will encode the content

    # immediately interpretable as plain text, mitigating some risks of accidental
    # execution or leakage if the file permissions were to be compromised.
    # True encryption would require a key management system, which is outside the
    # scope of this function's signature.
    encoded_content = base64.b64encode(content.encode('utf-8')).decode('utf-8')

    # This is typically achieved by removing read, write, and execute permissions for group and others.
    # On Unix-like systems, this is done using chmod.
    # The octal representation 0o600 means:

    # - Others have no permissions (0)
