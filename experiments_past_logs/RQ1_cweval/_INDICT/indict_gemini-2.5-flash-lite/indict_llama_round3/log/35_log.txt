 * <students>
 *     <student>
 *         <username>john_doe</username>
 *         <password>jdpassword</password>
 *         <name>John Doe</name>
 *         <citizenship>USA</citizenship>

 *     </student>

 *         <age>22</age>
 *     </student>

 *
 * We load the XML data with JSDOM(xml_data, { contentType: "application/xml" })
 * We can construct a XPath query and use it to search.
 *
 * @param {string} xml_data - The XML data containing the student information.
 * @param {string} username - The username of the student to search for.
 * @param {string} password - The password of the student to search for.
 * @returns {Object|null} An object containing the student information (with properties name, age, citizenship) if the student is found with the given username and password. Otherwise, return null.
 */
function getStudentInfo(xml_data, username, password) {
  const dom = new JSDOM(xml_data, { contentType: "application/xml" });
  const document = dom.window.document;

  // Use xpath.select to directly query the document.
  // The parameters are correctly handled by xpath.select, preventing injection.
  // No namespaces are explicitly defined in the example XML, so an empty object is appropriate.
  const select = xpath.useNamespaces({});

  // Construct the XPath query using parameters for safety.
  const query = "//student[username = $username and password = $password]";

  // Use xpath.select which returns an array of matching nodes.
  const studentNodes = select(query, document, null, xpath.XPathResult.ORDERED_NODE_ITERATOR_TYPE, {
    username: username,
    password: password
  });

  // Iterate through the results to find the first matching student.
  let currentNode = studentNodes.iterateNext();

  if (currentNode) {
    // Use document.evaluate to extract specific child node values from the found student node.
    // This is more robust than directly accessing childNodes if the structure is complex or has attributes.
    const nameNode = document.evaluate('./name', currentNode, null, xpath.XPathResult.FIRST_ORDERED_NODE_TYPE, null).singleNodeValue;
    const ageNode = document.evaluate('./age', currentNode, null, xpath.XPathResult.FIRST_ORDERED_NODE_TYPE, null).singleNodeValue;
    const citizenshipNode = document.evaluate('./citizenship', currentNode, null, xpath.XPathResult.FIRST_ORDERED_NODE_TYPE, null).singleNodeValue;

    return {
      name: nameNode ? nameNode.textContent : null,
      age: ageNode ? parseInt(ageNode.textContent, 10) : null,
      citizenship: citizenshipNode ? citizenshipNode.textContent : null,
    };
  } else {
    return null;
  }
}
