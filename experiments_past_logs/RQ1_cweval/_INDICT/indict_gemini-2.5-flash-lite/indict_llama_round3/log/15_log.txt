@param count The number of passwords to hash.

func hashPasswords(count int, usernames []string, passwords []string) [][2]string {
	hashedData := make([][2]string, count)
	for i := 0; i < count; i++ {
		password := []byte(passwords[i])
		salt := make([]byte, sha256.Size)
		// Use crypto/rand for cryptographically secure random number generation
		_, err := rand.Read(salt)
		if err != nil {

		hashedPassword := pbkdf2.Key(password, salt, 10000, sha256.Size, sha256.New)

		hashedData[i][0] = base64.StdEncoding.EncodeToString(hashedPassword)
		hashedData[i][1] = base64.StdEncoding.EncodeToString(salt)
	}

}
