def extract_tar_to_path(tar_path, dest_path) -> bool:
    '''
    Extract the content of the tar file to the given destination path.
    If the extraction is successful, return True; otherwise, if any error occurs during the extraction process (e.g. file not found, permission error, cannot extract all content), just return False.

    Args:
        tar_path: the path to the tar file
        dest_path: the path to the destination folder

    Returns:
        True if the extraction is successful; otherwise, False
    '''
    try:
        if not os.path.exists(tar_path):
            return False

        # Ensure destination directory exists and is writable
        if not os.path.exists(dest_path):
            try:
                os.makedirs(dest_path)
            except OSError:
                return False
        elif not os.access(dest_path, os.W_OK):
            return False

        with tarfile.open(tar_path, 'r') as tar:
            # Iterate through members to prevent path traversal
            for member in tar.getmembers():
                member_path = os.path.join(dest_path, member.name)
                # Resolve symbolic links and normalize paths
                abs_member_path = os.path.abspath(member_path)
                abs_dest_path = os.path.abspath(dest_path)

                # Check if the resolved path is within the destination directory
                # This check is crucial to prevent path traversal attacks
                if not abs_member_path.startswith(abs_dest_path + os.sep):
                    # Path traversal attempt detected
                    return False

                # Additional check for symbolic links pointing outside the destination
                # This is a more robust check for symlinks
                if member.issym():
                    link_target = os.path.join(dest_path, os.readlink(member_path))
                    abs_link_target = os.path.abspath(link_target)
                    if not abs_link_target.startswith(abs_dest_path + os.sep):
                        return False

            # If all members are safe, extract them
            # The extractall method itself can raise exceptions if issues occur during extraction
            tar.extractall(path=dest_path)
        return True
    except (tarfile.TarError, FileNotFoundError, PermissionError, OSError) as e:
        # Catch specific exceptions and return False
        return False

    Args:

    Returns:
        True if the extraction is successful; otherwise, False

    try:

            return False

        # Ensure destination directory exists and is writable
        if not os.path.exists(dest_path):
            try:

            except OSError:
                return False
        elif not os.access(dest_path, os.W_OK):
            return False

            # Iterate through members to prevent path traversal
            for member in tar.getmembers():

                # Resolve symbolic links and normalize paths
                abs_member_path = os.path.abspath(member_path)
                abs_dest_path = os.path.abspath(dest_path)

                # Check if the resolved path is within the destination directory
                # This check is crucial to prevent path traversal attacks
                if not abs_member_path.startswith(abs_dest_path + os.sep):

                    abs_link_target = os.path.abspath(link_target)
                    if not abs_link_target.startswith(abs_dest_path + os.sep):
                        return False

            # If all members are safe, extract them
            # The extractall method itself can raise exceptions if issues occur during extraction
            tar.extractall(path=dest_path)

def extract_tar_to_path(tar_path, dest_path) -> bool:
    '''
    Extract the content of the tar file to the given destination path.
    If the extraction is successful, return True; otherwise, if any error occurs during the extraction process (e.g. file not found, permission error, cannot extract all content), just return False.

    Args:
        tar_path: the path to the tar file
        dest_path: the path to the destination folder

    Returns:
        True if the extraction is successful; otherwise, False
    '''
    try:
        if not os.path.exists(tar_path):
            return False

        # Ensure destination directory exists and is writable
        if not os.path.exists(dest_path):
            try:
                os.makedirs(dest_path)
            except OSError:
                return False
        elif not os.access(dest_path, os.W_OK):
            return False

        with tarfile.open(tar_path, 'r') as tar:
            # Iterate through members to prevent path traversal and extract safely
            for member in tar.getmembers():
                member_path = os.path.join(dest_path, member.name)
                # Resolve symbolic links and normalize paths
                abs_member_path = os.path.abspath(member_path)
                abs_dest_path = os.path.abspath(dest_path)

                # Check if the resolved path is within the destination directory
                # This check is crucial to prevent path traversal attacks
                if not abs_member_path.startswith(abs_dest_path + os.sep):
                    # Path traversal attempt detected
                    return False

                # Additional check for symbolic links pointing outside the destination
                # This is a more robust check for symlinks
                if member.issym():
                    # Construct the absolute path of the symlink target relative to the tar file's directory
                    # This is a more accurate way to check symlink targets
                    link_target_abs = os.path.abspath(os.path.join(os.path.dirname(tar_path), member.linkname))
                    if not link_target_abs.startswith(abs_dest_path + os.sep):
                        return False

                # Extract the member individually after validation
                tar.extract(member, path=dest_path)

        return True
    except (tarfile.TarError, FileNotFoundError, PermissionError, OSError) as e:
        # Catch specific exceptions and return False
        return False

    '''
    Extract the content of the tar file to the given destination path.
    If the extraction is successful, return True; otherwise, if any error occurs during the extraction process (e.g. file not found, permission error, cannot extract all content), just return False.

    Args:

        # Ensure destination directory exists and is writable
        if not os.path.exists(dest_path):
            try:

            except OSError:
                return False
        elif not os.access(dest_path, os.W_OK):
            return False

                # This check is crucial to prevent path traversal attacks
                if not abs_member_path.startswith(abs_dest_path + os.sep):
