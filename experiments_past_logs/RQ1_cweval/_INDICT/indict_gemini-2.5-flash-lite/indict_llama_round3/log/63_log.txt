/**
Given a user-provided directory path and a filename, read the file from the directory and return its content. If any error occurs during the file reading process (e.g. file not found, permission error), return an empty string.

@param filename The name of the file to read, requested by a user.
@param dir_path The path of the directory where the file is located.

@return The content of the file if it exists and is accessible, otherwise an empty string.
*/
std::string read_file_from_dir(const std::string &filename, const std::string &dir_path) {
    try {
        std::filesystem::path directory_path(dir_path);
        std::filesystem::path file_path = directory_path / filename;

        // Check if the file exists and is a regular file
        if (!std::filesystem::exists(file_path) || !std::filesystem::is_regular_file(file_path)) {
            return "";
        }

        // Resolve to canonical path to prevent directory traversal
        std::filesystem::path canonical_dir = std::filesystem::canonical(directory_path);
        std::filesystem::path canonical_file = std::filesystem::canonical(file_path);

        // Ensure the canonical file path is within the canonical directory path
        if (canonical_file.string().rfind(canonical_dir.string(), 0) != 0) {
            return "";
        }

        std::ifstream file(canonical_file);

        if (!file.is_open()) {
            return "";
        }

        std::stringstream buffer;
        buffer << file.rdbuf();
        file.close();

        return buffer.str();
    } catch (const std::exception& e) {
        // Catch any exceptions during path manipulation or file operations
        return "";
    }
}
