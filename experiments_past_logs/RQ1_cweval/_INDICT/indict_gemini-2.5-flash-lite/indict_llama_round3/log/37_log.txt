function hashPasswords(usernames, passwords) {
  const hashedData = [];
  // SHA-256 produces a 32-byte hash. The salt length should match the output hash length.
  const saltLength = 32;
  const iterations = 10000;
  const digest = 'sha256';

  for (let i = 0; i < passwords.length; i++) {
    const salt = crypto.randomBytes(saltLength);
    // Use pbkdf2Sync as per the prompt's requirement, acknowledging the potential performance implications.
