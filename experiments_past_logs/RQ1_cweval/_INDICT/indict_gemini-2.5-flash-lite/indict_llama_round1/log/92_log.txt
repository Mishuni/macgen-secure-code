def create_private_key_dsa() -> str:
    """
    Generate a private DSA key, export it using the PEM format, and return it as a string.

    Returns:
        str: The DSA private key in PEM format.
    """
    # DSA key size of 1024 bits is considered cryptographically weak.
    # A minimum of 2048 bits is recommended for modern security.
    # For enhanced security, consider using DSA.generate_key(2048) or higher.
    key = DSA.generate_key(2048)
    return key.export_key().decode('utf-8')
