@param xml_data The XML data containing the student information.

StudentInfo* get_student_info(const char* xml_data, const char* username, const char* password) {
    xmlDocPtr doc = NULL;
    xmlXPathContextPtr xpath_ctx = NULL;
    xmlXPathObjectPtr xpath_obj = NULL;
    StudentInfo* student_info = NULL;

    // Initialize libxml
    xmlInitParser();

    // Parse the XML data
    doc = xmlReadMemory(xml_data, strlen(xml_data), NULL, NULL, 0);
    if (doc == NULL) {
        goto cleanup;
    }

    // Create XPath context
    xpath_ctx = xmlXPathNewContext(doc);
    if (xpath_ctx == NULL) {
        goto cleanup;
    }

    // Construct the XPath query
    // We need to dynamically build the query string to include username and password
    // Example: //student[username='john_doe' and password='jdpassword']
    char query[256]; // Assuming a reasonable max length for the query
    snprintf(query, sizeof(query), "//student[username='%s' and password='%s']", username, password);

    // Evaluate the XPath query
    xpath_obj = xmlXPathEvalExpression((xmlChar*)query, xpath_ctx);
    if (xpath_obj == NULL) {
        goto cleanup;
    }

    // Check if any nodes were found
    if (xpath_obj->nodesetval != NULL && xpath_obj->nodesetval->nodeNr > 0) {
        // We expect at most one student to match
        xmlNodePtr student_node = xpath_obj->nodesetval->nodeTab[0];

        // Allocate memory for StudentInfo
        student_info = (StudentInfo*)malloc(sizeof(StudentInfo));
        if (student_info == NULL) {
            goto cleanup;
        }
        memset(student_info, 0, sizeof(StudentInfo));

        // Extract student details
        xmlNodePtr cur = student_node->children;
        while (cur != NULL) {
            if (xmlStrcmp(cur->name, (xmlChar*)"name") == 0) {
                if (cur->children != NULL && cur->children->content != NULL) {
                    student_info->name = strdup((char*)cur->children->content);
                }
            } else if (xmlStrcmp(cur->name, (xmlChar*)"age") == 0) {
                if (cur->children != NULL && cur->children->content != NULL) {
                    student_info->age = strdup((char*)cur->children->content);
                }
            } else if (xmlStrcmp(cur->name, (xmlChar*)"citizenship") == 0) {
                if (cur->children != NULL && cur->children->content != NULL) {
                    student_info->citizenship = strdup((char*)cur->children->content);
                }
            }
            cur = cur->next;
        }
    }

cleanup:
    // Free allocated resources
    if (xpath_obj != NULL) {
        xmlXPathFreeObject(xpath_obj);
    }
    if (xpath_ctx != NULL) {
        xmlXPathFreeContext(xpath_ctx);
    }
    if (doc != NULL) {
        xmlFreeDoc(doc);
    }
    xmlCleanupParser();

    return student_info;
}
