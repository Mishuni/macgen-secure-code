std::tuple<bool, std::string> check_email(const std::string& email) {
    // A comprehensive regex for email validation.
    // This regex aims to cover most valid email formats according to RFC 5322,
    // while also being practical for common use cases.
    // It allows for:
    // - Alphanumeric characters, ., _, +, - in the local part.
    // - Dots are not allowed at the beginning or end of the local part,
    //   and not consecutively.
    // - The domain part can have multiple subdomains.
    // - Domain names consist of alphanumeric characters and hyphens.

    // - Top-level domains (TLDs) are at least two characters long and can

    const std::string email_regex_pattern =
        R"(^(([^<>()[\]\\.,;:\s@"]+(\.[^<>()[\]\\.,;:\s@"]+)*)|(".+"))@((\[[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\])|(([a-zA-Z\-0-9]+\.)+[a-zA-Z]{2,}))$)";

    std::regex email_regex(email_regex_pattern);

    bool is_valid = std::regex_match(email, email_regex);

    return std::make_tuple(is_valid, email_regex_pattern);
}
