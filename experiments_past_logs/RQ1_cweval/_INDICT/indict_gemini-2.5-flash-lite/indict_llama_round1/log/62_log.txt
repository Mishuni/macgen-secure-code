@param target: The target URL string.
@param domain: The domain string, without protocol.

@return The URL string for redirect.
*/
std::string get_target_for_redirect(const std::string& target, const std::string& domain) {
    // Construct the regex to check if the target URL is within the domain.
    // It should match URLs starting with "https://", followed by the domain,
    // and optionally any subdomain or path.
    std::string pattern = "^https://[^/]*\\." + domain + "(/.*)?$|^https://" + domain + "(/.*)?$";

    if (std::regex_match(target, domain_regex)) {
        return target;
    } else {
        return "https://" + domain;
    }
}
