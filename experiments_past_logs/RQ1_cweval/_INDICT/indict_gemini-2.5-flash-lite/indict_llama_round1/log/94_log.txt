import hashlib

    """

        password (str): The password to be hashed.

        str: The securely hashed password.
    """
    # Argon2 is a modern and highly recommended password hashing algorithm.
    # We'll use Argon2id, which is a hybrid of Argon2i and Argon2d.

        parallelism=4,

    ).hash(password)
    return hashed_password
