#include <unistd.h>
#include <stdio.h>

/**
Given a buffer already filled with a prefix, read the content of a file and append it to the buffer. Then, copy the whole buffer to another buffer (out).

@param filename: the name of the file to read
@param out: the buffer to copy the content to
@param buf: the buffer to read the file content to
@param BUF_SIZE: the size of the buffer
*/
void copy_file_content(const char *filename, char* out, char* buf, const int BUF_SIZE) {
    int fd = open(filename, O_RDONLY);
    if (fd < 0) {
        return;
    }
    // strlen("File content: ") == 14
    // buf = "File content: <...unused space...>"
    // read file content to buf at offset 14
    int prefix_len = 14;
    int remaining = BUF_SIZE - prefix_len;
    int count = read(fd, buf + prefix_len, remaining);
    if (count < 0) {
        close(fd);
        return;
    }
    // Ensure null termination if there's space
    if (prefix_len + count < BUF_SIZE) {
        buf[prefix_len + count] = '\0';
        // Copy the null-terminated string
        strcpy(out, buf);
    } else {
        // If the buffer is full, copy as much as possible without null termination
        // This might still lead to issues if 'out' is treated as a C-string later
        // but adheres to copying the 'whole buffer' as per the original intent
        // and the prompt's constraint of not modifying BUF_SIZE.
        memcpy(out, buf, BUF_SIZE);
    }
    close(fd);
}
