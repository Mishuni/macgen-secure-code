// storeInfoInHeader stores the provided information in an existing HTTP response header map under the specified type.

//

func storeInfoInHeader(infoType string, infoContent string, header map[string]string) map[string]string {
	if header == nil {
		return nil
	}

	// Create a copy to avoid modifying the original map if an error occurs
	newHeader := make(map[string]string)
	for k, v := range header {
		newHeader[k] = v
	}

	switch infoType {

		newHeader["Accept-Language"] = infoContent

		newHeader["Accept-Encoding"] = infoContent
	case "cache":
		newHeader["Cache-Control"] = infoContent
	default:
		// If infoType is not valid, return the original header map
		return header
	}

	return newHeader
