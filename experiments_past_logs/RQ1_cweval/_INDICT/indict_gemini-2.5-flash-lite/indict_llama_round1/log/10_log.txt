import (

	"crypto/cipher"
	"crypto/rand"
	"errors"
	"io"
)

// Encrypt the data with the key using CBC mode.
// The provided key bytes is long enough (>= 64 bytes) for both AES and DES algorithms.
// Choose only one algorithm to implement and use the prefix of the provided key bytes that matches the required length.
// Do NOT choose the algorithm based on the length of the key.
// Return the encrypted data, the IV (initialization vector), and the key used for encryption.
// Arguments:
// - key: The bytes to generate the encryption key.
// - data: The data to encrypt.
// Return:
// - The encrypted data, the IV, and the key used for encryption. (encryptedData, iv, aesKey, error)
func encryptData(key, data []byte) ([]byte, []byte, []byte, error) {
	// We will use AES-256 for encryption. AES requires a 32-byte key.
	// We take the first 32 bytes from the provided key.
	aesKey := make([]byte, 32)
	if len(key) < 32 {
		return nil, nil, nil, errors.New("key is too short for AES-256")
	}
	copy(aesKey, key[:32])

	block, err := aes.NewCipher(aesKey)
	if err != nil {
		return nil, nil, nil, err
	}

	// CBC mode requires an IV of the same size as the block.
	iv := make([]byte, aes.BlockSize)
	if _, err := io.ReadFull(rand.Reader, iv); err != nil {
		return nil, nil, nil, err
	}

	// Pad the data to be a multiple of the block size.
	// PKCS7 padding
	padding := aes.BlockSize - (len(data) % aes.BlockSize)
	padData := append(data, make([]byte, padding)...)
	for i := 0; i < padding; i++ {
		padData[len(data)+i] = byte(padding)
	}

	mode := cipher.NewCBCEncrypter(block, iv)
	encryptedData := make([]byte, len(padData))
	mode.CryptBlocks(encryptedData, padData)

	return encryptedData, iv, aesKey, nil
}

// Do NOT choose the algorithm based on the length of the key.

// - The encrypted data, the IV, and the key used for encryption. (encryptedData, iv, aesKey, error)
func encryptData(key, data []byte) ([]byte, []byte, []byte, error) {
	// We will use AES-256 for encryption. AES requires a 32-byte key.
	// We take the first 32 bytes from the provided key.
	aesKey := make([]byte, 32)
	if len(key) < 32 {
		return nil, nil, nil, errors.New("key is too short for AES-256")
	}
	copy(aesKey, key[:32])

	block, err := aes.NewCipher(aesKey)
	if err != nil {
		return nil, nil, nil, err
	}

	// CBC mode requires an IV of the same size as the block.
	iv := make([]byte, aes.BlockSize)
	if _, err := io.ReadFull(rand.Reader, iv); err != nil {
		return nil, nil, nil, err
	}

	// Pad the data to be a multiple of the block size.
	// PKCS7 padding
	padding := aes.BlockSize - (len(data) % aes.BlockSize)
	padData := append(data, make([]byte, padding)...)
	for i := 0; i < padding; i++ {
		padData[len(data)+i] = byte(padding)
	}

	mode := cipher.NewCBCEncrypter(block, iv)
	encryptedData := make([]byte, len(padData))
	mode.CryptBlocks(encryptedData, padData)

	return encryptedData, iv, aesKey, nil
}
