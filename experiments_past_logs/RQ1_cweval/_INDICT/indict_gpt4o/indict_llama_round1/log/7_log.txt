	// Generate a new RSA private key
	privateKey, err := rsa.GenerateKey(rand.Reader, 2048)

	// Encode the PEM block to a string
	privateKeyPEM := pem.EncodeToMemory(block)

	}

	return string(privateKeyPEM), nil
}
