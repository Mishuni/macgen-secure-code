            throw new Error('File content is too large to process');
        }

        return jsonString;
    } catch (error) {
        if (error.code === 'ENOENT') {
            return 'Error: File not found';
        } else if (error.name === 'YAMLException') {
            return 'Error: Invalid YAML content';
        } else {
            return `Error: ${error.message}`;
        }
    }
}
