    const infoTypeMap = {
        'lang': 'Accept-Language',
        'encoding': 'Accept-Encoding',
        'cache': 'Cache-Control'
    };

        const headerKey = infoTypeMap[infoType];

                break;

                // Accept-Encoding: e.g., "gzip, deflate, br"

                // Cache-Control: e.g., "no-cache, no-store, must-revalidate"
