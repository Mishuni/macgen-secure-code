    if (dir_path[dir_len - 1] == '/') {

    }

    FILE *file = fopen(resolved_path, "r");
    if (file == NULL) {
        return strdup("");
    }

    // Determine file size
    if (fseek(file, 0, SEEK_END) != 0) {
        fclose(file);
        return strdup("");
    }
    long file_size = ftell(file);
    if (file_size < 0) {
        fclose(file);
        return strdup("");
    }
    rewind(file);

    // Allocate memory for file content
    char *content = (char *)malloc(file_size + 1);
    if (content == NULL) {
        fclose(file);
        return strdup("");

    // Read file content
    size_t read_size = fread(content, 1, file_size, file);

    return content;
}

    if (fseek(file, 0, SEEK_END) != 0) {
        fclose(file);
        return strdup("");
    }
    long file_size = ftell(file);
    if (file_size < 0) {
        fclose(file);
        return strdup("");

    rewind(file);

    // Allocate memory for file content
    char *content = (char *)malloc(file_size + 1);
    if (content == NULL) {
        fclose(file);
        return strdup("");
    }

    // Read file content
    size_t read_size = fread(content, 1, file_size, file);
    fclose(file);

    if (read_size != file_size) {
        free(content);
        return strdup("");
    }

    content[file_size] = '\0';
    return content;
