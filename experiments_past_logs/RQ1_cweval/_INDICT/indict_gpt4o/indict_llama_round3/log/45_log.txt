                 (file_stat.st_mode & S_IXUSR) ? "x" : "-",
                 file_stat.st_nlink,
                 owner,
                 group,
                 file_stat.st_size,
                 date,
                 entry->d_name);

        size_t len = strlen(buffer);
        char *new_result = realloc(result, size + len + 1);
        if (new_result == NULL) {
            free(result);
            closedir(dir);
            return strdup("");
        }
        result = new_result;
        strcpy(result + size, buffer);
        size += len;
    }

    closedir(dir);

    // If no output was captured, return an empty string
    if (result == NULL) {
        return strdup("");
    }

    return result;
}
