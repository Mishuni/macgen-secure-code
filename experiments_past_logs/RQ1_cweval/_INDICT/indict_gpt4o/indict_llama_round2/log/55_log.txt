        fclose(file);
        unlink(template);
        return NULL;
    }
    if (fclose(file) == EOF) {
        unlink(template);
        return NULL;
    }

    // Return a dynamically allocated string containing the temporary file name
    char *filename = (char *)malloc(strlen(template) + 1);
    if (filename == NULL) {
        unlink(template);
        return NULL;
    }
    strcpy(filename, template);

    return filename;
}
