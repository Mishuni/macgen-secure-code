    closedir(dir);

    // Sort the file names alphabetically (similar to ls -l)
    std::sort(files.begin(), files.end());

    // Construct the output similar to 'ls -l'
    std::string result;
    for (const auto& file : files) {
        if (file == "." || file == "..") continue;  // Skip current and parent directories
        result += file + "\n";
    }

    return result.empty() ? "" : result.substr(0, result.size() - 1);  // Remove the last newline character
}

- **Sanitizing Input**: The function checks if the directory can be opened, preventing invalid paths from causing issues.
- **Error Handling**: If the directory cannot be opened (e.g., it doesn't exist or there's a permission issue), an empty string is returned.
- **Output Construction**: The function constructs output similar to `ls -l` by listing files alphabetically and formatting them appropriately.
