HeaderDict* store_info_in_header(const char *info_type, const char *info_content, HeaderDict *header) {
    // Check if info_type is valid
    if (strcmp(info_type, "lang") != 0 && strcmp(info_type, "encoding") != 0 && strcmp(info_type, "cache") != 0) {
        return header;
    }

    // Determine the corresponding key based on info_type
    const char *key;
    if (strcmp(info_type, "lang") == 0) {
        key = "Accept-Language";
    } else if (strcmp(info_type, "encoding") == 0) {
        key = "Accept-Encoding";
    } else { // cache case
        key = "Cache-Control";
    }

    // Check if the key already exists in the header
    int index = find_header_index(header, key);

    if (index != -1) {
        // Ensure info_content is not NULL and within bounds
        if (info_content != NULL && strlen(info_content) < sizeof(HeaderEntry::value)) {
            strncpy(header->entries[index].value, info_content, sizeof(HeaderEntry::value));
            header->entries[index].value[sizeof(HeaderEntry::value)-1] = '\0';
        }
    } else {
        // If the key does not exist and there's space, add a new entry
        if (header->count < HEADER_SIZE) {
            // Ensure info_content is not NULL and within bounds
