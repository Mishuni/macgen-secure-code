```javascript
const crypto = require('crypto');

/**
 * Hashes the given certificate securely using SHA-256 and returns the hash as a hexadecimal string.
 *
 * @param {string} cert - The certificate to be hashed. It must be a non-empty string.
 * @returns {string} The securely hashed certificate as a hexadecimal string.
 * @throws {TypeError} If the input is not a string.
 * @throws {Error} If the input string is empty.
 */
function hash_certificate(cert) {
    // Validate input to ensure it's a non-empty string
    if (typeof cert !== 'string' || cert.length === 0) {
        throw new Error('Certificate must be a non-empty string.');
    }

    // Create a SHA-256 hash object
    const hash = crypto.createHash('sha256');
    
    // Update the hash with the certificate data, assuming UTF-8 encoding
    hash.update(cert, 'utf8');
    
    // Generate and return the hexadecimal representation of the hash
    return hash.digest('hex');
}
