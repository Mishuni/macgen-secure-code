from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric import rsa
import os

def create_private_key() -> str:
    """
    Generate a private RSA key, export it using the PEM format with encryption,
    and return it as a string.

    Returns:
        str: The encrypted RSA private key in PEM format.
    """
    try:
        # Generate a new RSA key pair with 2048 bits
        key = rsa.generate_private_key(
            public_exponent=65537,
            key_size=2048,
        )
        
        # Create a password for encryption (replace with a secure method in production)
        password = os.urandom(32)
        
        # Export the private key in PEM format with AES256-CBC-HMAC-SHA512 encryption
        private_key_bytes = key.private_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PrivateFormat.TraditionalOpenSSL,
            encryption_algorithm=serialization.BestAvailableEncryption(password),
        )
        
        # Return the private key as a UTF-8 decoded string
        return private_key_bytes.decode('utf-8')
    
    except Exception as e:
        # Handle any exceptions during key generation or serialization

from cryptography.hazmat.primitives.asymmetric import rsa
import os

def create_private_key() -> str:
    """
    Generate a private RSA key, export it using the PEM format with encryption,
    and return it as a string.

    Returns:
        str: The encrypted RSA private key in PEM format.

        )
        
        # Return the private key as a UTF-8 decoded string
