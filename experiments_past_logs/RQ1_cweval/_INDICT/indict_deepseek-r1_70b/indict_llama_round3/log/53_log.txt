    }

    size_t encrypted_len = data_len;
    unsigned char* encrypted_data = malloc(encrypted_len);
    if(!encrypted_data) {
        return (encryption_result){NULL, 0, NULL, NULL};
    }

    unsigned char iv[AES_BLOCK_SIZE];
    RAND_bytes(iv, AES_BLOCK_SIZE);

    AES_cbc_encrypt(data, encrypted_data, data_len, &aes_key, iv, AES_ENCRYPT);

    // Copy the key for the result
    unsigned char* key_copy = malloc(key_len);
    if(!key_copy) {
        free(encrypted_data);
        return (encryption_result){NULL, 0, NULL, NULL};

    };

}
