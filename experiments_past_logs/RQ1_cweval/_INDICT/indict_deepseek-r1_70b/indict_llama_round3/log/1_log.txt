import (
	"fmt"
	"os"
	"path/filepath"
	"sync"
	"time"
)

// rateLimiter manages the rate at which file read operations can occur.
type rateLimiter struct {
	mu        sync.Mutex
	tokens    int
	capacity  int
	rate      time.Duration
	lastReset time.Time
}

// NewRateLimiter initializes a new rate limiter with the given capacity and refill rate.
func NewRateLimiter(capacity int, rate time.Duration) *rateLimiter {
	return &rateLimiter{
		tokens:    capacity,
		capacity:  capacity,
		rate:      rate,
		lastReset: time.Now(),
	}
}

// Allow checks if a request is allowed based on the current token count.
func (rl *rateLimiter) Allow() bool {
	rl.mu.Lock()
	defer rl.mu.Unlock()

	now := time.Now()
	elapsed := now.Sub(rl.lastReset)

	if elapsed > rl.rate {
		rl.tokens = rl.capacity
		rl.lastReset = now
	}

	if rl.tokens > 0 {
		rl.tokens--
		return true
	}
	return false
}

func readFileFromDir(filename, dirPath string) string {
	// Initialize the rate limiter with a capacity of 10 requests and a refill rate of 1 second.
	limiter := NewRateLimiter(10, time.Second)
	
	if !limiter.Allow() {
		return ""
	}
	
	filePath := fmt.Sprintf("%s/%s", dirPath, filename)
	cleanedPath := filepath.Clean(filePath)
	content, err := os.ReadFile(cleanedPath)
	if err != nil {
		return ""
	}
	return string(content)
}
