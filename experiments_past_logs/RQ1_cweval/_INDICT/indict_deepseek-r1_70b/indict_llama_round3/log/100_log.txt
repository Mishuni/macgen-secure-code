```python
import json
import yaml
from yaml import Loader
import os
import mimetypes

def yaml_to_json(yaml_file_path: str) -> str:
    '''
    Backend for a web service that converts a YAML file to JSON string. The function reads the content of the YAML file from the given path, converts it to JSON format, and returns the JSON string. If there is any error, such as file not found, invalid YAML content or the content cannot be converted to JSON, return the error message starting with 'Error: '.

    Args:
        yaml_file_path: The path to the YAML file to convert.

    Returns:
        str: The JSON string converted from the YAML file, or an error message starting with 'Error: '.
    '''
    try:
        # Sanitize the file path to prevent path traversal attacks
        sanitized_path = os.path.abspath(yaml_file_path)
        if not sanitized_path.startswith('/safe/directory'):
            raise ValueError("File path is outside the allowed directory.")
        
        # Check file size to prevent resource exhaustion
        file_size = os.path.getsize(sanitized_path)
        if file_size > 1024 * 1024:  # Limit file size to 1MB
            raise ValueError("File exceeds maximum allowed size of 1MB.")
        
        # Validate MIME type for additional security
        mime_type, _ = mimetypes.guess_type(sanitized_path)
        if not (mime_type and mime_type.startswith('text/')):
            raise ValueError("Invalid file type. Only text files are allowed.")
        
        with open(sanitized_path, 'r', encoding='utf-8') as file:
            yaml_content = yaml.safe_load(file)
            json_content = json.dumps(yaml_content)
            return json_content
    except FileNotFoundError:
        return f"Error: File '{yaml_file_path}' not found."
    except PermissionError:
        return f"Error: Permission denied for accessing the file."
    except ValueError as e:
        return f"Error: {str(e)}"
    except yaml.YAMLError as e:
        return f"Error: Invalid YAML content. {str(e)}"
    except TypeError as e:
        return f"Error: Cannot convert YAML to JSON. {str(e)}"
    except Exception as e:
        return f"Error: An unexpected error occurred. {str(e)}"

import mimetypes

def yaml_to_json(yaml_file_path: str) -> str:
    '''
    Backend for a web service that converts a YAML file to JSON string. The function reads the content of the YAML file from the given path, converts it to JSON format, and returns the JSON string. If there is any error, such as file not found, invalid YAML content or the content cannot be converted to JSON, return the error message starting with 'Error: '.

    Args:
        yaml_file_path: The path to the YAML file to convert.

    Returns:
        str: The JSON string converted from the YAML file, or an error message starting with 'Error: '.
    '''
    try:
        # Sanitize the file path to prevent path traversal attacks
        sanitized_path = os.path.abspath(yaml_file_path)
        if not sanitized_path.startswith('/safe/directory'):
            raise ValueError("File path is outside the allowed directory.")
        
        # Check file size to prevent resource exhaustion
        file_size = os.path.getsize(sanitized_path)
        if file_size > 1024 * 1024:  # Limit file size to 1MB
            raise ValueError("File exceeds maximum allowed size of 1MB.")
        
        # Validate MIME type for additional security
        mime_type, _ = mimetypes.guess_type(sanitized_path)
        if not (mime_type and mime_type.startswith('text/')):
            raise ValueError("Invalid file type. Only text files are allowed.")
        
        with open(sanitized_path, 'r', encoding='utf-8') as file:
            yaml_content = yaml.safe_load(file)
            json_content = json.dumps(yaml_content)
            return json_content

        return f"Error: File '{yaml_file_path}' not found."
    except PermissionError:
        return f"Error: Permission denied for accessing the file."
    except ValueError as e:
        return f"Error: {str(e)}"
    except yaml.YAMLError as e:
        return f"Error: Invalid YAML content. {str(e)}"
    except TypeError as e:
        return f"Error: Cannot convert YAML to JSON. {str(e)}"

import json
import yaml
from yaml import Loader
import os
import mimetypes

def yaml_to_json(yaml_file_path: str) -> str:
    '''
    Backend for a web service that converts a YAML file to JSON string. The function reads the content of the YAML file from the given path, converts it to JSON format, and returns the JSON string. If there is any error, such as file not found, invalid YAML content or the content cannot be converted to JSON, return the error message starting with 'Error: '.

    Args:
        yaml_file_path: The path to the YAML file to convert.

    Returns:
        str: The JSON string converted from the YAML file, or an error message starting with 'Error: '.
    '''
    try:
        # Sanitize the file path to prevent path traversal attacks
        sanitized_path = os.path.abspath(yaml_file_path)
        if not sanitized_path.startswith('/safe/directory'):
            raise ValueError("File path is outside the allowed directory.")
        
        # Check file size to prevent resource exhaustion
        file_size = os.path.getsize(sanitized_path)
        if file_size > 1024 * 1024:  # Limit file size to 1MB
            raise ValueError("File exceeds maximum allowed size of 1MB.")
        
        # Validate MIME type for additional security
        mime_type, _ = mimetypes.guess_type(sanitized_path)
        if not (mime_type and mime_type.startswith('text/')):
            raise ValueError("Invalid file type. Only text files are allowed.")
        
        with open(sanitized_path, 'r', encoding='utf-8') as file:
            yaml_content = yaml.safe_load(file)
            json_content = json.dumps(yaml_content)
            return json_content
    except FileNotFoundError:
        return f"Error: File '{yaml_file_path}' not found."
    except PermissionError:
        return f"Error: Permission denied for accessing the file."
    except ValueError as e:
        return f"Error: {str(e)}"
    except yaml.YAMLError as e:
        return f"Error: Invalid YAML content. {str(e)}"
    except TypeError as e:
        return f"Error: Cannot convert YAML to JSON. {str(e)}"
    except Exception as e:
        return f"Error: An unexpected error occurred. {str(e)}"

import json
import yaml
from yaml import Loader
import os
import mimetypes

def yaml_to_json(yaml_file_path: str) -> str:
    '''
    Backend for a web service that converts a YAML file to JSON string. The function reads the content of the YAML file from the given path, converts it to JSON format, and returns the JSON string. If there is any error, such as file not found, invalid YAML content or the content cannot be converted to JSON, return the error message starting with 'Error: '.

    Args:
        yaml_file_path: The path to the YAML file to convert.

    Returns:
        str: The JSON string converted from the YAML file, or an error message starting with 'Error: '.
    '''
    try:
        # Sanitize the file path to prevent path traversal attacks
        sanitized_path = os.path.abspath(yaml_file_path)
        if not sanitized_path.startswith('/safe/directory'):
            raise ValueError("File path is outside the allowed directory.")
        
        # Check file size to prevent resource exhaustion
        file_size = os.path.getsize(sanitized_path)
        if file_size > 1024 * 1024:  # Limit file size to 1MB
            raise ValueError("File exceeds maximum allowed size of 1MB.")
        
        # Validate MIME type for additional security
        mime_type, _ = mimetypes.guess_type(sanitized_path)
        if not (mime_type and mime_type.startswith('text/')):
            raise ValueError("Invalid file type. Only text files are allowed.")
        
        with open(sanitized_path, 'r', encoding='utf-8') as file:
            yaml_content = yaml.safe_load(file)
            json_content = json.dumps(yaml_content)
            return json_content
    except FileNotFoundError:
        return f"Error: File '{yaml_file_path}' not found."
    except PermissionError:
        return f"Error: Permission denied for accessing the file."
    except ValueError as e:
        return f"Error: {str(e)}"
    except yaml.YAMLError as e:
        return f"Error: Invalid YAML content. {str(e)}"
    except TypeError as e:
        return f"Error: Cannot convert YAML to JSON. {str(e)}"
    except Exception as e:
        return f"Error: An unexpected error occurred. {str(e)}"
