import (
	"crypto/aes"
	"crypto/cipher"
	"crypto/rand"
	"errors"
)

func encryptData(key, data []byte) ([]byte, []byte, []byte, error) {
	var aesKeyLength int
	switch {
	case len(key) >= 32:
		aesKeyLength = 32
	case len(key) >= 24:
		aesKeyLength = 24
	default:
		aesKeyLength = 16
	}

	if len(key) < aesKeyLength {
		return nil, nil, nil, errors.New("key must be at least 16 bytes long")
	}
	aesKey := key[:aesKeyLength]

	block, err := aes.NewCipher(aesKey)
	if err != nil {
		return nil, nil, nil, err
	}

	iv := make([]byte, block.BlockSize())
	if _, err := rand.Read(iv); err != nil {
		return nil, nil, nil, err
	}

	paddedData := make([]byte, ((len(data)+block.BlockSize()-1)/block.BlockSize())*block.BlockSize())
	copy(paddedData, data)

	mode := cipher.NewCBCEncrypter(block, iv)
	encryptedData := make([]byte, len(paddedData))
	mode.CryptBlocks(encryptedData, paddedData)

	return encryptedData, iv, aesKey, nil
}
```

### Approach
1. **Algorithm Selection**: Use AES as it is more secure than DES.
2. **Key Handling**: Ensure the key length is appropriate for AES (16, 24, or 32 bytes).
3. **IV Generation**: Generate a random IV of the same block size as AES.

```go
import (
	"crypto/aes"
	"crypto/cipher"
	"crypto/rand"
	"errors"
)

func encryptData(key, data []byte) ([]byte, []byte, []byte, error) {
	var aesKeyLength int
	switch {
	case len(key) >= 32:
		aesKeyLength = 32
	case len(key) >= 24:
		aesKeyLength = 24
	default:
		aesKeyLength = 16
	}

	if len(key) < aesKeyLength {
		return nil, nil, nil, errors.New("key must be at least 16 bytes long")
	}
	aesKey := key[:aesKeyLength]

	block, err := aes.NewCipher(aesKey)
	if err != nil {
		return nil, nil, nil, err
	}

	iv := make([]byte, block.BlockSize())
	if _, err := rand.Read(iv); err != nil {
		return nil, nil, nil, err
	}

	paddedData := make([]byte, ((len(data)+block.BlockSize()-1)/block.BlockSize())*block.BlockSize())
	copy(paddedData, data)

	mode := cipher.NewCBCEncrypter(block, iv)
	encryptedData := make([]byte, len(paddedData))
	mode.CryptBlocks(encryptedData, paddedData)

	return encryptedData, iv, aesKey, nil
}
```

### Explanation

1. **Algorithm Selection**: The function uses AES, which is a secure and modern encryption algorithm.
2. **Key Handling**: The function checks if the key length is appropriate for AES (16, 24, or 32 bytes) to ensure compatibility and security.
3. **IV Generation**: A random initialization vector (IV) is generated using `rand.Read` to ensure uniqueness and prevent predictable encryption outcomes.
4. **Data Padding**: The data is padded to fit the block size of AES, ensuring proper encryption without data loss.
5. **Encryption**: Using AES in CBC mode with the generated key and IV, the function encrypts the padded data securely.

This approach ensures that the data is encrypted using a secure algorithm with proper handling of keys, IV, and padding, making it robust against various security threats.

func encryptData(key, data []byte) ([]byte, []byte, []byte, error) {
