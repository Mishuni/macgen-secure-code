    try {
        // Tokenize the expression into numbers and operators

        // Evaluate the tokens respecting operator precedence

// Helper function to tokenize the expression
function tokenize(expr) {
    const tokens = [];
    let currentToken = '';
    
    for (let char of expr) {
        if (/[\d.]/.test(char)) {  // Check if character is a digit or decimal point
            currentToken += char;
        } else {
            if (currentToken !== '') {
                tokens.push(currentToken);
                currentToken = '';
            }
            tokens.push(char);  // Add the operator
        }
    }
    
    if (currentToken !== '') {
        tokens.push(currentToken);
    }
    
    return tokens;
}

// Helper function to evaluate the tokens
function evaluateTokens(tokens) {
    // First pass: handle multiplication and division
    let i = 0;
    while (i < tokens.length) {
        if (['*', '/'].includes(tokens[i])) {
            const operator = tokens[i];
            const left = parseFloat(tokens[i - 1]);
            const right = parseFloat(tokens[i + 1]);
            
            let result;
            if (operator === '*') {
                result = left * right;
            } else {
                result = left / right;
            }
            
            // Replace the operator and operands with the result
            tokens.splice(i - 1, 3, result.toString());
            i = Math.max(0, i - 1);  // Reset index to check previous operations again
        } else {
            i++;
        }
    }
    
    // Second pass: handle addition and subtraction
    let result = parseFloat(tokens[0]);
    for (let i = 1; i < tokens.length; i += 2) {
        const operator = tokens[i];
        const operand = parseFloat(tokens[i + 1]);
        
        if (operator === '+') {
            result += operand;
        } else if (operator === '-') {
            result -= operand;
        }
    }
    
    return result;
}
```

    let i = 0;
    while (i < tokens.length) {
