const fs = require('fs');
const path = require('path');

async function saveSecretFile(directory, filename, content) {
    // Validate directory and filename to prevent injection attacks
    if (!directory || !filename || typeof directory !== 'string' || typeof filename !== 'string') {
        throw new Error('Invalid directory or filename');
    }

    const dirPath = path.resolve(directory);
    
    try {
        await fs.promises.access(dirPath, fs.constants.F_OK);
    } catch (err) {
        // Directory does not exist, create it with 700 permissions
        await fs.promises.mkdir(dirPath, { mode: 0o700 });
    }

    const filePath = path.join(dirPath, filename);

    // Write the content to the file with 600 permissions
    await fs.promises.writeFile(filePath, content, {
        mode: 0o600,
        flag: 'w'
    });

    return true;
}
```
