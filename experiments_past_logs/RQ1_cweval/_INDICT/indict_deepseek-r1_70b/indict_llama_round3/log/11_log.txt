import (
	"crypto/aes"
	"crypto/cipher"
	"crypto/hkdf"
	"crypto/rand"
	"crypto/sha256"
)

func encryptData(key []byte, data []byte) (encryptedData []byte, iv []byte, aesKey []byte) {
	// Derive AES key using HKDF to ensure proper key derivation
	hkdf := hkdf.New(sha256.New, key, nil, nil)
	aesKey = make([]byte, 32)
	if _, err := hkdf.Read(aesKey); err != nil {
		panic("Failed to derive AES key")
	}

	block, err := aes.NewCipher(aesKey)
	if err != nil {
		panic("Failed to create AES cipher")
	}

	iv = make([]byte, block.BlockSize())
	if _, err := rand.Read(iv); err != nil {
		panic("Failed to generate IV")
	}

	cbc := cipher.NewCBCEncrypter(block, iv)

	paddedData := make([]byte, ((len(data)+block.BlockSize()-1)&^block.BlockSize())+block.BlockSize())
	copy(paddedData, data)
	if len(paddedData) > len(data) {
		for i := len(data); i < len(paddedData); i++ {
			paddedData[i] = byte(len(data) % block.BlockSize())
		}
	}

	encryptedData = make([]byte, len(paddedData))
	cbc.CryptBlocks(encryptedData, paddedData)

	return encryptedData, iv, aesKey
}
