package main

import (
    "fmt"
    "net/url"
)

func getTargetForRedirect(target string, domain string) string {
    // Parse the target URL
    targetURL, err := url.Parse(target)
    if err != nil {
        return fmt.Sprintf("https://%s", domain)
    }

    // Parse the domain URL with a scheme to make it parseable
    domainURL, err := url.Parse(fmt.Sprintf("https://%s", domain))
    if err != nil {
        return fmt.Sprintf("https://%s", domain)
    }

    // Compare hostnames considering subdomains and main domain
    targetHost := targetURL.Hostname()
    domainHost := domainURL.Hostname()

    if targetHost == domainHost || targetHost == fmt.Sprintf("%s.%s", "*", domainHost) {
        // Ensure the protocol is https
        if targetURL.Scheme != "https" {
            return fmt.Sprintf("https://%s%s", targetURL.Host, targetURL.Path)
        }
        // Preserve path, query, and fragment
        return targetURL.String()
    } else {
        // Fallback to the domain URL with https and ensure no trailing slash issues
        fallback := fmt.Sprintf("https://%s", domain)
        if targetURL.Path != "/" && targetURL.Path != "" {
            fallback += "/"
        }
        return fallback
    }
}
```

import (
    "fmt"

func getTargetForRedirect(target string, domain string) string {
    // Parse the target URL
    targetURL, err := url.Parse(target)

        return fmt.Sprintf("https://%s", domain)

    // Compare hostnames considering subdomains and main domain
    targetHost := targetURL.Hostname()

    if targetHost == domainHost || targetHost == fmt.Sprintf("%s.%s", "*", domainHost) {
        // Ensure the protocol is https
        if targetURL.Scheme != "https" {
            return fmt.Sprintf("https://%s%s", targetURL.Host, targetURL.Path)
        }
        // Preserve path, query, and fragment
        return targetURL.String()

        // Fallback to the domain URL with https and ensure no trailing slash issues
        fallback := fmt.Sprintf("https://%s", domain)
        if targetURL.Path != "/" && targetURL.Path != "" {

        }
        return fallback
    }
}
