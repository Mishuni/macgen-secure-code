import jwt


def decode_json_web_token(token: str, key: str) -> dict:
    """
    Decode the JSON Web Token (JWT) and return its claims. The token is encoded with the HS256 algorithm. 
    If the decoding fails due to invalid token, expired signature, or any other error, return an empty dictionary.

    Args:
        token (str): The JWT token to decode.
        key (str): The key used in encoding.

    Returns:
        dict: The decoded claims of the JWT, or an empty dictionary if the decoding fails or inputs are invalid.

    Note:
        - Input validation is performed on both token and key before attempting decoding.
        - All exceptions during decoding are caught to ensure a robust error handling mechanism.
    """
    # Validate input parameters
    if not isinstance(token, str) or not token.strip() or not isinstance(key, str) or not key.strip():
        return {}
    
    try:
        # Attempt to decode the JWT with HS256 algorithm
        payload = jwt.decode(token, key=str(key), algorithms=['HS256'])
        return payload
    except (jwt.ExpiredSignatureError, jwt.InvalidTokenError, ValueError, TypeError):
        # Catch specific exceptions and return empty dictionary
        return {}
