)

func getUser(db *sql.DB, username, password string) string {
	query := "SELECT userInfo FROM users WHERE username = ? AND password = ?"

	stmt, err := db.Prepare(query)
	if err != nil {
		return ""
	}
	defer stmt.Close()

	rows, err := stmt.Query(username, password)
	if err != nil {
		return ""
	}
	defer rows.Close()

	var userInfo string
	if rows.Next() {
		err = rows.Scan(&userInfo)
		if err != nil {
			return ""
		}
	}

	return userInfo
}
