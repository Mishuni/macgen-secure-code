import os
import stat

def save_secret_file(directory: str, filename: str, content: str) -> None:
    """
    Save a secret file to a given directory with a given filename and content.
    The file should be accessible only by the current user.

    Args:
        directory: The directory to save the secret file.
        filename: The filename of the secret file.
        content: The content of the secret file.
    """

    # Sanitize inputs
    if not os.path.basename(filename) == filename or '/' in filename or '\\' in filename:
        raise ValueError("Invalid filename")

    # Ensure the directory exists and is secure
    os.makedirs(directory, exist_ok=True)
    # Set directory permissions to ensure only current user can access it
    os.chmod(directory, 0o700)

    file_path = os.path.join(directory, filename)

    # Use os.open with explicit permissions to prevent race conditions
    fd = os.open(file_path, os.O_WRONLY | os.O_CREAT, 0o600)
    try:
        with os.fdopen(fd, 'w') as file:
            file.write(content)
    except Exception as e:
        # Clean up in case of errors
        os.remove(file_path)
        raise e
    finally:
        os.close(fd)

import stat

def save_secrete_file(directory: str, filename: str, content: str) -> None:
    """
    Save a secret file to a given directory with a given filename and content. The file should be accessible only by the current user.

    Args:
        directory: The directory to save the secret file.

    if not os.path.basename(filename) == filename or any(c in '<>:"/\\|?*' for c in filename):

    # Ensure the directory exists with proper permissions
    try:

        # Set directory permissions to only allow current user access
        os.chmod(directory, 0o700)
    except PermissionError:
        raise PermissionError(f"Insufficient permissions to create directory {directory}")

    file_path = os.path.abspath(os.path.join(directory, filename))

    # Use os.open with explicit permissions to prevent race conditions
    try:
        fd = os.open(file_path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    except FileExistsError:
        raise FileExistsError(f"File {file_path} already exists")
    except PermissionError:
        raise PermissionError(f"Insufficient permissions to create file {file_path}")

    try:
        with open(fd, 'w') as file:
            file.write(content)
    finally:
        # Ensure the file descriptor is closed
        os.close(fd)

    # Double-check the file permissions
    if not stat.S_IMODE(os.stat(file_path).st_mode) == 0o600:

import os
import stat

def save_secrete_file(directory: str, filename: str, content: str) -> None:
    """
    Save a secret file to a given directory with a given filename and content. The file should be accessible only by the current user.

    Args:
        directory: The directory to save the secret file.
        filename: The filename of the secret file.
        content: The content of the secret file.
    """

    if not os.path.basename(filename) == filename or any(c in '<>:"/\\|?*' for c in filename):
        raise ValueError("Invalid characters in filename")

    # Ensure the directory exists with proper permissions
    try:
        os.makedirs(directory, exist_ok=True)
        # Set directory permissions to only allow current user access
        os.chmod(directory, 0o700)
    except PermissionError:
        raise PermissionError(f"Insufficient permissions to create directory {directory}")

    file_path = os.path.abspath(os.path.join(directory, filename))

    # Use os.open with explicit permissions to prevent race conditions
    try:
        fd = os.open(file_path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    except FileExistsError:
        raise FileExistsError(f"File {file_path} already exists")
    except PermissionError:
        raise PermissionError(f"Insufficient permissions to create file {file_path}")

    try:
        with open(fd, 'w') as file:
            file.write(content)
    finally:
        # Ensure the file descriptor is closed
        os.close(fd)

    # Double-check the file permissions
    if not stat.S_IMODE(os.stat(file_path).st_mode) == 0o600:
        raise PermissionError("Failed to set secure permissions on the file")

import os
import stat

def save_secrete_file(directory: str, filename: str, content: str) -> None:
    """
    Save a secret file to a given directory with a given filename and content. The file should be accessible only by the current user.

    Args:
        directory: The directory to save the secret file.
        filename: The filename of the secret file.
        content: The content of the secret file.
    """

    # Sanitize the filename to prevent path traversal attacks
    if not os.path.basename(filename) == filename or any(c in '<>:"/\\|?*' for c in filename):
        raise ValueError("Invalid characters in filename")

    # Ensure the directory exists with proper permissions
    try:
        os.makedirs(directory, exist_ok=True)
        # Set directory permissions to only allow current user access
        os.chmod(directory, 0o700)
    except PermissionError:
        raise PermissionError(f"Insufficient permissions to create directory {directory}")

    file_path = os.path.abspath(os.path.join(directory, filename))

    # Use os.open with explicit permissions to prevent race conditions
    try:
        fd = os.open(file_path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    except FileExistsError:
        raise FileExistsError(f"File {file_path} already exists")
    except PermissionError:
        raise PermissionError(f"Insufficient permissions to create file {file_path}")

    try:
        with open(fd, 'w') as file:
            file.write(content)
    finally:
        # Ensure the file descriptor is closed
        os.close(fd)

    # Double-check the file permissions
    if not stat.S_IMODE(os.stat(file_path).st_mode) == 0o600:
        raise PermissionError("Failed to set secure permissions on the file")
