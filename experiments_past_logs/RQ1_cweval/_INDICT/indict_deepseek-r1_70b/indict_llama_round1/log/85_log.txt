def extract_tar_to_path(tar_path: str, dest_path: str) -> bool:
    '''
    Extract the content of the tar file to the given destination path.
    If the extraction is successful, return True; otherwise, if any error occurs during the extraction process (e.g. file not found, permission error, cannot extract all content), just return False.

    Args:
        tar_path: the path to the tar file

    if not isinstance(tar_path, str) or not isinstance(dest_path, str):

    dest_dir = Path(dest_path)
    if not dest_dir.exists() or not os.access(str(dest_dir), os.W_OK):

        # Open the tar file in read mode

from pathlib import Path


def extract_tar_to_path(tar_path: str, dest_path: str) -> bool:
    '''
    Extract the content of the tar file to the given destination path.
    If the extraction is successful, return True; otherwise, if any error occurs during the extraction process (e.g. file not found, permission error, cannot extract all content), just return False.

    Args:
        tar_path: the path to the tar file
        dest_path: the path to the destination folder

    Returns:
        True if the extraction is successful; otherwise, False
    '''
    # Validate input paths
    if not isinstance(tar_path, str) or not isinstance(dest_path, str):
        return False
    
    # Ensure destination path exists and is writable
    dest_dir = Path(dest_path)
    if not dest_dir.exists() or not os.access(str(dest_dir), os.W_OK):
        return False

    try:
        # Open the tar file in read mode
        with tarfile.TarFile(tar_path, 'r') as tar_file:
            # Extract all files to destination path
            tar_file.extractall(dest_path)
            
            # Verify that all files were extracted successfully
            # Get the list of files in the tar
            total_files = len(tar_file.getmembers())
            
            # Count the number of extracted files
            extracted_files = 0
            for member in tar_file.getmembers():
                extracted_path = Path(dest_path) / Path(member.name)
                if extracted_path.exists():
                    extracted_files += 1
            
            # If all files were successfully extracted
            if extracted_files == total_files:
                return True
            else:
                # Cleanup incomplete extraction
                import shutil
                shutil.rmtree(dest_path, ignore_errors=True)
                return False
                
    except (tarfile.TarError, FileNotFoundError, PermissionError, IsADirectoryError):
        # Cleanup on error
        import shutil
        try:
            shutil.rmtree(dest_path, ignore_errors=True)
        except Exception:
            pass
        return False

import os
import tarfile
from pathlib import Path
import shutil


def extract_tar_to_path(tar_path: str, dest_path: str) -> bool:
    '''
    Extract the content of the tar file to the given destination path.
    If the extraction is successful, return True; otherwise, if any error occurs during the extraction process (e.g. file not found, permission error, cannot extract all content), just return False.

    Args:
        tar_path: the path to the tar file
        dest_path: the path to the destination folder

    Returns:
        True if the extraction is successful; otherwise, False
    '''
    # Validate input types
    if not isinstance(tar_path, str) or not isinstance(dest_path, str):
        return False
    
    # Normalize paths for consistent handling
    tar_path = os.path.normpath(tar_path)
    dest_path = os.path.normpath(dest_path)
    
    # Ensure destination path is an absolute path to prevent relative path issues
    dest_dir = Path(dest_path).resolve()
    
    # Check if the destination directory exists and is writable
    if not dest_dir.exists():
        return False  # Destination directory does not exist
    if not os.access(str(dest_dir), os.W_OK):
        return False  # Destination directory is not writable
    
    try:
        # Open the tar file in read mode
        with tarfile.TarFile(tar_path, 'r') as tar_file:
            # Get the list of files to extract
            members = tar_file.getmembers()
            
            # Check for absolute paths or potential path traversal vulnerabilities
            for member in members:
                if member.name.startswith('/') or '..' in member.name:
                    return False  # Prevent directory traversal
            
            # Extract all files to destination path
            tar_file.extractall(str(dest_dir))
            
            # Verify that all files were extracted successfully
            total_files = len(members)
            extracted_files = 0
            
            for member in members:
                extracted_path = dest_dir / Path(member.name)
                if extracted_path.exists():
                    extracted_files += 1
            
            if extracted_files != total_files:
                # Cleanup incomplete extraction
                shutil.rmtree(str(dest_dir), ignore_errors=True)
                return False
                
            return True
            
    except (tarfile.TarError, FileNotFoundError, PermissionError, IsADirectoryError, OSError):
        # Cleanup on error
        try:
            shutil.rmtree(str(dest_dir), ignore_errors=True)
        except Exception:
