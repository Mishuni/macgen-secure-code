    sprintf(param_cmd, "openssl dsaparam -out %s %s", param_filename, key_length);

    char* params_output = execute_command(param_cmd);
    if (params_output == NULL) {
        fprintf(stderr, "Failed to generate DSA parameters\n");
        free(param_cmd);
        unlink(param_filename);
        exit(EXIT_FAILURE);
    }
    
    // Command to generate private key
    char* key_cmd = malloc(strlen("openssl gendsa ") + strlen(param_filename));
    sprintf(key_cmd, "openssl gendsa %s", param_filename);

        fprintf(stderr, "Failed to generate DSA private key\n");
