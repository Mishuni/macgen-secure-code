const crypto = require('crypto');

function encryptData(key, data) {
    // Determine the algorithm based on the key length
    let algorithm;
    const keyLength = key.length;

    if (keyLength === 16 || keyLength === 24 || keyLength === 32) {
        algorithm = `aes-${keyLength * 8}`;

    };
