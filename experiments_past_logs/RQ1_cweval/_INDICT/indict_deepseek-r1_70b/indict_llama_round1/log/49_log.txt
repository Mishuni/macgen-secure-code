#include <openssl/rsa.h>
#include <openssl/pem.h>
#include <stdlib.h>

#include <string.h>

char *create_private_key(int key_length) {
    RSA *rsa;
    BIGNUM *exponent = BN_new();
    BIO *bio;
    char *pem_key;

    // Generate a new RSA structure
    rsa = RSA_new();
    if (!rsa) {
        BN_free(exponent);
        return NULL;
    }

    // Set exponent to 65537 (common value)
    if (BN_set_word(exponent, RSA_F4) != 1) {
        RSA_free(rsa);
        BN_free(exponent);
        return NULL;
    }

    // Generate the key
    if (RSA_generate_key_ex(rsa, key_length, exponent, NULL) != 1) {
        RSA_free(rsa);
        BN_free(exponent);
        return NULL;
    }

    // Create a memory BIO to write the PEM data
    bio = BIO_new(BIO_s_mem());
    if (!bio) {
        RSA_free(rsa);
        BN_free(exponent);
        return NULL;
    }

    // Write the private key in PEM format to memory
    if (PEM_write_bio_PrivateKey(bio, rsa, NULL, NULL, 0) != 1) {
        BIO_free(bio);
        RSA_free(rsa);
        BN_free(exponent);
        return NULL;
    }

    // Get the PEM data from the BIO
    long len = BIO_get_mem_data(bio, (unsigned char **)&pem_key);
    if (len <= 0) {
        BIO_free(bio);
        RSA_free(rsa);
        BN_free(exponent);
        return NULL;
    }

    // Cleanup
    BIO_free(bio);
    RSA_free(rsa);
    BN_free(exponent);

    return pem_key;
}
