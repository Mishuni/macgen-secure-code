```go
func getURLToVisit(domain, subdomainPrefix string) string {
    // Check if the subdomain is valid (not empty and contains only alphanumeric characters and dots)
    if subdomainPrefix != "" && isValidSubdomain(subdomainPrefix) {
        return "https://" + subdomainPrefix + "." + domain
    }
    // If subdomain is invalid or empty, use the domain itself
    return "https://" + domain
}

func isValidSubdomain(subdomain string) bool {

            return false
        }
    }
    return true
}
