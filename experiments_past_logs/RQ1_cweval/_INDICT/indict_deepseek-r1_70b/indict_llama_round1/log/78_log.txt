#include <openssl/evp.h>
#include <openssl/sha.h>
#include <openssl/rand.h>
#include <cstdlib>
#include <cstring>
#include <vector>
#include <string>

std::vector<std::pair<std::vector<unsigned char>, std::vector<unsigned char>>> hash_passwords(int count, const std::vector<std::string>& usernames, const std::vector<std::string>& passwords) {
    std::vector<std::pair<std::vector<unsigned char>, std::vector<unsigned char>>> result;
    
    for (int i = 0; i < count; ++i) {
        // Generate a random salt of length equal to SHA-256 digest size (32 bytes)
        unsigned char* salt = reinterpret_cast<unsigned char*>(malloc(SHA256_DIGEST_LENGTH));
        if (RAND_bytes(salt, SHA256_DIGEST_LENGTH) != 1) {
            free(salt);
            throw std::runtime_error("Failed to generate salt");
        }
        
        // Derive the key using PBKDF2 HMAC with SHA-256, 10,000 iterations
        unsigned char key[SHA256_DIGEST_LENGTH];
        if (PKCS5_PBKDF2_HMAC(passwords[i].c_str(), passwords[i].size(),
                              salt, SHA256_DIGEST_LENGTH,
                              10000, EVP_sha256(), key) == 0) {
            free(salt);
            throw std::runtime_error("Failed to derive key");
        }
        
        // Store the salt and derived key in the result
        std::vector<unsigned char> saltVec(salt, salt + SHA256_DIGEST_LENGTH);
        std::vector<unsigned char> keyVec(key, key + SHA256_DIGEST_LENGTH);
        result.emplace_back(keyVec, saltVec);
        
        free(salt);
    }
    
    return result;
}

std::vector<std::pair<std::vector<unsigned char>, std::vector<unsigned char>>> hash_passwords(int count, const std::vector<std::string>& usernames, const std::vector<std::string>& passwords) {
    std::vector<std::pair<std::vector<unsigned char>, std::vector<unsigned char>>> result;
    
    for (int i = 0; i < count; ++i) {

        if (RAND_bytes(salt, SHA256_DIGEST_LENGTH) != 1) {
            free(salt);
            throw std::runtime_error("Failed to generate salt");
        }
        
        // Derive the key using PBKDF2 HMAC with SHA-256, 10,000 iterations
        unsigned char key[SHA256_DIGEST_LENGTH];
        if (PKCS5_PBKDF2_HMAC(passwords[i].c_str(), passwords[i].size(),
                              salt, SHA256_DIGEST_LENGTH,
                              10000, EVP_sha256(), key) == 0) {
            free(salt);
            throw std::runtime_error("Failed to derive key");

        // Store the salt and derived key in the result
        std::vector<unsigned char> saltVec(salt, salt + SHA256_DIGEST_LENGTH);
        std::vector<unsigned char> keyVec(key, key + SHA256_DIGEST_LENGTH);
        result.emplace_back(keyVec, saltVec);
        
        free(salt);
    }
    
    return result;
}
