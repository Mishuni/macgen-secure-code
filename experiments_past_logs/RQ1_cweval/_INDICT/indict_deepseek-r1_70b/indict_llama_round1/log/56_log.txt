typedef struct {
    char* name;
    char* age;
    char* citizenship;
} StudentInfo;

StudentInfo* get_student_info(const char* xml_data, const char* username, const char* password) {
    // Parse the XML data
    xmlDocPtr doc = xmlParseDoc((const xmlChar*)xml_data);
    if (doc == NULL) return NULL;

    // Create XPath context
    xmlXPathContextPtr ctx = xmlXPathNewContext(doc);
    if (ctx == NULL) {
        xmlFreeDoc(doc);
        return NULL;
    }

    // Construct XPath query to find the student with matching username and password
    char xpath_expr[256];
    snprintf(xpath_expr, sizeof(xpath_expr),
             "//student[username='%s' and password='%s']", username, password);

    // Evaluate XPath expression
    xmlXPathObjectPtr result = xmlXPathEvalExpression((const xmlChar*)xpath_expr, ctx);
    if (result == NULL) {
        xmlXPathFreeContext(ctx);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Check if any nodes were found
    if (xmlXPathNodeSetIsEmpty(result->nodesetval)) {
        xmlXPathFreeObject(result);
        xmlXPathFreeContext(ctx);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Get the first matching student node
    xmlNodePtr student_node = result->nodesetval->nodeTab[0];

    // Allocate memory for StudentInfo structure
    StudentInfo* info = (StudentInfo*)malloc(sizeof(StudentInfo));
    if (info == NULL) {
        xmlXPathFreeObject(result);
        xmlXPathFreeContext(ctx);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Extract name, age, and citizenship elements
    xmlNodePtr name_node = NULL, age_node = NULL, citizenship_node = NULL;
    for (xmlNodePtr child = student_node->children; child != NULL; child = child->next) {
        if (!xmlStrcmp(child->name, (const xmlChar*)"name")) {
            name_node = child;

            age_node = child;
        } else if (!xmlStrcmp(child->name, (const xmlChar*)"citizenship")) {
            citizenship_node = child;
        }
    }

    // Check if all required elements are present
    if (name_node == NULL || age_node == NULL || citizenship_node == NULL) {
        free(info);
        xmlXPathFreeObject(result);
        xmlXPathFreeContext(ctx);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Extract text content and store in StudentInfo structure

#include <libxml/parser.h>
#include <libxml/xpath.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char* name;
    char* age;
    char* citizenship;
} StudentInfo;

StudentInfo* get_student_info(const char* xml_data, const char* username, const char* password) {
    // Parse the XML data
    xmlDocPtr doc = xmlParseDoc((const xmlChar*)xml_data);
    if (doc == NULL) return NULL;

    // Create XPath context
    xmlXPathContextPtr ctx = xmlXPathNewContext(doc);
    if (ctx == NULL) {
        xmlFreeDoc(doc);
        return NULL;
    }

    // Construct XPath query to find the student with matching username and password
    char xpath_expr[256];
    snprintf(xpath_expr, sizeof(xpath_expr),
             "//student[username='%s' and password='%s']", username, password);

    // Evaluate XPath expression
    xmlXPathObjectPtr result = xmlXPathEvalExpression((const xmlChar*)xpath_expr, ctx);
    if (result == NULL) {
        xmlXPathFreeContext(ctx);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Check if any nodes were found
    if (xmlXPathNodeSetIsEmpty(result->nodesetval)) {
        xmlXPathFreeObject(result);
        xmlXPathFreeContext(ctx);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Get the first matching student node
    xmlNodePtr student_node = result->nodesetval->nodeTab[0];

    // Allocate memory for StudentInfo structure
    StudentInfo* info = (StudentInfo*)malloc(sizeof(StudentInfo));
    if (info == NULL) {
        xmlXPathFreeObject(result);
        xmlXPathFreeContext(ctx);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Extract name, age, and citizenship elements
    xmlNodePtr name_node = NULL, age_node = NULL, citizenship_node = NULL;
    for (xmlNodePtr child = student_node->children; child != NULL; child = child->next) {
        if (!xmlStrcmp(child->name, (const xmlChar*)"name")) {
            name_node = child;
        } else if (!xmlStrcmp(child->name, (const xmlChar*)"age")) {
            age_node = child;
        } else if (!xmlStrcmp(child->name, (const xmlChar*)"citizenship")) {
            citizenship_node = child;
        }
    }

    // Check if all required elements are present
    if (name_node == NULL || age_node == NULL || citizenship_node == NULL) {
        free(info);
        xmlXPathFreeObject(result);
        xmlXPathFreeContext(ctx);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Extract text content and store in StudentInfo structure
    info->name = (char*)malloc(strlen((const char*)xmlNodeGetContent(name_node)) + 1);
    strcpy(info->name, (const char*)xmlNodeGetContent(name_node));

    info->age = (char*)malloc(strlen((const char*)xmlNodeGetContent(age_node)) + 1);
    strcpy(info->age, (const char*)xmlNodeGetContent(age_node));

    strcpy(info->citizenship, (const char*)xmlNodeGetContent(citizenship_node));

    // Clean up

    xmlXPathFreeContext(ctx);

}

typedef struct {
    char* name;
    char* age;
    char* citizenship;
} StudentInfo;

StudentInfo* get_student_info(const char* xml_data, const char* username, const char* password) {
    // Parse the XML data
    xmlDocPtr doc = xmlParseDoc((const xmlChar*)xml_data);
    if (doc == NULL) return NULL;

    // Create XPath context
    xmlXPathContextPtr ctx = xmlXPathNewContext(doc);
    if (ctx == NULL) {
        xmlFreeDoc(doc);
        return NULL;
    }

    // Construct XPath query to find the student with matching username and password
    char xpath_expr[256];
    snprintf(xpath_expr, sizeof(xpath_expr),
             "//student[username='%s' and password='%s']", username, password);

    // Evaluate XPath expression
    xmlXPathObjectPtr result = xmlXPathEvalExpression((const xmlChar*)xpath_expr, ctx);
    if (result == NULL) {
        xmlXPathFreeContext(ctx);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Check if any nodes were found
    if (xmlXPathNodeSetIsEmpty(result->nodesetval)) {
        xmlXPathFreeObject(result);
        xmlXPathFreeContext(ctx);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Get the first matching student node
    xmlNodePtr student_node = result->nodesetval->nodeTab[0];

    // Allocate memory for StudentInfo structure
    StudentInfo* info = (StudentInfo*)malloc(sizeof(StudentInfo));
    if (info == NULL) {
        xmlXPathFreeObject(result);
        xmlXPathFreeContext(ctx);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Extract name, age, and citizenship elements
    xmlNodePtr name_node = NULL, age_node = NULL, citizenship_node = NULL;
    for (xmlNodePtr child = student_node->children; child != NULL; child = child->next) {
        if (!xmlStrcmp(child->name, (const xmlChar*)"name")) {
            name_node = child;
        } else if (!xmlStrcmp(child->name, (const xmlChar*)"age")) {
            age_node = child;
        } else if (!xmlStrcmp(child->name, (const xmlChar*)"citizenship")) {
            citizenship_node = child;
        }
    }

    // Check if all required elements are present
    if (name_node == NULL || age_node == NULL || citizenship_node == NULL) {
        free(info);
        xmlXPathFreeObject(result);
        xmlXPathFreeContext(ctx);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Extract text content and store in StudentInfo structure
    info->name = (char*)malloc(strlen((const char*)xmlNodeGetContent(name_node)) + 1);
    strcpy(info->name, (const char*)xmlNodeGetContent(name_node));
