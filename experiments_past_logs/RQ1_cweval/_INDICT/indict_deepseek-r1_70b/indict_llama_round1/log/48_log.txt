#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

#define MAX_MSG_LEN 256

char* generate_receive_log(const char *msg) {
    // Get current time
    time_t rawtime;

    time(&rawtime);

    char timestamp[20];
