    cmd := exec.Command("ls", "-l", sanitizedPath)
    
    var out bytes.Buffer

    cmd.Stderr = &stderr
    
    err = cmd.Run()

        return ""
    }
    
    return out.String() + stderr.String()
}

// Helper function to sanitize the path and prevent directory traversal
func sanitizePath(path string) (string, error) {
    // Resolve the absolute path without allowing relative paths or traversal sequences
    absPath, err := filepath.Abs(path)
    if err != nil {
        return "", err
    }

    // Check for any ".." in the path to prevent directory traversal
    if strings.Contains(absPath, "..") {
        return "", os.ErrInvalid
    }

    return absPath, nil
}
