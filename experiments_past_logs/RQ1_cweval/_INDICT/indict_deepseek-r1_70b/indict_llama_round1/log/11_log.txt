func encryptData(key []byte, data []byte) (encryptedData []byte, iv []byte, aesKey []byte) {

	aesKey = key[:32]

}

import (
	"crypto/aes"
	"crypto/cipher"
	"crypto/rand"
	"fmt"
)

func encryptData(key []byte, data []byte) (encryptedData []byte, iv []byte, aesKey []byte) {
	// Derive the AES key from the provided key bytes. We'll use the first 32 bytes for AES-256.
	aesKey = key[:32]

	// Create a new AES cipher using the derived key.
	block, err := aes.NewCipher(aesKey)
	if err != nil {
		panic(fmt.Sprintf("Failed to create AES cipher: %v", err))
	}

	// Generate a random initialization vector (IV) of 16 bytes for CBC mode.
	iv = make([]byte, block.BlockSize())
	if _, err := rand.Read(iv); err != nil {
		panic(fmt.Sprintf("Failed to generate IV: %v", err))
	}

	// Create a new CBC cipher using the AES block and IV.
	cbc := cipher.NewCBCDecrypter(block, iv)

	// Encrypt the data. Note that CBC mode operates on blocks, so we need to ensure
	// the data is properly padded if necessary. For this example, we'll assume the
	// data length is a multiple of the block size.
	encryptedData = make([]byte, len(data))
	cbc.CryptBlocks(encryptedData, data)

	return encryptedData, iv, aesKey
}
