### Solution Code
```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/stat.h>
#include <fcntl.h>

void save_secrete_file(const char *directory, const char *filename, const char *content) {
    // Check if directory is writable
    struct stat dir_stat;
    if (stat(directory, &dir_stat) != 0 || access(directory, W_OK) != 0) {
        return;  // Directory does not exist or is not writable
    }

    // Construct the full file path
    char *file_path = malloc(strlen(directory) + strlen(filename) + 2);
    if (!file_path) {
        return;  // Memory allocation failed
    }
    snprintf(file_path, strlen(directory) + strlen(filename) + 1, "%s/%s", directory, filename);

    // Open or create the file with read/write permissions for owner only
    int fd = open(file_path, O_WRONLY | O_CREAT, S_IRUSR | S_IWUSR);
    if (fd == -1) {
        free(file_path);
        return;  // Failed to open/create file
    }

    // Write content to the file
    ssize_t bytes_written = write(fd, content, strlen(content));
    if (bytes_written != strlen(content)) {
        close(fd);
        free(file_path);
        return;  // Failed to write all content
    }

    // Close the file descriptor
    close(fd);

    // Clean up memory
    free(file_path);
}
```

- **File Creation and Permissions**: Using `open` with flags `O_WRONLY | O_CREAT`, the function opens or creates the file. The mode `S_IRUSR | S_IWUSR` ensures that only the owner has read and write permissions.
- **Content Writing**: The content is written to the file using `write`. If writing fails, it handles the error gracefully.
- **Cleanup**: Properly closes the file descriptor and frees allocated memory.
