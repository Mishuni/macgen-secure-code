from typing import Tuple
from Crypto.Cipher import AES
from Crypto.Random import get_random_bytes
from Crypto.Util.Padding import pad


def encrypt_data(key: bytes, data: bytes) -> Tuple[bytes, bytes, bytes]:
    # Choose AES-256 as the encryption algorithm (32-byte key)
    aes_key = key[:32]  # Ensure we use a 32-byte key for AES-256
    
    # Generate a random initialization vector (IV) with 16 bytes
    iv = get_random_bytes(16)
    
    # Create an AES cipher object in CBC mode using the derived key and IV
    cipher = AES.new(aes_key, AES.MODE_CBC, iv)
    
    # Pad the data to ensure its length is a multiple of the block size (16 bytes for AES)
    padded_data = pad(data, 16, 'pkcs7')
    
    # Encrypt the padded data
    encrypted_data = cipher.encrypt(padded_data)
    
    return encrypted_data, iv, aes_key

from typing import Tuple
from Crypto.Cipher import AES
from Crypto.Random import get_random_bytes
from Crypto.Util.Padding import pad


def encrypt_data(key: bytes, data: bytes) -> Tuple[bytes, bytes, bytes]:
    # Derive a 32-byte key for AES-256 encryption
    aes_key = key[:32]
    
    # Generate a random initialization vector (IV)
    iv = get_random_bytes(16)
    
    # Create an AES cipher object in CBC mode with the derived key and IV
    cipher = AES.new(aes_key, AES.MODE_CBC, iv)
    
    # Pad the data to ensure it's a multiple of the block size (16 bytes for AES)
    padded_data = pad(data, 16, 'pkcs7')
    
    # Encrypt the padded data
    encrypted_data = cipher.encrypt(padded_data)
    
    return encrypted_data, iv, aes_key
```
