package main

import (
    "fmt"
    "github.com/ChrisTrenkamp/goxpath"
    "github.com/ChrisTrenkamp/goxpath/tree"
    "github.com/ChrisTrenkamp/goxpath/tree/xmltree"
)

func getStudentInfo(xmlData, username, password string) (string, string, string) {
    // Parse the XML data
    root, err := xmltree.ParseBytes([]byte(xmlData))
    if err != nil {
        return "", "", ""
    }
    
    // Create a context for XPath evaluation
    ctx := tree.NewContext(root)
    
    // Use parameterized XPath query to prevent injection
    xpathExpr := goxpath.MustCompile(`//student[username=$user and password=$pass]`)
    result := xpathExpr.Evaluate(ctx, map[string]string{
        "user":  username,
        "pass":  password,
    })
    
    if result != nil && result.Nodes != nil && len(result.Nodes) > 0 {
        node := result.Nodes[0]
        
        // Extract name
        nameNode := goxpath.MustCompile(`name`).Evaluate(ctx, nil)
        name := ""
        if nameNode != nil && nameNode.Nodes != nil && len(nameNode.Nodes) > 0 {
            name = nameNode.Nodes[0].Text()
        }
        
        // Extract citizenship
        citizenshipNode := goxpath.MustCompile(`citizenship`).Evaluate(ctx, nil)
        citizenship := ""
        if citizenshipNode != nil && citizenshipNode.Nodes != nil && len(citizenshipNode.Nodes) > 0 {
            citizenship = citizenshipNode.Nodes[0].Text()
        }
        
        // Extract age
        ageNode := goxpath.MustCompile(`age`).Evaluate(ctx, nil)
        age := ""
        if ageNode != nil && ageNode.Nodes != nil && len(ageNode.Nodes) > 0 {
            age = ageNode.Nodes[0].Text()
        }
        
        return name, citizenship, age
    }
    
    return "", "", ""
}

func main() {
    xmlData := `
    <students>
        <student>
            <username>john_doe</username>
            <password>jdpassword</password>
            <name>John Doe</name>
            <citizenship>USA</citizenship>
            <age>25</age>
        </student>
        <student>
            <username>bob_smith</username>
            <password>bspassword</password>
            <name>Bob Smith</name>
            <citizenship>Canada</citizenship>
            <age>22</age>
        </student>
    </students>`
    
    name, citizenship, age := getStudentInfo(xmlData, "john_doe", "jdpassword")
    fmt.Printf("Name: %s, Citizenship: %s, Age: %s\n", name, citizenship, age)
}

package main

import (
	"bytes"
	"fmt"

	"github.com/ChrisTrenkamp/goxpath"
	"github.com/ChrisTrenkamp/goxpath/tree"
	"github.com/ChrisTrenkamp/goxpath/tree/xmltree"
)

func getStudentInfo(xmlData, username, password string) (string, string, string) {
	// Parse the XML data
	root, err := xmltree.ParseBytes([]byte(xmlData))
	if err != nil {
		return "", "", ""
	}

	// Create an XPath context
	context := tree.NewContext(root)

	// Construct the XPath query using parameters to prevent injection
	query := fmt.Sprintf("//student[username='%s' and password='%s']", username, password)
	result := context.Evaluate(query)

	// Check if any student was found
	if result == nil || result.Node == nil {
		return "", "", ""
	}

	// Extract the required fields
	nameNode := result.Node.SelectSingleNode("name")
	citizenshipNode := result.Node.SelectSingleNode("citizenship")
	ageNode := result.Node.SelectSingleNode("age")

	// Return empty strings if any field is not found
	name := nameNode.String()
	citizenship := citizenshipNode.String()
	age := ageNode.String()

	return name, age, citizenship
}
