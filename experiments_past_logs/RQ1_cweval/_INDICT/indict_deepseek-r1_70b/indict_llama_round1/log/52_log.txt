#include <string.h>

typedef struct {
    unsigned char* encrypted_data;
    size_t encrypted_len;
    unsigned char* iv;
    unsigned char* key;
} encryption_result;

encryption_result encrypt_data(
    const unsigned char* key, size_t key_len,
    const unsigned char* data, size_t data_len
) {
    encryption_result result = {NULL, 0, NULL, NULL};
    
    // Generate random initialization vector (IV)
    unsigned char* iv = malloc(AES_BLOCK_SIZE);
    if (RAND_bytes(iv, AES_BLOCK_SIZE) != 1) {
        free(iv);
        return result;
    }
    
    // Use the first 16 bytes of the provided key for AES-128
    unsigned char* aes_key = malloc(AES_KEYLEN_128);
    memcpy(aes_key, key, AES_KEYLEN_128);
    
    AES_KEY aes_cbc_key;
    if (AES_set_encrypt_key(aes_key, 128, &aes_cbc_key) != 0) {
        free(iv);
        free(aes_key);
        return result;
    }
    
    // Calculate padding needed to make data length a multiple of block size
    int pad_len = (AES_BLOCK_SIZE - (data_len % AES_BLOCK_SIZE)) % AES_BLOCK_SIZE;
    size_t padded_data_len = data_len + pad_len;
    
    // Allocate memory for padded data and copy original data
    unsigned char* padded_data = malloc(padded_data_len);
    if (!padded_data) {
        free(iv);
        free(aes_key);
        return result;
    }
    memcpy(padded_data, data, data_len);
    
    // Add PKCS#7 padding
    for (int i = 0; i < pad_len; ++i) {
        padded_data[data_len + i] = pad_len;
    }
    
    // Allocate memory for encrypted data
    unsigned char* encrypted_data = malloc(padded_data_len);
    if (!encrypted_data) {
        free(iv);
        free(aes_key);
        free(padded_data);
        return result;
    }
    
    // Perform AES-CBC encryption
    AES_cbc_encrypt(padded_data, encrypted_data, padded_data_len, &aes_cbc_key, iv, AES_ENCRYPT);
    
    // Populate the result structure
    result.iv = iv;
    result.key = aes_key;
    result.encrypted_data = encrypted_data;
    result.encrypted_len = padded_data_len;
    
    free(padded_data);
    
    return result;
}

        free(aes_key);
        return result;

typedef struct {

        return result;
    }

    unsigned char* aes_key = malloc(AES_KEYLEN_128);
    memcpy(aes_key, key, AES_KEYLEN_128);
    
    AES_KEY aes_cbc_key;
    if (AES_set_encrypt_key(aes_key, 128, &aes_cbc_key) != 0) {
        free(iv);
        free(aes_key);
        return result;
    }
    
    // Calculate padding needed to make data length a multiple of block size
    int pad_len = (AES_BLOCK_SIZE - (data_len % AES_BLOCK_SIZE)) % AES_BLOCK_SIZE;
    size_t padded_data_len = data_len + pad_len;

    if (!padded_data) {

        free(aes_key);
        return result;
    }
    memcpy(padded_data, data, data_len);
    
    // Add PKCS#7 padding
    for (int i = 0; i < pad_len; ++i) {

    // Allocate memory for encrypted data
    unsigned char* encrypted_data = malloc(padded_data_len);
    if (!encrypted_data) {

        return result;

    // Perform AES-CBC encryption
    AES_cbc_encrypt(padded_data, encrypted_data, padded_data_len, &aes_cbc_key, iv, AES_ENCRYPT);
    
    // Populate the result structure
    result.iv = iv;
    result.key = aes_key;
    result.encrypted_data = encrypted_data;
    result.encrypted_len = padded_data_len;
