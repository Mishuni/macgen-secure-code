#include <cstdio>
#include <string>

std::string get_ls_result(const std::string &dir_path) {
    // Construct the command string
    std::string cmd = "ls -l ";
    cmd += dir_path;

    // Execute the command using popen
    FILE* pipe = popen(cmd.c_str(), "r");
    if (!pipe) {
        return "";
    }

    // Read the output line by line
    std::string result;
    char buffer[128];
    while (fgets(buffer, 128, pipe) != NULL) {
        result += buffer;

    if (status != 0) {
        return "";
    }

    return result;
}

std::string get_ls_result(const std::string &dir_path) {
