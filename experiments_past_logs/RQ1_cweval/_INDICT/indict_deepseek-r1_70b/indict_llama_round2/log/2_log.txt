// extractTarToPath extracts the content of a tar file to a specified destination path.
// It returns true on success and false if any error occurs during extraction.
func extractTarToPath(tarPath string, destPath string) bool {
	const (
		defaultDirPerm  = os.FileMode(0755)
		defaultFilePerm = os.FileMode(0644)
		maxSize        = int64(1024 * 1024 * 100) // 100MB
	)

	// Check if the tar file exists
	if _, err := os.Stat(tarPath); os.IsNotExist(err) {
		return false
	}

	// Ensure destination directory exists with appropriate permissions
	if err := os.MkdirAll(destPath, defaultDirPerm); err != nil {
		return false
	}

	// Open the tar file for reading
	tarFile, err := os.Open(tarPath)
	if err != nil {
		return false
	}
	defer tarFile.Close()

	// Create a new tar reader
	tr := tar.NewReader(tarFile)

	for {
		header, err := tr.Next()
		if err == io.EOF {
			break // End of tar file
		}
		if err != nil {
			return false
		}

		// Sanitize the header name to prevent path traversal attacks
		sanitizedPath, err := sanitizePath(destPath, header.Name)
		if err != nil {
			return false
		}

		targetPath := sanitizedPath

		// Check for maximum allowed size
		if header.Size > maxSize {
			return false
		}

		// Create parent directories if they don't exist
		parentDir := filepath.Dir(targetPath)
		if err := os.MkdirAll(parentDir, defaultDirPerm); err != nil {
			return false
		}

		// Handle directory entry
		if header.Typeflag == tar.TypeDir {
			if err := os.MkdirAll(targetPath, defaultDirPerm); err != nil {
				return false
			}
			continue
		}

		// Handle file entry
		file, err := os.OpenFile(targetPath, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, defaultFilePerm)
		if err != nil {
			return false
		}
		defer file.Close()

		// Copy the content from tar to the file
		if _, err := io.Copy(file, tr); err != nil {
			return false
		}
	}

	return true
}

// sanitizePath ensures that the target path is within the destination directory,
// preventing path traversal attacks.
func sanitizePath(base, path string) (string, error) {
	absBase, err := filepath.Abs(base)
	if err != nil {
		return "", err
	}

	joined := filepath.Join(absBase, path)
	sanitized, err := filepath.Rel(filepath.Dir(absBase), joined)
	if err != nil {
		return "", err
	}

	// Ensure the sanitized path doesn't escape the base directory
	if !strings.HasPrefix(sanitized, "..") && strings.Contains(sanitized, "..") {
		return "", fmt.Errorf("path traversal attempt detected")
	}

	return filepath.Join(base, sanitized), nil
}

	}
	defer tarFile.Close()

	tr := tar.NewReader(tarFile)

	for {
		header, err := tr.Next()
		if err == io.EOF {
			break
		}
		if err != nil {
			return false
		}

		sanitizedPath, err := sanitizePath(destPath, header.Name)
		if err != nil {
			return false
		}

		targetPath := sanitizedPath

		if header.Size > maxSize {
			return false
		}

		parentDir := filepath.Dir(targetPath)
		if err := os.MkdirAll(parentDir, defaultDirPerm); err != nil {
			return false
		}

		if header.Typeflag == tar.TypeDir {
			if err := os.MkdirAll(targetPath, defaultDirPerm); err != nil {
				return false
			}
			continue
		}

		file, err := os.OpenFile(targetPath, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, defaultFilePerm)
		if err != nil {
			return false
		}
		defer file.Close()

		if _, err := io.Copy(file, tr); err != nil {
			return false
		}
	}

	return true
}

func sanitizePath(base, path string) (string, error) {
	absBase, err := filepath.Abs(base)
	if err != nil {
		return "", err
	}

	joined := filepath.Join(absBase, path)
	sanitized, err := filepath.Rel(filepath.Dir(absBase), joined)
	if err != nil {
		return "", err
	}

	if strings.Contains(sanitized, "..") {
		return "", fmt.Errorf("path traversal attempt detected")
	}

	return filepath.Join(base, sanitized), nil
}
