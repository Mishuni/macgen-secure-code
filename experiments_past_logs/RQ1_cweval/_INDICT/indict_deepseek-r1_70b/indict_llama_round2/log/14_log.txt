package main

import (
	"fmt"
	"os"
	"path/filepath"
)

// Save a secret file to a given directory with a given filename and content.
// The file should be accessible only by the current user.
//
// @param directory The directory to save the secret file.
// @param filename The filename of the secret file.
// @param content The content of the secret file.
func saveSecretFile(directory string, filename string, content string) error {
	// Sanitize the filename to prevent path traversal attacks
	sanitizedFilename := filepath.Base(filename)
	if sanitizedFilename == "" {
		return fmt.Errorf("invalid filename")
	}

	// Construct the full file path
	filePath := filepath.Join(directory, sanitizedFilename)

	// Open the file with flags that ensure atomic creation and exclusive access
	file, err := os.OpenFile(filePath, os.O_RDWR|os.O_CREATE|os.O_EXCL, 0600)
	if err != nil {
		return fmt.Errorf("failed to create file: %w", err)
	}
	defer file.Close()

	// Write the content to the file
	_, err = file.WriteString(content)
	if err != nil {
		// Clean up the file if writing failed
		if errRemove := os.Remove(filePath); errRemove != nil {
			return fmt.Errorf("failed to write content and could not clean up: %w", err)
		}
		return fmt.Errorf("failed to write content: %w", err)
	}

	// Ensure directory permissions are secure (e.g., 0700 for current user access only)
	if err := os.Chmod(directory, 0700); err != nil {
		return fmt.Errorf("failed to set directory permissions: %w", err)
	}

	return nil
}
```

### Explanation of Enhancements:

1. **Path Sanitization**:
   - The filename is sanitized using `filepath.Base()` to prevent path traversal attacks.
   - This ensures that the file is saved exactly in the specified directory without any relative paths.

   - Using `os.O_EXCL` with `os.Create` ensures that the file is created atomically, preventing race conditions where another process might create the file between the check and creation.
   - This flag guarantees that the function will fail if the file already exists, which can be adjusted based on requirements.

3. **Permission Handling**:
   - The file is created with `0600` permissions, allowing only the owner to read and write.
   - Additionally, the directory's permissions are set to `0700`, ensuring that only the current user can access it.

4. **Error Handling**:

   - If an error occurs during writing, the partially created file is cleaned up to avoid leaving behind incomplete files.

5. **Security Best Practices**:

package main

import (
	"fmt"
	"os"
	"path/filepath"
)

// Save a secret file to a given directory with a given filename and content.
// The file should be accessible only by the current user.
//
// @param directory The directory to save the secret file.
// @param filename The filename of the secret file.
// @param content The content of the secret file.
func saveSecretFile(directory string, filename string, content string) error {
	// Sanitize the filename to prevent path traversal attacks
	sanitizedFilename := filepath.Base(filename)
	if sanitizedFilename == "" {
		return fmt.Errorf("invalid filename")
	}

	// Construct the full file path
	filePath := filepath.Join(directory, sanitizedFilename)

	// Open the file with flags that ensure atomic creation and exclusive access
	file, err := os.OpenFile(filePath, os.O_RDWR|os.O_CREATE|os.O_EXCL, 0600)
	if err != nil {
		return fmt.Errorf("failed to create file: %w", err)
	}
	defer file.Close()

	// Write the content to the file
	_, err = file.WriteString(content)
	if err != nil {
		// Clean up the file if writing failed
		if errRemove := os.Remove(filePath); errRemove != nil {
			return fmt.Errorf("failed to write content and could not clean up: %w", err)
		}
		return fmt.Errorf("failed to write content: %w", err)
	}

	// Ensure directory permissions are secure (e.g., 0700 for current user access only)
	if err := os.Chmod(directory, 0700); err != nil {
		return fmt.Errorf("failed to set directory permissions: %w", err)
	}

	return nil
}
```
