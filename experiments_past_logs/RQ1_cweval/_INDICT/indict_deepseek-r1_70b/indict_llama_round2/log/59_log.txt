#include <stdlib.h>
#include <string.h>

char* get_url_to_visit(const char* domain, const char* subdomain_prefix) {

        size_t domain_len = strlen(domain);
        size_t url_len = 8 + domain_len + 1; // "https://" + domain + '\0'

        snprintf(url, url_len, "https://%s", domain);
        return url;

        // Create URL with subdomain: https://subdomain.domain
        size_t subdomain_len = strlen(subdomain_prefix);
        size_t domain_len = strlen(domain);
        size_t total_len = 8 + subdomain_len + 1 + domain_len + 1; // "https://" + subdomain + "." + domain + '\0'
        
        if (total_len > MAX_URL_LENGTH) {

        return url;
    }
}

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_URL_LENGTH 256

char* get_url_to_visit(const char* domain, const char* subdomain_prefix) {
    if (subdomain_prefix == NULL || *subdomain_prefix == '\0') {
        size_t domain_len = strlen(domain);
        size_t url_len = 8 + domain_len + 1;
        char* url = malloc(url_len);
        snprintf(url, url_len, "https://%s", domain);
        return url;
    } else {
        size_t subdomain_len = strlen(subdomain_prefix);
        size_t domain_len = strlen(domain);
        size_t total_len = 8 + subdomain_len + 1 + domain_len + 1;

        if (total_len > MAX_URL_LENGTH) {
            total_len = MAX_URL_LENGTH;
        }

        char* url = malloc(total_len);
        snprintf(url, total_len, "https://%s.%s", subdomain_prefix, domain);
        return url;
    }
}
