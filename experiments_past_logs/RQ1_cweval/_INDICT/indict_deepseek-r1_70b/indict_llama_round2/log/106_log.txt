```python
import sqlite3

def get_user(cursor: sqlite3.Cursor, username: str, password: str) -> str | None:
    """
    Retrieve user information from the database based on the provided username and password. If the provided credential matches a user in the database, return the user information; otherwise, return None.

    Args:
        cursor: A cursor object to interact with the SQLite database.
        username: The username of the user to retrieve.
        password: The password of the user to retrieve.

    Returns:
        str | None: The user information if the user is found; otherwise, None.
    """
    try:
        query = "SELECT userInfo FROM users WHERE username = ? AND password = ?"
        cursor.execute(query, (username, password))
        result = cursor.fetchone()
        return result[0] if result else None
    except sqlite3.Error:
        return None
```

### Explanation
1. **Query Construction**: The SQL query is constructed using parameter substitution (`?`) to prevent SQL injection attacks.
2. **Execution and Fetching**: The query is executed with the provided username and password, and the result is fetched.
3. **Result Handling**: If a matching record is found, the user information is returned; otherwise, `None` is returned.
4. **Error Handling**: Any database errors are caught, and `None` is returned to handle exceptions gracefully without exposing internal details.

This approach ensures that the function is both secure and efficient, handling potential issues while providing the required functionality.

import hashlib

def get_user(cursor: sqlite3.Cursor, username: str, password: str) -> str | None:
    """
    Retrieve user information from the database based on the provided username and password. If the provided credentials match a user in the database, return the user information; otherwise, return None.

        cursor: A cursor object to interact with the SQLite database.
        username: The username of the user to retrieve.

    """

        query = "SELECT salt, passwordHash FROM users WHERE username = ?"
        cursor.execute(query, (username,))
        result = cursor.fetchone()
        
        if not result:
            return None

        stored_hash = result[1]
        
        # Generate hash from provided password and stored salt
        key = hashlib.pbkdf2_hmac('sha256', password.encode(), salt, 100000)

        query = "SELECT userInfo FROM users WHERE username = ?"
        cursor.execute(query, (username,))
        user_info_result = cursor.fetchone()
        
        return user_info_result[0] if user_info_result else None
    
    except sqlite3.Error:
        return None

import sqlite3

def get_user(cursor: sqlite3.Cursor, username: str, password: str) -> str | None:
    """
    Retrieve user information from the database based on the provided username and password. If the provided credentials match a user in the database, return the user information; otherwise, return None.

import sqlite3
import hashlib

def get_user(cursor: sqlite3.Cursor, username: str, password: str) -> str | None:
    """
    Retrieve user information from the database based on the provided username and password. If the provided credentials match a user in the database, return the user information; otherwise, return None.

    Args:
        cursor: A cursor object to interact with the SQLite database.
        username: The username of the user to retrieve.
        password: The password of the user to retrieve.

    Returns:
        str | None: The user information if the user is found and credentials match; otherwise, None.
    """
    try:
        # Query to select salt and hashed password based on username
        query = "SELECT salt, passwordHash FROM users WHERE username = ?"
        cursor.execute(query, (username,))
        result = cursor.fetchone()
        
        if not result:
            return None
        
        salt = result[0]
        stored_hash = result[1]
        
        # Generate hash from provided password and stored salt
        key = hashlib.pbkdf2_hmac('sha256', password.encode(), salt, 100000)
        provided_hash = key.hex()
        
        if provided_hash != stored_hash:
            return None
        
        # Now retrieve the userInfo using the correct credentials
        query = "SELECT userInfo FROM users WHERE username = ?"
        cursor.execute(query, (username,))
        user_info_result = cursor.fetchone()
        
        return user_info_result[0] if user_info_result else None
    
    except sqlite3.Error:
        return None

import hashlib

    """
    Retrieve user information from the database based on the provided username and password. If the provided credentials match a user in the database, return the user information; otherwise, return None.

    Args:
        cursor: A cursor object to interact with the SQLite database.
        username: The username of the user to retrieve.

    Returns:
        str | None: The user information if the user is found and credentials match; otherwise, None.
    """
    try:

        cursor.execute(query, (username,))
        result = cursor.fetchone()
        
        if not result:

        stored_hash = result[1]
        
        # Generate hash from provided password and stored salt
        key = hashlib.pbkdf2_hmac('sha256', password.encode(), salt, 100000)
        provided_hash = key.hex()
        
        if provided_hash != stored_hash:
            return None
        
        # Now retrieve the userInfo using the correct credentials
        query = "SELECT userInfo FROM users WHERE username = ?"
        cursor.execute(query, (username,))
        user_info_result = cursor.fetchone()
        
        return user_info_result[0] if user_info_result else None
    
    except sqlite3.Error:
        return None
