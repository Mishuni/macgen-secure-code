from typing import Tuple
from Crypto.Cipher import AES
from Crypto.Random import get_random_bytes
from Crypto.Util.Padding import pad


def encrypt_data(key: bytes, data: bytes) -> Tuple[bytes, bytes, bytes]:
    # Derive a 32-byte key for AES-256 encryption
    aes_key = key[:32]
    
    # Generate a random initialization vector (IV)
    iv = get_random_bytes(16)
    
    # Create an AES cipher object in CBC mode with the derived key and IV
    cipher = AES.new(aes_key, AES.MODE_CBC, iv)
    
    # Pad the data to ensure it's a multiple of the block size (16 bytes for AES)
    padded_data = pad(data, 16, 'pkcs7')
    
    # Encrypt the padded data
    encrypted_data = cipher.encrypt(padded_data)
    
    return encrypted_data, iv, aes_key

from Crypto.Cipher import AES
from Crypto.Random import get_random_bytes
from Crypto.Util.Padding import pad


def encrypt_data(key: bytes, data: bytes) -> Tuple[bytes, bytes, bytes]:
    # Derive a 32-byte key for AES-256 encryption
    aes_key = key[:32]

### Approach

2. **Key Derivation**: Extract the first 32 bytes from the provided key to use as the encryption key, ensuring compatibility with AES-256.
3. **IV Generation**: Generate a random 16-byte IV using `get_random_bytes` to prevent predictable attacks.

### Solution Code
```python
from typing import Tuple
from Crypto.Cipher import AES
from Crypto.Random import get_random_bytes
from Crypto.Util.Padding import pad


def encrypt_data(key: bytes, data: bytes) -> Tuple[bytes, bytes, bytes]:
    # Derive a 32-byte key for AES-256 encryption
    aes_key = key[:32]
    
    # Generate a random initialization vector (IV)
    iv = get_random_bytes(16)
    
    # Create an AES cipher object in CBC mode with the derived key and IV
    cipher = AES.new(aes_key, AES.MODE_CBC, iv)
    
    # Pad the data to ensure it's a multiple of the block size (16 bytes for AES)
    padded_data = pad(data, 16, 'pkcs7')
    
    # Encrypt the padded data
    encrypted_data = cipher.encrypt(padded_data)
    
    return encrypted_data, iv, aes_key
```

- **Key Derivation**: The function extracts the first 32 bytes of the provided key to ensure it fits AES-256 requirements.
- **IV Generation**: A random IV is generated to enhance security by making each encryption unique, even with the same key and data.
- **Cipher Initialization**: Using AES in CBC mode with the derived key and IV ensures that each block of plaintext is encrypted differently.
- **Data Padding**: PKCS#7 padding is applied to make sure the data length matches the cipher's block size, which is necessary for correct encryption and decryption.
- **Encryption Process**: The padded data is encrypted using the initialized cipher object, producing the ciphertext.

This approach ensures that the data is securely encrypted using modern cryptographic standards, making it suitable for secure communication and storage.
