#include <stdio.h>
#include <stdlib.h>

char *read_file_from_dir(const char *filename, const char *dir_path) {
    if (filename == NULL || dir_path == NULL) {
        return "";
    }

    // Calculate the required size for the full path
    size_t dir_len = strlen(dir_path);
    size_t file_len = strlen(filename);
    size_t full_path_size = dir_len + file_len + 2; // +1 for '/' and +1 for '\0'

    char *full_path = malloc(full_path_size);
    if (full_path == NULL) {
        return "";
    }

    // Safely construct the full path using snprintf
    int bytes_written = snprintf(full_path, full_path_size, "%s/%s", dir_path, filename);
    if (bytes_written < 0 || (size_t)bytes_written >= full_path_size) {
        free(full_path);
        return "";
    }

    // Open the file in read mode
    FILE *file = fopen(full_path, "r");
    if (!file) {
        free(full_path);
        return "";
    }

    // Determine the size of the file
    if (fseek(file, 0, SEEK_END) != 0) {
        fclose(file);
        free(full_path);
        return "";
    }
    long file_size = ftell(file);
    rewind(file);

    // Handle potential overflow by checking for negative sizes
    if (file_size < 0) {
        fclose(file);
        free(full_path);
        return "";
    }

    size_t content_size = (size_t)file_size;
    char *content = malloc(content_size + 1); // +1 for null terminator
    if (!content) {
        fclose(file);
        free(full_path);
        return "";
    }

    // Read the file in chunks to handle large files efficiently
    size_t total_read = 0;
    size_t chunk_size = 4096; // Adjust buffer size as needed
    char *buffer = malloc(chunk_size);
    if (!buffer) {
        free(content);
        fclose(file);
        free(full_path);
        return "";
    }

    while (total_read < content_size) {
        size_t bytes_to_read = (content_size - total_read < chunk_size) ? 
                             (content_size - total_read) : chunk_size;
        
        size_t read_bytes = fread(buffer, 1, bytes_to_read, file);
        if (read_bytes == 0 && ferror(file)) { // Check for read errors
            free(buffer);
            free(content);
            fclose(file);
            free(full_path);
            return "";
        }
        memcpy(content + total_read, buffer, read_bytes);
        total_read += read_bytes;
    }

    free(buffer);

    content[content_size] = '\0'; // Null-terminate the string

    // Clean up resources
    fclose(file);
    free(full_path);

    return content;
}
