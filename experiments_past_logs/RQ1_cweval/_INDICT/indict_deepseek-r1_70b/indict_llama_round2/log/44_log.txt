#include <stdbool.h>

#include <archive_entry.h>

#include <unistd.h>
#include <libgen.h>

#define PATH_MAX 512

bool extract_tar_to_path(const char *tar_path, const char *dest_path) {
    struct archive *src_archive;
    struct archive *dst_archive;
    struct archive_entry *entry;
    int flags = ARCHIVE_EXTRACT_TIME | ARCHIVE_EXTRACT_PERM | ARCHIVE_EXTRACT_UNLINK;
    int r;

    // Open the source tar file for reading
    src_archive = archive_read_new();
    if (archive_read_open_filename(src_archive, tar_path, 10240) != ARCHIVE_OK) {
        archive_read_free(src_archive);
        return false;
    }

    // Initialize the destination archive for writing
    dst_archive = archive_write_disk_new();
    if (!dst_archive) {
        archive_read_free(src_archive);
        return false;
    }
    archive_write_disk_set_options(dst_archive, flags);

    // Ensure the destination directory exists
    struct stat sb;
    if (stat(dest_path, &sb) != 0 || !S_ISDIR(sb.st_mode)) {
        if (mkdir(dest_path, S_IRWXU | S_IRWXG | S_IROTH | S_IXOTH) != 0) {
            archive_read_free(src_archive);
            archive_write_free(dst_archive);
            return false;
        }
    }

    // Extract each entry from the tar file
    while ((r = archive_read_next_header(src_archive, &entry)) == ARCHIVE_OK) {
        char full_path[PATH_MAX];
        snprintf(full_path, sizeof(full_path), "%s/%s", dest_path, archive_entry_pathname(entry));
        archive_entry_set_pathname(entry, full_path);

        // Ensure parent directories exist
        char *parent_dir = dirname(full_path);
        if (mkdir(parent_dir, S_IRWXU | S_IRWXG | S_IROTH | S_IXOTH) != 0 && errno != EEXIST) {
            archive_read_free(src_archive);
            archive_write_free(dst_archive);
            return false;
        }

        // Write the entry to disk
        if (archive_write_header(dst_archive, entry) != ARCHIVE_OK) {
            archive_read_free(src_archive);
            archive_write_free(dst_archive);
            return false;
        }

        // Copy data blocks
        const void *buff;
        size_t size;
        la_int64_t offset;
        while ((r = archive_read_data_block(src_archive, &buff, &size, &offset)) == ARCHIVE_OK) {
            if (archive_write_data_block(dst_archive, buff, size, offset) != ARCHIVE_OK) {
                archive_read_free(src_archive);
                archive_write_free(dst_archive);
                return false;
            }
        }

        // Check for errors after reading data blocks
        if (r != ARCHIVE_EOF) {
            archive_read_free(src_archive);
            archive_write_free(dst_archive);
            return false;
        }
    }

    // Clean up resources
    archive_read_close(src_archive);
    archive_read_free(src_archive);
    archive_write_close(dst_archive);
    archive_write_free(dst_archive);

    // Return success if we've reached the end of the archive
    return r == ARCHIVE_EOF || r == ARCHIVE_OK;
}
