    if (!out) {
        return false;

    archive_write_disk_set_options(out.get(), flags);

        if (sanitized_path.substr(0, normalized_dest.length()) != normalized_dest) {
            return false; // Prevent directory traversal
        }

        archive_entry_set_pathname(entry, sanitized_path.c_str());

        if (archive_write_header(out.get(), entry) != ARCHIVE_OK) {
            return false;
