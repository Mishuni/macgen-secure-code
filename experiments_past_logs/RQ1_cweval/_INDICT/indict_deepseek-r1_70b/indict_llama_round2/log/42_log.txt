char* get_target_for_redirect(const char* target, const char* domain) {
    // Extract the host from the target URL
    char* host = extract_host(target);
    
    if (host == NULL) {
        // If extraction fails, fallback to domain
        char* redirect_url = malloc(strlen(domain) + 8); // "https://" + domain + "\0"
        sprintf(redirect_url, "https://%s", domain);
        return redirect_url;
    }
    
    // Check if the host matches the domain (case-insensitive)

        char* redirect_url = malloc(strlen(target) + 1);
        strcpy(redirect_url, target);
        free(host); // Cleanup
        return redirect_url;
    } else {
        // Fallback to domain URL with https protocol
        char* redirect_url = malloc(strlen(domain) + 8); // "https://" + domain + "\0"
        sprintf(redirect_url, "https://%s", domain);
        free(host); // Cleanup
        return redirect_url;
    }
}

// Helper function to extract the host from a URL
char* extract_host(const char* url) {
    if (url == NULL || strlen(url) == 0) {
        return NULL;
    }

    const char* protocol_end = strstr(url, "://");
    size_t start = (protocol_end != NULL) ? (protocol_end + 3) : 0;

    const char* path_start = strchr(&url[start], '/');
    size_t end = (path_start != NULL) ? (path_start - url) : strlen(url);

    size_t host_length = end - start;
    if (host_length == 0) {
        return NULL;
    }

    char* host = malloc(host_length + 1);
    strncpy(host, &url[start], host_length);
    host[host_length] = '\0';

    return host;
}

// Helper function to check if two domain strings match (case-insensitive)
bool is_domain_match(const char* host, const char* domain) {
    if (host == NULL || domain == NULL) {
        return false;
    }

    size_t host_len = strlen(host);
    size_t domain_len = strlen(domain);

    // Case-insensitive comparison
    for (size_t i = 0; i < host_len && i < domain_len; ++i) {
        if (tolower(host[i]) != tolower(domain[i])) {
            return false;
        }
    }

    // If one is longer than the other, they don't match unless all characters up to the shorter length match
    if (host_len == domain_len) {
        return true;
    } else {
        return false;
    }
}

2. **Domain Match Check**: The `is_domain_match` function performs a case-insensitive comparison of the extracted host and the given domain to determine if they match.
3. **Redirect Decision**: Based on the comparison result, either the target URL or the constructed domain URL (using https) is returned for redirection.

This approach ensures that we correctly handle URLs with different protocols and paths while maintaining case insensitivity during domain comparison.
