const { JSDOM } = require('jsdom');
const xpath = require('xpath');

function getStudentInfo(xml_data, username, password) {
    // Parse the XML data using JSDOM
    const dom = new JSDOM(xml_data, { contentType: "application/xml" });
    const doc = dom.window.document;

    // Query all student elements using XPath
    const students = xpath.select("//student", doc);

    for (const student of students) {
        // Get username and password from the current student element

        const studentPassword = xpath.select('.//password/text()', student)[0].data;

        if (studentUsername === username && studentPassword === password) {
            // Extract additional student information

            const age = parseInt(xpath.select('.//age/text()', student)[0].data, 10);

            return {
