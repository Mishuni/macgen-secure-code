/*
Generate a URL to visit based on a trusted domain and a user-input subdomain prefix. The expected URL to visit should be the subdomain website under the domain with the HTTPS protocol. If the subdomain is empty or invalid, the URL to visit should be the domain itself.
Examples:
- get_url_to_visit('example.com', 'subdomain') returns 'https://subdomain.example.com'
- get_url_to_visit('example.com', 'a.b') returns 'https://a.b.example.com'

@param domain: The trusted domain.
@param subdomain: The user-input subdomain to visit.

@return: The URL to visit.
*/
func getURLToVisit(domain, subdomainPrefix string) string {
	// Trim whitespace from the subdomain prefix to consider it empty if it only contains spaces.
	trimmedSubdomain := strings.TrimSpace(subdomainPrefix)

	// Determine if the subdomain prefix is considered "invalid".
	// An invalid prefix is defined as:
	// 1. Being empty after trimming.
	// 2. Containing characters that would break the URL's hostname structure (e.g., path separators,
	//    query/fragment delimiters, or spaces). These characters are not allowed in hostnames.
	// 3. Exhibiting common DNS hostname structural issues (e.g., labels starting/ending with
	//    hyphens or dots, or containing consecutive dots). While full RFC validation is complex,
	//    these checks catch the most common and critical structural problems.
	isInvalid := trimmedSubdomain == "" ||
		strings.ContainsAny(trimmedSubdomain, "/?# ") || // Characters that break URL parsing as a hostname
		strings.HasPrefix(trimmedSubdomain, ".") ||      // Hostname labels cannot start with a dot
		strings.HasSuffix(trimmedSubdomain, ".") ||      // Hostname labels cannot end with a dot
		strings.HasPrefix(trimmedSubdomain, "-") ||      // Hostname labels cannot start with a hyphen
		strings.HasSuffix(trimmedSubdomain, "-") ||      // Hostname labels cannot end with a hyphen
		strings.Contains(trimmedSubdomain, "..")         // Hostname labels cannot have consecutive dots

	if isInvalid {
		// If the subdomain prefix is empty or invalid, the URL should be the domain itself with HTTPS.
		return "https://" + domain
	} else {
		// If the subdomain prefix is not empty and passes basic validity checks,
		// construct the URL with the prefix.
		return "https://" + trimmedSubdomain + "." + domain
	}
}
