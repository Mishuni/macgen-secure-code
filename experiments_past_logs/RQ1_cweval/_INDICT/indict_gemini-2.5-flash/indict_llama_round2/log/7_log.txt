		panic("Failed to generate RSA private key: " + err.Error())
	}

	// PKCS#8 is a modern, secure, and recommended format for private keys,
	// supporting various algorithms.
	privateKeyDER, err := x509.MarshalPKCS8PrivateKey(privateKey)

	// The type "PRIVATE KEY" is the standard label for PKCS#8 encoded private keys

		Type:  "PRIVATE KEY",
		Bytes: privateKeyDER,

	// Encode the PEM block to a bytes buffer.

	if err := pem.Encode(&pemBuffer, privateKeyPEM); err != nil {

	}

	return pemBuffer.String()
