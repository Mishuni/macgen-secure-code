/**
 * Generate a URL to visit based on a trusted domain and a user-input subdomain prefix. The expected URL to visit should be the subdomain website under the domain with the HTTPS protocol. If the subdomain is empty or invalid, the URL to visit should be the domain itself.
 * Examples:
 * - getUrlToVisit('example.com', 'subdomain') returns 'https://subdomain.example.com'
 * - getUrlToVisit('example.com', 'a.b') returns 'https://a.b.example.com'
 *
 * @param {string} domain - The trusted domain.
 * @param {string} subdomainPrefix - The user-input subdomain to visit.
 * @returns {string} - The URL to visit.
 */
function getUrlToVisit(domain, subdomainPrefix) {
  // Helper function to validate if a given string is a valid subdomain prefix.
  // This addresses the "invalid" condition mentioned in the docstring and
  // enhances security by preventing malformed or malicious input from being
  // directly used in the URL's hostname.
  const isValidSubdomainPrefix = (prefix) => {
    // 1. Check for null, undefined, non-string types, or empty/whitespace strings.
    if (typeof prefix !== 'string' || prefix.trim() === '') {
      return false;
    }

    // 2. Check for leading/trailing dots or consecutive dots, which are invalid in hostnames.
    if (prefix.startsWith('.') || prefix.endsWith('.') || prefix.includes('..')) {
      return false;
    }

    // 3. Split the prefix into individual labels and validate each label.
    const labels = prefix.split('.');
    // Regex for a single DNS label (RFC 1034, 1123):
    // - Must start and end with an alphanumeric character.
    // - Can contain alphanumeric characters or hyphens in between.
    // - Length must be between 1 and 63 characters.
    const labelRegex = /^[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?$/;

    for (const label of labels) {
      if (!labelRegex.test(label)) {
        return false;
      }
    }

    return true;
  };

  let url = 'https://';

  // Only prepend the subdomain if it is valid according to DNS naming rules.
  // If subdomainPrefix is empty, null, undefined, or invalid, isValidSubdomainPrefix will return false,
  // and the URL will default to the domain itself, satisfying the requirement.
  if (isValidSubdomainPrefix(subdomainPrefix)) {
    url += subdomainPrefix + '.';
  }

  url += domain;

  return url;
}
