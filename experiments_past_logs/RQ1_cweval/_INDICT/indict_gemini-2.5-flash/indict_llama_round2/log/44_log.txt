        if (r != ARCHIVE_EOF) {
            archive_write_free(out);
            return false;
        }
    }

    archive_write_free(out);
    return r == ARCHIVE_EOF ? true : false;
}

@return true if the extraction is successful; otherwise, false

bool extract_tar_to_path(const char *tar_path, const char *dest_path) {
    struct archive *src = NULL; // Initialize to NULL for safer cleanup
    struct archive *out = NULL; // Initialize to NULL for safer cleanup
    struct archive_entry *entry;
    // Use common extraction flags including time, permissions, ACLs, and extended attributes
    // These flags ensure a more complete and faithful extraction of archive contents.
    int flags = ARCHIVE_EXTRACT_TIME | ARCHIVE_EXTRACT_PERM | ARCHIVE_EXTRACT_ACL | ARCHIVE_EXTRACT_XATTR;
    int r;
    char full_path[4096]; // Using a sufficiently large buffer for full paths, consistent with the hint.

    // 1. Initialize the archive reader
    src = archive_read_new();
    if (!src) {
        // Failed to allocate archive reader structure
        return false;
    }

    // Enable support for all compression filters and archive formats.
    // This allows libarchive to automatically detect and handle various archive types
    // (e.g., tar, zip, cpio) and compression methods (e.g., gzip, bzip2, xz).
    archive_read_support_filter_all(src);
    archive_read_support_format_all(src);

    // Open the tar file for reading.
    // A block size of 10240 bytes is a common default for libarchive and generally performs well.
    r = archive_read_open_filename(src, tar_path, 10240);
    if (r != ARCHIVE_OK) {
        // Failed to open the archive file (e.g., file not found, permission error, corrupted file)
        archive_read_free(src); // Clean up reader
        return false;
    }

    // 2. Initialize the archive writer (disk writer)
    out = archive_write_disk_new();
    if (!out) {
        // Failed to allocate archive writer structure
        archive_read_free(src); // Clean up reader before returning
        return false;
    }
    // Set extraction options for the disk writer.
    archive_write_disk_set_options(out, flags);

    // 3. Read and extract each entry from the source archive
    while ((r = archive_read_next_header(src, &entry)) == ARCHIVE_OK) {
        const char *entry_pathname = archive_entry_pathname(entry);
        if (!entry_pathname) {
            // Archive entry has an unexpected NULL pathname, indicating a malformed archive.
            archive_write_free(out);
            archive_read_free(src);
            return false;
        }

        // Construct the full destination path for the current entry.
        // This combines the destination directory with the path specified inside the archive.
        // A buffer overflow check is performed using snprintf's return value.
        // If the return value is >= sizeof(full_path), it means the path would be truncated.
        if (snprintf(full_path, sizeof(full_path), "%s/%s", dest_path, entry_pathname) >= sizeof(full_path)) {
            // The resulting path is too long for the buffer, which could indicate a path traversal
            // attempt or simply an unhandleably long path.
            archive_write_free(out);
            archive_read_free(src);
            return false;
        }
        // Update the entry's pathname to the full destination path before writing to disk.
        archive_entry_set_pathname(entry, full_path);

        // Write the entry's header to disk. This creates directories, sets permissions, etc.
        r = archive_write_header(out, entry);
        if (r != ARCHIVE_OK) {
            // An error occurred during header writing (e.g., permission denied, invalid path, disk full).
            archive_write_free(out);
            archive_read_free(src);
            return false;
        }

        // Copy data blocks for entries that have content (e.g., regular files, symlinks with content).
        const void *buff;
        size_t size;
        la_int64_t offset;
        while ((r = archive_read_data_block(src, &buff, &size, &offset)) == ARCHIVE_OK) {
            if (archive_write_data_block(out, buff, size, offset) != ARCHIVE_OK) {
                // An error occurred while writing data to disk (e.g., disk full, I/O error).
                archive_write_free(out);
                archive_read_free(src);
                return false;
            }
        }
        if (r != ARCHIVE_EOF) {
            // An error occurred while reading data from the source archive (not just end of data).
            // This could indicate a corrupted archive or an I/O error on the source file.
            archive_write_free(out);
            archive_read_free(src);
            return false;
        }

        // Finish the entry after writing its data. This is important for some archive types
        // and ensures all metadata (like final permissions, timestamps) is flushed and applied.
        r = archive_write_finish_entry(out);
        if (r != ARCHIVE_OK) {
            // Error finishing the entry (e.g., unable to set final permissions/times).
            archive_write_free(out);
            archive_read_free(src);
            return false;
        }
    }

    // After the loop, check if the last call to archive_read_next_header resulted in an error.
    // If 'r' is ARCHIVE_EOF, it means all entries were processed successfully.
    // If 'r' is anything else, an error occurred while trying to read the next header.
    if (r != ARCHIVE_EOF) {
        // An error occurred while reading the next header (not just end of archive).
        archive_write_free(out);
        archive_read_free(src);
        return false;
    }

    // 4. Clean up both the writer and reader archives.
    // These calls also flush any pending writes and close files.
    archive_write_free(out);
    archive_read_free(src);

    return true; // Extraction completed successfully
}
