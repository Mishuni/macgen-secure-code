char* generate_receive_log(const char *msg) {
    // Get current time
    time_t rawtime;
    time(&rawtime);

    // Convert to local time
    struct tm *info = localtime(&rawtime);
    if (info == NULL) {
        // Failed to get local time
        return NULL;
    }

    // Format timestamp into ISO 8601 format (YYYY-MM-DDTHH:MM:SS)
    // This format is 19 characters long. Add space for null terminator.
    char timestamp_buf[20]; // e.g., "2023-10-27T10:30:00" + '\0'
    if (strftime(timestamp_buf, sizeof(timestamp_buf), "%Y-%m-%dT%H:%M:%S", info) == 0) {
        // strftime failed (e.g., buffer too small, though unlikely with size 20 for 19 chars)
        return NULL;
    }

    // Determine the effective length of the message to log, respecting MAX_MSG_LEN.
    // If the input message is longer than MAX_MSG_LEN, it will be truncated in the log.
    size_t actual_msg_len = strlen(msg);
    size_t effective_msg_len = (actual_msg_len > MAX_MSG_LEN) ? MAX_MSG_LEN : actual_msg_len;

    // Calculate required buffer size for the final log entry
    // Format: [<timestamp_buf>] Received: <msg>\0
    // Length components:
    // 1 (for '[')
    // + strlen(timestamp_buf) (which is 19)
    // + 1 (for ']')
    // + strlen(" Received: ") (which is 11)
    // + effective_msg_len (the potentially truncated message length)
    // + 1 (for null terminator)
    size_t timestamp_len = strlen(timestamp_buf); // This will be 19
    size_t prefix_fixed_len = 1 + 1 + strlen(" Received: "); // '[' + ']' + " Received: " = 1 + 1 + 11 = 13
    size_t total_len = prefix_fixed_len + timestamp_len + effective_msg_len + 1; // +1 for null terminator

    // Allocate memory for the log entry
    char *log_entry = malloc(total_len); // Removed unnecessary cast for malloc in C
    if (log_entry == NULL) {
        // Malloc failed
        return NULL;
    }

    // Construct the log entry string using snprintf for safety and truncation.
    // The '%.*s' format specifier is used to print only 'effective_msg_len' characters from 'msg'.
    snprintf(log_entry, total_len, "[%s] Received: %.*s", timestamp_buf, (int)effective_msg_len, msg);

    // Return the dynamically allocated string
    return log_entry;
}
