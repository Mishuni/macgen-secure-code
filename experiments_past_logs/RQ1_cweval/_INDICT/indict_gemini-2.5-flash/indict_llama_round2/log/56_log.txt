    // This is crucial for preventing XPath injection by treating username/password as literal strings.
    username_var_obj = xmlXPathNewString((const xmlChar*)username);
    if (username_var_obj == NULL) {
        goto cleanup;
    }
    password_var_obj = xmlXPathNewString((const xmlChar*)password);
    if (password_var_obj == NULL) {
        goto cleanup;

    // The variable names 'user' and 'pass' are used in the xpath_query string.
    // xmlXPathRegisterVariable returns 0 on success.
    if (xmlXPathRegisterVariable(xpathCtx, (const xmlChar*)"user", username_var_obj) != 0) {
        goto cleanup;
    }
    if (xmlXPathRegisterVariable(xpathCtx, (const xmlChar*)"pass", password_var_obj) != 0) {
        goto cleanup;
    }

    xpathObj = xmlXPathEvalExpression((const xmlChar*)xpath_query, xpathCtx);
    if (xpathObj == NULL || xpathObj->nodesetval == NULL || xpathObj->nodesetval->nodeNr == 0) {
        // No matching student found or XPath evaluation failed.
        goto cleanup;
    }

    // We expect only one student to match the unique username/password combination.
    xmlNodePtr student_node = xpathObj->nodesetval->nodeTab[0];
    if (student_node == NULL) {
        goto cleanup;
    }

    // 6. Allocate memory for the StudentInfo structure
    student_info = (StudentInfo*)malloc(sizeof(StudentInfo));
    if (student_info == NULL) {
        goto cleanup;
    }
    // Initialize members to NULL to safely handle partial allocation/strdup failures
    student_info->name = NULL;
    student_info->age = NULL;
    student_info->citizenship = NULL;

    xmlNodePtr current_node = student_node->children;
    while (current_node != NULL) {
        if (current_node->type == XML_ELEMENT_NODE) {
            xmlChar* content = xmlNodeGetContent(current_node); // Get the text content of the element
            if (content != NULL) {
                if (xmlStrcmp(current_node->name, (const xmlChar*)"name") == 0) {
                    student_info->name = strdup((char*)content);
                } else if (xmlStrcmp(current_node->name, (const xmlChar*)"age") == 0) {
                    student_info->age = strdup((char*)content);
                } else if (xmlStrcmp(current_node->name, (const xmlChar*)"citizenship") == 0) {
                    student_info->citizenship = strdup((char*)content);
                }
                xmlFree(content); // Free memory allocated by xmlNodeGetContent
            }
            // Check for strdup failure immediately
            if ((xmlStrcmp(current_node->name, (const xmlChar*)"name") == 0 && student_info->name == NULL) ||
                (xmlStrcmp(current_node->name, (const xmlChar*)"age") == 0 && student_info->age == NULL) ||
                (xmlStrcmp(current_node->name, (const xmlChar*)"citizenship") == 0 && student_info->citizenship == NULL)) {
                // strdup failed for one of the fields. Free any fields that were successfully duplicated.
                if (student_info->name) free(student_info->name);
                if (student_info->age) free(student_info->age);
                if (student_info->citizenship) free(student_info->citizenship);
                free(student_info);
                student_info = NULL; // Indicate failure
                goto cleanup;
            }
        }
        current_node = current_node->next;
    }

    // Final check: ensure all required fields were successfully populated.
    // This handles cases where a field might be missing in the XML for the matched student.
    if (student_info != NULL && (student_info->name == NULL || student_info->age == NULL || student_info->citizenship == NULL)) {
        if (student_info->name) free(student_info->name);
        if (student_info->age) free(student_info->age);
        if (student_info->citizenship) free(student_info->citizenship);
        free(student_info);
        student_info = NULL; // Indicate failure
    }

cleanup:
    // 8. Clean up libxml2 resources
    if (xpathObj != NULL) {
        xmlXPathFreeObject(xpathObj);
    }
    // Free the variable objects created by xmlXPathNewString.
    // xmlXPathRegisterVariable increments their ref count, so we must free our local copies.
    if (username_var_obj != NULL) {
        xmlXPathFreeObject(username_var_obj);
    }
    if (password_var_obj != NULL) {
        xmlXPathFreeObject(password_var_obj);
    }
    if (xpathCtx != NULL) {
        xmlXPathFreeContext(xpathCtx);
    }
    if (doc != NULL) {
        xmlFreeDoc(doc);
    }

    return student_info;
}
