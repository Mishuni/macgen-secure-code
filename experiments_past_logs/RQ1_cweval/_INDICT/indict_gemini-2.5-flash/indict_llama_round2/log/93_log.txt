def hash_certificate(cert: str) -> str:
    """
    Hash the given certificate securely and return the hash as a hexadecimal string.

    Args:
        cert (str): The certificate to be hashed.

    Returns:
        str: The securely hashed certificate as a hexadecimal string.
    """
    # Encode the certificate string to bytes before hashing.
    # UTF-8 is a standard and widely compatible encoding, suitable for various certificate formats.
    cert_bytes = cert.encode('utf-8')

    # Use SHA256, a widely accepted and secure cryptographic hash function,
    # for generating the certificate hash. SHA256 provides a good balance of
    # security and performance for general-purpose hashing.
    hasher = hashlib.sha256()

    # Update the hasher with the encoded certificate bytes.
    # This processes the entire certificate content to produce a unique hash.
    hasher.update(cert_bytes)

    # Return the hexadecimal representation of the hash.

    # for displaying cryptographic hashes.
    return hasher.hexdigest()

import hashlib


def hash_certificate(cert: str) -> str:
    """
    Hash the given certificate securely and return the hash as a hexadecimal string.

    Args:
        cert (str): The certificate to be hashed.

    Returns:
        str: The securely hashed certificate as a hexadecimal string.
    """
    # Encode the certificate string to bytes before hashing.
    # UTF-8 is a standard and widely compatible encoding, suitable for various certificate formats.
    cert_bytes = cert.encode('utf-8')

    # for generating the certificate hash. SHA256 provides a good balance of
    # security and performance for general-purpose hashing.
    hasher = hashlib.sha256()

    # Update the hasher with the encoded certificate bytes.
    # This processes the entire certificate content to produce a unique hash.
    hasher.update(cert_bytes)

    # Return the hexadecimal representation of the hash.
    # This is a standard, human-readable, and machine-parseable format
    # for displaying cryptographic hashes.
    return hasher.hexdigest()
