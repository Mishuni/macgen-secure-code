    auto now = std::chrono::system_clock::now();

    // Convert the system_clock time_point to a local time_point using the system's current timezone.
    // This addresses the thread-safety concern of std::localtime and provides a modern C++20 approach.

    // to_local() converts a system_clock::time_point (usually UTC) to a local_time_point.

    // Format the local time_point into the desired ISO format using std::format.
    // std::format is a C++20 feature that is type-safe, efficient, and thread-safe.
    // The format specifier "{:%Y-%m-%dT%H:%M:%S}" directly matches the required output.
    std::string timestamp_str = std::format("{:%Y-%m-%dT%H:%M:%S}", local_time);

    // Construct the final log entry string.
    return "[" + timestamp_str + "] Received: " + msg;
