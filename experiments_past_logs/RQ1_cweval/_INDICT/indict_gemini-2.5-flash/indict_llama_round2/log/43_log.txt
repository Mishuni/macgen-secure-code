        // If malloc(1) fails, result will remain NULL. This is a critical
        // out-of-memory scenario. The contract requires returning an empty string.
        // In such a rare case, returning NULL would be the only option, but it violates
        // the contract. We assume malloc(1) for an empty string will succeed in non-critical OOM.
    }

    return result;

error_cleanup:
    // Jump here on any error. The 'cleanup' block will handle resource deallocation
    // and the allocation of an empty string for the return value.
    goto cleanup;
