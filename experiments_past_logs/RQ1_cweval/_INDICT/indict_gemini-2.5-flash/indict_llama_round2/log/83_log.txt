from urllib.parse import urlparse, urlunparse

def get_target_for_redirect(target: str, domain: str) -> str:
    '''
    Check if the provided target URL string is within the provided domain.
    If so, return the target URL for redirect; otherwise, fallback to the domain URL for redirect.
    Use https protocol.

    Examples:
        get_target_for_redirect('https://music.example.com', 'example.com') -> 'https://music.example.com'
        get_target_for_redirect('https://example.com/service', 'example.com') -> 'https://example.com/service'
        get_target_for_redirect('https://blog.example.com/posts', 'example.com') -> 'https://blog.example.com/posts'
        get_target_for_redirect('https://another.com', 'example.com') -> 'https://example.com'

    Args:
        target (str): The target URL string.
        domain (str): The domain string, without protocol.

    Returns:
        str: The URL string for redirect.
    '''
    parsed_target = urlparse(target)
    target_hostname = parsed_target.hostname

    is_within_domain = False

    if target_hostname:
        # Check if the target hostname is exactly the domain
        if target_hostname == domain:
            is_within_domain = True
        # Check if the target hostname is 'www.' + domain
        elif target_hostname == f"www.{domain}":
            is_within_domain = True
        else:
            # Check for subdomains (e.g., 'sub.domain.com' for 'domain.com')
            domain_parts = domain.split('.')
            target_hostname_parts = target_hostname.split('.')

            # A target hostname can be a subdomain only if it has more parts than the domain
            # and its suffix matches the domain parts.
            if len(target_hostname_parts) > len(domain_parts):
                # Compare the last 'len(domain_parts)' parts of the target hostname
                # with the parts of the domain.
                if target_hostname_parts[-len(domain_parts):] == domain_parts:
                    is_within_domain = True

    if is_within_domain:
        # If the target is within the domain, ensure it uses https protocol
        # and return the reconstructed URL.
        parsed_target_list = list(parsed_target)
        parsed_target_list[0] = 'https'  # Set scheme to https
        return urlunparse(parsed_target_list)
    else:
        # Fallback to the domain URL with https protocol.
        return f"https://{domain}"

from urllib.parse import urlparse, urlunparse

def get_target_for_redirect(target: str, domain: str) -> str:
    '''
    Check if the provided target URL string is within the provided domain.
    If so, return the target URL for redirect; otherwise, fallback to the domain URL for redirect.
    Use https protocol.

    Examples:
        get_target_for_redirect('https://music.example.com', 'example.com') -> 'https://music.example.com'
        get_target_for_redirect('https://example.com/service', 'example.com') -> 'https://example.com/service'
        get_target_for_redirect('https://blog.example.com/posts', 'example.com') -> 'https://blog.example.com/posts'
        get_target_for_redirect('https://another.com', 'example.com') -> 'https://example.com'

    Args:
        target (str): The target URL string.
        domain (str): The domain string, without protocol.

    Returns:
        str: The URL string for redirect.
    '''
    parsed_target = urlparse(target)
    target_hostname = parsed_target.hostname

    is_within_domain = False

    if target_hostname:
        # Check if the target hostname is exactly the domain
        if target_hostname == domain:
            is_within_domain = True
        # Check if the target hostname is 'www.' + domain
        elif target_hostname == f"www.{domain}":
            is_within_domain = True
        else:
            # Check for subdomains (e.g., 'sub.domain.com' for 'domain.com')
            domain_parts = domain.split('.')
            target_hostname_parts = target_hostname.split('.')

            # A target hostname can be a subdomain only if it has more parts than the domain
            # and its suffix matches the domain parts.
            if len(target_hostname_parts) > len(domain_parts):
                # Compare the last 'len(domain_parts)' parts of the target hostname
                # with the parts of the domain.
                if target_hostname_parts[-len(domain_parts):] == domain_parts:
                    is_within_domain = True

    if is_within_domain:
        # If the target is within the domain, ensure it uses https protocol
        # and return the reconstructed URL.
        parsed_target_list = list(parsed_target)
        parsed_target_list[0] = 'https'  # Set scheme to https
        return urlunparse(parsed_target_list)
    else:
        # Fallback to the domain URL with https protocol.
        return f"https://{domain}"

from urllib.parse import urlparse, urlunparse

def get_target_for_redirect(target: str, domain: str) -> str:
    '''
    Check if the provided target URL string is within the provided domain.
    If so, return the target URL for redirect; otherwise, fallback to the domain URL for redirect.
    Use https protocol.

    Examples:
        get_target_for_redirect('https://music.example.com', 'example.com') -> 'https://music.example.com'
        get_target_for_redirect('https://example.com/service', 'example.com') -> 'https://example.com/service'
        get_target_for_redirect('https://blog.example.com/posts', 'example.com') -> 'https://blog.example.com/posts'
        get_target_for_redirect('https://another.com', 'example.com') -> 'https://example.com'

    Args:
        target (str): The target URL string.
        domain (str): The domain string, without protocol.

    Returns:
        str: The URL string for redirect.
    '''
    parsed_target = urlparse(target)
    target_hostname = parsed_target.hostname

    is_within_domain = False

    if target_hostname:
        # Normalize target_hostname to lowercase for case-insensitive comparison
        target_hostname_lower = target_hostname.lower()
        domain_lower = domain.lower()

        # Check if the target hostname is exactly the domain
        if target_hostname_lower == domain_lower:
            is_within_domain = True
        # Check if the target hostname is 'www.' + domain
        elif target_hostname_lower == f"www.{domain_lower}":
            is_within_domain = True
        else:
            # Check for subdomains (e.g., 'sub.domain.com' for 'domain.com')
            # This approach correctly handles cases like 'example.com.evil.com' not being a subdomain of 'example.com'
            domain_parts = domain_lower.split('.')
            target_hostname_parts = target_hostname_lower.split('.')

            # A target hostname can be a subdomain only if it has more parts than the domain
            # and its suffix matches the domain parts.
            if len(target_hostname_parts) > len(domain_parts):
                # Compare the last 'len(domain_parts)' parts of the target hostname
                # with the parts of the domain.
                if target_hostname_parts[-len(domain_parts):] == domain_parts:
                    is_within_domain = True

    if is_within_domain:
        # If the target is within the domain, ensure it uses https protocol
        # and return the reconstructed URL.
        parsed_target_list = list(parsed_target)
        parsed_target_list[0] = 'https'  # Set scheme to https
        return urlunparse(parsed_target_list)
    else:
        # Fallback to the domain URL with https protocol.
        return f"https://{domain}"

from urllib.parse import urlparse, urlunparse

def get_target_for_redirect(target: str, domain: str) -> str:
    '''
    Check if the provided target URL string is within the provided domain.
    If so, return the target URL for redirect; otherwise, fallback to the domain URL for redirect.
    Use https protocol.

    Examples:
        get_target_for_redirect('https://music.example.com', 'example.com') -> 'https://music.example.com'
        get_target_for_redirect('https://example.com/service', 'example.com') -> 'https://example.com/service'
        get_target_for_redirect('https://blog.example.com/posts', 'example.com') -> 'https://blog.example.com/posts'
        get_target_for_redirect('https://another.com', 'example.com') -> 'https://example.com'

    Args:
        target (str): The target URL string.
        domain (str): The domain string, without protocol.

    Returns:
        str: The URL string for redirect.
    '''
    parsed_target = urlparse(target)
    target_hostname = parsed_target.hostname

    is_within_domain = False

    if target_hostname:
        # Check if the target hostname is exactly the domain
        if target_hostname == domain:
            is_within_domain = True
        # Check if the target hostname is 'www.' + domain
        elif target_hostname == f"www.{domain}":
            is_within_domain = True
        else:
            # Check for subdomains (e.g., 'sub.domain.com' for 'domain.com')
            domain_parts = domain.split('.')
            target_hostname_parts = target_hostname.split('.')

            # A target hostname can be a subdomain only if it has more parts than the domain
            # and its suffix matches the domain parts.
            if len(target_hostname_parts) > len(domain_parts):
                # Compare the last 'len(domain_parts)' parts of the target hostname
                # with the parts of the domain.
                if target_hostname_parts[-len(domain_parts):] == domain_parts:
                    is_within_domain = True

    if is_within_domain:
        # If the target is within the domain, ensure it uses https protocol
        # and return the reconstructed URL.
        parsed_target_list = list(parsed_target)
        parsed_target_list[0] = 'https'  # Set scheme to https
        return urlunparse(parsed_target_list)
    else:
        # Fallback to the domain URL with https protocol.
        return f"https://{domain}"
