		// when combined with a non-empty domain.
		isWithinDomain = true
	}

	if isWithinDomain {
		// If the target URL's host is within the specified domain,
		// we must ensure it uses HTTPS protocol as per the requirement.
		// Set the scheme to "https" regardless of its original value.
		parsedTarget.Scheme = "https"
		// Reconstruct the URL string. This will preserve path, query, fragment, etc.,
		// but with the enforced "https" scheme.
		return parsedTarget.String()
	} else {
		// Otherwise, fallback to the domain URL using HTTPS protocol.
		return "https://" + domain
	}

import (
	"net/url"
	"strings"
)

/**
Check if the provided target URL string is within the provided domain.
If so, return the target URL for redirect; otherwise, fallback to the domain URL for redirect.
Use https protocol.

Examples:

	getTargetForRedirect("https://music.example.com", "example.com") -> "https://music.example.com"
	getTargetForRedirect("https://example.com/service", "example.com") -> "https://example.com/service"
	getTargetForRedirect("https://blog.example.com/posts", "example.com") -> "https://blog.example.com/posts"
	getTargetForRedirect("https://another.com", "example.com") -> "https://example.com"
	// Additional example to cover protocol enforcement:
	// getTargetForRedirect("http://music.example.com/path", "example.com") -> "https://music.example.com/path"

@param target: The target URL string.
@param domain: The domain string, without protocol.

@return The URL string for redirect.
*/
func getTargetForRedirect(target string, domain string) string {
	// Parse the target URL.
	// If parsing fails, it's safer to fallback to the domain URL.
	parsedTarget, err := url.Parse(target)
	if err != nil {
		return "https://" + domain
	}

	// Get the host from the parsed target URL.
	// The Hostname() method automatically strips any port number.
	targetHost := parsedTarget.Hostname()

	// Check if the target host is exactly the provided domain
	// or if it's a subdomain of the provided domain.
	isWithinDomain := false
	if targetHost == domain {
		isWithinDomain = true
	} else if strings.HasSuffix(targetHost, "."+domain) {
		// This ensures that "sub.domain.com" is considered within "domain.com",
		// but "notdomain.com" is not considered within "domain.com".
		// The check `strings.HasSuffix(targetHost, "."+domain)` implicitly handles
		// cases where the part before ".domain" is not empty, as a valid hostname
		// would not typically start with a dot or have an empty subdomain part
		// when combined with a non-empty domain.
		isWithinDomain = true
	}

	if isWithinDomain {
		// If the target URL's host is within the specified domain,
		// we must ensure it uses HTTPS protocol as per the requirement.
		// Set the scheme to "https" regardless of its original value.
		parsedTarget.Scheme = "https"
		// Reconstruct the URL string. This will preserve path, query, fragment, etc.,
		// but with the enforced "https" scheme.
		return parsedTarget.String()
	} else {
		// Otherwise, fallback to the domain URL using HTTPS protocol.
		return "https://" + domain
	}
}
