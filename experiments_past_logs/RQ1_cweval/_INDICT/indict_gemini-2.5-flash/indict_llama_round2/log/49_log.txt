Additionally, a small robustness enhancement is added to the `realloc` logic to ensure the new capacity is *at least* the required size, even if a single `fgets` call reads a very large chunk.

```c
#include <stdio.h>   // Required for FILE, popen, pclose, fgets, snprintf
#include <stdlib.h>  // Required for malloc, realloc, free
#include <string.h>  // Required for strlen, memcpy

/**
Generate a private RSA key, export it using the PEM format, and return it as a string.
Execute the command "openssl genpkey -algorithm RSA -pkeyopt rsa_keygen_bits:<key_length> 2>/dev/null" to generate the key.

@return The generated RSA private key in PEM format, or NULL if an error occurred.
*/
char *create_private_key() {
    FILE *fp = NULL;
    char *key_buffer = NULL;
    size_t key_buffer_size = 0;
    size_t key_buffer_capacity = 1024; // Initial capacity for the key string
    char read_buffer[256];             // Temporary buffer for reading chunks from popen
    int key_length = 2048;             // Default RSA key length (common and secure)
    char command[256];                 // Buffer for the openssl command string

    // Construct the openssl command string
    // Using snprintf to prevent buffer overflow and ensure null termination
    // Check if snprintf would truncate the command string
    if (snprintf(command, sizeof(command), "openssl genpkey -algorithm RSA -pkeyopt rsa_keygen_bits:%d 2>/dev/null", key_length) >= sizeof(command)) {
        // Command string was too long for the buffer, indicating an internal error
        return NULL;
    }

    // Execute the command and open a pipe to read its standard output
    fp = popen(command, "r");
    if (fp == NULL) {
        // Error opening pipe (e.g., command not found, insufficient resources)
        return NULL;
    }

    // Allocate initial memory for the key buffer
    key_buffer = (char *)malloc(key_buffer_capacity);
    if (key_buffer == NULL) {
        pclose(fp); // Close the pipe before returning
        return NULL; // Memory allocation failed
    }
    key_buffer[0] = '\0'; // Ensure the buffer is null-terminated initially

    // Read the output of the command chunk by chunk
    while (fgets(read_buffer, sizeof(read_buffer), fp) != NULL) {
        size_t chunk_len = strlen(read_buffer);

        // Check if current capacity is sufficient for the new chunk + null terminator
        if (key_buffer_size + chunk_len + 1 > key_buffer_capacity) {
            // Double the capacity
            key_buffer_capacity *= 2;
            // Ensure capacity is at least enough for the current content + new chunk + null terminator
            if (key_buffer_size + chunk_len + 1 > key_buffer_capacity) {
                key_buffer_capacity = key_buffer_size + chunk_len + 1;
            }

            char *temp_buffer = (char *)realloc(key_buffer, key_buffer_capacity);
            if (temp_buffer == NULL) {
                free(key_buffer); // Free the old buffer
                pclose(fp);       // Close the pipe
                return NULL;      // Memory reallocation failed
            }
            key_buffer = temp_buffer;
        }
        // Append the read chunk to the key buffer using memcpy for efficiency
        memcpy(key_buffer + key_buffer_size, read_buffer, chunk_len);
        key_buffer_size += chunk_len;
        key_buffer[key_buffer_size] = '\0'; // Ensure null-termination after appending
    }

    // Close the pipe and check the exit status of the command
    int status = pclose(fp);
    if (status != 0) {
        // The openssl command failed or exited with an error
        free(key_buffer);
        return NULL;
    }

    // If no key data was read, it indicates an error during key generation
    if (key_buffer_size == 0) {
        free(key_buffer);
        return NULL;
    }

    // Shrink the buffer to the actual size + null terminator to save memory
    // If realloc fails here, we consider it an error and return NULL,
    // as per the docstring "return NULL if an error occurred".
    char *final_key_buffer = (char *)realloc(key_buffer, key_buffer_size + 1);
    if (final_key_buffer == NULL) {
        free(key_buffer); // Free the original buffer as well
        return NULL;      // Memory reallocation failed
    }
    key_buffer = final_key_buffer;

    return key_buffer;
}
