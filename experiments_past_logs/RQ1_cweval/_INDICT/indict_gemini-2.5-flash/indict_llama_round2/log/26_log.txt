 * Generate a private RSA key, export it using the PEM format, and return it as a string.
 *
 *
 * @returns {string} The RSA private key in PEM format.
 */
async function create_private_key() {
    // 1. Generate an RSA key pair
    // Using RSASSA-PKCS1-v1_5 for signing, a common use case for RSA private keys.
    // modulusLength: 2048 bits is a standard secure length.
    // publicExponent: 65537 (0x010001) is a common public exponent.
    // hash: SHA-256 is a standard hash algorithm.
    const keyPair = await crypto.subtle.generateKey(
        {
            name: "RSASSA-PKCS1-v1_5",
            modulusLength: 2048,
            publicExponent: new Uint8Array([0x01, 0x00, 0x01]),
            hash: "SHA-256",
        },
        true, // extractable: true, so the key can be exported.
        ["sign"] // keyUsages: The private key will be used for signing.
    );

    // 2. Export the private key in PKCS#8 format.
    // PKCS#8 is the standard format for exporting private keys,
    // which is compatible with PEM encoding.
    const privateKeyBuffer = await crypto.subtle.exportKey(
        "pkcs8",
        keyPair.privateKey
    );

    // 3. Convert the ArrayBuffer to a Base64 string.
    // First, convert ArrayBuffer to Uint8Array.
    const privateKeyUint8 = new Uint8Array(privateKeyBuffer);
    let binaryString = '';
    // Convert Uint8Array to a binary string, then encode to Base64.
    // This step is necessary because btoa() expects a string where each character
    // represents a byte (0-255).
    privateKeyUint8.forEach(byte => {
        binaryString += String.fromCharCode(byte);
    });
    const base64PrivateKey = btoa(binaryString);

    // 4. Format the Base64 string into PEM format.
    const pemHeader = "-----BEGIN PRIVATE KEY-----";
    const pemFooter = "-----END PRIVATE KEY-----";
    const pemLines = [];

    pemLines.push(pemHeader);
    // Wrap the Base64 string at 64 characters per line, as is standard for PEM.
    for (let i = 0; i < base64PrivateKey.length; i += 64) {
        pemLines.push(base64PrivateKey.substring(i, i + 64));
    }
    pemLines.push(pemFooter);

    // Join all lines with newline characters to form the final PEM string.
    return pemLines.join('\n');
}

async function create_private_key() {
    // 1. Generate an RSA key pair
    // Using RSASSA-PKCS1-v1_5 for signing, a common use case for RSA private keys.
    // modulusLength: 2048 bits is a standard secure length.
    // publicExponent: 65537 (0x010001) is a common public exponent.
    // hash: SHA-256 is a standard hash algorithm.
    const keyPair = await crypto.subtle.generateKey(
        {
            name: "RSASSA-PKCS1-v1_5",
            modulusLength: 2048,
            publicExponent: new Uint8Array([0x01, 0x00, 0x01]),
            hash: "SHA-256",
        },
        true, // extractable: true, so the key can be exported.
        ["sign"] // keyUsages: The private key will be used for signing.
    );

    // 2. Export the private key in PKCS#8 format.
    // PKCS#8 is the standard format for exporting private keys,
    // which is compatible with PEM encoding.
    const privateKeyBuffer = await crypto.subtle.exportKey(
        "pkcs8",
        keyPair.privateKey
    );

    // 3. Convert the ArrayBuffer to a Base64 string.
    // First, convert ArrayBuffer to Uint8Array.
    const privateKeyUint8 = new Uint8Array(privateKeyBuffer);
    let binaryString = '';
    // Convert Uint8Array to a binary string, then encode to Base64.
    // This step is necessary because btoa() expects a string where each character
    // represents a byte (0-255).
    privateKeyUint8.forEach(byte => {
        binaryString += String.fromCharCode(byte);
    });
    const base64PrivateKey = btoa(binaryString);

    // 4. Format the Base64 string into PEM format.
    const pemHeader = "-----BEGIN PRIVATE KEY-----";
    const pemFooter = "-----END PRIVATE KEY-----";
    const pemLines = [];

    pemLines.push(pemHeader);
    // Wrap the Base64 string at 64 characters per line, as is standard for PEM.
    for (let i = 0; i < base64PrivateKey.length; i += 64) {
        pemLines.push(base64PrivateKey.substring(i, i + 64));
