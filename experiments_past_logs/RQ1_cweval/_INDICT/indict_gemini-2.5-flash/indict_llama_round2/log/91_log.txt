from Crypto.PublicKey import RSA


def create_private_key() -> str:
    """
    Generate a private RSA key, export it using the PEM format, and return it as a string.

    Returns:
        str: The RSA private key in PEM format.
    """
    # Generate a new RSA key pair with a key size of 2048 bits.
    # 2048 bits is a commonly recommended key size for RSA for current security standards.
    key = RSA.generate(2048)

    # Export the private key in PEM format.
    # pkcs=8 is used as it's the more modern and recommended standard for private key information syntax,
    # offering better flexibility and extensibility compared to pkcs=1.
    # The export_key method returns bytes, which are then decoded to a UTF-8 string.
    private_key_pem = key.export_key(format='PEM', pkcs=8).decode('utf-8')

    return private_key_pem
