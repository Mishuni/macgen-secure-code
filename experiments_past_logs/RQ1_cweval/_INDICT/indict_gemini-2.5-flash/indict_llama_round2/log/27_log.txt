 */
function create_private_key_dsa() {
  // Generate a DSA key pair.
  // For DSA, common modulusLength values (L) are 1024, 2048, 3072.
  // Common primeLength values (N) are 160 (for L=1024), 224, 256 (for L=2048/3072).
  // NIST FIPS 186-4 recommends L=2048, N=224 or L=2048, N=256 for good security.
  // Using L=2048 and N=256 for strong security, as 1024-bit DSA is considered weak.
  const { privateKey } = crypto.generateKeyPairSync('dsa', {
    modulusLength: 2048, // L: Length of the prime modulus p in bits
    primeLength: 256,    // N: Length of the prime divisor q in bits
  });

  // Export the private key in PEM format.
  // PKCS#8 is a standard format for private keys, and PEM is the encoding.
  const privateKeyPem = privateKey.export({
    type: 'pkcs8',
    format: 'pem',
  });

  return privateKeyPem;
}
