from Crypto.PublicKey import DSA


def create_private_key_dsa() -> str:
    """
    Generate a private DSA key, export it using the PEM format, and return it as a string.

    Returns:
        str: The DSA private key in PEM format.
    """
    # Generate a new DSA key pair. A common key size is 1024 bits.
    # For production, consider 2048 bits or higher for stronger security.
    # The DSA.generate method creates a key object containing both public and private components.
    key = DSA.generate(1024)

    # Export the private key in PEM format.
    # The export_key method, when called on a DSA key object, exports the private key by default.
    # It returns bytes, which are then decoded to a UTF-8 string.
    private_key_pem = key.export_key(format='PEM').decode('utf-8')

    return private_key_pem
