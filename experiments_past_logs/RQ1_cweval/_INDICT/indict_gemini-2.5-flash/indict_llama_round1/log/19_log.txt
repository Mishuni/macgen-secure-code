function get_target_for_redirect(target, domain) {
    const fallbackUrl = `https://${domain}`;

    try {
        const targetUrl = new URL(target);

        // Check if the target URL's hostname is the domain itself or a subdomain of it.
        // For example:
        // - If domain is 'example.com', 'example.com' matches.

        // - If domain is 'example.com', 'anotherexample.com' does NOT match.
