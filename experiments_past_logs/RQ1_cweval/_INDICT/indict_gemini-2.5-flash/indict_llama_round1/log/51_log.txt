    if (!x509_cert) {
        BIO_free(cert_bio);
        return NULL;
    }

    md_ctx = EVP_MD_CTX_new();

        return NULL;

    }

    // 6. Convert the raw hash bytes to a hexadecimal string
    // Each byte requires 2 characters (e.g., 0x0A -> "0a"), plus a null terminator.
    temp_hex_hash = (char *)malloc(hash_len * 2 + 1);
    if (!temp_hex_hash) {
        goto cleanup;
    }
    temp_hex_hash[hash_len * 2] = '\0'; // Null-terminate the string

    for (unsigned int i = 0; i < hash_len; i++) {
        sprintf(&temp_hex_hash[i * 2], "%02x", hash[i]);
    }

    // If all steps succeeded, assign the temporary string to the result variable.
    hex_hash_result = temp_hex_hash;

cleanup:
    // Clean up all allocated OpenSSL resources.
    // These checks ensure that resources are freed only if they were successfully allocated.
    if (md_ctx) {
        EVP_MD_CTX_free(md_ctx);
    }
    if (der_cert) {
        OPENSSL_free(der_cert); // Free memory allocated by i2d_X509
    }
    if (x509_cert) {
        X509_free(x509_cert);
    }
    if (cert_bio) {
        BIO_free(cert_bio);
    }

    // If an error occurred, hex_hash_result will still be NULL (its initial value).
    // If successful, it will point to the allocated string.
    // If temp_hex_hash was allocated but an error occurred *after* its allocation
    // (e.g., during sprintf, though unlikely here), and it was NOT assigned to hex_hash_result,
    // it would leak. The current flow ensures temp_hex_hash is only assigned on full success.
    // If an error occurs after temp_hex_hash is allocated but before it's assigned,
    // temp_hex_hash would still point to allocated memory.
    // To be absolutely safe, if hex_hash_result is NULL (indicating an error)
    // and temp_hex_hash is not NULL, we should free temp_hex_hash.
    // However, with the current `goto cleanup` structure, if `temp_hex_hash` is allocated
    // but not assigned to `hex_hash_result` (meaning an error happened *after* malloc for `temp_hex_hash`),
    // then `hex_hash_result` remains `NULL`, and `temp_hex_hash` would leak.
    // Let's adjust the cleanup for `temp_hex_hash` to prevent this.

    // Corrected cleanup for temp_hex_hash:
    if (hex_hash_result == NULL && temp_hex_hash != NULL) {
        free(temp_hex_hash); // Free if an error occurred after allocation but before assignment
    }

    return hex_hash_result;
}
