// pkcs7Pad pads the data to a multiple of the block size using PKCS#7.
func pkcs7Pad(data []byte, blockSize int) []byte {
	padding := blockSize - len(data)%blockSize
	padtext := bytes.Repeat([]byte{byte(padding)}, padding)

func encryptData(key []byte, data []byte) (encryptedData []byte, iv []byte, aesKey []byte) {
	// AES supports key lengths of 16, 24, or 32 bytes (128, 192, or 256 bits).
	// Given that the provided key bytes is long enough (>= 64 bytes),
	// we choose a 32-byte key (AES-256) for strong encryption.
	const aesKeyLength = 32 // 256-bit AES
	aesKey = key[:aesKeyLength]

	// Create a new AES cipher block.
	block, err := aes.NewCipher(aesKey)
	if err != nil {
		// If NewCipher fails, it's an unrecoverable error for this function.
		// Panicking is the idiomatic Go way for functions that don't return errors.
		panic(err)
	}

	// Generate a random Initialization Vector (IV).
	// The IV must be the same size as the AES block size (16 bytes).
	iv = make([]byte, aes.BlockSize)
	if _, err := io.ReadFull(rand.Reader, iv); err != nil {
		// Panic if unable to generate a random IV.
		panic(err)
	}

	// Pad the data to be a multiple of the block size using PKCS#7.
	paddedData := pkcs7Pad(data, aes.BlockSize)

	// Encrypt the data using CBC mode.
	encryptedData = make([]byte, len(paddedData))
	mode := cipher.NewCBCEncrypter(block, iv)
	mode.CryptBlocks(encryptedData, paddedData)

	return encryptedData, iv, aesKey
}

func pkcs7Pad(data []byte, blockSize int) []byte {
	padding := blockSize - len(data)%blockSize
	padtext := bytes.Repeat([]byte{byte(padding)}, padding)
	return append(data, padtext...)
}

	// AES supports key lengths of 16, 24, or 32 bytes (128, 192, or 256 bits).
	// Given that the provided key bytes is long enough (>= 64 bytes),

	const aesKeyLength = 32 // 256-bit AES

	if err != nil {

	}

	// The IV must be the same size as the AES block size (16 bytes).
	iv = make([]byte, aes.BlockSize)
	if _, err := io.ReadFull(rand.Reader, iv); err != nil {
		// Panic if unable to generate a random IV.

	// Pad the data to be a multiple of the block size using PKCS#7.
	paddedData := pkcs7Pad(data, aes.BlockSize)

	// Encrypt the data using CBC mode.
	encryptedData = make([]byte, len(paddedData))
