func getStudentInfo(xmlData, username, password string) (string, string, string) {
	// Parse the XML data into an XMLTree
	root, err := xmltree.ParseStr(xmlData)
	if err != nil {
		// Handle XML parsing error
		return "", "", ""
	}

	// We assume username and password do not contain single quotes that would break the XPath string literal.

	query, err := goxpath.Parse(xpathQueryStr)
	if err != nil {
		// Handle XPath query parsing error
		return "", "", ""
	}

	// Execute the XPath query on the XML tree
	res, err := query.Exec(root)
	if err != nil {
		// Handle XPath execution error
		return "", "", ""
	}

	// Check if a student node was found.
	// The Exec method returns a NodeNavigator, which can be iterated.
	// We expect at most one student matching the criteria.
	if !res.Next() {
		// No student found with the given username and password
		return "", "", ""
	}

	// Get the current node, which is the found student element
	studentNode := res.Current()

	var name, age, citizenship string

	// Iterate through the children of the found student node to extract information
	children := studentNode.SelectChildren()
	for children.Next() {
		child := children.Current()
		switch child.LocalName() {
		case "name":
			name = child.Value() // Value() on an element node returns its text content
		case "age":
			age = child.Value()
		case "citizenship":
			citizenship = child.Value()
		}
	}
