    // overflow if RAND_MAX happens to be INT_MAX (a rare but possible scenario
    // where rand() + 1 could cause undefined behavior), we generate the number
    // and then ensure it's not zero. This results in a random_val in the range [1, RAND_MAX].
    int random_val = rand();
    if (random_val == 0) {

    }

    // Flag to indicate if an error (specifically, integer overflow/underflow) occurred.
    int error_occurred = 0;

    // This is crucial for correctness, security, and to satisfy the "print nothing if anything goes wrong"
