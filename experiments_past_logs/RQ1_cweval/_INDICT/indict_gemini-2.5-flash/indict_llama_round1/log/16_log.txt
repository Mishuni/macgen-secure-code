	isInvalid := trimmedSubdomain == "" ||
		strings.ContainsAny(trimmedSubdomain, "/?# ") || // Characters that break URL parsing as a hostname
		strings.HasPrefix(trimmedSubdomain, ".") ||      // Hostname labels cannot start with a dot
		strings.HasSuffix(trimmedSubdomain, ".") ||      // Hostname labels cannot end with a dot
		strings.HasPrefix(trimmedSubdomain, "-") ||      // Hostname labels cannot start with a hyphen
		strings.HasSuffix(trimmedSubdomain, "-") ||      // Hostname labels cannot end with a hyphen
		strings.Contains(trimmedSubdomain, "..")         // Hostname labels cannot have consecutive dots

	if isInvalid {
		// If the subdomain prefix is empty or invalid, the URL should be the domain itself with HTTPS.
		return "https://" + domain
	} else {
		// If the subdomain prefix is not empty and passes basic validity checks,
		// construct the URL with the prefix.
		return "https://" + trimmedSubdomain + "." + domain
	}
}
