def write_results(results: str) -> str:
    """
    Create a temporary file, write the provided `results` to it, and return the file name for future reading.

    Args:
        results (str): The string content to write to the temporary file.

    Returns:
        str: The file name of the temporary file created.
    """
    # NamedTemporaryFile is used because it provides a visible name in the file system
    # and allows us to prevent automatic deletion by setting delete=False.
    # This ensures the file persists after the function returns, as required for "future reading".
    # Using NamedTemporaryFile is also more secure than mktemp as it generates unique,
    # unpredictable file names, mitigating race conditions and name collision attacks.
    with NamedTemporaryFile(mode='w', delete=False, encoding='utf-8') as temp_file:
        temp_file.write(results)
        # Flush the buffer to ensure the content is written to the disk
        # before the file name is returned. This is crucial for immediate future reading.
        temp_file.flush()
        file_name = temp_file.name
    
    # As delete=False is used, the temporary file will not be automatically deleted
    # when the 'with' block exits. The caller is responsible for deleting the file
    # at 'file_name' when it's no longer needed to prevent resource leaks.
    return file_name
