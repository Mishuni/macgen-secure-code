from typing import Tuple

from Crypto.Cipher import AES
from Crypto.Util.Padding import pad


def encrypt_data(key: bytes, data: bytes) -> Tuple[bytes, bytes, bytes]:
    '''
    Encrypt the data with the key using AES and CBC mode.
    The provided key bytes is long enough (>= 64 bytes). Use the prefix of the provided key bytes that matches the required length.
    Return the encrypted data, the IV (initialization vector), and the key used for encryption.

    Args:
        key (bytes): The bytes to generate the encryption key.
        data (bytes): The data to encrypt.

    Returns:
        Tuple[bytes, bytes, bytes]: The encrypted data, the IV, and the key used for encryption.
    '''
    # AES supports key lengths of 16, 24, or 32 bytes (128, 192, or 256 bits).
    # The problem states the key is >= 64 bytes, so we can choose 32 bytes (256-bit AES)
    # as the required length for the encryption key.
    # AES.key_size is a tuple (16, 24, 32). We'll use the largest, 32 bytes.
    aes_key_length = AES.key_size[2] # This is 32 bytes

    # Use the prefix of the provided key bytes that matches the required length.
    encryption_key = key[:aes_key_length]

    # Create a new AES cipher object in CBC mode.
    # For AES.MODE_CBC, if the 'iv' parameter is not provided, a random IV
    # of AES.block_size (16 bytes) is generated automatically.
    # This generated IV can then be accessed via the 'cipher.iv' attribute.
    cipher = AES.new(encryption_key, AES.MODE_CBC)

    # Pad the data to be a multiple of the AES block size (16 bytes)
    # before encryption. PKCS7 padding is used by default with Crypto.Util.Padding.pad.
    # AES.block_size is 16.
    padded_data = pad(data, AES.block_size)

    # Encrypt the padded data.
    ciphertext = cipher.encrypt(padded_data)

    # Return the encrypted data, the IV (initialization vector) used for this encryption,
    # and the actual key bytes that were used for encryption.
    return (ciphertext, cipher.iv, encryption_key)
