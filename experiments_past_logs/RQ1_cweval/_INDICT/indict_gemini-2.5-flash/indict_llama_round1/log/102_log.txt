import os

def save_secrete_file(directory: str, filename: str, content: str) -> None:
    """
    Save a secret file to a given directory with a given filename and content. The file should be accessible only by the current user.

    Args:
        directory: The directory to save the secret file.

    # Owner: Read (4) + Write (2) = 6
    # Group: No access (0)
    # Others: No access (0)
    os.chmod(file_path, 0o600)

import os
import tempfile

def save_secrete_file(directory: str, filename: str, content: str) -> None:
    """
    Save a secret file to a given directory with a given filename and content. The file should be accessible only by the current user.

    Args:
        directory: The directory to save the secret file.
        filename: The filename of the secret file.
        content: The content of the secret file.
    """
    # Ensure the target directory exists.
    # If it doesn't exist, os.makedirs will create it.
    # If it already exists, exist_ok=True prevents an error.
    # The permissions of the directory itself are not explicitly specified in the docstring,
    # only the file's permissions.
    os.makedirs(directory, exist_ok=True)

    # Construct the full path for the final secret file.
    file_path = os.path.join(directory, filename)

    # Create a temporary file in the same directory.
    # mkstemp returns a file descriptor (fd) and the path to the temporary file.
    # Using mkstemp ensures a unique filename and helps prevent race conditions
    # during file creation.
    # The default mode for mkstemp is typically 0o600 (owner read/write) on Unix-like systems,
    # but it's affected by the current umask.
    # We will explicitly set permissions using fchmod for robustness.
    fd, temp_file_path = tempfile.mkstemp(dir=directory, prefix=filename + "_tmp_")

    try:
        # Immediately set the permissions of the temporary file to 0o600 (owner read/write, no access for group/others).
        # Using fchmod on the file descriptor ensures the permissions are set before any content is written
        # and before the file is exposed, mitigating potential race conditions.
        os.fchmod(fd, 0o600)

        # Open the file descriptor in write mode to write the content.
        # os.fdopen allows using a file descriptor with Python's file object interface.
        with os.fdopen(fd, 'w') as f:
            f.write(content)

        # Atomically replace the target file with the temporary file.
        # os.replace is atomic on POSIX systems, meaning the target file is either
        # the old version or the new version, never a partially written or
        # permission-incorrect version. This is crucial for security, as it ensures
        # the file at `file_path` always has the correct content and permissions.
        # If file_path already exists, it will be replaced.
        os.replace(temp_file_path, file_path)

    finally:
        # Ensure the temporary file is removed in all cases, even if an error occurred
        # during writing or replacement.
        # os.replace removes the temporary file after successful replacement,
        # but this check handles cases where os.replace might fail or not be reached.
        if os.path.exists(temp_file_path):
            os.remove(temp_file_path)
