    if (count <= 0) {
        return results; // Return empty for no passwords
    }

    // Determine the length of the output hash (derived key) and salt.
    // SHA-256 produces a 32-byte hash.
    const int key_len = EVP_MD_size(EVP_sha256());

        // This indicates a serious OpenSSL configuration error or invalid digest.
        throw std::runtime_error("Failed to get SHA-256 digest size.");
    }
    const int salt_len = key_len; // Salt length is specified to be the same as output hash length.

    // Iterate through each password to hash it.
    for (int i = 0; i < count; ++i) {
        // Ensure we don't go out of bounds if 'count' is larger than 'passwords' vector size.
        // Although the problem implies 'count' is correct, this adds robustness.
        if (i >= passwords.size()) {
            throw std::out_of_range("Password list is shorter than specified count.");
        }

        // 1. Generate a cryptographically secure random salt.
        std::vector<unsigned char> salt(salt_len);
        if (RAND_bytes(salt.data(), salt_len) != 1) {
            throw std::runtime_error("Failed to generate random salt using RAND_bytes.");
        }

        std::vector<unsigned char> hashed_password(key_len);

        //   passlen: int password_len
        //   salt: const unsigned char *salt

        //   keylen: int key_len (length of the derived key)
        //   out: unsigned char *out (buffer to store the derived key)
