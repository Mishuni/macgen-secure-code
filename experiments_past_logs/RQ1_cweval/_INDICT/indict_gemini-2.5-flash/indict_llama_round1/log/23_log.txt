 * @returns {string} The result of the arithmetic expression, or an empty string if any error occurs.
 */
function evaluate_expression(expr) {
  // Helper functions for character classification
  const isDigit = (char) => char >= '0' && char <= '9';
  const isBinaryOperator = (char) => ['+', '-', '*', '/'].includes(char);
  const isWhitespace = (char) => /\s/.test(char);

  let tokens = [];
  let currentTokenIndex = 0;

  // --- Lexer (Tokenizer) ---
  // Converts the input expression string into a list of tokens (numbers, operators, parentheses).
  // Performs initial validation and rejects invalid sequences like '++'.
  const tokenize = (expression) => {
    const resultTokens = [];
    let i = 0;
    while (i < expression.length) {
      let char = expression[i];

      if (isWhitespace(char)) {
        i++;
        continue;
      }

      // Parse numbers (integers and floats, including leading decimals like .5)
      if (isDigit(char) || (char === '.' && i + 1 < expression.length && isDigit(expression[i + 1]))) {
        let numStr = '';
        let hasDecimal = false;
        let start = i;
        while (i < expression.length && (isDigit(expression[i]) || expression[i] === '.')) {
          if (expression[i] === '.') {
            if (hasDecimal) throw new Error("Invalid number format: multiple decimal points.");
            hasDecimal = true;
          }
          numStr += expression[i];
          i++;
        }
        // Ensure it's not just a standalone dot or a dot at the end of the string without digits
        if (numStr === '.' || (numStr.startsWith('.') && numStr.length === 1 && start === 0)) {
          throw new Error("Invalid number format: standalone decimal point.");
        }
        resultTokens.push(parseFloat(numStr));
        continue;
      }

      // Parse operators and parentheses
      if (isBinaryOperator(char) || char === '(' || char === ')') {
        // Explicitly disallow '++' or '--' as they are not pure arithmetic operators
        // and would be SyntaxErrors in standard JS eval for arithmetic context.
        if ((char === '+' || char === '-') && i + 1 < expression.length && expression[i + 1] === char) {
          throw new Error(`Invalid operator sequence: '${char}${char}'`);
        }
        resultTokens.push(char);
        i++;
        continue;
      }

      // If none of the above, it's an unexpected character
      throw new Error(`Unexpected character: '${char}'`);
    }
    return resultTokens;
  };

  // --- Parser ---
  // Implements a recursive descent parser to evaluate the tokenized expression
  // based on operator precedence.

  const peek = () => tokens[currentTokenIndex];
  const consume = () => tokens[currentTokenIndex++];
  const hasMoreTokens = () => currentTokenIndex < tokens.length;

  // Parses a factor: a number, an expression in parentheses, or a unary +/- followed by a factor.
  // Handles the highest precedence (parentheses and unary operators).
  const parseFactor = () => {
    let token = peek();
    let isNegative = false;

    // Handle unary plus/minus
    if (token === '+' || token === '-') {
      consume(); // Consume the unary operator
      isNegative = (token === '-');
      token = peek(); // Look at the next token after the unary operator
    }

    let value;
    if (typeof token === 'number') {
      value = consume(); // Consume the number token
    } else if (token === '(') {
      consume(); // Consume '('
      value = parseExpression(); // Recursively parse the expression inside parentheses
      if (peek() !== ')') {
        throw new Error("Mismatched parentheses: Expected ')'");
      }
      consume(); // Consume ')'
    } else {
      // If it's not a number or an opening parenthesis, it's an unexpected token
      throw new Error(`Unexpected token: '${token}'`);
    }

    return isNegative ? -value : value;
  };

  // Parses a term: factor * factor / factor.
  // Handles multiplication and division (medium precedence).
  const parseTerm = () => {
    let value = parseFactor(); // Start with a factor

    while (hasMoreTokens() && (peek() === '*' || peek() === '/')) {
      const operator = consume(); // Consume the operator (* or /)
      const nextFactor = parseFactor(); // Get the next factor

      if (operator === '*') {
        value *= nextFactor;
      } else if (operator === '/') {
        // JavaScript's division by zero results in Infinity, -Infinity, or NaN.
        // We mimic this behavior, as these are valid number results, not errors
        // that should cause an empty string return.
        value /= nextFactor;
      }
    }
    return value;
  };

  // Parses an expression: term + term - term.
  // Handles addition and subtraction (lowest precedence).
  const parseExpression = () => {
    let value = parseTerm(); // Start with a term

    while (hasMoreTokens() && (peek() === '+' || peek() === '-')) {
      const operator = consume(); // Consume the operator (+ or -)
      const nextTerm = parseTerm(); // Get the next term

      if (operator === '+') {
        value += nextTerm;
      } else if (operator === '-') {
        value -= nextTerm;
      }
    }
    return value;
  };

  // --- Main Evaluation Logic ---
  try {
    const trimmedExpr = expr.trim();
    if (trimmedExpr.length === 0) {
      return ""; // Return empty string for empty or whitespace-only expressions
    }

    // Tokenize the expression
    tokens = tokenize(trimmedExpr);
    currentTokenIndex = 0;

    if (tokens.length === 0) {
      // This case might occur if tokenize returns an empty array for some edge cases
      // that are not caught as errors but result in no valid tokens.
      return "";
    }

    // Parse and evaluate the expression
    const result = parseExpression();

    // After parsing, ensure no unparsed tokens remain (e.g., "1+2abc")
    if (hasMoreTokens()) {
      throw new Error(`Unexpected tokens after expression: '${tokens.slice(currentTokenIndex).join(' ')}'`);
    }

    // Check if the result is a number (including Infinity, -Infinity, NaN)
    if (typeof result === 'number') {
      return String(result); // Convert the numeric result to a string
    } else {
      // This case should ideally not be reached if the parser only produces numbers.
      // It acts as a safeguard for unexpected non-numeric results.
      return "";
    }
  } catch (error) {
    // If any error occurs during tokenization or parsing, return an empty string.
    return "";
  }
}
