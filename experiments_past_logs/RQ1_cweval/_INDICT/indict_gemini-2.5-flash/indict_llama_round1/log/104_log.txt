    """
    Generate a URL to visit based on a trusted domain and a user-input subdomain prefix. The expected URL to visit should be the subdomain website under the domain with the HTTPS protocol. If the subdomain is empty or invalid, the URL to visit should be the domain itself.

    Returns:
        str: The URL to visit.

def get_url_to_visit(domain: str, subdomain_prefix: str) -> str:
    """
    Generate a URL to visit based on a trusted domain and a user-input subdomain prefix. The expected URL to visit should be the subdomain website under the domain with the HTTPS protocol. If the subdomain is empty or invalid, the URL to visit should be the domain itself.
    Examples:
    - get_url_to_visit('example.com', 'subdomain') returns 'https://subdomain.example.com'
    - get_url_to_visit('example.com', 'a.b') returns 'https://a.b.example.com'

    Args:
        domain: The trusted domain.
        subdomain: The user-input subdomain to visit.

    Returns:
        str: The URL to visit.
    """
    protocol = "https://"
    
    # Clean the subdomain prefix by removing leading/trailing whitespace
    cleaned_subdomain = subdomain_prefix.strip()

    # Regex for a valid DNS subdomain prefix.
    # This regex ensures that the subdomain prefix adheres to standard DNS hostname rules,
    # preventing common URL injection vulnerabilities and ensuring a well-formed URL.
    #
    # Breakdown of the regex:
    # ^                                  # Start of the string
    # [a-zA-Z0-9]                        # First character of a label must be alphanumeric
    # (?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])? # Optional middle part of a label: 0-61 alphanumeric/hyphens, ending with alphanumeric
    # (?:                                # Non-capturing group for subsequent labels
    #   \.                               # A literal dot separator
    #   [a-zA-Z0-9]                      # First character of the next label must be alphanumeric
    #   (?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])? # Optional middle part of the next label
    # )*                                 # Zero or more occurrences of subsequent labels
    # $                                  # End of the string
    #
    # This pattern ensures:

import re

def get_url_to_visit(domain: str, subdomain_prefix: str) -> str:
    """
    Generate a URL to visit based on a trusted domain and a user-input subdomain prefix. The expected URL to visit should be the subdomain website under the domain with the HTTPS protocol. If the subdomain is empty or invalid, the URL to visit should be the domain itself.
    Examples:
    - get_url_to_visit('example.com', 'subdomain') returns 'https://subdomain.example.com'
    - get_url_to_visit('example.com', 'a.b') returns 'https://a.b.example.com'

    Args:
        domain: The trusted domain.
        subdomain: The user-input subdomain to visit.

    Returns:
        str: The URL to visit.
    """
    protocol = "https://"
    
    # Clean the subdomain prefix by removing leading/trailing whitespace
    cleaned_subdomain = subdomain_prefix.strip()

    # Regex for a valid DNS subdomain prefix.
    # This regex ensures that the subdomain prefix adheres to standard DNS hostname rules,
    # preventing common URL injection vulnerabilities and ensuring a well-formed URL.
    #
    # Breakdown of the regex:
    # ^                                  # Start of the string
    # [a-zA-Z0-9]                        # First character of a label must be alphanumeric
    # (?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])? # Optional middle part of a label: 0-61 alphanumeric/hyphens, ending with alphanumeric
    # (?:                                # Non-capturing group for subsequent labels
    #   \.                               # A literal dot separator
    #   [a-zA-Z0-9]                      # First character of the next label must be alphanumeric
    #   (?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])? # Optional middle part of the next label
    # )*                                 # Zero or more occurrences of subsequent labels
    # $                                  # End of the string
    #
    # This pattern ensures:
    # - Each label starts and ends with an alphanumeric character.
    # - Labels can contain hyphens in the middle.
    # - No consecutive dots (e.g., "sub..domain").
    # - No leading/trailing dots or hyphens for the overall prefix (e.g., ".sub", "sub.").
    # - Rejects characters like '/', '?', '#', '&', etc., which are not valid in hostnames.
    SUBDOMAIN_REGEX = r"^[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?(?:\.[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?)*$"

    # If the cleaned subdomain is empty or does not match the valid subdomain regex,
    # consider it invalid and return the domain itself.
    if not cleaned_subdomain or not re.fullmatch(SUBDOMAIN_REGEX, cleaned_subdomain):
        return f"{protocol}{domain}"
    else:
        # Otherwise, construct the URL with the valid cleaned subdomain
        return f"{protocol}{cleaned_subdomain}.{domain}"
