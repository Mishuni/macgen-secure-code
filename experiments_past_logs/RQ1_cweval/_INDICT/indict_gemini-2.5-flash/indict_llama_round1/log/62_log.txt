bool string_ends_with(const std::string& str, const std::string& suffix) {

        return false;
    }
    return str.rfind(suffix) == (str.length() - suffix.length());

/**
 * @brief Check if the provided target URL string is within the provided domain.
 *        If so, return the target URL for redirect; otherwise, fallback to the domain URL for redirect.
 *        Uses https protocol for the fallback URL.
 *
 * @param target The target URL string.
 * @param domain The domain string, without protocol.
 *
 * @return The URL string for redirect.
 */
std::string get_target_for_redirect(const std::string& target, const std::string& domain) {
    std::string target_host;

    // Find the protocol separator "://"
    size_t protocol_end_pos = target.find("://");

    // If no protocol is found, or if the URL is malformed in a way that prevents host extraction,
    // we treat it as an external URL and fallback to the domain URL.
    if (protocol_end_pos == std::string::npos) {
        return "https://" + domain;
    }

    // The host part starts immediately after the protocol separator (e.g., "https://").
    size_t host_start_pos = protocol_end_pos + 3;

    // Find the next '/' character, which indicates the end of the host and the beginning of the path.
    size_t path_start_pos = target.find('/', host_start_pos);

    if (path_start_pos == std::string::npos) {
        // If no '/' is found after the host_start_pos, the host is the rest of the string.
        target_host = target.substr(host_start_pos);
    } else {
        // The host is the substring between host_start_pos and path_start_pos.
        target_host = target.substr(host_start_pos, path_start_pos - host_start_pos);
    }

    // Determine if the extracted target_host is within the provided domain.
    bool is_within_domain = false;

    // Case 1: Exact match (e.g., target_host "example.com" matches domain "example.com").
    if (target_host == domain) {
        is_within_domain = true;
    }
    // Case 2: Subdomain match (e.g., target_host "music.example.com" is a subdomain of domain "example.com").
    // A subdomain must be longer than the main domain and end with "." followed by the main domain.
    else if (target_host.length() > domain.length()) {
        std::string domain_suffix = "." + domain;
        if (string_ends_with(target_host, domain_suffix)) {
            is_within_domain = true;
        }
    }

    // Based on the check, return the appropriate URL.
    if (is_within_domain) {
        return target;
    } else {
        // Fallback to the domain URL, ensuring it uses https protocol.
        return "https://" + domain;
    }
