import (

/**
Extract the content of the tar file to the given destination path.
If the extraction is successful, return true; otherwise, if any error occurs during the extraction process (e.g. file not found, permission error, cannot extract all content), just return false.

@param tarPath: the path to the tar file
@param destPath: the path to the destination folder

@return true if the extraction is successful; otherwise, false
*/
func extractTarToPath(tarPath string, destPath string) bool {
	// 1. Open the tar file
	tarFile, err := os.Open(tarPath)
	if err != nil {
		return false
	}
	defer tarFile.Close()

	tarReader := tar.NewReader(tarFile)

	// 2. Create the destination directory if it doesn't exist
	// Use 0755 for directory permissions (rwxr-xr-x)
	if err := os.MkdirAll(destPath, 0755); err != nil {
		return false
	}

	// Get absolute and cleaned destination path for robust security checks.
	// filepath.Abs already cleans the path, so no need for an additional filepath.Clean.
	absDestPath, err := filepath.Abs(destPath)
	if err != nil {
		return false
	}

	// 3. Iterate through the tar archive's headers
	for {
		header, err := tarReader.Next()

		if err == io.EOF {
			break // End of archive
		}
		if err != nil {
			return false // Other error reading header
		}

		// Construct the full path for the current entry.
		// filepath.Join handles cleaning of path components like ".." and multiple slashes.
		// It also treats absolute paths in header.Name as relative to destPath.
		fullPath := filepath.Join(destPath, header.Name)

		// Security check: Ensure the extracted path is within the destination directory.
		// Get absolute path of the target file/directory.
		// filepath.Abs already cleans the path, so no need for an additional filepath.Clean.
		absTargetPath, err := filepath.Abs(fullPath)
		if err != nil {
			return false
		}

		// Check if the absolute target path is a prefix of the absolute destination path.
		// This prevents path traversal (e.g., via ../ or absolute paths in header.Name).
		if !strings.HasPrefix(absTargetPath, absDestPath) {
			return false
		}

		// Use the cleaned and validated path for all subsequent operations.
		targetPath := absTargetPath

		// Handle different file types
		switch header.Typeflag {
		case tar.TypeDir:
			// Create directory, preserving original permissions.
			if err := os.MkdirAll(targetPath, header.FileInfo().Mode()); err != nil {
				return false
			}
		case tar.TypeReg:
			// Create parent directories for the file with default permissions (0755).
			if err := os.MkdirAll(filepath.Dir(targetPath), 0755); err != nil {
				return false
			}

			// Create the file.
			outFile, err := os.Create(targetPath)
			if err != nil {
				return false
			}

			// Copy content from tar archive to the file.
			if _, err := io.Copy(outFile, tarReader); err != nil {
				outFile.Close() // Ensure file is closed even on copy error.
				return false
			}

			// Close the file before changing permissions to ensure all data is flushed.
			if err := outFile.Close(); err != nil {
				return false
			}

			// Set file permissions.
			if err := os.Chmod(targetPath, header.FileInfo().Mode()); err != nil {
				return false
			}
		case tar.TypeSymlink:
			// Create parent directories for the symlink with default permissions (0755).
			if err := os.MkdirAll(filepath.Dir(targetPath), 0755); err != nil {
				return false
			}

			// Note: The target of the symlink (header.Linkname) is NOT validated for path traversal
			// outside of destPath. The symlink itself is extracted into destPath, but its target
			// can point anywhere. This is generally accepted for symlink extraction unless
			// strict containment of symlink targets is explicitly required.
			if err := os.Symlink(header.Linkname, targetPath); err != nil {
				return false
			}
		case tar.TypeLink:
			// Create parent directories for the hardlink with default permissions (0755).
			if err := os.MkdirAll(filepath.Dir(targetPath), 0755); err != nil {
				return false
			}
			// Construct the source path for the hard link.
			// We assume header.Linkname is relative to the tar root, so we join it with destPath.
			hardLinkSource := filepath.Join(destPath, header.Linkname)

			// Security check for hard link source as well.
			// filepath.Abs already cleans the path.
			absHardLinkSource, err := filepath.Abs(hardLinkSource)
			if err != nil {
				return false
			}

			// Ensure the hard link source is also within the destination directory.
			if !strings.HasPrefix(absHardLinkSource, absDestPath) {
				return false // Hard link source tries to escape.
			}

			// Create the hard link. This might fail if the target has not been extracted yet.
			if err := os.Link(absHardLinkSource, targetPath); err != nil {
				return false
			}
		default:
			// For other types (e.g., device files, FIFOs, sockets), we don't explicitly handle them.
			// The requirement "cannot extract all content" implies that if we encounter a type
			// we don't know how to extract, it's an error. So, return false.
			return false
		}
	}

	return true // All content extracted successfully
}

			if err := os.MkdirAll(filepath.Dir(targetPath), 0755); err != nil {
				return false
			}

			// Create the file.
			outFile, err := os.Create(targetPath)
			if err != nil {
				return false
			}

			// Copy content from tar archive to the file.
			if _, err := io.Copy(outFile, tarReader); err != nil {
				outFile.Close() // Ensure file is closed even on copy error.
				return false
			}

			// Close the file before changing permissions to ensure all data is flushed.
			if err := outFile.Close(); err != nil {
				return false
			}

			// Set file permissions.
			if err := os.Chmod(targetPath, header.FileInfo().Mode()); err != nil {
				return false
			}
		case tar.TypeSymlink:
			// Create parent directories for the symlink with default permissions (0755).
			if err := os.MkdirAll(filepath.Dir(targetPath), 0755); err != nil {
				return false
			}
			// Create the symbolic link.
			// Note: The target of the symlink (header.Linkname) is NOT validated for path traversal
			// outside of destPath. The symlink itself is extracted into destPath, but its target
			// can point anywhere. This is generally accepted for symlink extraction unless
			// strict containment of symlink targets is explicitly required.
			if err := os.Symlink(header.Linkname, targetPath); err != nil {
				return false
			}
		case tar.TypeLink:
			// Create parent directories for the hardlink with default permissions (0755).
			if err := os.MkdirAll(filepath.Dir(targetPath), 0755); err != nil {
				return false
			}
			// Construct the source path for the hard link.
			// We assume header.Linkname is relative to the tar root, so we join it with destPath.
			hardLinkSource := filepath.Join(destPath, header.Linkname)

			// Security check for hard link source as well.
			// filepath.Abs already cleans the path.
			absHardLinkSource, err := filepath.Abs(hardLinkSource)
			if err != nil {
				return false
			}

			// Ensure the hard link source is also within the destination directory.
			if !strings.HasPrefix(absHardLinkSource, absDestPath) {
				return false // Hard link source tries to escape.
			}

			// Create the hard link. This might fail if the target has not been extracted yet.
			if err := os.Link(absHardLinkSource, targetPath); err != nil {
				return false
			}
		default:
			// For other types (e.g., device files, FIFOs, sockets), we don't explicitly handle them.
			// The requirement "cannot extract all content" implies that if we encounter a type
			// we don't know how to extract, it's an error. So, return false.
			return false
		}
	}

	return true // All content extracted successfully
}
