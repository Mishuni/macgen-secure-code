        if (r != ARCHIVE_OK) {

            archive_write_free(out);
            archive_read_close(src);
            archive_read_free(src);

    // If r is not ARCHIVE_EOF, it means an error occurred during header reading (e.g., corrupted archive structure).
    if (r != ARCHIVE_EOF) {
        archive_write_free(out);
        archive_read_close(src);
        archive_read_free(src);
        return false;

    archive_write_free(out);
    archive_read_close(src); // Close the input archive file.
    archive_read_free(src);  // Free the input archive structure.

            archive_write_free(out);
            archive_read_close(src);
            archive_read_free(src);
            return false;
        }
    }

    // Check the final result of the archive_read_next_header loop.
    // If r is not ARCHIVE_EOF, it means an error occurred during header reading (e.g., corrupted archive structure).
    if (r != ARCHIVE_EOF) {
        archive_write_free(out);
        archive_read_close(src);
        archive_read_free(src);
        return false;
    }

    // 4. Finalize and clean up resources.
    // archive_write_free also closes the output archive and frees its resources.
    archive_write_free(out);
    archive_read_close(src); // Close the input archive file.
    archive_read_free(src);  // Free the input archive structure.

    return true; // All content extracted successfully.
}
