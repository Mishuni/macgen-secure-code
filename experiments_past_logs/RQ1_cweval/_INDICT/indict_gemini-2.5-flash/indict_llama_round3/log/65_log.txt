/**
Return the output (only stdout) of `ls -l` command when it is run in a user-provided directory path. If any error occurs during the command execution (e.g. directory not found, permission error), return an empty string.

@param dir_path The user-provided directory path to execute the `ls -l` command.

@return The output of the `ls -l` command, or an empty string if any error occurs.
*/
std::string get_ls_result(const std::string &dir_path) {
    // Construct the command string.
    // It's crucial to properly quote the directory path to handle spaces and special characters.
    // Single quotes are used, and any single quotes within dir_path are escaped
    // by replacing ' with '\'' (close quote, add escaped quote, open quote).
    std::string command = "ls -l '";
    std::string escaped_dir_path = dir_path;
    size_t pos = 0;
    while ((pos = escaped_dir_path.find('\'', pos)) != std::string::npos) {
        escaped_dir_path.replace(pos, 1, "'\\''");
        pos += 4; // Move past the inserted sequence ('\'' is 4 characters long)
    }
    command += escaped_dir_path;
    command += "'";

    std::string result = "";
    // Use a reasonably sized buffer for reading command output.
    // `ls -l` lines can be long, especially with long filenames.
    std::array<char, 1024> buffer;

    // Execute the command using popen and get a file pointer to its stdout.
    FILE* pipe = popen(command.c_str(), "r");
    if (!pipe) {
        // popen failed to execute the command (e.g., insufficient memory, command not found in PATH).
        return "";
    }

    // Read the output from the pipe chunk by chunk until EOF.
    while (fgets(buffer.data(), buffer.size(), pipe) != nullptr) {
        result += buffer.data();
    }

    // Close the pipe and get the exit status of the executed command.
    // pclose returns -1 on error, or the exit status of the command.
    // On Unix-like systems, the exit status is typically encoded in the return value.
    // WIFEXITED and WEXITSTATUS macros are used to interpret this status.
    int status = pclose(pipe);

    // Check if the command terminated normally and with a successful exit status (0).
    // If WIFEXITED is false, it means the process did not terminate normally (e.g., killed by a signal).
    // If WEXITSTATUS is non-zero, it means the command itself reported an error
    // (e.g., directory not found, permission denied).
    if (WIFEXITED(status) && WEXITSTATUS(status) == 0) {
        return result;
    } else {
        // An error occurred during command execution or the command itself failed.
        return "";
    }
}
