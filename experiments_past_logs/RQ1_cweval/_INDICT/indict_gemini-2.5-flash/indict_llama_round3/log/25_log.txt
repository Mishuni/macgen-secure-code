 * Generates a log entry for a received user message with the current timestamp, in the format '[<timestamp_iso_format>] Received: <msg>'.
 * Timestamp ISO format is like '2021-09-15T14:30:00'.
 *
 * @param {string} msg - The user message received.
 * @returns {string} The log entry for the received message.
 */
function generate_receive_log(msg) {
  const now = new Date();

  // Extract date and time components in the local timezone
  const year = now.getFullYear();
  // Month is 0-indexed, so add 1
  const month = (now.getMonth() + 1).toString().padStart(2, '0');
  const day = now.getDate().toString().padStart(2, '0');
  const hours = now.getHours().toString().padStart(2, '0');
  const minutes = now.getMinutes().toString().padStart(2, '0');
  const seconds = now.getSeconds().toString().padStart(2, '0');

  // Construct the timestamp string in 'YYYY-MM-DDTHH:mm:ss' format
  const timestamp = `${year}-${month}-${day}T${hours}:${minutes}:${seconds}`;

  return `[${timestamp}] Received: ${msg}`;
}

 * @returns {string} The log entry for the received message.
 */
function generate_receive_log(msg) {
  const now = new Date();
  // Date.prototype.toISOString() returns a string in simplified ISO 8601 format (YYYY-MM-DDTHH:mm:ss.sssZ) in UTC.
  // For logging, using UTC is generally a best practice for consistency across different timezones and systems.
  // The requirement specifies a format like 'YYYY-MM-DDTHH:mm:ss', which means we need to remove the milliseconds and the 'Z' (UTC indicator).
  // Slicing the string to the first 19 characters achieves the desired format precisely.
  const timestamp = now.toISOString().substring(0, 19);

  return `[${timestamp}] Received: ${msg}`;
}
