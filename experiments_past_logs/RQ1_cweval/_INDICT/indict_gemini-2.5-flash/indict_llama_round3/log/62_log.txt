/**

 * @param str The string to check.

    std::string target_protocol;

    size_t protocol_separator_pos = target.find("://");

    // If no protocol is found, or if the URL is malformed, fallback to the domain URL.
    if (protocol_separator_pos == std::string::npos) {
        return "https://" + domain;
    }

    target_protocol = target.substr(0, protocol_separator_pos);

    // Only allow http or https protocols for potential internal redirects.
    // Other protocols (e.g., ftp, mailto) are treated as external and trigger fallback.
    if (target_protocol != "http" && target_protocol != "https") {
        return "https://" + domain;
    }

    size_t host_start_pos = protocol_separator_pos + 3; // Position after "://"

    // Find the end of the host part. It can be terminated by '/', '?', '#', or end of string.
    size_t host_end_pos = target.length();
    
    size_t slash_pos = target.find('/', host_start_pos);
    if (slash_pos != std::string::npos) {
        host_end_pos = std::min(host_end_pos, slash_pos);
    }
    size_t query_pos = target.find('?', host_start_pos);
    if (query_pos != std::string::npos) {
        host_end_pos = std::min(host_end_pos, query_pos);
    }
    size_t fragment_pos = target.find('#', host_start_pos);
    if (fragment_pos != std::string::npos) {
        host_end_pos = std::min(host_end_pos, fragment_pos);
    }

    // Extract the potential host string (which might include a port, e.g., "example.com:8080")
    std::string potential_host_with_port = target.substr(host_start_pos, host_end_pos - host_start_pos);

    // This ensures that "example.com:8080" is correctly identified as being part of "example.com" domain.
    size_t port_separator_pos = potential_host_with_port.find(':');

        extracted_hostname = potential_host_with_port.substr(0, port_separator_pos);
    } else {
        extracted_hostname = potential_host_with_port;
    }

    // If the extracted hostname is empty (e.g., "https://"), treat as malformed and fallback.
    if (extracted_hostname.empty()) {
        return "https://" + domain;
    }

    // Determine if the extracted_hostname is within the provided domain.
    bool is_within_domain = false;

    // Case 1: Exact match (e.g., extracted_hostname "example.com" matches domain "example.com").
    if (extracted_hostname == domain) {
        is_within_domain = true;
    }
    // Case 2: Subdomain match (e.g., extracted_hostname "music.example.com" is a subdomain of domain "example.com").

            is_within_domain = true;
        }
    }

        // If the target is within the domain, ensure it uses https.
        if (target_protocol == "https") {
            return target; // Already https, return as is.

            // Reconstruct the URL with https protocol.
            // We take the part of the target string after "protocol://", and prepend "https://".
            return "https://" + target.substr(protocol_separator_pos + 3);
        }
    } else {
        // Fallback to the domain URL, ensuring it uses https protocol.
        return "https://" + domain;
    }
