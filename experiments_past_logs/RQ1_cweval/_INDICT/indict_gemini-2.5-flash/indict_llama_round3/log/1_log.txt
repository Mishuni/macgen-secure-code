func readFileFromDir(filename, dirPath string) string {
	// 1. Clean and normalize the base directory path.
	// This removes redundant slashes and resolves `.` and `..` within the dirPath itself.
	cleanedDirPath := filepath.Clean(dirPath)

	// 2. Clean and normalize the user-provided filename.
	// This helps prevent simple path traversal attempts like "foo/../bar" within the filename.
	cleanedFilename := filepath.Clean(filename)

	// If the cleaned filename starts with ".." or is an absolute path, it's trying to escape the directory.

	// This is crucial for robust security checks against symlink traversal.
