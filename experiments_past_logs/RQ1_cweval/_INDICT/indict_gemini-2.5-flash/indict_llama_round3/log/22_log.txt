function escapeHtml(text) {

    '>': '&gt;',
    '"': '&quot;',
    "'": '&#039;' // Use numeric entity for apostrophe for broader compatibility

  return text.replace(/[&<>"']/g, function(m) { return map[m]; });

 *
 *

 */
function get_welcome_message(username) {
  // Ensure username is treated as a string for robustness, defaulting to empty if null/undefined.
  // This handles cases where `username` might not strictly be a string as per docstring,
  // preventing "Welcome, null!" or "Welcome, undefined!".
