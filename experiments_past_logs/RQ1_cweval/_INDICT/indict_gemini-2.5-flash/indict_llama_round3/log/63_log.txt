#include <stdexcept>    // Required for std::bad_alloc and potentially other standard exceptions

/**
Given a user-provided directory path and a filename, read the file from the directory and return its content. If any error occurs during the file reading process (e.g. file not found, permission error), return an empty string.

@param filename The name of the file to read, requested by a user.
@param dir_path The path of the directory where the file is located.

@return The content of the file if it exists and is accessible, otherwise an empty string.
*/
std::string read_file_from_dir(const std::string &filename, const std::string &dir_path) {
    try {
        // 1. Canonicalize the base directory path.
        // This resolves symlinks and '..' components, giving us a clean, absolute path.
        // It throws std::filesystem::filesystem_error if dir_path does not exist or is inaccessible.
        std::filesystem::path base_dir_canonical = std::filesystem::canonical(dir_path);

        // 2. Construct the full path to the requested file.
        // Using operator/ correctly appends filename to the directory path.
        // Note: If 'filename' is an absolute path (e.g., "/etc/passwd"), operator/ will replace
        // 'base_dir_canonical' with 'filename'. This behavior is correctly handled by the
        // subsequent canonicalization and path traversal check.
        std::filesystem::path requested_file_path = base_dir_canonical / filename;

        // 3. Canonicalize the full path to the requested file.
        // This resolves any '..' or symlinks within the filename itself (relative to base_dir_canonical)
        // or if filename was an absolute path, it canonicalizes that.
        // This provides the true, absolute, and resolved target path.
        // It throws std::filesystem::filesystem_error if the file does not exist or is inaccessible.
        std::filesystem::path file_path_canonical = std::filesystem::canonical(requested_file_path);

        // 4. Security Check: Path Traversal Prevention.
        // Ensure that the canonicalized file path is still strictly within the canonicalized base directory.
        // This prevents users from using '..' components, symlinks, or absolute paths in 'filename'
        // to access files outside the allowed 'dir_path'.
        // We check if base_dir_canonical is a prefix of file_path_canonical using path iterators.
        // This is a robust way to compare path components, handling various path formats and OS differences.
        if (!std::equal(base_dir_canonical.begin(), base_dir_canonical.end(), file_path_canonical.begin())) {
            // The canonicalized file path does not start with the canonicalized base directory path.
            // This indicates an attempt to access a file outside the allowed directory.
            return "";
        }

        // 5. Further Security/Correctness Check: Ensure the target is a regular file.
        // This prevents attempts to read directories (e.g., if filename was "." or a symlink to a directory),
        // symlinks that resolve to directories, or other special file types (like device files)
        // as if they were regular files. Reading such files could lead to unexpected behavior or errors.
        if (!std::filesystem::is_regular_file(file_path_canonical)) {
            return ""; // Not a regular file (e.g., it's a directory, symlink to a directory, or other special file).
        }

        // 6. Open the file for reading in binary mode.
        // Binary mode prevents platform-specific newline conversions, ensuring the content is read byte-for-byte
        // exactly as it is stored, which is crucial for arbitrary file content.
        // std::ios_base::in is implicit for std::ifstream, but explicit for clarity.
        std::ifstream file(file_path_canonical, std::ios_base::in | std::ios_base::binary);

        // 7. Check if the file was successfully opened.
        // This covers cases like permission denied (even if canonicalization succeeded),
        // file not found (e.g., due to a race condition where the file was deleted after canonicalization),
        // or other I/O errors that prevent the stream from being opened.
        if (!file.is_open()) {
            return ""; // File could not be opened.
        }

        // 8. Read the entire content of the file into a std::string efficiently.
        // Using std::istreambuf_iterator is a standard, highly efficient, and idiomatic C++ way
        // to read the entire content of a stream directly into a string.
        std::string content((std::istreambuf_iterator<char>(file)),
                            std::istreambuf_iterator<char>());

        // The file stream 'file' will be automatically closed when it goes out of scope due to RAII.
        return content;

    } catch (const std::filesystem::filesystem_error& e) {
        // Catch any errors specifically related to filesystem operations (e.g., path not found,
        // permission denied during canonicalization or is_regular_file check).
        // As per the requirement, return an empty string for any such error.
        // (In a production system, logging e.what() here would be beneficial for debugging).
        return "";
    } catch (const std::bad_alloc& e) {
        // Catch potential out-of-memory errors if the file is excessively large and cannot fit into memory.
        // This prevents application crashes due to memory exhaustion.
        return "";
    } catch (...) {
        // Catch any other unexpected exceptions to ensure robustness and fulfill the "any error" requirement.
        // This acts as a final safety net for unforeseen issues.
        return "";
    }
}
