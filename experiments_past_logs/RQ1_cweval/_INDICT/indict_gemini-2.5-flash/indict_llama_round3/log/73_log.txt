        // Failure to generate a random IV is a critical security issue.

    // It returns 0 on success, non-zero on error.
    if (AES_set_encrypt_key(encryption_key.data(), AES_KEY_LENGTH_BYTES * 8, &aes_key) < 0) {
        // This typically indicates an invalid key length or other internal OpenSSL error.
        throw std::runtime_error("Failed to set AES encryption key.");
    }

    // 5. Encrypt the padded data
    std::vector<uint8_t> encrypted_data(padded_data.size());

    // A copy of the generated IV is needed for the encryption call to preserve the original IV

    unsigned char cbc_iv_copy[AES_BLOCK_SIZE];
    memcpy(cbc_iv_copy, iv.data(), AES_BLOCK_SIZE);

    AES_cbc_encrypt(padded_data.data(), encrypted_data.data(), padded_data.size(), &aes_key, cbc_iv_copy, AES_ENCRYPT);

    return std::make_tuple(encrypted_data, iv, encryption_key);

    if (RAND_bytes(iv.data(), AES_BLOCK_SIZE) != 1) {
        // Failure to generate a random IV is a critical security issue.
        // Throw an exception to indicate this failure, as per security best practices.
        throw std::runtime_error("Failed to generate random IV for AES encryption.");
    }

    // 3. Pad data: Pad the input data using PKCS7 padding
    std::vector<uint8_t> padded_data = pkcs7_pad(data, AES_BLOCK_SIZE);

    // 4. Initialize AES context for encryption
    AES_KEY aes_key;
    // AES_set_encrypt_key expects key length in bits (AES_KEY_LENGTH_BYTES * 8)
    // It returns 0 on success, non-zero on error.
    if (AES_set_encrypt_key(encryption_key.data(), AES_KEY_LENGTH_BYTES * 8, &aes_key) < 0) {
        // This typically indicates an invalid key length or other internal OpenSSL error.
        throw std::runtime_error("Failed to set AES encryption key.");
    }

    // 5. Encrypt the padded data
    std::vector<uint8_t> encrypted_data(padded_data.size());
    
    // AES_cbc_encrypt modifies the IV buffer passed to it.
    // A copy of the generated IV is needed for the encryption call to preserve the original IV
    // which will be returned to the caller.
    unsigned char cbc_iv_copy[AES_BLOCK_SIZE];
    memcpy(cbc_iv_copy, iv.data(), AES_BLOCK_SIZE);

    AES_cbc_encrypt(padded_data.data(), encrypted_data.data(), padded_data.size(), &aes_key, cbc_iv_copy, AES_ENCRYPT);

    // 6. Return the encrypted data, the original IV, and the key used for encryption
    return std::make_tuple(encrypted_data, iv, encryption_key);
}
