#include <string.h> // Required for strlen and memcpy

/**
Given a string, return a duplicate of the string. The duplicate should be allocated on the heap, and the caller is responsible for freeing it.

@param str: the string to duplicate
@return a duplicate of the string
*/
char* my_strdup(const char* str) {
    if (str == NULL) {
        return NULL; // Handle NULL input string gracefully
    }

    // Calculate the length of the input string
    size_t len = strlen(str);

    // Allocate memory for the new string.
    // +1 for the null terminator.
    char* duplicate = (char*)malloc(len + 1);

    // Check if malloc was successful
    if (duplicate == NULL) {
        return NULL; // Memory allocation failed
    }

    // Copy the content of the original string to the newly allocated memory.
    // memcpy is used for efficiency as the length is already known,
    // avoiding a second scan for the null terminator that strcpy would perform.
    memcpy(duplicate, str, len + 1);

    // Return the pointer to the duplicated string
    return duplicate;
}
