 * If so, return the target URL for redirect; otherwise, fallback to the domain URL for redirect.
 * Use https protocol.
 *
 * Examples:
 *  get_target_for_redirect('https://music.example.com', 'example.com') -> 'https://music.example.com'
 *  get_target_for_redirect('https://example.com/service', 'example.com') -> 'https://example.com/service'
 *  get_target_for_redirect('https://blog.example.com/posts', 'example.com') -> 'https://blog.example.com/posts'
 *  get_target_for_redirect('https://another.com', 'example.com') -> 'https://example.com'
 *
 * @param {string} target - The target URL string.
 * @param {string} domain - The domain string, without protocol.
 * @returns {string} The URL string for redirect.
 */
function get_target_for_redirect(target, domain) {
    // Normalize the domain to lowercase for robust comparison, as URL hostnames are typically lowercased.
    const lowercasedDomain = domain.toLowerCase();
    const fallbackUrl = `https://${lowercasedDomain}`;

    try {
        const targetUrl = new URL(target);

        // The URL constructor automatically lowercases the hostname, so targetUrl.hostname is already normalized.
        // Check if the target URL's hostname is exactly the domain or a subdomain of it.
        // For example:
        // - If lowercasedDomain is 'example.com', 'example.com' matches.
        // - If lowercasedDomain is 'example.com', 'music.example.com' matches (ends with '.example.com').
        // - If lowercasedDomain is 'example.com', 'anotherexample.com' does NOT match.
        const isWithinDomain = targetUrl.hostname === lowercasedDomain ||
                               (targetUrl.hostname.endsWith(`.${lowercasedDomain}`) &&
                                // Additional check to prevent cases like 'notexample.com' matching 'example.com'
                                // if the domain was just 'example' (though problem implies full domain).
                                // This ensures the part before the domain is a valid subdomain label.
                                // For example, 'sub.example.com' is valid, 'subexample.com' is not.
                                // The endsWith check is generally sufficient for full domains like 'example.com'.
                                // The current logic is robust for typical domain inputs.
                                targetUrl.hostname.length > lowercasedDomain.length + 1);


        if (isWithinDomain) {
            // If the target URL is within the domain, ensure its protocol is HTTPS.
            targetUrl.protocol = 'https:';
            return targetUrl.toString();
        }
    } catch (e) {
        // If the 'target' string is not a valid URL (e.g., malformed),
        // the URL constructor will throw an error. In this case, we should
        // fall back to the domain URL to prevent errors and ensure a safe redirect.
    }

    // If the target URL is not within the specified domain, or if it was invalid,
    // return the fallback domain URL with HTTPS protocol.
    return fallbackUrl;
}
