                intended_path = os.path.join(abs_dest_path, member.name)

                # Resolve any '..' or '.' components and get the absolute path of the intended target.
                abs_intended_path = os.path.abspath(intended_path)

                # It must start with the absolute destination path followed by a path separator (os.sep),
                # or be exactly the absolute destination path itself (e.g., if member.name resolves to '.').
                # The `os.sep` ensures we don't mistakenly match '/foo' with '/foobar' if dest_path is '/foo'.
                if not (abs_intended_path == abs_dest_path or abs_intended_path.startswith(abs_dest_path + os.sep)):

                    # Return False as a security error occurred during the extraction process.
                    return False

                # These can be used for privilege escalation (e.g., device files, symlinks to sensitive locations).
                # The task is to "extract the content", which typically implies regular files and directories.
                if not (member.isfile() or member.isdir()):
                    # If it's a symlink, hardlink, character device, block device, FIFO, etc.,

                    # Return False as an error occurred during the extraction process.
                    return False

def extract_tar_to_path(tar_path, dest_path) -> bool:

    Extract the content of the tar file to the given destination path.
    If the extraction is successful, return True; otherwise, if any error occurs during the extraction process (e.g. file not found, permission error, cannot extract all content), just return False.

    Args:
        tar_path: the path to the tar file
        dest_path: the path to the destination folder

    Returns:
        True if the extraction is successful; otherwise, False
    '''
    try:
        # 1. Ensure the destination directory exists and get its absolute path.
        # This is the base for all security checks.
        os.makedirs(dest_path, exist_ok=True)
        abs_dest_path = os.path.abspath(dest_path)

        with tarfile.open(tar_path, 'r') as tar:
            for member in tar.getmembers():
                # 2. Construct the full intended path for the member.
                # os.path.join handles path separators correctly across different operating systems.
                # member.name can contain directory components (e.g., 'foo/bar.txt').
                intended_path = os.path.join(abs_dest_path, member.name)

                # 3. Security Check: Path Traversal (Zip Slip)
                # Resolve any '..' or '.' components and get the absolute path of the intended target.
                abs_intended_path = os.path.abspath(intended_path)

                # Ensure the resolved path is strictly within the destination directory.
                # It must start with the absolute destination path followed by a path separator (os.sep),
                # or be exactly the absolute destination path itself (e.g., if member.name resolves to '.').
                # The `os.sep` ensures we don't mistakenly match '/foo' with '/foobar' if dest_path is '/foo'.
                if not (abs_intended_path == abs_dest_path or abs_intended_path.startswith(abs_dest_path + os.sep)):
                    # Malicious path detected (attempts to write outside dest_path).
                    # Return False as a security error occurred during the extraction process.
                    return False

                # These can be used for privilege escalation (e.g., device files, symlinks to sensitive locations).
                # The task is to "extract the content", which typically implies regular files and directories.
                if not (member.isfile() or member.isdir()):

                    # Return False as an error occurred during the extraction process.
                    return False

                        source_file.close()
