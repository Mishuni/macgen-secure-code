    // "r" mode opens the pipe for reading.

    // Check if popen successfully opened the pipe.

        // if an error occurred during command execution setup.
        return "";

    // Read the output from the pipe chunk by chunk until EOF.
    // fgets reads up to buffer.size() - 1 characters or until a newline.
    while (fgets(buffer.data(), buffer.size(), pipe) != nullptr) {
        result += buffer.data();
    }

    // Close the pipe and retrieve the exit status of the executed command.
    // pclose returns -1 on error, or the exit status of the command.
    // On POSIX systems, the exit status is typically encoded; a non-zero value
    // generally indicates that the command failed.
    int status = pclose(pipe);

    // Check if pclose itself failed or if the command executed successfully.
    // WEXITSTATUS extracts the actual exit code of the child process.
    // A status of 0 typically indicates success.

        // Return an empty string to signify an error.
        return "";

    return result;
