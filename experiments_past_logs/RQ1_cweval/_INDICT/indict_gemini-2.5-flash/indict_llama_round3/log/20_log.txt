/**
 * Reads the content of a file from a specified directory. If any error occurs during the file reading process (e.g., file not found, permission error), returns an empty string.
 *
 * @param {string} filename - The name of the file to read, requested by a user.
 * @param {string} dir_path - The directory path where the file is located.
 * @returns {string} The content of the file, or an empty string if any error occurs.
 */
function read_file_from_dir(filename, dir_path) {
  try {
    // Resolve the base directory path to its absolute, normalized form.
    // This is crucial for robust path traversal prevention.
    const baseDirPath = path.resolve(dir_path);

    // Construct the full path by joining the base directory and the filename.
    // path.join handles relative segments like '..' but does not prevent traversal
    // if the resulting path goes outside the intended base directory when resolved.
    // It also handles absolute filenames by ignoring preceding path segments.
    const fullPathCandidate = path.join(baseDirPath, filename);

    // Resolve the full path candidate to its absolute, normalized form.
    const resolvedFullPath = path.resolve(fullPathCandidate);

    // Crucial Path Traversal Check:
    // Use path.relative to determine the relationship between the base directory
    // and the resolved full path.
    const relativePath = path.relative(baseDirPath, resolvedFullPath);

    // If relativePath starts with '..', it means resolvedFullPath is outside baseDirPath.
    // If relativePath is an empty string, it means resolvedFullPath is identical to baseDirPath,
    // which implies trying to read the directory itself, not a file within it.
    if (relativePath.startsWith('..') || relativePath === '') {
      // This indicates a path traversal attempt or an attempt to read the directory itself.
      // In either case, we should not proceed.
      return '';
    }

    // If all security checks pass, proceed to read the file content.
    const content = fs.readFileSync(resolvedFullPath, 'utf8');
    return content;
  } catch (error) {
    // If any error occurs during path resolution or file reading (e.g., file not found,
    // permission denied, invalid path), return an empty string as specified.
    return '';
  }
}
