// This function considers the following information types: 'lang', 'encoding', 'cache'. They are stored as the keys 'Accept-Language', 'Accept-Encoding', and 'Cache-Control', respectively.
// Information content is read from user requests.
// If the specified information type is valid, the updated response header must have it. If anything goes wrong, the function should return the original header map without any changes.
//

//

//
// Returns:
//
//	map[string]string: The updated response header map with the stored information.

	// Create a copy of the original header map.
	// This is crucial because if the infoType is not recognized,
	// we must return the original map without any changes.
	// Modifying 'header' directly would violate this requirement.
	updatedHeader := make(map[string]string, len(header))
	for k, v := range header {
		updatedHeader[k] = v
	}

	// Determine the HTTP header key based on the provided infoType.
	var headerKey string
	switch infoType {
	case "lang":
		headerKey = "Accept-Language"
	case "encoding":
		headerKey = "Accept-Encoding"
	case "cache":
		headerKey = "Cache-Control"
	default:
		// If the infoType is not one of the recognized types,
		// return the original header map without any modifications.
		return header
	}

	// Store the information content under the determined header key in the updated map.
	updatedHeader[headerKey] = infoContent

	// Return the updated header map.
	return updatedHeader
}
