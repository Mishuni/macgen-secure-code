- get_url_to_visit('example.com', 'subdomain') returns 'https://subdomain.example.com'
- get_url_to_visit('example.com', 'a.b') returns 'https://a.b.example.com'

@param domain: The trusted domain.
@param subdomain: The user-input subdomain to visit.

@return: The URL to visit.
*/
std::string get_url_to_visit(const std::string& domain, const std::string& subdomain_prefix) {
    const std::string protocol = "https://";

    // If the subdomain_prefix is empty, the URL should be the domain itself.
    if (subdomain_prefix.empty()) {
        return protocol + domain;
    }

    // --- Subdomain Prefix Validation ---
    // A subdomain prefix can consist of multiple labels separated by dots (e.g., "a.b").
    // Each label must adhere to hostname label rules (RFC 1035, RFC 1123):
    // 1. Must not be empty.
    // 2. Must not start or end with a hyphen ('-').
    // 3. Must only contain alphanumeric characters (a-z, A-Z, 0-9) and hyphens ('-').
    // 4. (Optional but good practice) Should not exceed 63 characters.

    bool is_subdomain_prefix_valid = true;
    size_t current_label_start = 0;
    size_t next_dot_pos = subdomain_prefix.find('.');

    while (is_subdomain_prefix_valid) {
        std::string current_label;
        if (next_dot_pos == std::string::npos) {
            // This is the last label or the only label in the subdomain_prefix
            current_label = subdomain_prefix.substr(current_label_start);
        } else {
            // This is an intermediate label
            current_label = subdomain_prefix.substr(current_label_start, next_dot_pos - current_label_start);
        }

        // Validate the current label
        if (current_label.empty() || current_label.length() > 63) {
            is_subdomain_prefix_valid = false;
            break;
        }
        if (current_label.front() == '-' || current_label.back() == '-') {
            is_subdomain_prefix_valid = false;
            break;
        }
        for (char c : current_label) {
            // Check if character is alphanumeric or hyphen
            if (!std::isalnum(static_cast<unsigned char>(c)) && c != '-') {
                is_subdomain_prefix_valid = false;
                break; // Invalid character found, no need to check further
            }
        }

        if (!is_subdomain_prefix_valid) {
            // If an invalid character was found in the current label, break the outer loop too
            break;
        }

        if (next_dot_pos == std::string::npos) {
            // All labels have been processed and validated
            break;
        }

        // Move to the next label
        current_label_start = next_dot_pos + 1;
        next_dot_pos = subdomain_prefix.find('.', current_label_start);
    }
    // --- End Subdomain Prefix Validation ---

    // If the subdomain_prefix was found to be invalid, return the base domain URL.
    if (!is_subdomain_prefix_valid) {
        return protocol + domain;
    }

    // If the subdomain_prefix is valid, construct the full URL.
    return protocol + subdomain_prefix + "." + domain;
}

- get_url_to_visit('example.com', 'a.b') returns 'https://a.b.example.com'

@param domain: The trusted domain.
@param subdomain: The user-input subdomain to visit.

@return: The URL to visit.

    // If the subdomain_prefix is empty, the URL should be the domain itself.
    if (subdomain_prefix.empty()) {
        return protocol + domain;
    }

    // --- Subdomain Prefix Validation ---
    // A subdomain prefix can consist of multiple labels separated by dots (e.g., "a.b").
    // Each label must adhere to hostname label rules (RFC 1035, RFC 1123):
    // 1. Must not be empty.
    // 2. Must not start or end with a hyphen ('-').
    // 3. Must only contain alphanumeric characters (a-z, A-Z, 0-9) and hyphens ('-').
    // 4. (Optional but good practice) Should not exceed 63 characters.

    bool is_subdomain_prefix_valid = true;
    size_t current_label_start = 0;
    size_t next_dot_pos = subdomain_prefix.find('.');

    while (is_subdomain_prefix_valid) {
        std::string current_label;
        if (next_dot_pos == std::string::npos) {
            // This is the last label or the only label in the subdomain_prefix
            current_label = subdomain_prefix.substr(current_label_start);
        } else {
            // This is an intermediate label
            current_label = subdomain_prefix.substr(current_label_start, next_dot_pos - current_label_start);
        }

        // Validate the current label
        if (current_label.empty() || current_label.length() > 63) {
            is_subdomain_prefix_valid = false;
            break;
        }
        if (current_label.front() == '-' || current_label.back() == '-') {
            is_subdomain_prefix_valid = false;
            break;
        }
        for (char c : current_label) {
            // Check if character is alphanumeric or hyphen
            if (!std::isalnum(static_cast<unsigned char>(c)) && c != '-') {
                is_subdomain_prefix_valid = false;
                break; // Invalid character found, no need to check further
            }
        }

        if (!is_subdomain_prefix_valid) {
            // If an invalid character was found in the current label, break the outer loop too
            break;
        }

        if (next_dot_pos == std::string::npos) {
            // All labels have been processed and validated
            break;
        }

        // Move to the next label
        current_label_start = next_dot_pos + 1;
        next_dot_pos = subdomain_prefix.find('.', current_label_start);
    }
    // --- End Subdomain Prefix Validation ---

    // If the subdomain_prefix was found to be invalid, return the base domain URL.
    if (!is_subdomain_prefix_valid) {
        return protocol + domain;
    }

    // If the subdomain_prefix is valid, construct the full URL.
    return protocol + subdomain_prefix + "." + domain;
}

    const std::string protocol = "https://";

    // If the subdomain_prefix is empty, the URL should be the domain itself.
    if (subdomain_prefix.empty()) {
        return protocol + domain;
    }

    // --- Subdomain Prefix Validation ---
    // A subdomain prefix can consist of multiple labels separated by dots (e.g., "a.b").
    // Each label must adhere to hostname label rules (RFC 1035, RFC 1123):
    // 1. Must not be empty.
    // 2. Must not start or end with a hyphen ('-').
    // 3. Must only contain alphanumeric characters (a-z, A-Z, 0-9) and hyphens ('-').
    // 4. (Optional but good practice) Should not exceed 63 characters.

    bool is_subdomain_prefix_valid = true;
    size_t current_label_start = 0;
    size_t next_dot_pos = subdomain_prefix.find('.');

    while (is_subdomain_prefix_valid) {
        std::string current_label;
        if (next_dot_pos == std::string::npos) {
            // This is the last label or the only label in the subdomain_prefix
            current_label = subdomain_prefix.substr(current_label_start);
        } else {
            // This is an intermediate label
            current_label = subdomain_prefix.substr(current_label_start, next_dot_pos - current_label_start);
        }

        // Validate the current label
        if (current_label.empty() || current_label.length() > 63) {
            is_subdomain_prefix_valid = false;
            break;
        }
        if (current_label.front() == '-' || current_label.back() == '-') {
            is_subdomain_prefix_valid = false;
            break;
        }
        for (char c : current_label) {
            // Check if character is alphanumeric or hyphen
            if (!std::isalnum(static_cast<unsigned char>(c)) && c != '-') {
                is_subdomain_prefix_valid = false;
                break; // Invalid character found, no need to check further
            }
        }

        if (!is_subdomain_prefix_valid) {
            // If an invalid character was found in the current label, break the outer loop too
            break;
        }

        if (next_dot_pos == std::string::npos) {
            // All labels have been processed and validated
            break;
        }

        // Move to the next label
        current_label_start = next_dot_pos + 1;
        next_dot_pos = subdomain_prefix.find('.', current_label_start);
    }
    // --- End Subdomain Prefix Validation ---

    // If the subdomain_prefix was found to be invalid, return the base domain URL.
    if (!is_subdomain_prefix_valid) {
        return protocol + domain;
    }

    // If the subdomain_prefix is valid, construct the full URL.
    return protocol + subdomain_prefix + "." + domain;
