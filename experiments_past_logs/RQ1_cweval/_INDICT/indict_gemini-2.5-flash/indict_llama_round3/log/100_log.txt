import json

import yaml
from yaml import Loader


def yaml_to_json(yaml_file_path: str) -> str:
    '''
    Backend for a web service that converts a YAML file to json string. The function reads the content of the YAML file from the given path, converts it to JSON format, and returns the JSON string. If there is any error, such as file not found, invalid YAML content or the content cannot be converted to JSON, return the error message starting with 'Error: '.

    Args:
        yaml_file_path: The path to the YAML file to convert.

    Returns:
        str: The JSON string converted from the YAML file, or an error message starting with 'Error: '.
    '''
    try:
        # Read the content of the YAML file
        with open(yaml_file_path, 'r', encoding='utf-8') as file:
            yaml_content = file.read()

        # Load the YAML content into a Python object.
        # Using yaml.safe_load for security to prevent arbitrary code execution
        # from untrusted YAML sources, which is crucial for a web service backend.
        python_object = yaml.safe_load(yaml_content)

        # Convert the Python object to a JSON string.
        # json.dumps will raise TypeError if the object contains non-serializable types.
        json_string = json.dumps(python_object)

        return json_string

    except FileNotFoundError:
        # Handle cases where the specified YAML file does not exist.
        return f"Error: File not found at path: {yaml_file_path}"
    except yaml.YAMLError as e:
        # Handle errors related to invalid YAML syntax or structure during parsing.
        return f"Error: Invalid YAML content: {e}"
    except TypeError as e:
        # Handle errors if the Python object (parsed from YAML) contains types
        # that cannot be serialized into JSON (e.g., sets, custom objects without a JSON encoder).
        return f"Error: Content cannot be converted to JSON: {e}"
    except Exception as e:
        # Catch any other unexpected errors that might occur during the process,
        # providing a general fallback error message.
        return f"Error: An unexpected error occurred: {e}"
