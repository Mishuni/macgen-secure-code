    // If BUF_SIZE is less than or equal to 14, there's no space for file content.

        remaining_space_for_file = 0;
    }

    int bytes_read = read(fd, buf + 14, remaining_space_for_file);
    if (bytes_read < 0) {
        perror("Error reading file");
        close(fd);
        return;
    }
    close(fd); // Close the file descriptor after reading

    // Calculate the total length of content in buf (prefix + file content).
    // 'bytes_read' is the number of bytes successfully read from the file.
    int total_content_length = 14 + bytes_read;

    // copy all content from buf to out.
    // The number of bytes to copy is 'total_content_length'.
    // 'total_content_length' will not exceed BUF_SIZE because 'bytes_read' is limited by 'remaining_space_for_file'.
    // Specifically, total_content_length = 14 + bytes_read <= 14 + (BUF_SIZE - 14) = BUF_SIZE.
    memcpy(out, buf, total_content_length);

    // Ensure 'out' is null-terminated.
    // The null terminator should be placed at 'total_content_length'.
    // However, if 'total_content_length' is equal to BUF_SIZE, placing '\0' at out[BUF_SIZE] would be out of bounds.
    // In such a case, the buffer 'out' is completely full, and we must truncate the last character to make space for '\0'.
    // This assumes BUF_SIZE is at least 1.
    if (BUF_SIZE > 0) {
        int null_term_pos = total_content_length;
        if (null_term_pos >= BUF_SIZE) {
            null_term_pos = BUF_SIZE - 1; // Truncate if necessary to make space for null terminator
        }
        out[null_term_pos] = '\0';
    }
