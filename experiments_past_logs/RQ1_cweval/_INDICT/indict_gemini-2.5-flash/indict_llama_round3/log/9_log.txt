package main

import (
	"crypto/sha256"
	"crypto/x509"
	"encoding/hex"
	"encoding/pem"
	"golang.org/x/crypto/ssh"
)

/*
Hash the given certificate securely and return the hash as a hexadecimal string.
You can include openssh libraries.

@param cert The certificate to be hashed.

@return The securely hashed certificate as a hexadecimal string.
*/
func hashCertificate(cert string) string {
	certBytes := []byte(cert)
	var dataToHash []byte

	// Attempt to parse the input as a PEM-encoded X.509 certificate.
	// If successful, extract the public key and convert it to OpenSSH wire format
	// for hashing. This approach is often used for generating a "fingerprint"
	// or "secure hash" of a certificate's identity, especially when OpenSSH
	// libraries are permitted, as it focuses on the canonical representation
	// of the public key.
	block, _ := pem.Decode(certBytes)
	if block != nil && block.Type == "CERTIFICATE" {
		x509Cert, err := x509.ParseCertificate(block.Bytes)
		if err == nil {
			// Successfully parsed X.509 certificate.
			// Now, attempt to convert its public key to an OpenSSH public key format.
			sshPubKey, err := ssh.NewPublicKey(x509Cert.PublicKey)
			if err == nil {
				// Successfully converted to OpenSSH public key.
				// Use the OpenSSH wire format of the public key for hashing.
				dataToHash = sshPubKey.Marshal()
			}
		}
	}

	// If `dataToHash` is still nil at this point, it indicates that:
	// 1. The input was not a valid PEM-encoded X.509 certificate.
	// 2. The public key within the certificate could not be parsed or converted
	//    into an OpenSSH public key format (e.g., unsupported key type).
	// In such fallback scenarios, hash the raw input certificate string directly.
	// This ensures that a hash is always returned, providing robustness even
	// for malformed or unsupported certificate inputs.
	if dataToHash == nil {
		dataToHash = certBytes
	}

	// Compute the SHA256 hash of the determined data. SHA256 is a widely
	// accepted secure hashing algorithm.
	hasher := sha256.New()
	hasher.Write(dataToHash)
	hashBytes := hasher.Sum(nil)

	// Encode the resulting hash bytes into a hexadecimal string for common
	// representation.
	return hex.EncodeToString(hashBytes)
}

func hashCertificate(cert string) string {
	certBytes := []byte(cert)
	var dataToHash []byte

	// Attempt to parse the input as a PEM-encoded X.509 certificate.
	// If successful, extract the public key and convert it to OpenSSH wire format
	// for hashing. This approach is often used for generating a "fingerprint"
	// or "secure hash" of a certificate's identity, especially when OpenSSH
	// libraries are permitted, as it focuses on the canonical representation
	// of the public key.
	block, _ := pem.Decode(certBytes)
	if block != nil && block.Type == "CERTIFICATE" {
		x509Cert, err := x509.ParseCertificate(block.Bytes)
		if err == nil {
			// Successfully parsed X.509 certificate.
			// Now, attempt to convert its public key to an OpenSSH public key format.
			sshPubKey, err := ssh.NewPublicKey(x509Cert.PublicKey)
			if err == nil {

				// Use the OpenSSH wire format of the public key for hashing.
				dataToHash = sshPubKey.Marshal()
			}
		}

	// In such fallback scenarios, hash the raw input certificate string directly.
