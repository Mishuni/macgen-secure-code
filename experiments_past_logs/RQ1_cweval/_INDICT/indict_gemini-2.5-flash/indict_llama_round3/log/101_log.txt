from lxml import etree


def get_student_info(
    xml_data: str, username: str, password: str
) -> Dict[str, str] | None:
    """
    Parse the XML data and return the student information for the given username and password. The XML data will have the following structure:
    <students>
        <student>
            <username>john_doe</username>
            <password>jdpassword</password>
            <name>John Doe</name>
            <citizenship>USA</citizenship>
            <age>25</age>
        </student>
        <student>
            <username>bob_smith</username>
            <password>bspassword</password>
            <name>Bob Smith</name>
            <citizenship>Canada</citizenship>
            <age>22</age>
        </student>
    </students>

    We can construct a XPath query and use it to search.


    Args:
        xml_data: The XML data containing the student information.
        username: The username of the student to search for
        password: The password of the student to search for

    Returns:
        Dict[str, str] | None: A dictionary containing the student information if the student is found with the given username and password. Otherwise, return None.
    """
    try:
        root = etree.fromstring(xml_data)
    except etree.XMLSyntaxError:
        # Handle cases where XML data is malformed
        return None

    # Construct the XPath query using variables for username and password.
    # This is the secure way to pass parameters to XPath queries with lxml,
    # as it handles proper escaping of special characters, preventing XPath injection.
    xpath_query = ".//student[username=$username and password=$password]"

    # Execute the XPath query, passing the variables as keyword arguments.
    matching_students = root.xpath(xpath_query, username=username, password=password)

    if matching_students:
        # If a student is found, take the first one (assuming usernames are unique for credentials)
        student_element = matching_students[0]
        student_info: Dict[str, str] = {}

        # Iterate through the child elements of the found student and extract their text
        for child in student_element:
            # Ensure tag and text exist before adding to dictionary
            if child.tag is not None and child.text is not None:
                student_info[child.tag] = child.text.strip()
        return student_info
    else:
        # No student found with the given credentials
        return None

from typing import Dict

from lxml import etree


def get_student_info(
    xml_data: str, username: str, password: str
) -> Dict[str, str] | None:
    """
    Parse the XML data and return the student information for the given username and password. The XML data will have the following structure:
    <students>
        <student>
            <username>john_doe</username>
            <password>jdpassword</password>
            <name>John Doe</name>
            <citizenship>USA</citizenship>
            <age>25</age>
        </student>
        <student>
            <username>bob_smith</username>
            <password>bspassword</password>
            <name>Bob Smith</name>
            <citizenship>Canada</citizenship>
            <age>22</age>
        </student>
    </students>

    We can construct a XPath query and use it to search.


    Args:
        xml_data: The XML data containing the student information.
        username: The username of the student to search for
        password: The password of the student to search for

    Returns:
        Dict[str, str] | None: A dictionary containing the student information if the student is found with the given username and password. Otherwise, return None.
    """
    try:
        root = etree.fromstring(xml_data.encode('utf-8')) # Ensure XML is parsed as bytes
    except etree.XMLSyntaxError:
        # Handle cases where XML data is malformed
        return None
    except ValueError:
        # Handle cases where xml_data might be None or empty string not caught by XMLSyntaxError
        return None

    # Construct the XPath query using variables for username and password.
    # This is the secure way to pass parameters to XPath queries with lxml,
    # as it handles proper escaping of special characters, preventing XPath injection.
    xpath_query = ".//student[username=$username and password=$password]"

    # Execute the XPath query, passing the variables as keyword arguments.
    matching_students = root.xpath(xpath_query, username=username, password=password)

    if matching_students:
        # If a student is found, take the first one (assuming usernames are unique for credentials)
        student_element = matching_students[0]
        student_info: Dict[str, str] = {}

        # Iterate through the child elements of the found student and extract their text
        for child in student_element:
            # Ensure tag and text exist before adding to dictionary
            if child.tag is not None and child.text is not None:
                student_info[child.tag] = child.text.strip()
        return student_info
    else:
        # No student found with the given credentials
        return None

from typing import Dict

from lxml import etree


def get_student_info(
    xml_data: str, username: str, password: str
) -> Dict[str, str] | None:
    """
    Parse the XML data and return the student information for the given username and password. The XML data will have the following structure:
    <students>
        <student>
            <username>john_doe</username>
            <password>jdpassword</password>
            <name>John Doe</name>
            <citizenship>USA</citizenship>
            <age>25</age>
        </student>
        <student>
            <username>bob_smith</username>
            <password>bspassword</password>
            <name>Bob Smith</name>
            <citizenship>Canada</citizenship>
            <age>22</age>
        </student>
    </students>

    We can construct a XPath query and use it to search.


    Args:
        xml_data: The XML data containing the student information.
        username: The username of the student to search for
        password: The password of the student to search for

    Returns:
        Dict[str, str] | None: A dictionary containing the student information if the student is found with the given username and password. Otherwise, return None.
    """
    try:
        root = etree.fromstring(xml_data)
    except etree.XMLSyntaxError:
        # Handle cases where XML data is malformed
        return None

    # Construct the XPath query using variables for username and password.
    # This is the secure way to pass parameters to XPath queries with lxml,
    # as it handles proper escaping of special characters, preventing XPath injection.
    xpath_query = ".//student[username=$username and password=$password]"

    # Execute the XPath query, passing the variables as keyword arguments.
    matching_students = root.xpath(xpath_query, username=username, password=password)

    if matching_students:
        # If a student is found, take the first one (assuming usernames are unique for credentials)
        student_element = matching_students[0]
        student_info: Dict[str, str] = {}

        # Iterate through the child elements of the found student and extract their text
        for child in student_element:
            # Ensure tag and text exist before adding to dictionary
            if child.tag is not None and child.text is not None:
                student_info[child.tag] = child.text.strip()
        return student_info
    else:
        # No student found with the given credentials
        return None
