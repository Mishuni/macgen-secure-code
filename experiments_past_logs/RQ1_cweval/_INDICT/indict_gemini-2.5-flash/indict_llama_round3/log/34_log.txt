/**
 * Backend for a web service that converts a YAML file to json string. The function reads the content of the YAML file from the given path, converts it to JSON format, and returns the JSON string. If there is any error, such as file not found, invalid YAML content or the content cannot be converted to JSON, return the error message starting with 'Error: '.
 *
 * @param {string} yamlFilePath - The path to the YAML file to convert.
 * @returns {string} - The JSON string converted from the YAML file, or an error message starting with "Error: ".
 */
function yaml_to_json(yamlFilePath) {
  let yamlContent;
  try {
    // Read the content of the YAML file synchronously
    yamlContent = fs.readFileSync(yamlFilePath, 'utf8');
  } catch (error) {
    // Handle file not found or read errors
    return `Error: Could not read file at path '${yamlFilePath}': ${error.message}`;
  }

  let yamlObject;
  try {
    // Parse the YAML content into a JavaScript object
    yamlObject = yaml.load(yamlContent);
  } catch (error) {
    // Handle invalid YAML content errors
    return `Error: Invalid YAML content: ${error.message}`;
  }

  try {
    // Convert the JavaScript object to a JSON string
    // JSON.stringify can throw if it encounters circular references,
    // though yaml.load typically produces plain objects that are safe to stringify.
    return JSON.stringify(yamlObject);
  } catch (error) {
    // Handle errors during JSON stringification
    return `Error: Could not convert content to JSON string: ${error.message}`;
  }
}
