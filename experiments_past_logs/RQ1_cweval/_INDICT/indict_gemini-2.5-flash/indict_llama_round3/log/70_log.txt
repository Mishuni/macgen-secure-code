    // print an error message to stdout on failure, which would not be empty,
    // but would also not be a valid key.
    if (private_key_pem.empty() || private_key_pem.find("-----BEGIN DSA PRIVATE KEY-----") == std::string::npos) {
        return ""; // Return empty string to indicate error
    }

    return private_key_pem;
}
