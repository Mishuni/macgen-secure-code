    char param_filename[] = "/tmp/dsa_params_XXXXXX"; // Template for a unique temporary file name
    int fd = -1; // Initialize fd to an invalid value
    int key_length = 2048; // Recommended key length for DSA
    char command_buf[512]; // Buffer for constructing shell commands
    char* dsaparam_output = NULL;
    char* gendsa_output = NULL;
    char* result = NULL; // To store the final key or NULL
    int param_file_created = 0; // Flag to indicate if param_filename was successfully created

    // 1. Create a unique temporary file for DSA parameters
    fd = mkstemp(param_filename);
    if (fd == -1) {
        // Failed to create temporary file
        goto cleanup;
    }
    close(fd); // Close the file descriptor; openssl will open the file by name
    param_file_created = 1; // File successfully created

    // 2. Generate DSA parameters and store them in the temporary file
    // Command: openssl dsaparam -out <param_filename> <key_length>
    snprintf(command_buf, sizeof(command_buf), "openssl dsaparam -out %s %d", param_filename, key_length);
    dsaparam_output = execute_command(command_buf);

    if (dsaparam_output == NULL) {
        // execute_command failed or returned NULL
        goto cleanup;
    }

    // openssl dsaparam typically prints nothing to stdout on success.
    // If it prints something, it's likely an error message.
    if (strlen(dsaparam_output) > 0) {
        // Error occurred during dsaparam generation (output indicates error)
        goto cleanup;
    }
    free(dsaparam_output); // Free the output as it's not needed
    dsaparam_output = NULL; // Reset pointer

    // 3. Generate the DSA private key using the parameters from the temporary file
    // Command: openssl gendsa -outform PEM <param_filename>
    snprintf(command_buf, sizeof(command_buf), "openssl gendsa -outform PEM %s", param_filename);
    gendsa_output = execute_command(command_buf);

    if (gendsa_output == NULL) {
        // execute_command failed or returned NULL for gendsa
        goto cleanup;
    }

    // Check if the output looks like a PEM key (starts with BEGIN DSA PRIVATE KEY)
    // An empty string or one not containing the expected PEM header indicates an error.
    if (strlen(gendsa_output) == 0 || strstr(gendsa_output, "BEGIN DSA PRIVATE KEY") == NULL) {
        goto cleanup;
    }

    // If we reach here, gendsa_output is the valid key.
    result = gendsa_output;
    gendsa_output = NULL; // Prevent freeing in cleanup, as it's now the result

cleanup:
    // Clean up temporary files
    if (param_file_created) {
        unlink(param_filename);
    }

    // Free any allocated memory that hasn't been returned
    if (dsaparam_output != NULL) {
        free(dsaparam_output);
    }
    if (gendsa_output != NULL) { // This will be NULL if result was set and gendsa_output transferred
        free(gendsa_output);
    }

    return result;
}
