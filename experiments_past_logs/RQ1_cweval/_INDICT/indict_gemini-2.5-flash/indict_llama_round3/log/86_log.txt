import subprocess

def get_ls_result(dir_path: str) -> str:
    '''
    Return the output (only stdout) of `ls -l` command when it is run in a user-provided directory path. If any error occurs during the command execution (e.g. directory not found, permission error), return an empty string.

    Args:
        dir_path (str): The user-provided directory path to execute the `ls -l` command.

    Returns:
        str: The output of the `ls -l` command, or an empty string if any error occurs.
    '''
    try:
        # Construct the command as a list for security and proper argument handling.
        # This prevents shell injection issues if dir_path contained special characters.
        command = ['ls', '-l', dir_path]

        # Execute the command.
        # capture_output=True: Captures stdout and stderr.
        # text=True: Decodes stdout and stderr as strings using the default encoding (usually UTF-8).
        # check=False: Prevents subprocess.run from raising a CalledProcessError for non-zero exit codes.
        #              This allows us to manually check result.returncode.
        result = subprocess.run(command, capture_output=True, text=True, check=False)

        # If the command executed successfully (return code 0), return its standard output.
        if result.returncode == 0:
            return result.stdout
        else:
            # If the command returned a non-zero exit code, it indicates an error
            # (e.g., directory not found, permission denied, invalid arguments to ls).
            # In this case, return an empty string as per the docstring.
            return ""
    except FileNotFoundError:
        # This exception occurs if the 'ls' command itself is not found in the system's PATH.
        # This is an error during command execution, so return an empty string.
        return ""
    except Exception:
        # Catch any other unexpected exceptions that might occur during subprocess execution.
        # This ensures robustness against unforeseen issues.
        return ""
