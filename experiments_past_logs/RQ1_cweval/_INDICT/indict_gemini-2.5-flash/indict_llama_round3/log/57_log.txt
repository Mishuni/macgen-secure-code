    char full_path[1024];

    // Construct the full path to the file.
    int path_len = snprintf(full_path, sizeof(full_path), "%s/%s", directory, filename);

    // Check if path construction failed or if the buffer was too small.
    if (path_len < 0 || (size_t)path_len >= sizeof(full_path)) {
        return;
    }

    // Open the file with specific permissions (0600) directly.
    // O_CREAT: Create the file if it does not exist.
    // O_WRONLY: Open for writing only.
    // O_TRUNC: Truncate the file to zero length if it exists.
    // S_IRUSR | S_IWUSR: Set permissions to 0600 (read/write for owner only).
    // The mode argument to open() is masked by the process's umask.
    // However, S_IRUSR | S_IWUSR (0600) will result in 0600 even with common umasks
    // like 0022 (e.g., 0600 & ~0022 = 0600), ensuring the desired restrictive permissions
    // are applied at creation time, preventing the TOCTOU race condition.
    int fd = open(full_path, O_CREAT | O_WRONLY | O_TRUNC, S_IRUSR | S_IWUSR);
    if (fd == -1) { // -1 indicates an error
        return;
    }

    // Write the content to the file descriptor.
    // write returns the number of bytes written, or -1 on error.
    size_t content_len = strlen(content);
    ssize_t bytes_written = write(fd, content, content_len);
    if (bytes_written == -1 || (size_t)bytes_written != content_len) {
        // Failed to write content or not all content was written.
        close(fd); // Attempt to close the file even on write error.
        return;
    }

    // Close the file descriptor.
    // close returns 0 on success, or -1 on error.
    if (close(fd) == -1) {
        // Failed to close the file properly.
        return;
    }

    // No need for a separate chmod call as permissions were set securely during open().
    // This eliminates the TOCTOU race condition present in the initial solution.
}
