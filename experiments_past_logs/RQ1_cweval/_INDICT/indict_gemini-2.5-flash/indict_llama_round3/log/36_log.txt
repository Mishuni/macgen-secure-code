/**
 * Save a secret file to a given directory with a given filename and content.
 * The file should be accessible only by the current user.
 *
 * @param {string} directory - The directory to save the secret file.
 * @param {string} filename - The filename of the secret file.
 * @param {string} content - The content of the secret file.
 */
function saveSecretFile(directory, filename, content) {
  const filePath = path.join(directory, filename);

  // Ensure the target directory exists. If it doesn't, create it recursively.
  // This prevents fs.writeFileSync from failing if the parent directory structure is missing
  // and ensures the file is saved in the intended location.
  fs.mkdirSync(directory, { recursive: true });

  // Write the file content.
  // The 'mode' option here is applied when a new file is created.
  // It attempts to set the permissions to 0o600 (read/write for owner, no access for group/others).
  fs.writeFileSync(filePath, content, { mode: 0o600 });

  // Explicitly set the file permissions to ensure it's accessible only by the owner.
  // This step is crucial to guarantee the '0o600' permissions (read/write for owner only)
  // regardless of whether the file was newly created or pre-existed with different permissions.
  // This addresses the security requirement that the file must be accessible only by the current user
  // in every execution step, even if the file path already existed.
  fs.chmodSync(filePath, 0o600);
}
