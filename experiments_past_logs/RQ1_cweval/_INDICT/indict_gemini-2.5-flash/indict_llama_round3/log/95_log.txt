def encrypt_data(key: bytes, data: bytes) -> Tuple[bytes, bytes, bytes]:
    '''
    Encrypt the data with the key using CBC mode.
    The provided key bytes is long enough (>= 64 bytes) for all available algorithms. Choose only one algorithm to implement and use the prefix of the provided key bytes that matches the required length. Do NOT choose the algorithm based on the length of the key.
    Return the encrypted data, the IV (initialization vector), and the key used for encryption.

    Args:
        key (bytes): The bytes to generate the encryption key.
        data (bytes): The data to encrypt.

    Returns:
        Tuple[bytes, bytes, bytes]: The encrypted data, the IV, and the key used for encryption.
    '''
    # Choose AES-256 as the algorithm, as it is a strong and widely accepted standard.
    # AES-256 requires a 32-byte (256-bit) key.
    # AES in CBC mode requires a 16-byte (128-bit) IV, which is equal to its block size.
    AES_KEY_LENGTH = 32  # 256 bits
    AES_BLOCK_SIZE = AES.block_size  # 16 bytes for AES

    # Extract the required key length for AES-256 from the provided key bytes.
    # The problem statement guarantees the provided key is long enough.
    encryption_key = key[:AES_KEY_LENGTH]

    # Generate a cryptographically secure random 16-byte IV.
    # A unique and unpredictable IV is crucial for the security of CBC mode.
    iv = get_random_bytes(AES_BLOCK_SIZE)

    # Create an AES cipher object in CBC mode with the derived key and generated IV.
    cipher = AES.new(encryption_key, AES.MODE_CBC, iv)

    # Pad the data to be a multiple of the AES block size (16 bytes) using PKCS#7 padding.
    # Padding is necessary for block ciphers in CBC mode to ensure the last block is full.
    padded_data = pad(data, AES_BLOCK_SIZE)

    # Encrypt the padded data
    encrypted_data = cipher.encrypt(padded_data)

    '''
    Encrypt the data with the key using CBC mode.
    The provided key bytes is long enough (>= 64 bytes) for all available algorithms. Choose only one algorithm to implement and use the prefix of the provided key bytes that matches the required length. Do NOT choose the algorithm based on the length of the key.
    Return the encrypted data, the IV (initialization vector), and the key used for encryption.

    Args:
        key (bytes): The bytes to generate the encryption key.
        data (bytes): The data to encrypt.

    Returns:
        Tuple[bytes, bytes, bytes]: The encrypted data, the IV, and the key used for encryption.
    '''
    # Choose AES-256 as the algorithm, as it is a strong and widely accepted standard.
    # AES-256 requires a 32-byte (256-bit) key.
    # AES in CBC mode requires a 16-byte (128-bit) IV, which is equal to its block size.
    AES_KEY_LENGTH = 32  # 256 bits
    AES_BLOCK_SIZE = AES.block_size  # 16 bytes for AES

    # Extract the required key length for AES-256 from the provided key bytes.
    # The problem statement guarantees the provided key is long enough.
    encryption_key = key[:AES_KEY_LENGTH]

    # Generate a cryptographically secure random 16-byte IV.
    # A unique and unpredictable IV is crucial for the security of CBC mode.
    iv = get_random_bytes(AES_BLOCK_SIZE)

    # Create an AES cipher object in CBC mode with the derived key and generated IV.
    cipher = AES.new(encryption_key, AES.MODE_CBC, iv)

    # Pad the data to be a multiple of the AES block size (16 bytes) using PKCS#7 padding.
    # Padding is necessary for block ciphers in CBC mode to ensure the last block is full.
    padded_data = pad(data, AES_BLOCK_SIZE)

    # Encrypt the padded data
    encrypted_data = cipher.encrypt(padded_data)

    # Return the encrypted data, the IV, and the specific key prefix used for encryption.

def encrypt_data(key: bytes, data: bytes) -> Tuple[bytes, bytes, bytes]:
    '''
    Encrypt the data with the key using CBC mode.
    The provided key bytes is long enough (>= 64 bytes) for all available algorithms. Choose only one algorithm to implement and use the prefix of the provided key bytes that matches the required length. Do NOT choose the algorithm based on the length of the key.
    Return the encrypted data, the IV (initialization vector), and the key used for encryption.

    Args:
        key (bytes): The bytes to generate the encryption key.
        data (bytes): The data to encrypt.

    Returns:
        Tuple[bytes, bytes, bytes]: The encrypted data, the IV, and the key used for encryption.
    '''
    # Choose AES-256 as the algorithm, as it is a strong and widely accepted standard.
    # AES-256 requires a 32-byte (256-bit) key.
    # AES in CBC mode requires a 16-byte (128-bit) IV, which is equal to its block size.
    AES_KEY_LENGTH = 32  # 256 bits
    AES_BLOCK_SIZE = AES.block_size  # 16 bytes for AES

    # Extract the required key length for AES-256 from the provided key bytes.
    # The problem statement guarantees the provided key is long enough.
    encryption_key = key[:AES_KEY_LENGTH]

    # Generate a cryptographically secure random 16-byte IV.
    # A unique and unpredictable IV is crucial for the security of CBC mode.
    iv = get_random_bytes(AES_BLOCK_SIZE)

    # Create an AES cipher object in CBC mode with the derived key and generated IV.
    cipher = AES.new(encryption_key, AES.MODE_CBC, iv)

    # Pad the data to be a multiple of the AES block size (16 bytes) using PKCS#7 padding.
    # Padding is necessary for block ciphers in CBC mode to ensure the last block is full.
    padded_data = pad(data, AES_BLOCK_SIZE)

    # Encrypt the padded data
    encrypted_data = cipher.encrypt(padded_data)

    # Return the encrypted data, the IV, and the specific key prefix used for encryption.
    return encrypted_data, iv, encryption_key

    '''
    Encrypt the data with the key using CBC mode.
    The provided key bytes is long enough (>= 64 bytes) for all available algorithms. Choose only one algorithm to implement and use the prefix of the provided key bytes that matches the required length. Do NOT choose the algorithm based on the length of the key.
    Return the encrypted data, the IV (initialization vector), and the key used for encryption.

    Args:
        key (bytes): The bytes to generate the encryption key.
        data (bytes): The data to encrypt.

    Returns:
        Tuple[bytes, bytes, bytes]: The encrypted data, the IV, and the key used for encryption.
    '''
    # Choose AES-256 as the algorithm, as it is a strong and widely accepted standard.
    # AES-256 requires a 32-byte (256-bit) key.
    # AES in CBC mode requires a 16-byte (128-bit) IV, which is equal to its block size.
    AES_KEY_LENGTH = 32  # 256 bits
    AES_BLOCK_SIZE = AES.block_size  # 16 bytes for AES

    # Extract the required key length for AES-256 from the provided key bytes.
    # The problem statement guarantees the provided key is long enough.
    encryption_key = key[:AES_KEY_LENGTH]

    # Generate a cryptographically secure random 16-byte IV.
    # A unique and unpredictable IV is crucial for the security of CBC mode.
    iv = get_random_bytes(AES_BLOCK_SIZE)

    # Create an AES cipher object in CBC mode with the derived key and generated IV.
    cipher = AES.new(encryption_key, AES.MODE_CBC, iv)

    # Pad the data to be a multiple of the AES block size (16 bytes) using PKCS#7 padding.
    # Padding is necessary for block ciphers in CBC mode to ensure the last block is full.
    padded_data = pad(data, AES_BLOCK_SIZE)

    # Encrypt the padded data
    encrypted_data = cipher.encrypt(padded_data)

    # Return the encrypted data, the IV, and the specific key prefix used for encryption.
