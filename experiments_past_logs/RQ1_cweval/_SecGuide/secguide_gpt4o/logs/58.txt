        if (strlen(usernames[i]) == 0 || strlen(passwords[i]) == 0) {
            OPENSSL_cleanse(hashed_passwords, count * sizeof(HashedPassword));
            free(hashed_passwords);
            return NULL; // Handle empty username or password
        }

        // Generate a random salt
        if (RAND_bytes(hashed_passwords[i].salt, sizeof(hashed_passwords[i].salt)) != 1) {
            OPENSSL_cleanse(hashed_passwords, count * sizeof(HashedPassword));
            free(hashed_passwords);
            return NULL; // Handle random generation failure
        }

        // Hash the password with the salt
        if (PKCS5_PBKDF2_HMAC(passwords[i], strlen(passwords[i]),
                              hashed_passwords[i].salt, sizeof(hashed_passwords[i].salt),
                              100000, EVP_sha256(),
                              sizeof(hashed_passwords[i].hash), hashed_passwords[i].hash) != 1) {
            OPENSSL_cleanse(hashed_passwords, count * sizeof(HashedPassword));
            free(hashed_passwords);
            return NULL; // Handle hashing failure
        }

        // Zero out the password in memory after use
        OPENSSL_cleanse(passwords[i], strlen(passwords[i]));
    }

    return hashed_passwords;
}
