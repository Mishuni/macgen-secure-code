char *read_file_from_dir(const char *filename, const char *dir_path) {

        }
    }

    // Ensure filename contains only one dot
    if (strchr(filename, '.') != strrchr(filename, '.')) {
        return strdup(""); // More than one dot found
    }

    // Validate directory path: ensure it doesn't contain ".." or directory separators
    if (strstr(dir_path, "..") != NULL || strchr(dir_path, '/') != NULL || strchr(dir_path, '\\') != NULL) {
        return strdup(""); // Invalid directory path
    }

    // Construct the full path
    char full_path[PATH_MAX];
    if (snprintf(full_path, sizeof(full_path), "%s/%s", dir_path, filename) >= sizeof(full_path)) {
        return strdup(""); // Path too long
    }

    // Canonicalize the path to ensure it's safe
    char real_path[PATH_MAX];
    if (realpath(full_path, real_path) == NULL) {
        return strdup(""); // Error in resolving path
    }

    // Ensure the file is within the intended directory
    if (strncmp(real_path, dir_path, strlen(dir_path)) != 0) {
        return strdup(""); // File is outside the intended directory
    }

    // Open the file
    FILE *file = fopen(real_path, "r");
    if (!file) {
        return strdup(""); // Error opening file
    }

    // Determine file size
    struct stat st;
    if (fstat(fileno(file), &st) != 0) {
        fclose(file);
        return strdup(""); // Error getting file size
    }

    // Allocate memory for file content
    char *content = (char *)malloc(st.st_size + 1);
    if (!content) {
        fclose(file);
        return strdup(""); // Memory allocation failed
    }

    // Read file content
    size_t read_size = fread(content, 1, st.st_size, file);
    if (read_size != st.st_size) {
        free(content);
        fclose(file);
        return strdup(""); // Error reading file
    }

    // Null-terminate the content
    content[read_size] = '\0';

    // Close the file

    return content;
}
