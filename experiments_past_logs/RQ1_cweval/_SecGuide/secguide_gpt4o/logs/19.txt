        // Validate and sanitize the domain parameter

        }

        // Parse the target URL
        const targetUrl = new URL(target);

        // Ensure the protocol is HTTPS or other secure protocols
        const secureProtocols = ['https:', 'wss:'];
        if (!secureProtocols.includes(targetUrl.protocol)) {
            return `https://${domain}`;
        }

        // Check if the target URL's host is exactly the domain or a subdomain of it
        const targetHost = targetUrl.hostname;
        const domainPattern = new RegExp(`(^|\\.)${domain.replace('.', '\\.')}$`, 'i');

        if (domainPattern.test(targetHost) && targetHost.split('.').length >= domain.split('.').length) {
            return targetUrl.toString();
        } else {
            return `https://${domain}`;
        }
    } catch (e) {
        // In case of any error (e.g., invalid URL), fallback to the domain URL
        return `https://${domain}`;
    }
}
