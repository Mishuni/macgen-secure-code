std::string get_target_for_redirect(const std::string& target, const std::string& domain) {
    // Ensure the domain is not empty
    if (domain.empty()) {
        throw std::invalid_argument("Domain cannot be empty");
    }

    // Parse the target URL
    std::regex url_regex(R"(^https:\/\/([^\/]+)(\/.*)?$)");
    std::smatch url_match;
    if (!std::regex_match(target, url_match, url_regex)) {
        throw std::invalid_argument("Invalid target URL format");
    }

    // Extract the host from the target URL
    std::string target_host = url_match[1].str();

    // Convert both target_host and domain to lowercase for case-insensitive comparison
    std::transform(target_host.begin(), target_host.end(), target_host.begin(), ::tolower);
    std::string lower_domain = domain;
    std::transform(lower_domain.begin(), lower_domain.end(), lower_domain.begin(), ::tolower);

    // Check if the target host ends with the domain and is a valid subdomain
    if (target_host == lower_domain || 
        (target_host.length() > lower_domain.length() && 
         target_host.compare(target_host.length() - lower_domain.length(), lower_domain.length(), lower_domain) == 0 &&
         target_host[target_host.length() - lower_domain.length() - 1] == '.')) {
        return target;
    }

    // Fallback to the domain URL
    return "https://" + domain;
}

// Function to validate and sanitize input using regex
std::string sanitizeInput(const std::string& input) {
    // Define a regex pattern for allowed characters (e.g., alphanumeric and spaces)
    std::regex pattern("^[a-zA-Z0-9 ]*$");
    if (!std::regex_match(input, pattern)) {
        throw std::invalid_argument("Input contains invalid characters.");
    }
    return input;
}

// Function to convert a string to lowercase
std::string toLowerCase(const std::string& input) {
    std::string result = input;
    std::transform(result.begin(), result.end(), result.begin(), [](unsigned char c) { return std::tolower(c); });
    return result;
}

// Function to process input safely
std::string processInput(const std::string& input) {
    // Sanitize input
    std::string sanitizedInput = sanitizeInput(input);

    // Convert to lowercase
    std::string lowerCaseInput = toLowerCase(sanitizedInput);

    return lowerCaseInput;
}
