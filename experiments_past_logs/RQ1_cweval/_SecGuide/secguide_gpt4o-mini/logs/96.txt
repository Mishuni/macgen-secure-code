def encrypt_data(key: bytes, data: bytes) -> Tuple[bytes, bytes]:
    '''
    Encrypt the data with the key using AES and GCM mode.

    iv = get_random_bytes(AES.block_size)

    cipher = AES.new(encryption_key, AES.MODE_GCM, nonce=iv)
