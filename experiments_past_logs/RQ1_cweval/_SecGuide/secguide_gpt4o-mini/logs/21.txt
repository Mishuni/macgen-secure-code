const { spawn } = require('child_process');
const path = require('path');

function get_ls_result(dir_path) {
    return new Promise((resolve) => {
        const baseDir = path.resolve('/base/directory'); // Set your base directory here
        const normalizedPath = path.normalize(dir_path);
        const resolvedPath = path.resolve(baseDir, normalizedPath);

        const isValidPath = resolvedPath.startsWith(baseDir) && 
                            normalizedPath === dir_path && 
                            !normalizedPath.includes('..') && 
                            normalizedPath.length <= 255; // Limit input length

        if (!isValidPath) {
            resolve('');
            return;
        }

        const ls = spawn('ls', ['-l', resolvedPath]);

        let output = '';
        ls.stdout.on('data', (data) => {
            output += data.toString();
        });

        ls.stderr.on('data', () => {
            resolve('');
        });

        ls.on('close', (code) => {
            if (code !== 0) {
                resolve('');
                return;
            }
            resolve(output);
        });
    });
}

function get_ls_result(dir_path) {

        const normalizedPath = path.normalize(dir_path);
        const resolvedPath = path.resolve(baseDir, normalizedPath);

        if (!isValidPath) {
            resolve([]);
            return;
        }

        try {
            const files = await fs.readdir(resolvedPath);
            resolve(files);
        } catch (error) {
            resolve([]);
        }
    });
}
