#include <string>
#include <openssl/rsa.h>
#include <openssl/pem.h>
#include <openssl/err.h>

std::string create_private_key(int key_length = 2048) {
    // Validate the key length is within a reasonable range
    if (key_length < 512 || key_length > 4096) {
        return "";
    }

    RSA* rsa;
    rsa = RSA_new();
    if (!rsa) {
        return "";
    }

    BIGNUM* exponent = BN_new();
    if (!exponent) {
        RSA_free(rsa);
        return "";
    }
    
    // Set public exponent (typically 65537 for RSA)
    if (!BN_set_word(exponent, RSA_F4)) {
        BN_free(exponent);
        RSA_free(rsa);
        return "";
    }

    // Generate the key
    if (!RSA_generate_key_ex(rsa, key_length, exponent, NULL)) {
        BN_free(exponent);
        RSA_free(rsa);
        return "";
    }
    
    BN_free(exponent);

    BIO* bio;
    bio = BIO_new(BIO_s_mem());
    if (!bio) {
        RSA_free(rsa);
        return "";
    }

    // Write the private key in PEM format
    if (!PEM_write_bio_RSAPrivateKey(bio, rsa, NULL, NULL, 0)) {
        BIO_free(bio);
        RSA_free(rsa);
        return "";
    }

    char* buffer;
    long length = BIO_get_mem_data(bio, &buffer);
    std::string private_key(buffer, length);

    // Clean up
    BIO_free(bio);
    RSA_free(rsa);

    return private_key;
}

    // Validate the key length is within a secure range (>=2048 bits)
    if (key_length < 2048 || key_length > 4096) {
        return "ERROR: Invalid key length. Must be between 2048 and 4096 bits.";
    }

        char* err_str = ERR_reason_error_string(err);
        std::string error("ERROR: Failed to create RSA structure. ");
        error += err_str;
        OPENSSL_free(err_str);
        return error;
    }

    BIGNUM* exponent = BN_new();
    if (!exponent) {
        unsigned long err = ERR_peek_last_error();
        char* err_str = ERR_reason_error_string(err);
        std::string error("ERROR: Failed to create exponent BIGNUM. ");
        error += err_str;
        RSA_free(rsa);
        OPENSSL_free(err_str);
        return error;

    if (!BN_set_word(exponent, RSA_F4)) {
        unsigned long err = ERR_peek_last_error();

        std::string error("ERROR: Failed to set exponent. ");
        error += err_str;
        BN_free(exponent);
        RSA_free(rsa);
        OPENSSL_free(err_str);
        return error;
    }

    // Seed the random number generator
    RAND_poll();

    if (!RSA_generate_key_ex(rsa, key_length, exponent, NULL)) {

        RSA_free(rsa);
        OPENSSL_free(err_str);
        return error;

    // Write the private key in encrypted PEM format

    unsigned char key[32];
