    '''
    Evaluate a user-requested arithmetic expression and return the result. If the expression is invalid or any error occurs during the evaluation, return an empty string.

    Args:
        expr (str): The user-requested arithmetic expression to evaluate.

    Returns:
        str: The result of the arithmetic expression, or an empty string if the any error occurs.
    '''
    import numbers
    from simpleeval import SimpleEval

    try:
        # Define a safe evaluator with limited operations
        s = SimpleEval()

            'sqrt': lambda x: (x ** 0.5)
        }

        # Evaluate the expression in a controlled environment
        result = s.eval(expr)

        # Check if the result is a boolean value
        if isinstance(result, bool):
            return str(result).lower()

        if isinstance(result, numbers.Number):

            if isinstance(result, complex):
                if result.imag == 0:
                    return str(int(result.real)) if result.real.is_integer() else str(result.real)
