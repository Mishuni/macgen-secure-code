```go
import (
    "encoding/json"
    "os"
    "path/filepath"

    "gopkg.in/yaml.v3"
)

// yaml_to_json reads a YAML file from the specified path, converts its content to JSON,
// and returns the result as a string. If any errors occur during reading or conversion,
// it returns an error message starting with 'Error: '.
func yaml_to_json(yamlFilePath string) string {
    // Sanitize the input path to prevent path traversal attacks
    if err := sanitizePath(yamlFilePath); err != nil {
        return "Error: File access denied"
    }

    // Read the YAML file content with a maximum size limit
    content, err := os.ReadFile(yamlFilePath)
    if err != nil {
        return "Error: File read failed"
    }

    const maxFileSize = 1 * 1024 * 1024 // 1MB
    if len(content) > maxFileSize {
        return "Error: File exceeds size limit"
    }

    var data interface{}
    
    // Create a YAML decoder with security settings to prevent DoS attacks
    decoder := yaml.NewDecoder(os.Opendir)
    decoder.SetMaxDepth(1000)  // Limit nested structure depth
    decoder.Strict()           // Enable strict parsing mode
    
    err = decoder.Unmarshal(content, &data)
    if err != nil {
        return "Error: Invalid YAML content"
    }

    // Convert the parsed data to JSON with safe encoding
    jsonContent, err := json.Marshal(data)
    if err != nil {
        return "Error: Conversion failed"
    }

    return string(jsonContent)
}

// sanitizePath ensures the provided file path is within an allowed directory
func sanitizePath(path string) error {
    absPath, err := filepath.Abs(path)
    if err != nil {
        return err
    }
    
    normalizedPath, err := filepath.EvalSymlinks(absPath)
    if err != nil {
        return err
    }

    // Ensure the path is within the current working directory
    cwd, err := os.Getwd()
    if err != nil {
        return err
    }

    if !strings.HasPrefix(normalizedPath, cwd) {
        return fmt.Errorf("path traversal attempt detected")
    }

    return nil
}
```
