#include <jwt.h>
#include <nlohmann/json.hpp>

using json = nlohmann::json;

std::string verify_jwt_token(const std::string& token, const std::string& key) {
    if (key.empty()) {
        return "{\"error\": \"key_empty\"}";
    }

    jwt_t* token_struct = nullptr;
    int ret;

    // Decode without verification to get header claims
    ret = jwt_decode_without_verify(&token_struct, token.c_str(), static_cast<size_t>(token.length()));
    if (ret != 0) {
        return "{\"error\": \"invalid_token\"}";
    }

    char* alg;
    size_t alg_len;
    ret = jwt_get_claim(token_struct, "alg", &alg, &alg_len);
    if (ret != 0 || alg == nullptr) {
        jwt_free(token_struct);
        return "{\"error\": \"missing_algorithm\"}";
    }

    std::string algorithm(alg, alg_len);
    jwt_free(alg);

    // Check if the algorithm is HMAC
    if (algorithm.find("HS") != 0) {
        jwt_free(token_struct);
        return "{\"error\": \"unsupported_algorithm\"}";
    }

    // Set the algorithm and key for verification
    ret = jwt_setalg(token_struct, algorithm.c_str());
    if (ret != 0) {
        jwt_free(token_struct);
        return "{\"error\": \"key_error\"}";
    }

    ret = jwt_setverifkey(token_struct, JWT_ALG_INSECURE_NO_PUBKEY, key.c_str(), nullptr);
    if (ret != 0) {
        jwt_free(token_struct);
        return "{\"error\": \"key_error\"}";
    }

    // Verify the token
    ret = jwt_verify(token_struct);
    if (ret != 0) {
        jwt_free(token_struct);
        return "{\"error\": \"signature_invalid\"}";
    }

    // Extract payload claims
    json payload_json;
    char* claim_name;
    size_t claim_name_len;
    char* claim_value;
    size_t claim_value_len;

    while (jwt_getclaim(token_struct, &claim_name, &claim_name_len, &claim_value, &claim_value_len) == 0) {
        if (claim_name != nullptr && claim_value != nullptr) {
            payload_json[std::string(claim_name, claim_name_len)] = std::string(claim_value, claim_value_len);
        }
        jwt_free(claim_name);
        jwt_free(claim_value);
    }

    jwt_free(token_struct);

    return payload_json.dump();
}

    json payload_json;
    char* claim_name;
    size_t claim_name_len;
    char* claim_value;
    size_t claim_value_len;

    while (jwt_getclaim(token_struct, &claim_name, &claim_name_len, &claim_value, &claim_value_len) == 0) {
        if (claim_name != nullptr && claim_value != nullptr) {
            payload_json[std::string(claim_name, claim_name_len)] = std::string(claim_value, claim_value_len);
        }
        jwt_free(claim_name);
        jwt_free(claim_value);
    }

    jwt_free(token_struct);

    return payload_json.dump();
}
