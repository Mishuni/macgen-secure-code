func encrypt_data(key, data []byte) ([]byte, []byte, []byte, error) {
	if len(key) != 16 && len(key) != 24 && len(key) != 32 {
		return nil, nil, nil, errors.New("invalid key length")
	}

	keyUsed := make([]byte, 16)
	copy(keyUsed, key)

	nonce := make([]byte, 12)
	if _, err := rand.Read(nonce); err != nil {
		return nil, nil, nil, errors.New("failed to generate nonce: " + err.Error())
	}

	cipherBlock, err := aes.NewCipher(keyUsed)
	if err != nil {
		return nil, nil, nil, errors.New("failed to create cipher block: " + err.Error())
	}

	gcm, err := cipher.NewGCM(cipherBlock)
	if err != nil {
		return nil, nil, nil, errors.New("failed to create GCM: " + err.Error())
	}

	ciphertext := gcm.Seal(nil, nonce, data, nil)

	return ciphertext, nonce, keyUsed, nil
}
