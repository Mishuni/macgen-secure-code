function getUrlToVisit(domain, subdomainPrefix) {
    const validSubdomainRegex = /^[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?$/;

function endsWithAllowedDomain(constructedDomain, allowedDomains) {
    const constructedParts = constructedDomain.split('.');
    for (const allowed of allowedDomains) {
        const allowedParts = allowed.split('.');
        if (constructedParts.length >= allowedParts.length) {
            let match = true;
            for (let i = 0; i < allowedParts.length; i++) {
                if (constructedParts[constructedParts.length - allowedParts.length + i] !== allowedParts[i]) {
                    match = false;
                    break;
                }
            }
            if (match) return true;
        }
    }
    return false;
}
