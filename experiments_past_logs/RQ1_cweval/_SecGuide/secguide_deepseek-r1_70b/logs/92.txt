from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.primitives.asymmetric.rsa import RSAPublicExponent
from cryptography.hazmat.primitives import serialization, hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
from cryptography.hazmat.primitives import padding as symmetric_padding
from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
from cryptography.hazmat.backends import default_backend
import os

def create_private_key() -> tuple:
    """
    Generate a secure RSA private key with encryption and passphrase protection.

    Returns:
        tuple: A tuple containing the encrypted private key (str), public key (str),
              and the generated passphrase (str).
    """
    try:
        # Generate RSA parameters with a secure key size (4096 bits)
        key = rsa.generate_private_key(
            public_exponent=RSAPublicExponent.F4_8,
            key_size=4096,
            backend=default_backend()
        )
        
        # Generate a random 32-byte passphrase
        passphrase = os.urandom(32)
        
        # Derive encryption key using PBKDF2 with SHA256
        kdf = PBKDF2HMAC(
            algorithm=hashes.SHA256(),
            length=32,
            salt=os.urandom(16),
            iterations=100000,
            backend=default_backend()
        )
        key_derivation = kdf.derive(passphrase)
        
        # Encrypt the private key with AES-256-CBC
        cipher = Cipher(
            algorithm=algorithms.AES(key_derivation),
            mode=modes.CBC(),
            backend=default_backend()
        )
        encryptor = cipher.encryptors()
        padder = symmetric_padding.PKCS7(128).padder()
        
        private_pem = key.private_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PrivateFormat.TraditionalOpenSSL,
            encryption_algorithm=serialization.NoEncryption()
        )
        
        padded_data = padder.update(private_pem) + padder.finalize()
        ct = encryptor[0].update(padded_data) + encryptor[0].finalized()
        
        # Serialize public key
        public_pem = key.public_key().public_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PublicFormat.SubjectPublicKeyInfo
        ).decode('utf-8')
        
        return (ct.decode('latin1'), public_pem.decode('utf-8'), passphrase.hex())
        
    except Exception as e:
        raise RuntimeError(f"Failed to generate RSA private key: {str(e)}") from e
