```go
package main

import (
	"bytes"
	"fmt"
	"html"
)

// Sanitizes the username by replacing invalid characters with a placeholder.
func sanitizeUsername(username string) string {
	var buf bytes.Buffer
	for _, r := range username {
		if (r >= 'a' && r <= 'z') ||
		   (r >= 'A' && r <= 'Z') ||
		   (r >= '0' && r <= '9') ||
		   r == '_' || r == '-' || r == '.' || r == ' ' {
			buf.WriteRune(r)
		} else {
			buf.WriteRune('?')
		}
	}
	return buf.String()
}

// getWelcomeMessage generates a secure welcome message for a user.
// It escapes the username to prevent XSS attacks when rendering in HTML.
// The function returns html.SafeString to ensure proper handling by Go templates,
// avoiding double-escaping while maintaining security.
func getWelcomeMessage(username string) html.SafeString {
	sanitizedUsername := sanitizeUsername(username)
	escapedUsername := html.EscapeString(sanitizedUsername)
	return html.SafeString(fmt.Sprintf("Welcome, %s!", escapedUsername))
}
```
