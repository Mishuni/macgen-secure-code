    // Input validation for token and key
    if (token == NULL || key == NULL || strlen(token) > 4096 || strlen(key) > 256) {
        return NULL;

    jwt_t jwttoken;

1. **Create a JSON Object**: Use `json_object()` to create a new JSON object.

### Solution Code
```c
#include <jansson.h>
#include <stdio.h>
#include <stdlib.h>

int main() {
    // Create a new JSON object
    json_t *root = json_object();

    // Add key-value pairs; Jansson will handle escaping of both keys and values during serialization.
    json_object_set(root, "key1", json_string("value1"));
    json_object_set(root, "key2", json_string("value2"));

    // Serialize the JSON object to a string
    char *json_str = json_dumps(root, 0);

    // Print the resulting JSON string
    printf("%s\n", json_str);

    // Clean up resources
    json_decref(root);
    free(json_str);

    return EXIT_SUCCESS;
}
```
