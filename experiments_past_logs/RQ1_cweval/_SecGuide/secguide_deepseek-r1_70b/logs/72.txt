encryption_result encrypt_data(
    const std::vector<unsigned char>& key,
    const std::vector<unsigned char>& data

    }
    result.iv.assign(iv, iv + AES_BLOCK_SIZE);

    // Create and initialize AES-GCM cipher context
    EVP_CIPHER_CTX* ctx = EVP_CIPHER_CTX_new();
    if (!ctx) {
        throw std::runtime_error("Failed to create cipher context");
    }

    const unsigned char* key_ptr = key.data();
    if (EVP_EncryptInit_ex(ctx, EVP_aes_256_gcm(), nullptr, key_ptr, iv) != 1) {

                          static_cast<int>(data.size())) != 1) {
        EVP_CIPHER_CTX_free(ctx);
        throw std::runtime_error("Encryption failed");

    // Finalize encryption to get the authentication tag
    if (EVP_EncryptFinal_ex(ctx, encrypted.data() + len, &len) != 1) {
        EVP_CIPHER_CTX_free(ctx);
        throw std::runtime_error("Failed to finalize encryption");
    }

    result.encrypted_data = encrypted;
    EVP_CIPHER_CTX_free(ctx);

    return result;
}
