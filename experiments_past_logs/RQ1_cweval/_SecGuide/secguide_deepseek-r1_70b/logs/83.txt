        target_hostname = target_hostname.encode('idna').decode('utf-8').lower()
    except UnicodeError:
        # Invalid IDNA encoding, treat as mismatch
        return f"https://{domain}"

    # Parse and normalize the provided domain
    try:
        domain_parts = tldextract.extract(domain)
        provided_domain = f"{domain_parts.domain}.{domain_parts.suffix}".lower()
    except Exception:
        # Fallback to provided domain in case of parsing issues
        provided_domain = domain.lower()

    # Compare domains
    if target_hostname == provided_domain:
        # Use HTTPS and preserve path, query, etc., but ensure scheme is https
        new_scheme = 'https'
        # Rebuild the URL with new scheme
        new_target = parsed_target._replace(scheme=new_scheme)
        return urlunparse(new_target)
    else:
        # Not within domain, redirect to provided domain's HTTPS
        return f"https://{domain}"
