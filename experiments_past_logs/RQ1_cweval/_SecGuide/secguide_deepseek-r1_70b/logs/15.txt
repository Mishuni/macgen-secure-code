import (
    "crypto/pbkdf2"
    "crypto/sha256"
    "crypto/rand"
    "encoding/hex"
)

func hashPasswords(count int, usernames []string, passwords []string, pepper []byte) ([][2]string, error) {
    var hashedCredentials [][2]string
    
    for i := 0; i < count; i++ {
        // Generate salt with username included
        usernameSalt := sha256.Sum256([]byte(usernames[i]))
        salt := make([]byte, 32)
        if _, err := rand.Read(salt); err != nil {
            return nil, err
        }
        
        // Combine username hash and random salt
        combinedSalt := make([]byte, 0, len(usernameSalt) + len(salt))
        combinedSalt = append(combinedSalt, usernameSalt[:]...)
        combinedSalt = append(combinedSalt, salt...)

        // Generate password hash with increased iterations
        passwordHash := pbkdf2.Key(
            []byte(passwords[i]),
            combinedSalt,
            100000,
            32,
            sha256.New,
        )

        if _, err := rand.Read(passwordHash); err != nil {
            return nil, err
        }

        // Combine hash and salt for storage
        hashedPassword := hex.EncodeToString(passwordHash)
        saltString := hex.EncodeToString(salt)
        
        // Add pepper to the final hash
        pepperedHash := sha256.Sum256(append([]byte(pepper), []byte(hashedPassword)...))
        finalHash := hex.EncodeToString(pepperedHash[:])

        hashedCredentials = append(hashedCredentials, [2]string{finalHash, saltString})
    }
    
    return hashedCredentials, nil
}
