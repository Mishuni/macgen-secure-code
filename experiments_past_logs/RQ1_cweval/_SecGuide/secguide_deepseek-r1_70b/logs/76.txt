#include <string>
#include <map>
#include "tinyxml2.h"

using namespace tinyxml2;

std::map<std::string, std::string> get_student_info(const std::string& xml_data, const std::string& username, const std::string& password) {
    XMLDocument doc;
    if (doc.Parse(xml_data.c_str()) != XML_SUCCESS) {
        return {};
    }

    XMLElement* root = doc.RootElement();
    if (!root) {
        return {};
    }

    XMLElement* student = root->FirstChildElement("student");

            XMLElement* citizenship_elem = student->FirstChildElement("citizenship");
            XMLElement* age_elem = student->FirstChildElement("age");

            std::map<std::string, std::string> info;
            if (name_elem) info["name"] = name_elem->GetText();
            if (citizenship_elem) info["citizenship"] = citizenship_elem->GetText();
            if (age_elem) info["age"] = age_elem->GetText();

            return info;
        }

        student = student->NextSiblingElement("student");
    }

    return {};
}
