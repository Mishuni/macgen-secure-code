const checkEmail = (email) => {
    /**
     * Email validation pattern based on RFC 5322 and RFC 6531 standards,
     * allowing for internationalized domain names and most common email formats.
     * The pattern ensures:
     * - Local part (before @) can include letters, numbers, dots, underscores, 
     *   percent signs, plus signs, hyphens, and quoted strings
     * - Domain part supports internationalized domain names (IDNs)
     * - Proper handling of subdomains, multiple top-level domains,
     *   IPv6 addresses, and longer TLDs
     */
    const emailPattern = [
        '^',
        // Local part (before @) with support for quoted strings and RFC 5322
        '(?:[a-zA-Z0-9.!#$%&\'*+/=?^_`{|}~-]+|"(?:[\\x20\\t\\r\\n\\f\\v]|\\\\.|[^\\"])*")@',
        // Domain part (after @) with support for IDNs and IPv6
        '(?:[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?\\.)*' +  // Subdomains
        '[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?' +  // Domain
        '$'
    ].join('');

    const emailRegex = new RegExp(emailPattern);
    
    if (!emailRegex.test(email)) {
        return [false, "Invalid email format"];
    }

    // Check overall email length (RFC 5321: max 254 characters)
    if (email.length > 254) {
        return [false, "Email address exceeds maximum allowed length"];
    }

    // Split email into local and domain parts
    const [localPart, domainPart] = email.split('@');

    // Check local part length (RFC 5321: max 64 characters)
    if (localPart.length > 64) {
        return [false, "Local part of email address exceeds maximum allowed length"];
    }

    // Check domain part length (RFC 5321: max 255 characters)
    if (domainPart.length > 255) {
        return [false, "Domain part of email address exceeds maximum allowed length"];
    }

    // Sanitize the email by escaping special characters
    const sanitizedEmail = email
        .replace(/"/g, '\\"')
        .replace(/'/g, '\\\'')
        .replace(/\\/g, '\\\\');

    return [true, sanitizedEmail];
};
