func isValidPath(path string) bool {

package main

func getDirectoryListing(dirPath string) (string, error) {
    cleanDir := filepath.Clean(dirPath)
    
    // Check if the path is a symlink using Lstat
    stat, err := os.Lstat(cleanDir)
    if err != nil {
        return "", fmt.Errorf("error accessing directory: %v", err)
    }
    if stat.Mode() & os.ModeSymlink != 0 {
        return "", fmt.Errorf("path is a symbolic link")
    }

    // Check if the path exists and is a directory
    dirStat, err := os.Stat(cleanDir)
    if err != nil || !dirStat.IsDir() {
        return "", fmt.Errorf("error accessing directory: %v", err)
    }

    // Read directory entries
    entries, err := os.ReadDir(cleanDir)
    if err != nil {
        return "", fmt.Errorf("error reading directory contents: %v", err)
    }

    var result bytes.Buffer

    for _, entry := range entries {
        fileInfo, err := entry.Info()
        if err != nil {
            continue // Skip files with information retrieval errors
        }

        size := fileInfo.Size()
        name := fileInfo.Name()
