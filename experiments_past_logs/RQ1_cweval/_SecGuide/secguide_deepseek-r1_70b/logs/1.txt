import (
	"fmt"
	"os"
	"path/filepath"
	"regexp"
	"strings"
)

func readFileFromDir(filename, dirPath string) (string, error) {
	// Validate filename to prevent path traversal and invalid characters
	if !regexp.MatchString("^[a-zA-Z0-9_]+$", filename) || strings.Contains(filename, "\x00") {
		return "", fmt.Errorf("invalid characters in filename")
	}

	cleanDirPath := filepath.Clean(dirPath)
	fullPath := filepath.Join(cleanDirPath, filename)

	// Resolve symlinks and normalize the path
	resolvedPath, err := filepath.EvalSymlinks(fullPath)
	if err != nil {
		return "", fmt.Errorf("failed to resolve path: %w", err)
	}
	cleanResolvedPath := filepath.Clean(resolvedPath)

	// Verify the resolved path is within the intended directory
	if _, err := filepath.Rel(cleanDirPath, cleanResolvedPath); err != nil {
		return "", fmt.Errorf("path traversal attempt detected")
	}

	// Open the file to ensure it exists and handle TOCTOU vulnerability
	file, err := os.Open(cleanResolvedPath)
	if err != nil {
		return "", err
	}
	defer file.Close()

	// Check if the opened file is a symbolic link
	info, err := file.Stat()
	if err != nil {
		return "", err
	}
	if info.Mode()&os.ModeSymlink != 0 {
		return "", fmt.Errorf("refusing to read symbolic link")
	}

	// Read the file content after successful validation
	content, err := os.ReadFile(cleanResolvedPath)
	if err != nil {
		return "", err
	}

	return string(content), nil
