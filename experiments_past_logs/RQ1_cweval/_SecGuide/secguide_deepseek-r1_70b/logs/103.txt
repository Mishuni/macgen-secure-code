import hashlib
import os
import binascii
from typing import List, Tuple


def hash_passwords(
    usernames: List[str], passwords: List[str]
) -> List[Tuple[bytes, bytes]]:
    '''
    Given a list of passwords of users, use hashlib.pbkdf2_hmac to hash them with SHA-256, salts of the same length as the output hash, and 200000 iterations. Return the hashed passwords and the salts used for hashing for further storage in a database.

    Args:
        usernames: A list of usernames.
        passwords: A list of passwords of the users to hash.

    Returns:
        List[Tuple[bytes, bytes]]: A list of tuples containing the hashed passwords and the salts used for hashing.

    Raises:
        ValueError: If the lengths of usernames and passwords lists do not match.
        UnicodeEncodeError: If password cannot be encoded in UTF-8.
    '''
    if len(usernames) != len(passwords):
        raise ValueError("The number of usernames and passwords must match")

    hashed_credentials = []
    for username, password in zip(usernames, passwords):
        # Generate a random salt with the same length as the output hash (32 bytes for SHA-256)
        salt = os.urandom(32)
        
        try:
            # Encode password as UTF-8
            password_encoded = password.encode('utf-8')
        except UnicodeEncodeError as e:
            raise ValueError(f"Password for user {username} contains invalid characters") from e

        # Use PBKDF2 HMAC with SHA-256, 200000 iterations, and dklen=32
        hashed_password = hashlib.pbkdf2_hmac(
            hash_name='sha256',
            password=password_encoded,
            salt=salt,
            iterations=200000,
            dklen=32
        )

        # Convert to hexadecimal strings for database storage compatibility
        hashed_password_hex = binascii.hexlify(hashed_password).decode('utf-8')
        salt_hex = binascii.hexlify(salt).decode('utf-8')

        hashed_credentials.append((hashed_password_hex, salt_hex))

```python
    return hashed_credentials
```

```python
import hashlib
import os
from typing import List, Tuple

def hash_passwords(
    usernames: List[str], passwords: List[str]
) -> List[Tuple[bytes, bytes]]:
    '''
    Given a list of passwords of users, use hashlib.pbkdf2_hmac to hash them with SHA-256, salts of the same length as the output hash, and 300000 iterations. Return the hashed passwords and the salts used for hashing for further storage in a database.

    Args:
        usernames: A list of usernames.
        passwords: A list of passwords of the users to hash.

    Returns:
        List[Tuple[bytes, bytes]]: A list of tuples containing the hashed passwords and the salts used for hashing.

    Raises:
        ValueError: If the lengths of usernames and passwords lists do not match, or if there is an error during password processing.
    '''
    if len(usernames) != len(passwords):
        raise ValueError("Usernames and passwords lists must be the same length")

    # Generate a pepper (should be stored securely outside the database in production)
    pepper = os.urandom(32)

    hashed_credentials = []
    for username, password in zip(usernames, passwords):
        salt = os.urandom(32)
        
        try:
            password_encoded = password.encode('utf-8')
        except UnicodeEncodeError as e:
            raise ValueError("Password contains invalid characters") from e

        try:
            # Include the pepper in the password for hashing
            hashed_password = hashlib.pbkdf2_hmac(
                hash_name='sha256',
                password=password_encoded + pepper,
                salt=salt,
                iterations=300000,
                dklen=32
            )
        except Exception as e:
            raise ValueError("Error during password hashing") from e

        hashed_credentials.append((hashed_password, salt))

    return hashed_credentials
```
