package main

import (
	"encoding/json"
	"errors"
	"time"

	"github.com/golang-jwt/jwt/v5"
)

type Claims struct {
	Issuer    string `jwt:"iss"`
	Audience  string `jwt:"aud"`
	ExpiresAt int64  `jwt:"exp"`
	IssuedAt  int64  `jwt:"iat"`
}

func decodeJsonWebToken(token string, key string) (string, error) {
	if len(key) < 32 {
		return "", errors.New("invalid key: key must be at least 32 bytes long")
	}

	parser := jwt.NewParserWithMethods(jwt.SigningMethodHS256)
	claims := Claims{}
	err := parser.ParseWithClaims(token, &claims, []byte(key))
	if err != nil {
		return "", err
	}

	// Validate standard claims
	if claims.ExpiresAt == 0 || claims.ExpiresAt < jwt.TimeNow().Unix() {
		return "", errors.New("token is expired")
	}

	// Validate audience and issuer (these should be configurable)
	expectedIssuer := "your_issuer"
	expectedAudience := "your_audience"

	if claims.Issuer != expectedIssuer {
		return "", errors.New("invalid issuer")
	}

	if claims.Audience != expectedAudience {
		return "", errors.New("invalid audience")
	}

	// Validate issued at time
	if claims.IssuedAt == 0 || claims.IssuedAt > jwt.TimeNow().Unix() {
		return "", errors.New("invalid issued at time")
	}

	// Marshal claims to JSON
	jsonClaims, err := json.Marshal(claims)
	if err != nil {
		return "", err
	}

	return string(jsonClaims), nil
}

```go
package main

import (
	"encoding/json"
	"errors"
	"time"

	"github.com/golang-jwt/jwt/v5"
)

type Claims struct {
	Issuer     string `jwt:"iss"`
	Audience   string `jwt:"aud"`
	ExpiresAt  int64  `jwt:"exp"`
	IssuedAt   int64  `jwt:"iat"`
	NotBefore  int64  `jwt:"nbf"`
}

func decodeJsonWebToken(token string, key string, expectedIssuer string, expectedAudience string) (string, error) {
	if len(key) < 32 {
		return "", errors.New("invalid key")
	}

	parser := jwt.NewParserWithMethods(jwt.SigningMethodHS256)
	claims := Claims{}
	err := parser.ParseWithClaims(token, &claims, []byte(key))
	if err != nil {
		return "", errors.New("invalid token")
	}

	leeway := 1 * time.Minute
	now := jwt.TimeNow().Add(-leeway)

	// Validate expiration with leeway
	if claims.ExpiresAt < now.Unix() {
		return "", errors.New("token expired")
	}

	// Validate not-before claim
	if claims.NotBefore > now.Unix()+leeway.Seconds() {
		return "", errors.New("token not yet valid")
	}

	// Validate issued at time with leeway
	if claims.IssuedAt > now.Unix()+leeway.Seconds() {
		return "", errors.New("invalid token")
	}

	// Validate issuer and audience
	if claims.Issuer != expectedIssuer {
		return "", errors.New("invalid token")
	}

	if claims.Audience != expectedAudience {
		return "", errors.New("invalid token")
	}

	return "Token is valid", nil
}
```
