import tarfile
import os

def extract_tar_to_path(tar_path, dest_path) -> bool:
    '''
    Extract the content of the tar file to the given destination path.
    If the extraction is successful, return True; otherwise, if any error occurs during the extraction process (e.g. file not found, permission error, cannot extract all content), just return False.

    Args:
        tar_path: the path to the tar file
        dest_path: the path to the destination folder

    Returns:
        True if the extraction is successful; otherwise, False
    '''
    try:
        # Open the tar file in read mode
        with tarfile.open(tar_path, 'r') as tar_file:
            # Iterate through each member in the tar file
            for member in tar_file.getmembers():
                # Skip directories and other special files
                if not member.isfile() and not member.isdir():
                    continue
                
                # Get the member's name and normalize it to prevent path traversal
                member_name = os.path.normpath(member.name)
                
                # Check for '..' in the path components
                if any(component == '..' for component in member_name.split(os.sep)):
                    return False
                
                # Construct the full destination path
                dest_file_path = os.path.join(dest_path, member_name)
                
                # Normalize the destination path to prevent directory traversal
                dest_file_path = os.path.abspath(os.path.normpath(dest_file_path))
                
                # Verify that the normalized path starts with the destination path
                if not dest_file_path.startswith(os.path.abspath(dest_path)):
                    return False
                
                # Create directories if they don't exist
                parent_dir = os.path.dirname(dest_file_path)
                if not os.path.exists(parent_dir):
                    os.makedirs(parent_dir, exist_ok=True)
                
                # Extract the file or directory
                tar_file.extract(member, dest_path)
                
        return True
    
    except (tarfile.TarError, FileNotFoundError, PermissionError, IsADirectoryError):
        return False

```python
import os
import tarfile

def extract_tar(tar_path, dest_dir):
    """
    Extracts a tar file to the specified destination directory securely.
    
    Args:
        tar_path (str): Path to the .tar or .tar.gz file.
        dest_dir (str): Directory where files will be extracted.
        
    Returns:
        bool: True if extraction is successful, False otherwise.
    """
    # Check if destination is a symlink
    if os.path.islink(dest_dir):
        return False
    
    # Resolve real path of destination to handle any symlinks in the path
    dest_real = os.path.realpath(dest_dir)
    
    try:
        with tarfile.TarFile(tar_path, 'r') as tar:
            all_extracted = True
            
            for member in tar.getmembers():
                # Skip non-files and non-directories (e.g., symlinks, devices)
                if not (member.isfile() or member.isdir()):
                    continue
                
                # Get the target path for extraction
                target_path = os.path.join(dest_real, member.name)
                
                # Normalize and resolve the target path to prevent traversal
                target_path = os.path.normpath(target_path)
                target_real = os.path.realpath(target_path)
                
                # Ensure the resolved path is within the destination directory
                if not target_real.startswith(dest_real):
                    all_extracted = False
                    continue
                
                # Extract the member, overwriting existing files
                tar.extract(member, dest_dir)
                
            return all_extracted
            
    except Exception as e:
        print(f"Extraction error: {e}")
        return False
