const crypto = require('crypto');

function encryptData(key, data) {
    // Choose AES algorithm with GCM mode for authenticated encryption
    const aesKeyLength = 32; // 256 bits (32 bytes)
    
    // Validate that the key is exactly 32 bytes long
    if (key.length !== aesKeyLength) {
        throw new Error('Key must be exactly 32 bytes long');
    }
    
    // Extract the key - take the first 32 bytes from the provided key bytes
    const aesKey = key.slice(0, aesKeyLength);
    
    // Generate a random initialization vector (IV) with length equal to block size
    const iv = crypto.randomBytes(16); // 16 bytes for AES
    
    // Create cipher object with AES-256-GCM algorithm
    const cipher = crypto.createCipheriv('aes-256-gcm', aesKey, iv, {
        authTagLength: 16 // Use 128 bits (16 bytes) for authentication tag
    });
    
    // Encrypt the data
    const encryptedData = Buffer.concat([cipher.update(data), cipher.final()]);
    
    // Get the authentication tag
    const authTag = cipher.getAuthTag();
    
    // Concatenate IV, encrypted data, and authentication tag
    const completeEncryptedData = Buffer.concat([iv, encryptedData, authTag]);
    
    // Return only necessary components - exclude the key for security
    return {
        encryptedData: completeEncryptedData
    };
}
