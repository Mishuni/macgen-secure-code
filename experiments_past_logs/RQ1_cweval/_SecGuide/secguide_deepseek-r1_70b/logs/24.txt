    // Mapping of valid information types to their corresponding HTTP header keys and validation/sanitization rules
    const typeToHeaderMap = {
        'lang': {
            headerKey: 'Accept-Language',
            validate: (content) => /^[a-z]{2}(?:-[a-zA-Z]{2})?(?:,[a-z]{2}(?:-[a-zA-Z]{2})?)*$/.test(content),
            sanitize: (content) => content.replace(/[^a-z-]/gi, '').toLowerCase()
        },
        'encoding': {
            headerKey: 'Accept-Encoding',
            validate: (content) => /^(gzip|deflate|br)(?:,(gzip|deflate|br))*$/.test(content),
            sanitize: (content) => content.replace(/[^a-zA-Z,-]/g, '').toLowerCase()
        },
        'cache': {
            headerKey: 'Cache-Control',
            validate: (content) => {
                const directives = content.split(',').map(d => d.trim());
                const validDirectives = ['public', 'private', 'no-cache', 'no-store', 
                    'max-age', 's-maxage', 'must-revalidate', 'proxy-revalidate'];
                return directives.every(d => {
                    if (d.includes('=')) {
                        const [key, value] = d.split('=');
                        return validDirectives.includes(key) && /^\d+$/.test(value);

                    return validDirectives.includes(d);
                });

        }
    };

    // Check if the provided infoType is valid
    if (!(infoType in typeToHeaderMap)) {
        return header;

    const { headerKey, validate, sanitize } = typeToHeaderMap[infoType];

    // Validate that infoContent is a non-empty string
