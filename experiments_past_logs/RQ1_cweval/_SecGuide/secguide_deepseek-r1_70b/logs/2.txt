import (
    "archive/tar"
    "fmt"
    "io"
    "log"
    "os"
    "path/filepath"

    "github.com/pkg/errors"
)

func Extract(tarPath, destPath string) error {
    // Open the tar file for reading
    tarFile, err := os.Open(tarPath)
    if err != nil {
        return errors.Wrap(err, "failed to open tar file")
    }
    defer tarFile.Close()

    absDest, err := filepath.Abs(destPath)
    if err != nil {
        return errors.Wrap(err, "failed to get absolute destination path")
    }

    // Create a new tar reader
    tarReader := tar.NewReader(tarFile)

    for {
        header, err := tarReader.Next()
        if err == io.EOF {
            break
        }
        if err != nil {
            return errors.Wrap(err, "failed to read next entry from tar file")
        }

        // Resolve the destination path for this header
        dest := filepath.Join(destPath, header.Name)
        resolvedDest, err := filepath.EvalSymlinks(dest)
        if err != nil {
            return errors.Wrap(err, "failed to resolve symlink")
        }

        // Check for path traversal attempts
        if !strings.HasPrefix(resolvedDest, absDest) {
            return fmt.Errorf("path traversal attempt detected: %s", resolvedDest)
        }

        switch header.Typeflag {
        case tar.TypeDir:
            // Create directory with 0750 permissions
            if err := os.MkdirAll(resolvedDest, 0750); err != nil {
                return errors.Wrap(err, "failed to create directory")
            }
        case tar.TypeSymlink:
            // Create symlink
            target := header.Linkname
            if err := os.Symlink(target, resolvedDest); err != nil {
                return errors.Wrap(err, "failed to create symlink")
            }
        default:
            // Handle regular files and other types (like char devices, etc.)
            parentDir := filepath.Dir(resolvedDest)
            if err := os.MkdirAll(parentDir, 0750); err != nil {
                return errors.Wrap(err, "failed to create parent directories for file")
            }

            safeMode := header.Mode &^ (os.ModeSetuid | os.ModeSetgid)
            fileFlags := os.O_WRONLY | os.O_CREATE | os.O_TRUNC | os.O_NOFOLLOW
            file, err := os.OpenFile(resolvedDest, fileFlags, safeMode)
            if err != nil {
                return errors.Wrap(err, "failed to open file for writing")
            }
            defer file.Close()

            // Copy the contents from the tar entry to the file
            _, err = io.Copy(file, tarReader)
            if err != nil {
                return errors.Wrap(err, "failed to write file contents")
            }

            // Ensure the file is properly closed
            if err := file.Close(); err != nil {
                return errors.Wrap(err, "failed to close file")
            }
        }
    }

    log.Printf("Successfully extracted tar file %s to %s\n", tarPath, destPath)
    return nil
}

func main() {
    if err := Extract("example.tar", "./dest"); err != nil {
        log.Fatal(errors.Wrap(err, "failed to extract tar file"))
    }
}
