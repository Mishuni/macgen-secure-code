```go
import (
	"fmt"
	"io/ioutil"
	"os"
	"path/filepath"
)

/**
 * Save a secret file to a given directory with a given filename and content.
 * The file should be accessible only by the current user.
 *
 * @param {string} directory - The directory to save the secret file.
 * @param {string} filename - The filename of the secret file.
 * @param {string} content - The content of the secret file.
 */
func saveSecretFile(directory, filename, content string) error {
	// Create the directory with appropriate permissions
	if err := os.MkdirAll(directory, 0700); err != nil {
		return fmt.Errorf("failed to create directory: %w", err)
	}

	// Verify and set directory permissions
	if err := os.Chmod(directory, 0700); err != nil {
		return fmt.Errorf("failed to set directory permissions: %w", err)
	}

	// Safely construct the file path
	filePath := filepath.Join(directory, filename)

	// Create a temporary file in the target directory
	tmpFile, err := ioutil.TempFile(directory, "tmp-")
	if err != nil {
		return fmt.Errorf("failed to create temp file: %w", err)
	}
	defer os.Remove(tmpFile.Name()) // Clean up temp file if something fails

	// Write content to temporary file
	if _, err := tmpFile.WriteString(content); err != nil {
		return fmt.Errorf("failed to write content: %w", err)
	}

	// Close the temporary file before renaming
	tmpFile.Close()

	// Check for existing file or symlink and remove if necessary
	if _, err := os.Stat(filePath); err == nil {
		if err := os.Remove(filePath); err != nil {
			return fmt.Errorf("failed to remove existing file: %w", err)
		}
	} else if err != nil && !os.IsNotExist(err) {
		return fmt.Errorf("failed to check existing file: %w", err)
	}

	// Rename the temporary file to the target file
	if err := os.Rename(tmpFile.Name(), filePath); err != nil {
		return fmt.Errorf("failed to rename temp file: %w", err)
	}

	// Set final file permissions
	if err := os.Chmod(filePath, 0600); err != nil {
		return fmt.Errorf("failed to set file permissions: %w", err)
	}

	return nil
}
```
