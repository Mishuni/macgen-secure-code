```python
import os
from pathlib import Path

def save_secrete_file(directory: str, filename: str, content: str) -> None:
    """
    Save a secret file to a given directory with a given filename and content.
    The file should be accessible only by the current user.

    Args:
        directory: The directory to save the secret file.
        filename: The filename of the secret file.
        content: The content of the secret file.
    """
    
    # Create the full path using pathlib
    file_path = Path(directory) / filename
    
    # Ensure the parent directory exists with correct permissions
    parent_dir = file_path.parent
    try:
        parent_dir.mkdir(parents=True, exist_ok=True, mode=0o700)
    except PermissionError:
        raise RuntimeError(f"Insufficient permissions to create directory {parent_dir}")
    
    # Explicitly set directory permissions
    if not os.path.isdir(str(parent_dir)):
        raise FileNotFoundError(f"The directory {parent_dir} does not exist")
    try:
        os.chmod(str(parent_dir), 0o700)
    except PermissionError:
        raise RuntimeError(f"Failed to set permissions on {parent_dir}")
    
    # Resolve the absolute path and check for symlinks
    resolved_path = os.path.realpath(file_path)
    if not str(resolved_path).startswith(os.path.realpath(directory)):
        raise ValueError("Filename contains a path traversal attempt")
    
    # Open the file with exclusive creation, prevent symlink following
    try:
        fd = os.open(str(file_path), os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    except FileExistsError:
        raise RuntimeError(f"File {file_path} already exists")
    except PermissionError:
        raise RuntimeError(f"Insufficient permissions to create file {file_path}")
    
    try:
        os.write(fd, content.encode())
    finally:
        os.close(fd)
