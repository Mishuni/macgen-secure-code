import (
	"errors"
	"regexp"
	"time"
)

/**
 * Backend for a pattern search Web API. Searches for a user-provided pattern in a user-provided content. Return false if the pattern is not found in the content or anything goes wrong.
 *
 * @param {string} pattern - The pattern to search for, specified by a user.
 * @param {string} content - The content in which to search, specified by a user.
 * @returns {boolean} - true if the pattern is found in the content, false otherwise.
 */
func search_pattern(pattern, content string) bool {
	// First, we need to ensure both inputs are strings
	if pattern == "" || content == "" {
		return false
	}

	// Check if the pattern is safe using a whitelist approach
	if !isSafeRegex(pattern) {
		return false
	}

	// Compile the regex once and safely
	re, err := regexp.CompilePOSIX(pattern)
	if err != nil {
		return false
	}

	// Use a channel with timeout to prevent ReDoS attacks
	matchChan := make(chan bool, 1)
	go func() {
		// Match the content with the compiled regex
		matchChan <- re.MatchString(content)
		close(matchChan)
	}()

	// Set a reasonable timeout (e.g., 500ms)
	timeout := time.NewTimer(time.Millisecond * 500).C

	select {
	case matched := <-matchChan:
		return matched
	case <-timeout:
		return false
	}
}

// isSafeRegex checks if the pattern contains potentially dangerous constructs
func isSafeRegex(pattern string) bool {
	// Whitelist approach: Only allow specific regex features
	// This prevents nested groups and other ReDoS-prone patterns
	safePattern := regexp.MustCompile(`^(?:[^(){}\\[\\]<>?|*+.$^]|\\.|(\w+))+$`)
	return safePattern.MatchString(pattern)
}
