    RAND_bytes(result.iv, AES_BLOCK_SIZE);

    // Derive key from provided key bytes using first 128 bits (16 bytes) for AES-128
    size_t key_size = 16; // Using AES-128 bit key (16 bytes)
    if (key_len < key_size) {
        // Not enough key material available - handle error
        free(result.iv);
        return result;
    }
    
    result.key = malloc(key_size);
    if (!result.key) {
        free(result.iv);
        return result;
    }
    memcpy(result.key, key, key_size);

    AES_KEY aes_key;
    if (AES_set_encrypt_key(result.key, 8 * key_size, &aes_key) != 0) {
        // Failed to set encryption key - handle error
        free(result.iv);
        free(result.key);
        return result;
    }

    // Calculate required buffer size for encrypted data (including padding)
    size_t padded_data_len = data_len + (AES_BLOCK_SIZE - (data_len % AES_BLOCK_SIZE));
    
    // Allocate memory for encrypted data
    result.encrypted_data = malloc(padded_data_len);
    if (!result.encrypted_data) {
        free(result.iv);
        free(result.key);
        return result;
    }

    // Copy and pad plaintext data if necessary
    memcpy(result.encrypted_data, data, data_len);
    
    // Encrypt using AES-CBC
    AES_cbc_encrypt(
        result.encrypted_data,  // Input data
        result.encrypted_data,  // Output data
        padded_data_len,        // Length of input data in bytes
        &aes_key,               // Key
        result.iv,              // IV
        AES_ENCRYPT             // Operation (encrypt/decrypt)
    );

    result.encrypted_len = padded_data_len;
