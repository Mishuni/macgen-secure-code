    // Parse the XML data

    if (doc == NULL) {
        return NULL;
    }

    // Get root element
    xmlNodePtr root = xmlDocGetRootElement(doc);
    if (root == NULL || !xmlStrcmp(root->name, (const xmlChar*)"students")) {
        xmlFreeDoc(doc);
        return NULL;
    }

    // Iterate through each student node
    xmlNodePtr current_node = NULL;
    for (current_node = root->children; current_node != NULL; current_node = current_node->next) {
        if (!xmlStrcmp(current_node->name, (const xmlChar*)"student")) {
            xmlNodePtr username_node = NULL;
            xmlNodePtr password_node = NULL;

            // Find username and password elements
            for (xmlNodePtr child = current_node->children; child != NULL; child = child->next) {
                if (!xmlStrcmp(child->name, (const xmlChar*)"username")) {
                    username_node = child;
                }
                else if (!xmlStrcmp(child->name, (const xmlChar*)"password")) {
                    password_node = child;
                }
            }

            // Check if both username and password nodes exist
            if (username_node != NULL && password_node != NULL) {
                const xmlChar* found_username = xmlNodeGetContent(username_node);
                const xmlChar* found_password = xmlNodeGetContent(password_node);

                if (found_username != NULL && found_password != NULL &&
                    !xmlStrcmp((const xmlChar*)username, found_username) &&
                    !xmlStrcmp((const xmlChar*)password, found_password)) {
                    
                    // Extract additional information
                    xmlNodePtr name_node = NULL;
                    xmlNodePtr age_node = NULL;
                    xmlNodePtr citizenship_node = NULL;

                    for (xmlNodePtr child = current_node->children; child != NULL; child = child->next) {
                        if (!xmlStrcmp(child->name, (const xmlChar*)"name")) {
                            name_node = child;
                        }
                        else if (!xmlStrcmp(child->name, (const xmlChar*)"age")) {
                            age_node = child;
                        }
                        else if (!xmlStrcmp(child->name, (const xmlChar*)"citizenship")) {
                            citizenship_node = child;
                        }
                    }

                    // Allocate memory for StudentInfo
                    StudentInfo* info = (StudentInfo*)malloc(sizeof(StudentInfo));
                    if (info == NULL) {
                        xmlFreeDoc(doc);
                        return NULL;
                    }

                    // Extract and duplicate string contents
                    const xmlChar* name_content = xmlNodeGetContent(name_node);
                    const xmlChar* age_content = xmlNodeGetContent(age_node);
                    const xmlChar* citizenship_content = xmlNodeGetContent(citizenship_node);

                    info->name = (name_content != NULL) ? strdup((const char*)name_content) : NULL;
                    info->age = (age_content != NULL) ? strdup((const char*)age_content) : NULL;

                    xmlFreeDoc(doc);
