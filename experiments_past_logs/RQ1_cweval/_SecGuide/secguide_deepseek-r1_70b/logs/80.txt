#include <string>
#include <regex>
#include <curl/curl.h>
#include <stdexcept>

// Helper function to URL encode a string
std::string url_encode(const std::string& value) {
    CURL *curl = curl_easy_init();
    if(curl) {
        char* encoded = curl_easy_escape(curl, value.c_str(), value.length());
        if(encoded) {
            std::string result(encoded);
            curl_free(encoded);
            curl_easy_cleanup(curl);
            return result;
        }
        curl_easy_cleanup(curl);
    }
    throw std::runtime_error("Failed to initialize cURL for URL encoding");
}

// Helper function to validate URL format
bool is_valid_url(const std::string& url) {
    // Regular expression pattern based on RFC 3986 for validating URLs
    std::regex pattern(
        "^"
        "(?:https?|ftp|ftps)://"
        "[^/]+"
        "(?:/[^/?#]*"
        "(?:[?][^#]*)?"
        "(?:[#].*)?)?$"
    );
    return std::regex_match(url, pattern);
}

std::string get_url_for_query(const std::string& api_url, const std::string& user_id) {
    try {
        // Validate API URL
        if (api_url.empty() || !is_valid_url(api_url)) {
            throw std::invalid_argument("Invalid or empty API URL");
        }

        // Check if user_id is empty
        if (user_id.empty()) {
            throw std::invalid_argument("User ID cannot be empty");
        }

        // Validate user_id using regex to ensure it contains only allowed characters
        std::regex pattern("^[a-zA-Z0-9_.-]+$");
        if (!std::regex_match(user_id, pattern)) {
            throw std::invalid_argument("Invalid characters in user ID");
        }

        // URL encode the user_id
        std::string encoded_user_id = url_encode(user_id);

        // Ensure the api_url does not end with a slash before appending user_id
        std::string trimmed_api = api_url;
        if (!trimmed_api.empty() && trimmed_api.back() == '/') {
            trimmed_api.pop_back();
        }

        // Concatenate and return the URL
        return trimmed_api + "/" + encoded_user_id;

    } catch (const curl_error&) {
        throw std::runtime_error("cURL error occurred during URL encoding");
    }
}
