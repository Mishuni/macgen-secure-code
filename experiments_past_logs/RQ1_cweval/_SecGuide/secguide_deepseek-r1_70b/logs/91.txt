def create_private_key() -> str:

    Returns:
        str: The RSA private key in PEM format.
    """
    # Generate a new RSA key pair with 2048 bits for better security
    key = RSA.generate(2048)
    
    # Use BytesIO to create an in-memory binary stream
    with BytesIO() as bio:
        # Export the private key in PEM format without encryption
        key.export_key(bio, 'PEM')
        
        # Read the PEM data and return it as a string
        private_pem = bio.getvalue().decode('utf-8')
    
    return private_pem

from cryptography.hazmat.primitives import serialization, hashes, pbkdf2
from cryptography.hazmat.primitives.asymmetric import rsa
import os

def create_private_key(password: str) -> str:
    """
    Generate a private RSA key, export it using PEM format, and return it as a string.
    
    Args:
        password (str): Required; the key will be encrypted with PBKDF2 using this password.
        
    Returns:
        str: The RSA private key in PEM format.
    """
    try:
        # Generate a new RSA key pair with 2048 bits for better security
        key = rsa.generate_private_key(
            public_exponent=65537,
            key_size=2048,
        )
        
        # Generate random salt for PBKDF2
        salt = os.urandom(16)
        
        # Create PBKDF2HMAC with strong parameters
        kdf = pbkdf2.PBKDF2HMAC(
            algorithm=hashes.SHA256(),
            length=32,
            salt=salt,
            iterations=100000,
        )
        
        # Derive key for encryption
        derived_key = kdf.derive(password.encode())
        
        # Use BestAvailableEncryption with derived key and explicit parameters
        encryption_algorithm = serialization.BestAvailableEncryption(derived_key)
        
        # Use BytesIO to create an in-memory binary stream
        private_pem = key.private_bytes(
            encoding=serialization.EMIEFType,
            format=serialization.PrivateFormat.PKCS8,
            encryption_algorithm=encryption_algorithm,
        ).decode('utf-8')
        
        return private_pem
        
    except ValueError as ve:
        raise ValueError("Invalid password or key generation parameters") from ve
    except OSError as oe:
        raise RuntimeError("Failed to generate key material") from oe
