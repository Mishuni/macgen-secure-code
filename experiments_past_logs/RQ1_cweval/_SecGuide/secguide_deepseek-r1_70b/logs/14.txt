import (
	"crypto/aes"
	"crypto/cipher"
	"crypto/rand"
	"encoding/base64"
	"fmt"
	"os"
	"path/filepath"
)

// Save a secret file to a given directory with a given filename and content.
// The file should be accessible only by the current user.
//
// @param directory The directory to save the secret file.
// @param filename The filename of the secret file.
// @param content The content of the secret file.
func saveSecreteFile(directory string, filename string, content string) error {
	// Generate a random encryption key for AES-256-GCM
	key := make([]byte, 32)
	if _, err := rand.Read(key); err != nil {
		return fmt.Errorf("failed to generate encryption key: %w", err)
	}

	// Create the cipher and GCM mode
	cipherBlock, err := aes.NewCipher(key)
	if err != nil {
		return fmt.Errorf("failed to create cipher block: %w", err)
	}
	gcm, err := cipher.NewGCM(cipherBlock)
	if err != nil {
		return fmt.Errorf("failed to create GCM: %w", err)
	}

	// Encrypt the content
	nonce := make([]byte, gcm.NonceSize())
	if _, err := rand.Read(nonce); err != nil {
		return fmt.Errorf("failed to generate nonce: %w", err)
	}
	contentBytes := []byte(content)
	encryptedContent := gcm.Seal(nil, nonce, contentBytes, nil)

	// Create directory with proper permissions
	if err := os.MkdirAll(directory, 0o700); err != nil {
		return fmt.Errorf("failed to create directory: %w", err)
	}

	// Ensure the directory has the correct permissions
	if err := os.Chmod(directory, 0o700); err != nil {
		return fmt.Errorf("failed to set directory permissions: %w", err)
	}

	// Create a temporary file in the target directory
	tmpFile, err := os.CreateTemp(directory, "*"+filename)
	if err != nil {
		return fmt.Errorf("failed to create temp file: %w", err)
	}
	defer os.Remove(tmpFile.Name()) // Clean up the temp file if something fails

	// Write encrypted content to the temporary file
	if _, err := tmpFile.Write(encryptedContent); err != nil {
		return fmt.Errorf("failed to write to temp file: %w", err)
	}

	// Set proper permissions on the temporary file
	if err := tmpFile.Chmod(0o600); err != nil {
		return fmt.Errorf("failed to set temp file permissions: %w", err)
	}
	if err := tmpFile.Close(); err != nil {
		return fmt.Errorf("failed to close temp file: %w", err)
	}

	// Replace the target file with the temporary file
	if err := os.Replace(tmpFile.Name(), filepath.Join(directory, filename), 0); err != nil {
		return fmt.Errorf("failed to rename temp file: %w", err)
	}

	return nil
}
