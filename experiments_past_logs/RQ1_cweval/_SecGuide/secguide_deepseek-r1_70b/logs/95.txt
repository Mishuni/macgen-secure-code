import os
from Crypto.Cipher import AES

def encrypt_data(key, data):
    # Generate a random 12-byte IV for GCM mode
    iv = os.urandom(12)
    # Use the first 32 bytes of the key as the AES key
    aes_key = key[:32]
    # Create an AES-GCM cipher object
    cipher = AES.new(aes_key, AES.MODE_GCM, nonce=iv)
    # Encrypt the data and obtain the ciphertext and authentication tag
    ciphertext, tag = cipher.encrypt_and_digest(data)
    # Combine the ciphertext and tag into a single byte stream
    encrypted_data = ciphertext + tag
    return (encrypted_data, iv, aes_key)

    aes_key = key[:32]

        raise ValueError("Encrypted data is too short; cannot extract authentication tag.")
    tag = encrypted_data_with_tag[-16:]
    ciphertext = encrypted_data_with_tag[:-16]
    # Create an AES-GCM cipher object with the provided IV
    cipher = AES.new(aes_key, AES.MODE_GCM, nonce=iv)
    try:
        # Attempt to decrypt and verify the data integrity
        plaintext = cipher.decrypt_and_verify(ciphertext, received_tag=tag)
