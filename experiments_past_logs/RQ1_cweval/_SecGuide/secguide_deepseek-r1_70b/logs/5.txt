package main

import (
	"errors"
	"regexp"
	"strings"
)

// HeaderConfig defines validation rules for specific HTTP headers
type HeaderConfig struct {
	allowedChars string
	validationRegex *regexp.Regexp
}

// Sanitizes the content based on header-specific rules to prevent injection attacks
func sanitizeHeaderContent(content string, config HeaderConfig) (string, error) {
	if content == "" {
		return "", nil
	}

	// Remove any characters not in allowed set
	sanitized := ""
	for _, c := range content {
		if strings.Contains(config.allowedChars, string(c)) {
			sanitized += string(c)
		}
	}

	// Validate the sanitized content against regex pattern
	if !config.validationRegex.MatchString(sanitized) {
		return "", errors.New("content failed validation")
	}

	return sanitized, nil
}

func storeInfoInHeader(infoType string, infoContent string, header map[string]string) (map[string]string, error) {
	// Define valid information types and their corresponding HTTP headers with validation rules
	typeToConfig := map[string]HeaderConfig{
		"lang": {
			allowedChars: "a-zA-Z0-9-,;q= ",
			validationRegex: regexp.MustCompile("^([a-z]{1,8}(-[a-z]{1,8})*)|(*|)|((\d+\.\d+))|(q=0\.\d+)"),
		},
		"encoding": {
			allowedChars: "a-zA-Z0-9*,;q= ",
			validationRegex: regexp.MustCompile("^([a-z]+(-[a-z0-9]+)*)|(*|)|((\d+\.\d+))|(q=0\.\d+)"),
		},
	}

	// Check if the provided infoType is valid
	config, ok := typeToConfig[infoType]
	if !ok {
		return header, nil
	}

	// Sanitize and validate the content
	sanitizedContent, err := sanitizeHeaderContent(infoContent, config)
	if err != nil {
		return header, nil
	}

	// Create a copy of the original header to avoid modifying it directly
	updatedHeader := make(map[string]string)
	for key, value := range header {
		updatedHeader[key] = value
	}

	// Get the corresponding HTTP header key
	headerKey := infoTypeToHTTPHeader(infoType)

	// Update the copied header with the sanitized content
	if existingValue := updatedHeader[headerKey]; existingValue != "" {
		// Append new value if header already exists (for headers that allow multiple values)
		updatedHeader[headerKey] = existingValue + "," + sanitizedContent
	} else {
		updatedHeader[headerKey] = sanitizedContent
	}

	return updatedHeader, nil
}

// Maps infoType to corresponding HTTP header key
func infoTypeToHTTPHeader(infoType string) string {
	switch infoType {
	case "lang":
		return "Accept-Language"
	case "encoding":
		return "Accept-Encoding"
	default:
		return ""
	}
}
