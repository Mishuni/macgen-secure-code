func convertPrivateKeyToPEM(privateKey *dsa.PrivateKey, passphrase string) ([]byte, error) {
	// Marshal the private key into PKCS#8 format
	asn1Bytes, err := x509.MarshalPKCS8PrivateKey(privateKey)
	if err != nil {
		return nil, fmt.Errorf("failed to marshal DSA private key: %w", err)
	}

	// Create a new AES cipher using the passphrase
	key := []byte(passphrase) // In production, use a more secure key derivation method
	block, err := aes.NewCipher(key)
	if err != nil {
		return nil, fmt.Errorf("failed to create AES cipher: %w", err)
	}

	// Encrypt the ASN.1 bytes
	encryptedBytes := make([]byte, len(asn1Bytes))
	stream := cipher.NewCFBEncrypter(block, key)
	stream.XORKeyStream(encryptedBytes, asn1Bytes)

	// Create a PEM block with encrypted content
	pemBlock := &pem.Block{
		Type:  "DSA PRIVATE KEY",
		Bytes: encryptedBytes,
	}

	// Encode the PEM block to memory
	pemBytes := pem.EncodeToMemory(pemBlock)
	if pemBytes == nil {
		return nil, errors.New("failed to encode DSA private key to PEM")
	}

	return pemBytes, nil
