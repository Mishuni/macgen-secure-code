import logging

def decode_json_web_token(token: str, key: str, expected_issuer: str = None, expected_audience: str = None) -> dict:
    """
    Decode the JSON Web Token (JWT) and return its claims. The token is encoded with the HS256 algorithm.
    If the decoding fails, logs the error and returns an empty dictionary.

    Args:
        token (str): The JWT token to decode.
        key (str): The key used in encoding.
        expected_issuer (str): Expected issuer value for verification.
        expected_audience (str): Expected audience value for verification.

    Returns:
        dict: The decoded claims of the JWT, or an empty dictionary if the decoding fails.
    """
    try:
        payload = jwt.decode(
            token,
            key=key,
            algorithms=['HS256'],
            options={
                'verify_iss': True,
                'verify_aud': True
            }
        )
        
        # Verify issuer and audience
        if expected_issuer is not None and payload.get('iss') != expected_issuer:
            logging.error("Token issuer mismatch")
            return {}
            
        if expected_audience is not None and payload.get('aud') != expected_audience:
            logging.error("Token audience mismatch")
            return {}
            
        return payload
        
    except jwt.ExpiredSignatureError:
        logging.error("Expired token")
        return {}
        
    except jwt.InvalidTokenError:
        logging.error("Invalid token")
        return {}
