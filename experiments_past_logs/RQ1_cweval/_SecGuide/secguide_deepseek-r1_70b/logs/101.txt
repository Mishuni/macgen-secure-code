from typing import Dict, Optional
from lxml import etree
import bcrypt

def get_student_info(
    xml_data: str, username: str, password: str
) -> Optional[Dict[str, str]]:
    """
    Parse the XML data and return the student information for the given username and password.
    
    Args:
        xml_data: The XML data containing the student information.
        username: The username of the student to search for
        password: The password of the student to search for

    Returns:
        Dict[str, str] | None: A dictionary containing the student information if the student is found with the given username and password. Otherwise, return None.
    """
    
    # Parse XML data securely without resolving entities to prevent XXE attacks
    parser = etree.XMLParser(resolve_entities=False)
    try:
        root = etree.fromstring(xml_data, parser=parser)
    except etree.ParseError:
        return None
    
    # Use bcrypt for secure password hashing and verification
    # Hash the input password for secure comparison
    hashed_password = bcrypt.hashpw(password.encode(), bcrypt.gensalt()).decode()
    
    # Use XPath to find the student element with matching username and hashed password
    students = root.xpath(
        ".//student[username=$username and password=$hashed_password]",
        namespace=None,
        username=username,
        hashed_password=hashed_password
    )
    
    if len(students) != 1:
        return None
    
    student = students[0]
    
    # Safely retrieve text, handling missing elements gracefully
    info = {}
    for key in ["name", "citizenship", "age"]:
        element = student.find(key)
        info[key] = element.text if element is not None else ""
    
    return info
