/**
 * Removes trailing spaces from a given string in place.
 *
 * @param str The input C-style string to be trimmed. The string must be null-terminated.
 */
void trim_spaces(char* str) {
    // Check for NULL pointer to prevent undefined behavior
    if (str == NULL) {
        return;
    }

    // Find the last non-space character
    size_t length = strlen(str);
    size_t last_non_space = 0;

    // Iterate through each character starting from the end
    for (size_t i = 0; i < length; ++i) {
        if (str[length - 1 - i] != ' ') {
            last_non_space = length - i;
            break;
        }
    }

    // If all characters are spaces, set to empty string
    if (last_non_space == 0) {
        str[0] = '\0';
    } else {
        // Null-terminate after the last non-space character
        str[last_non_space] = '\0';
    }
}
