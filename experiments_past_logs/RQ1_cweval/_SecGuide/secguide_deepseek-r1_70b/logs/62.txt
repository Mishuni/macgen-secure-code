bool startsWith(const string& str, const string& prefix) {

    return str.substr(0, prefix.length()) == prefix;
}

string sanitizeUrl(const string& url) {
    string sanitized;
    for (char c : url) {
        if (isalnum(c) || c == '-' || c == '_' || c == '.' || c == '/' || c == '?' || 
            c == '=' || c == '&' || c == '#' || c == '%' || c == '+' || c == '@') {
            sanitized += c;
        } else {
            // Remove invalid characters (can be enhanced for more comprehensive sanitization)
            continue;
        }
    }
    return sanitized;
}

string generateSecureCode(string url) {
    if (url.empty()) {
        return url;
    }

    size_t pos = url.find("://");
    if (pos != string::npos) {
        string protocol = url.substr(0, pos);
        transform(protocol.begin(), protocol.end(), protocol.begin(), ::tolower);
        
        if (protocol == "http") {
            // Change HTTP to HTTPS while preserving the rest of the URL
            return "https://" + sanitizeUrl(url.substr(pos + 3));
        } else if (protocol == "https" || protocol != "http") {
            // For other protocols, keep them as-is and sanitize
            return sanitizeUrl(url);
        }
    } else {
        // Handle URLs without a protocol
        if (startsWith(url, "//")) {
            // Protocol-relative URL, convert to HTTPS
            return "https:" + sanitizeUrl(url);
        } else if (url[0] == '/' || url[0] == '\\') {
            // Relative path, keep as-is and sanitize
            return sanitizeUrl(url);
        } else {
            // No protocol specified, prepend HTTPS and sanitize
            return "https://" + sanitizeUrl(url);

    }
