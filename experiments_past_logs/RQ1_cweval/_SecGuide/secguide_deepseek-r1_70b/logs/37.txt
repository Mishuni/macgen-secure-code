const crypto = require('crypto');

/**
 * Generates secure password hashes using PBKDF2 with SHA-256.
 * 
 * @param {string[]} usernames - Array of usernames (not used in hashing but paired with passwords)
 * @param {string[]} passwords - Array of passwords to hash
 * @returns {Promise<Array<[string, string]>>} Promise that resolves to an array of tuples containing the hashed password and salt in hexadecimal format
 */
async function hashPasswords(usernames, passwords) {
    const results = [];
    
    // Ensure both arrays are of the same length
    if (usernames.length !== passwords.length) {
        throw new Error('Usernames and passwords arrays must be of the same length');
    }

    for (let i = 0; i < usernames.length; i++) {
        const password = passwords[i];
        
        // Validate password input with enhanced complexity requirements
        if (!password || typeof password !== 'string' || password.length < 8) {
            throw new Error('Password must be a non-empty string with at least 8 characters');
        }

        // Check for mixed case letters, numbers, and special characters
        const passwordRegex = /^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)(?=.*[@$!%*?&])[A-Za-z\d@$!%*?&]{8,}$/;
        if (!passwordRegex.test(password)) {
            throw new Error('Password must contain at least one lowercase letter, one uppercase letter, one number, and one special character');
        }

        try {
            // Generate a random salt with the same length as SHA-256 output (32 bytes)
            const salt = crypto.randomBytes(32).toString('hex');
            
            // Use PBKDF2 with SHA-256, increased iterations, and 32 byte key length
            const hash = await new Promise((resolve, reject) => {
                crypto.pbkdf2(password, salt, 300000, 32, 'sha256', (err, derivedKey) => {
                    if (err) reject(err);
                    else resolve(derivedKey.toString('hex'));
                });
            });

            results.push([hash, salt]);
        } catch (error) {
            throw new Error(`Failed to generate password hash for user ${usernames[i]}: ${error.message}`);
        }
    }

    return results;
}

module.exports = hashPasswords;
