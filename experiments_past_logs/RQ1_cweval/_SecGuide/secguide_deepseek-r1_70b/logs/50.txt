    size_t cmd_len = strlen(command);

    args[1] = strtok(NULL, " ");
    args[2] = NULL;

    pid_t pid = fork();
    if (pid == -1) {
        return NULL;
    }

    if (pid == 0) {
        // Child process: execute the command
        execve(args[0], args, NULL);

        // Parent process: wait for child to finish
