```go
}

func isDigit(c byte) bool {
    return c >= '0' && c <= '9'
}

func tokenize(s string) ([]Token, error) {
    tokens := make([]Token, 0)
    i := 0
    n := len(s)
    for i < n {
        if s[i] == ' ' {
            i++
            continue
        }
        if isDigit(s[i]) || (s[i] == '-' && (i == 0 || "+-*/(".Contains(string(s[i-1])))) {
            j := i
            for j < n && (isDigit(s[j]) || s[j] == '.') {
                j++
            }
            numStr := s[i:j]
            if len(numStr) == 0 {
                return nil, errors.New("invalid number")
            }
            num, err := strconv.ParseFloat(numStr, 64)
            if err != nil {
                return nil, err
            }
            tokens = append(tokens, Token{Type: "number", Value: num})
            i = j
        } else if "+-*/".Contains(string(s[i])) {
            op := string(s[i])
            tokens = append(tokens, Token{Type: "operator", Value: op})
            i++
        } else if s[i] == '(' || s[i] == ')' {
            t := string(s[i])
            tokens = append(tokens, Token{Type: "parenthesis", Value: t})
            i++
        } else {
            return nil, errors.New("invalid character")
        }
    }
    return tokens, nil
}

func evaluateTokens(tokens []Token) (float64, error) {
    output := make([]Token, 0)
    opStack := make([]Token, 0)

    precedence := map[string]int{
        "+": 1,
        "-": 1,
        "*": 2,
        "/": 2,
    }

    for _, token := range tokens {
        if token.Type == "number" {
            output = append(output, token)
        } else if token.Type == "operator" {
            opVal := token.Value.(string)
            for len(opStack) > 0 && opStack[len(opStack)-1].Type == "operator" &&
                precedence[opStack[len(opStack)-1].Value.(string)] >= precedence[opVal] {
                output = append(output, opStack[len(opStack)-1])
                opStack = opStack[:len(opStack)-1]
            }
            opStack = append(opStack, token)
        } else if token.Type == "parenthesis" {
            pVal := token.Value.(string)
            if pVal == "(" {
                opStack = append(opStack, token)
            } else { // ')'
                for opStack[len(opStack)-1].Value != "(" {
                    output = append(output, opStack[len(opStack)-1])
                    opStack = opStack[:len(opStack)-1]
                    if len(opStack) == 0 {
                        return 0, errors.New("Mismatched parentheses")
                    }
                }
                // Pop the '(' from stack
                opStack = opStack[:len(opStack)-1]
            }
        } else {
            return 0, errors.New("Invalid token type")
        }
    }

    for len(opStack) > 0 {
        output = append(output, opStack[len(opStack)-1])
        opStack = opStack[:len(opStack)-1]
    }

    stack := make([]float64, 0)
    for _, token := range output {
        if token.Type == "number" {
            stack = append(stack, token.Value.(float64))
        } else if token.Type == "operator" {
            opVal := token.Value.(string)
            if len(stack) < 2 {
                return 0, errors.New("Not enough operands")
            }
            b := stack[len(stack)-1]
            a := stack[len(stack)-2]
            switch opVal {
            case "+":
                stack[len(stack)-2] = a + b
            case "-":
                stack[len(stack)-2] = a - b
            case "*":
                stack[len(stack)-2] = a * b
            case "/":
                if b == 0 {
                    return 0, errors.New("Division by zero")
                }
                stack[len(stack)-2] = a / b
            default:
                return 0, errors.New("Unknown operator")
            }
        } else {
            return 0, errors.New("Invalid token in postfix")
        }
    }

    if len(stack) != 1 {
        return 0, errors.New("Invalid expression")
    }

    return stack[0], nil
}

func main() {
    expr := "3 + 5 * (2 - 4)"
    tokens, err := tokenize(expr)
    if err != nil {
        fmt.Println(err)
        return
    }
    result, err := evaluateTokens(tokens)
    if err != nil {
        fmt.Println(err)
        return
    }
    fmt.Printf("Result: %.2f\n", result)
}
```

### Explanation
1. **Tokenization**: The `tokenize` function processes the input string character by character to identify numbers, operators, and parentheses. Each identified token is added to a list.
2. **Shunting-Yard Algorithm**: The `evaluateTokens` function uses this algorithm to convert the infix expression (standard notation) into postfix notation. This involves using two stacks: one for output and one for operators. Operators are moved to the output stack based on their precedence, ensuring that higher precedence operators are processed first.
3. **Postfix Evaluation**: After converting to postfix notation, a stack is used to evaluate the expression. Each number is pushed onto the stack, and when an operator is encountered, the necessary operands are popped from the stack, the operation is performed, and the result is pushed back onto the stack.

This approach ensures that mathematical expressions are evaluated correctly, respecting operator precedence and handling parentheses appropriately.
