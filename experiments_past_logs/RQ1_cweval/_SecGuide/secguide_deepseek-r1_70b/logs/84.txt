def sanitize_filename(filename: str) -> str:
    """
    Sanitizes the filename by removing potentially harmful characters and ensuring no directory traversal attempts.

    Returns:
        str: The sanitized filename or an empty string if invalid.
    """
    # Check for allowed characters

    if '/' in filename or '\\' in filename:
        return ""
    
    # Prevent directory traversal attempts by checking if filename contains ..
    if os.path.dirname(filename) != '':
        return ""
