def process_header(header, content, info_type):
    # Define valid types and corresponding allowed characters
    valid_types = ['lang', 'encoding', 'cache']
    if info_type not in valid_types:
        return header.copy()
    
    allowed_chars_mapping = {
        'lang': r'[^a-zA-Z0-9-]',
        'encoding': r'[^a-zA-Z0-9-, ]',
        'cache': r'[^a-zA-Z0-9-,=]'
    }

        # Validate language tag format (e.g., en-US, fr-CA)

        allowed_encodings = {'gzip', 'deflate', 'br'}
        # Split and validate each encoding
        parts = [p.strip() for p in sanitized_content.split(',')]
        valid_parts = []
        for part in parts:
            if part.lower() in allowed_encodings:
                valid_parts.append(part)
        if not valid_parts:
            return header.copy()
        sanitized_content = ', '.join(valid_parts)
    elif info_type == 'cache':
        valid_directives = {'max-age', 's-maxage', 'no-cache', 'no-store', 'private', 'public'}
        # Split and validate each directive
        parts = [p.strip() for p in sanitized_content.split(',')]
        valid_parts = []
        for part in parts:
            if '=' in part:
                key, value = part.split('=', 1)
                if key in valid_directives:
                    valid_parts.append(part)
            else:
                if part in valid_directives:
                    valid_parts.append(part)
        if not valid_parts:
            return header.copy()
        sanitized_content = ', '.join(valid_parts)
    
    # Update the header with validated content
    new_header = header.copy()
    new_header[info_type] = sanitized_content
    return new_header
