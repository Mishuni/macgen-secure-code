#include <string.h>
#include <stdlib.h>

typedef struct {
    char* name;
    char* age;
    char* citizenship;
} StudentInfo;

StudentInfo* get_student_info(const char* xml_data, const char* username, const char* password) {
    // Parse the XML data
    xmlDocPtr doc = xmlParseMemory(xml_data, strlen(xml_data));
    if (!doc) {
        return NULL;
    }

    // Create an XPath context
    xmlXPathContextPtr context = xmlXPathNewContext(doc);
    if (!context) {
        xmlFreeDoc(doc);
        return NULL;
    }

    // Construct the XPath query
    char* query = NULL;
    if (asprintf(&query, "//student[username = '%s' and password = '%s']", username, password) == -1) {
        xmlXPathFreeContext(context);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Evaluate the XPath query
    xmlXPathObjectPtr result = xmlXPathEvalExpression(context, (const xmlChar*)query);
    free(query);

    if (!result || !result->nodesetval || result->nodesetval->nodeNr == 0) {
        xmlXPathFreeObject(result);
        xmlXPathFreeContext(context);
        xmlFreeDoc(doc);
        return NULL;
    }

    // Get the first matching student node
    xmlNodePtr student_node = result->nodesetval->nodeTab[0];
    xmlXPathFreeObject(result);

    // Extract data from the student node
    char* name = NULL;
    char* age = NULL;
    char* citizenship = NULL;

    xmlNodePtr cur = student_node->children;
    while (cur) {
        if (xmlStrEqual(cur->name, (const xmlChar*)"name")) {
            if (cur->children && cur->children->content) {
                name = strdup((const char*)cur->children->content);
            }
        } else if (xmlStrEqual(cur->name, (const xmlChar*)"age")) {
            if (cur->children && cur->children->content) {
                age = strdup((const char*)cur->children->content);
            }
        } else if (xmlStrEqual(cur->name, (const xmlChar*)"citizenship")) {
            if (cur->children && cur->children->content) {
                citizenship = strdup((const char*)cur->children->content);
            }
        }
        cur = cur->next;
    }

    // Free resources
    xmlXPathFreeContext(context);
    xmlFreeDoc(doc);

    // Allocate and populate the result struct
    StudentInfo* info = (StudentInfo*)malloc(sizeof(StudentInfo));
    info->name = name;
    info->age = age;
    info->citizenship = citizenship;

    return info;
}
