void saveSecreteFile(const std::string& directory, const std::string& filename, const std::string& content) {

    if (filename.find('/') != std::string::npos || filename.find('\\') != std::string::npos) {
        // Invalid characters (e.g., directory separators)
        return;
    }

    if (filename.find("..") != std::string::npos) {
        // Prevent directory traversal
        return;
    }

    if (filename.find('.') == std::string::npos) {
        // Must have at least one dot
        return;
    }

    int dot_count = 0;
    for (char c : filename) {
        if (c == '.') dot_count++;
    }

    if (dot_count != 1) {
        // Only one dot allowed
        return;
    }

    if (filename[0] == '.' || filename.back() == '.') {
        // Dot cannot be at start or end
        return;
    }

    // Construct full path
    fs::path dir_path(directory);
    fs::path file_path(filename);
    fs::path full_path = dir_path / file_path;

    // Normalize paths
    fs::path dir_canonical = dir_path.make_preferred().canonical();
    fs::path full_canonical = full_path.make_preferred().canonical();

    // Ensure the file is within the specified directory
    if (full_canonical.parent_path() != dir_canonical) {
        return;
    }

    // Create directory if it doesn't exist
    if (!fs::exists(dir_path)) {
        if (!fs::create_directories(dir_path)) {
            return;
        }
    }

    // Write content to file
    std::ofstream file(full_path);
    if (!file) {
        return;
    }

    file << content;
    file.close();

    // Set file permissions to 0600 (read/write for owner only)
    chmod(full_path.string().c_str(), S_IRUSR | S_IWUSR);
}
