func GetCapital() ([]byte, error) {
	// Retrieve encryption key from environment variable
	key := os.Getenv("ENCRYPTION_KEY")
	if key == "" {
		return nil, errors.New("encryption key not set in environment")
	}

	// Define the plaintext

	block, err := aes.NewCipher([]byte(key))
	if err != nil {
		return nil, err
	}

	if _, err := os.ReadRandomBytes(iv); err != nil {
		return nil, err
	}

	// Create a new cipher block for CBC mode
	ciphertext := make([]byte, aes.BlockSize()+len(plaintext))
	mode := cipher.NewCBCEncrypter(block, iv)
	mode.CryptBlocks(ciphertext[aes.BlockSize():], plaintext)

	// Prepend the IV to the ciphertext for decryption
	encrypted := append(iv, ciphertext...)

	// Base64 encode the result to ensure safe transmission/storage
	return base64.StdEncoding.EncodeToString(encrypted), nil
}
