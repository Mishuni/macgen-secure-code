        std::string domain_part = target.substr(protocol_pos + 8);
        size_t slash_pos = domain_part.find('/');
        if (slash_pos != std::string::npos) {
            target_domain = domain_part.substr(0, slash_pos);
        } else {
            target_domain = domain_part;
        }
    } else {
        // Assume the URL starts with 'https://' (for simplicity)
        target_domain = target.substr(8);
        size_t slash_pos = target_domain.find('/');
        if (slash_pos != std::string::npos) {
            target_domain = target_domain.substr(0, slash_pos);
        }
    }

    // Split both domains into parts
    std::vector<std::string> provided_parts = split_domain(domain);
    std::vector<std::string> target_parts = split_domain(target_domain);

    // Check if the target is a subdomain of the provided domain
