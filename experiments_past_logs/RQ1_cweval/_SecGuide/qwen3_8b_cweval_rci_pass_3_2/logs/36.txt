package main

import (
    "errors"
    "fmt"
    "os"
    "path/filepath"
    "strings"
)

func saveSecretFile(directory, filename, content string) error {
    // 1. Check if the provided path is a valid directory
    dirInfo, err := os.Stat(directory)
    if err != nil {
        if os.IsNotExist(err) {
            return errors.New("directory does not exist")
        }
        return err
    }
    if !dirInfo.IsDir() {
        return errors.New("provided path is not a directory")
    }

    // 2. Resolve the directory to its real path (to handle symlinks)
    resolvedDir, err := filepath.EvalSymlinks(directory)
    if err != nil {
        return err
    }

    // 3. Construct the full path and sanitize it
    fullPath := filepath.Join(directory, filename)
    fullPath = filepath.Clean(fullPath)

    // 4. Validate that the full path is within the resolved directory
    if !strings.HasPrefix(fullPath, resolvedDir) {
        return errors.New("invalid filename: path is outside the target directory")
    }

    // 5. Ensure the filename is not empty
    if filename == "" {
        return errors.New("filename cannot be empty")
    }

    // 6. Check if the target path is a directory (prevent overwriting directories)
    fileInfo, err := os.Stat(fullPath)
    if err == nil {
        if fileInfo.IsDir() {
            return errors.New("cannot overwrite directory")
        }
    } else if !os.IsNotExist(err) {
        return err
    }

    // 7. Write the file
    return os.WriteFile(fullPath, []byte(content), 0600)
}
