### ✅ **Final Code**

```go
package main

import (
    "fmt"
    "net/http"
    "time"
    "text/template"
)

var recentInputs = make(map[string]time.Time)

func loopDetectionHandler(w http.ResponseWriter, r *http.Request) {
    // 1. Validate input length
    userInput := r.URL.Query().Get("input")
    if userInput != "" && len(userInput) > 100 {
        http.Error(w, "Input too long", http.StatusBadRequest)
        return
    }

    // 2. Loop detection
    if userInput != "" {
        if time.Since(recentInputs[userInput]) < 5*time.Second {
            http.Error(w, "Loop detected", http.StatusTooManyRequests)
            return
        }
        recentInputs[userInput] = time.Now()
    }

    // 3. Set Content Security Policy
    w.Header().Set("Content-Security-Policy", "default-src 'self'; script-src 'self'; style-src 'self';")

    // 4. Parse and render template
    tmpl, err := template.New("template").Parse(`<html><body><h1>Hello</h1></body></html>`)
    if err != nil {
        http.Error(w, "Template error", http.StatusInternalServerError)
        return
    }

    if err := tmpl.Execute(w, nil); err != nil {
        http.Error(w, "Template execution error", http.StatusInternalServerError)
        return
    }
}

func main() {
    http.HandleFunc("/", loopDetectionHandler)
    http.ListenAndServe(":8080", nil)
}
```
