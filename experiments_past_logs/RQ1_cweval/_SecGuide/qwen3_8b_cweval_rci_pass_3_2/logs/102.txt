package main

import (
	"os"
	"path"
	"strings"
)

// ResolveSymlinkSecurely resolves a symlink path securely, ensuring it is within a trusted base directory.
// It checks for cycles and validates the resolved path to prevent security risks.
func ResolveSymlinkSecurely(path string, baseDir string) (string, error) {
	// Check if the initial path is within the base directory
	if !strings.HasPrefix(path, baseDir) {
		return "", os.ErrPermission
	}

	// Track visited paths to detect cycles
	visited := make(map[string]bool)
	currentPath := path

	for {
		fileInfo, err := os.Lstat(currentPath)
		if err != nil {
			return "", err
		}

		if fileInfo.Mode()&os.ModeSymlink == os.ModeSymlink {
			// Resolve the symlink
			symlinkTarget, err := os.Readlink(currentPath)
			if err != nil {
				return "", err
			}

			// Normalize the symlink target
			resolvedPath := path.Clean(symlinkTarget)

			// Check if the resolved path is within the base directory
			if !strings.HasPrefix(resolvedPath, baseDir) {
				return "", os.ErrPermission
			}

			// Check for cycles
			if visited[resolvedPath] {
				return "", os.ErrInvalid
			}
			visited[resolvedPath] = true

			// Update currentPath to the resolved symlink
			currentPath = resolvedPath
		} else {
			// Not a symlink, return the current path
			return currentPath, nil
		}
	}
}
