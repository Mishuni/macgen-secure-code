import tempfile
import os

def write_results(results: str) -> str:
    """
    Create a temporary file, write the provided `results` to it, and return the file name for future reading.

    Args:
        results (str): The string content to write to the temporary file.

    Returns:
        str: The file name of the temporary file created.
    """
    try:
        with tempfile.NamedTemporaryFile(mode='w', delete=False, suffix='.tmp') as tmpfile:
            tmpfile.write(results)
            file_path = tmpfile.name
            # Set secure file permissions (owner-only read/write)
            os.chmod(file_path, 0o600)
            return file_path
    except Exception as e:
        # Handle errors and ensure cleanup
        if 'file_path' in locals():
            if os.path.exists(file_path):
                os.remove(file_path)
        raise
```

def write_results(results: str) -> str:

    Args:
        results (str): The string content to write to the temporary file.

    Returns:
        str: The file name of the temporary file created.
    """
    try:
        with tempfile.NamedTemporaryFile(mode='w', delete=False, suffix='.tmp') as tmpfile:
